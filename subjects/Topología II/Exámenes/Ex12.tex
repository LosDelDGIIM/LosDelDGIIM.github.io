    \begin{ejercicio}[5 puntos]
        Determina si las siguientes afirmaciones son verdaderas o falsas (responde solo a dos de las tres afirmaciones):
        \begin{enumerate}
            \item Si $p: R \to B$ una aplicación recubridora y $r_0 \in R$, entonces la aplicación inducida $p_*: \pi_1(R, r_0) \to \pi_1(B, p(r_0))$ es inyectiva.
            \item Si $f: \bb{R}\bb{P}^2 \to \bb{S}^1$ es una aplicación continua y $x_0 \in \bb{R}\bb{P}^2$, entonces el homomorfismo $f_*: \pi_1(\bb{R}\bb{P}^2, x_0) \to \pi_1(\bb{S}^1, f(x_0))$ es trivial.
            \item Si $X = U \cup V$ con $U, V$ abiertos simplemente conexos entonces $X$ es simplemente conexo.
        \end{enumerate}
    \end{ejercicio}

    \textbf{Solución.}

    \begin{ejercicio}[5 puntos]
        Determina si las siguientes afirmaciones son verdaderas o falsas (responde solo a dos de las tres afirmaciones):
        \begin{enumerate}
            \item Si $p: R \to B$ una aplicación recubridora y $r_0 \in R$, entonces la aplicación inducida $p_*: \pi_1(R, r_0) \to \pi_1(B, p(r_0))$ es inyectiva.
            
            Las aplicaciones recubridoras más comunes que conocemos son:
            \begin{itemize}
                \item Los homeomorfismos, cuya aplicación inducida es un isomorfismo, por lo que es inyectiva.
                \item La aplicación recubridora estándar $p: \bb{R} \to \bb{S}^1$, que como $\pi_1(\bb{R}) \cong \{1\}$ se tiene que es inyectiva.
                \item La aplicación recubridora dada por:
                \Func{p}{\bb{S}^1}{\bb{S}^1}{z}{z^n}
                
                Si $\deg([\alpha]) = k\in \bb{Z}$, entonces $\deg(p_*([\alpha])) = nk$. Por lo que es inyectiva.
            \end{itemize}

            Por tanto, a priori podemos pensar que la afirmación es verdadera. Veámoslo.\\
            
            Sean $[\alpha], [\beta] \in \pi_1(R, r_0)$ tales que $p_*([\alpha]) = p_*([\beta])$. Por definición de $p_*$, se tiene que:
            \begin{equation*}
                [p \circ \alpha] = [p \circ \beta]\in \pi_1(B, p(r_0)).
            \end{equation*}

            Tenemos la siguiente situación:
            \begin{center}
                \begin{tikzcd}
                    & R \arrow[d, "p"] \\
                    \left[0, 1\right] \arrow[ur, "{\alpha, \beta}"] & B
                \end{tikzcd}
            \end{center}

            Como $[p \circ \alpha] = [p \circ \beta]$, entonces:
            \begin{equation*}
                (p \circ \alpha)(0) = (p \circ \beta)(0) = p(r_0)
            \end{equation*}

            Por tanto, como los arcos se pueden levantar, si elegimos $\wt{p\circ \alpha}$ y $\wt{p\circ \beta}$ como los levantamientos de $p \circ \alpha$ y $p \circ \beta$ que cumplen $\wt{p\circ \alpha}(0) = \wt{p\circ \beta}(0) = r_0$, entonces:
            \begin{equation*}
                [\wt{p\circ \alpha}] = [\wt{p\circ \beta}] \in \pi_1(R, r_0)
            \end{equation*}

            Por la unicidad de los levantamientos, se tiene que $\wt{p\circ \alpha} = \alpha$ y $\wt{p\circ \beta} = \beta$. Por tanto:
            \begin{equation*}
                [\alpha] = [\beta] \in \pi_1(R, r_0)
            \end{equation*}

            Por tanto, $p_*$ es inyectiva y la afirmación es verdadera.

            \item Si $f: \bb{R}\bb{P}^2 \to \bb{S}^1$ es una aplicación continua y $x_0 \in \bb{R}\bb{P}^2$, entonces el homomorfismo $f_*: \pi_1(\bb{R}\bb{P}^2, x_0) \to \pi_1(\bb{S}^1, f(x_0))$ es trivial.
            
            Sabemos que $\pi_1(\bb{R}\bb{P}^2, x_0) \cong \bb{Z}_2$ y $\pi_1(\bb{S}^1, f(x_0)) \cong \bb{Z}$. Por tanto, existen isomorfismos $\phi: \pi_1(\bb{R}\bb{P}^2, x_0) \to \bb{Z}_2$ y $\psi: \pi_1(\bb{S}^1, f(x_0)) \to \bb{Z}$. Consideramos el siguiente diagrama conmutativo:
            \begin{center}
                \begin{tikzcd}
                    \pi_1(\bb{R}\bb{P}^2, x_0) \arrow[r, "f_*"] \arrow[d, "\phi", "\cong"'] & \pi_1(\bb{S}^1, f(x_0)) \arrow[d, "\psi", "\cong"'] \\
                    \bb{Z}_2 \arrow[r, "g"] & \bb{Z}
                \end{tikzcd}
            \end{center}

            Definimos por tanto $g=\psi \circ f_* \circ \phi^{-1}: \bb{Z}_2 \to \bb{Z}$. Como $g$ es un homomorfismo de grupos, se tiene que $g(0)=0$. Además:
            \begin{equation*}
                0 = g(0) = g(1 + 1) = g(1) + g(1) = 2g(1)\Longrightarrow g(1) = 0
            \end{equation*}

            Por tanto, $g$ es el homomorfismo trivial. Como $g = \psi \circ f_* \circ \phi^{-1}$ y $\psi$ y $\phi$ son isomorfismos, se tiene que $f_*$ es el homomorfismo trivial. Por tanto, la afirmación es verdadera.
            \item Si $X = U \cup V$ con $U, V$ abiertos simplemente conexos entonces $X$ es simplemente conexo.
            
            Veamos un contraejemplo. Consideremos $X=\bb{S}^1$, $U = \bb{S}^1 \setminus \{(0, 1)\}$ y $V = \bb{S}^1 \setminus \{(0, -1)\}$. Los conjuntos $U,V$ se muestran en la Figura~\ref{fig:contraejemplo}.
            \begin{figure}[h]
                \centering
                \begin{tikzpicture}
                    % Conjunto U
                    \draw[thick] (0, 0) circle (2);
                    \draw[thick, fill=white] (0, 2) circle (0.1);
                    \node at (0, -2.5) {$U$};

                    % Conjunto V
                    \draw[thick] (6, 0) circle (2);
                    \draw[thick, fill=white] (6, -2) circle (0.1);
                    \node at (6, -2.5) {$V$};
                \end{tikzpicture}
                \caption{Conjuntos $U,V$.}
                \label{fig:contraejemplo}
            \end{figure}

            Tenemos que $X=U \cup V = \bb{S}^1$, con $U,V$ abiertos y homeomorfos a $\bb{R}$, por lo que son simplemente conexos. Sin embargo, $X$ no es simplemente conexo, ya que $\pi_1(X) \cong \bb{Z}$. Por tanto, la afirmación es falsa. A la hora de aplicar el teorema de Van Kampen, necesitamos que $U \cap V$ sea arcoconexo, lo cual no se cumple en este caso.
        \end{enumerate}
    \end{ejercicio}
