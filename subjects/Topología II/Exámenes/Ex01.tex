    \begin{ejercicio}[7.5 puntos]
        Sea $X_n = \{x\in \mathbb{R}^{n+1} : \|x\| \geq 2\}$ con la topología usual inducida.
        \begin{enumerate}[label=\alph*)]
            \item Encontrar un retracto de deformación de $X_n$ distinto de $X_n$ (detallar la retracción y la homotopía).

                Consideramos el conjunto:
                \begin{equation*}
                    S_n = \{x\in \mathbb{R}^{n+1} : \|x\| = 2\}
                \end{equation*}

                Antes de demostrar formalmente que, efectivamente, $S_n$ es un retracto de deformación de $X_n$, veamos cómo debe ser el retracto. La idea es contraer cada punto $x\in X_n$ hacia su punto correspondiente en la esfera $S_n$ a lo largo de la recta que une el origen con $x$, como se puede observar en la Figura~\ref{fig:retracto_deformacion}.
                \begin{figure}[ht]
                    \centering
                    \begin{tikzpicture}
                        % Región X_1: exterior del círculo de radio 2
                        \fill[blue!8] (-3.2,-3.2) rectangle (3.2,3.2);
                        % Agujero abierto ||x|| < 2
                        \fill[white] (0,0) circle (1.6cm);

                        % Ejes coordenados
                        \draw[->, thin, gray!70] (-3.4,0) -- (3.4,0) node[right, black] {$x_1$};
                        \draw[->, thin, gray!70] (0,-3.4) -- (0,3.4) node[above, black] {$x_2$};

                        % Circunferencia S_1 (radio 2 representado a escala como 1.6cm)
                        \draw[thick, blue!80!black] (0,0) circle (1.6cm);
                        \node[blue!80!black, font=\small] at (135:1.95cm) {$S_1$};
                        \node[blue!60!black, font=\small] at (-2.5, 2.5) {$X_1$};

                        % Radio 2 indicado
                        \draw[dashed, gray] (0,0) -- (-15:1.6cm) node[midway, below, font=\scriptsize, black] {$2$};

                        % Bucle con varios puntos (ángulo / radio)
                        \foreach \ang/\rad in {25/3.0, 72/2.3, 118/2.8, 165/2.1, 210/3.1, 255/2.4, 305/2.9} {

                            \draw[->, thick, red!75!black] (\ang:\rad cm) -- (\ang:1.6cm);
                            \fill[red!75!black] (\ang:\rad cm) circle (1.7pt);
                            \fill[blue!80!black] (\ang:1.6cm) circle (1.7pt);
                        }

                        % Punto destacado con etiquetas explícitas
                        \def\angMain{42}
                        \def\radMain{2.85}
                        \draw[dotted, thick, gray!80] (0,0) -- (\angMain:1.6cm);
                        \draw[->, very thick, red!75!black] (\angMain:\radMain cm) -- (\angMain:1.6cm);
                        \fill[red!75!black] (\angMain:\radMain cm) circle (2.2pt) node[above right, font=\small] {$x$};

                        % Origen
                        \fill[black] (0,0) circle (1.5pt) node[below left, font=\scriptsize] {$0$};
                    \end{tikzpicture}
                    \caption{Representación de la retracción por deformación radial de $X_1$ sobre $S_1$ en $\mathbb{R}^2$.}
                    \label{fig:retracto_deformacion}
                \end{figure}

                Para definir la homotopía que transforma $X_n$ en $S_n$, la idea es unir mediante un segmento cada punto $x\in X_n$ con su correspondiente punto en $S_n$, que es $2\frac{x}{\|x\|}$. Definimos por tanto la siguiente aplicación:
                \Func{H}{X_n\times [0,1]}{X_n}{(x,t)}{(1-t)x + 2t\frac{x}{\|x\|}}
                Observemos que $\|x\|\neq 0$ para todo $x\in X_n$. Además, para comprobar que la aplicación $H$ está bien definida, debemos verificar que $H(x,t)\in X_n$ para todo $(x,t)\in X_n\times [0,1]$; es decir, que $\|H(x,t)\|\geq 2$ para todo $(x,t)\in X_n\times [0,1]$.
                \begin{align*}
                    \|H(x,t)\| &= \left\|(1-t)x + 2t\frac{x}{\|x\|}\right\| = \|x\| \left(1-t+\frac{2t}{\|x\|}\right) = \|x\|(1-t) + 2t  \\
                               &\geq 2(1-t) + 2t = 2 \qquad \forall x\in X_n, \quad \forall t\in [0,1]
                \end{align*}
                Por lo que $H(x,t)\in X_n$ para todo $(x,t)\in X_n\times [0,1]$, por lo que $H$ está bien definida. Es claro también que $H$ es continua, y observamos además que:
                \begin{itemize}
                    \item $H(x,0) = x\quad \forall x\in X_n$.
                    \item $H(x,1) = \dfrac{2x}{\|x\|}\in S_n \quad \forall x\in X_n$.
                    \item Si $y\in S_n$, entonces $\|y\|=2$, y por tanto:
                        \begin{equation*}
                            H(y,1) = 2\frac{y}{\|y\|} = y 
                        \end{equation*}
                \end{itemize}

                Por tanto, consideramos la siguiente aplicación:
                \Func{r}{X_n}{S_n}{x}{H(x,1) = 2\frac{x}{\|x\|}}
                
                Esta aplicación es continua, y además cumple que $r(y) = y$ para todo $y\in S_n$, por lo que $r$ es una retracción de $X_n$ sobre $S_n$. Para ver que $S_n$ es un retracto de deformación de $X_n$, basta con observar que $H$ es una homotopía entre $\id_{X_n}$ y $i\circ r$, donde $i:S_n\hookrightarrow X_n$ es la inclusión canónica.
            \item Estudiar si $X_1 \cong X_n$ cuando $n\geq 2$.

                Para cada $m\geq 1$ tenemos que $S_m=\{x\in \mathbb{R}^{m+1}:\|x\|=2\}$ es un retracto de deformación de $X_m$. Observamos además que la aplicación $f:\mathbb{S}^m\to S_m$ dada por:
                \begin{equation*}
                    f(x) = 2x
                \end{equation*}
                es un homeomorfismo, de donde deducimos que:
                \begin{equation*}
                    \pi_1(X_m) \cong \pi_1(S_m) \cong \pi_1(\mathbb{S}^m) \qquad \forall m\geq 1
                \end{equation*}
                Por tanto, para $n=1$ tenemos que:
                \begin{equation*}
                    \pi_1(X_1) \cong \pi_1(\mathbb{S}^1) \cong \mathbb{Z}
                \end{equation*}
                Por otro lado, para $n\geq 2$ tenemos que:
                \begin{equation*}
                    \pi_1(X_n)\cong \pi_1(\mathbb{S}^n) = \{1\}
                \end{equation*}
                Como $\{1\}\not\cong \mathbb{Z}$, deducimos que sus grupos fundamentales no son isomorfos, y por tanto $X_1\not\cong X_n$ para todo $n\geq 2$.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}[2.5 puntos]
        Sea $X$ un conjunto con $|X|\geq 2$. Fijado $x_0\in X$ se considera la topología
        \begin{equation*}
            T = \{U\subseteq X : x_0\notin U\}\cup \{X\}
        \end{equation*}
        ¿Es $(X,T)$ contráctil?\\

        \noindent
        Sí es contráctil, pues si consideramos $H:X\times [0,1]\to X$ dada por:
        \begin{equation*}
            H(x,t) = \left\{\begin{array}{ll}
                x & \text{si\ } t\in \left[0,\nicefrac{1}{2}\right[ \\
                    x_0 & \text{si\ } t\in [\nicefrac{1}{2},1]
            \end{array}\right. 
        \end{equation*}
        Si tomamos $U\in T$:
        \begin{itemize}
            \item Si $U=X$ tenemos trivialmente que $H^{-1}(X) = X\times [0,1]$, que es un conjunto abierto.
            \item Si $U\neq X$ tenemos entonces que $x_0\notin U$, y:
                \begin{align*}
                    H^{-1}(U) &= \{(x,t)\in X\times [0,1] : H(x,t) \in U\} \\
                              &= \{(x,t)\in X\times [0,1]: (t\geq \nicefrac{1}{2}\land x_0 \in U) \lor (t<\nicefrac{1}{2}\land x\in U)\} \\
                              &=\{(x,t)\in X\times [0,1]: t<\nicefrac{1}{2}\land x\in U\} = U\times \left[0,\nicefrac{1}{2}\right[
                \end{align*}
                Con $U\times \left[0,\nicefrac{1}{2}\right[$ un conjunto abierto, como producto de dos conjuntos abiertos.
        \end{itemize}
        En definitiva, tenemos que $H$ es continua, así como que:
        \begin{itemize}
            \item $H(x,0) = x \quad \forall x\in X$.
            \item $H(x,1) = x_0\quad \forall x\in X$.
        \end{itemize}
        Por lo que $H$ es una homotopía entre $\id_X$ y la función constantemente igual a $x_0$, de donde $\{x_0\}$ es un retracto de deformación de $(X,T)$, por lo que este espacio topológico es contráctil.
    \end{ejercicio}
