\begin{ejercicio}
    Sea $\alpha=\sqrt[3]{2}+\sqrt{3}$. Calcular:
    \begin{enumerate}
        \item $\Irr(\alpha,\mathbb{Q})$.
        \item $\Irr(\alpha,\mathbb{Q}(\sqrt{3}))$.
        \item $\Irr(\sqrt[3]{2},\mathbb{Q}(\sqrt{3}))$.
    \end{enumerate}
\end{ejercicio}

\begin{ejercicio}
    Sea $\alpha=\sqrt[3]{2}+\sqrt{3}$. Calcular:
    \begin{enumerate}
        \item $\Irr(\alpha,\mathbb{Q})$.
        
        Veamos en primer lugar que $\bb{Q}(\alpha) = \bb{Q}(\sqrt[3]{2},\sqrt{3})$. Por un lado:
        \begin{align*}
            \alpha-\sqrt{3} = \sqrt[3]{2} &\Longrightarrow
            \alpha^3 - 3\alpha^2\sqrt{3} + 9\alpha - 3\sqrt{3} = 2\\
            &\Longrightarrow \dfrac{\alpha^3 + 9\alpha - 2}{3(\alpha+1)} = \sqrt{3} \in \bb{Q}(\alpha)
        \end{align*}

        Por otro lado, $\sqrt[3]{2} = \alpha - \sqrt{3} \in \bb{Q}(\alpha)$, luego $\bb{Q}(\alpha) = \bb{Q}(\sqrt[3]{2},\sqrt{3})$. Buscamos un polinomio mónico en $\bb{Q}[x]$ del que $\alpha$ sea raíz. Para ello, elevamos al cuadrado la expresión de $\sqrt{3}$:
        \begin{align*}
            \dfrac{\alpha^6+2\alpha^3(9\alpha-2)+81\alpha^2+4-36\alpha}{9(\alpha^2+2\alpha+1)} &= 3 \\
            \alpha^6 + 18\alpha^4 - 4\alpha^3 + 81\alpha^2 - 36\alpha + 4 &= 27(\alpha^2+2\alpha+1) \\
            \alpha^6 + 18\alpha^4 - 4\alpha^3 + 54\alpha^2 - 90\alpha - 23 &= 0
        \end{align*}

        Por tanto, $\alpha$ es raíz de $f = x^6 + 18x^4 - 4x^3 + 54x^2 - 90x - 23\in \bb{Q}[x]$. No obstante, no es fácil ver si es irreducible o no. Para ello, consideramos las siguientes torres:
        \begin{itemize}
            \item $\bb{Q} \leq \bb{Q}(\sqrt{3}) \leq \bb{Q}(\sqrt[3]{2},\sqrt{3}) = \bb{Q}(\alpha)$
            
            Por el Lema de la Torre, tenemos que:
            \begin{equation*}
                [\bb{Q}(\alpha):\bb{Q}] = [\bb{Q}(\alpha):\bb{Q}(\sqrt{3})][\bb{Q}(\sqrt{3}):\bb{Q}]
            \end{equation*}
            \begin{itemize}
                \item $\Irr(\sqrt{3},\mathbb{Q}) = x^2-3$, que es irreducible por Eisenstein para $p=3$, luego $[\bb{Q}(\sqrt{3}):\bb{Q}] = 2$.
                \item Dado $g=x^3-2\in \bb{Q}(\sqrt{3})[x]$, tenemos que $g(\sqrt[3]{2}) = 0$, luego $\Irr(\sqrt[3]{2},\bb{Q}(\sqrt{3}))$ divide a $g$, con lo que $[\bb{Q}(\alpha):\bb{Q}(\sqrt{3})] \leq 3$. Notemos que no es fácil ver si es irreducible o no, por lo que a priori no podemos afirmar si es $1$, $2$ o $3$.
            \end{itemize}
            \item $\bb{Q} \leq \bb{Q}(\sqrt[3]{2}) \leq \bb{Q}(\sqrt[3]{2},\sqrt{3}) = \bb{Q}(\alpha)$
            
            Por el Lema de la Torre, tenemos que:
            \begin{equation*}
                [\bb{Q}(\alpha):\bb{Q}] = [\bb{Q}(\alpha):\bb{Q}(\sqrt[3]{2})][\bb{Q}(\sqrt[3]{2}):\bb{Q}]
            \end{equation*}
            \begin{itemize}
                \item $\Irr(\sqrt[3]{2},\mathbb{Q}) = x^3-2$, que es irreducible por Eisenstein para $p=2$, luego $[\bb{Q}(\sqrt[3]{2}):\bb{Q}] = 3$.
                \item Dado $h=x^2-3\in \bb{Q}(\sqrt[3]{2})[x]$, tenemos que $h(\sqrt{3}) = 0$, luego $\Irr(\sqrt{3},\bb{Q}(\sqrt[3]{2}))$ divide a $h$, con lo que $[\bb{Q}(\alpha):\bb{Q}(\sqrt[3]{2})] \leq 2$. Notemos que no es fácil ver si es irreducible o no, por lo que a priori no podemos afirmar si es $1$ o $2$.
            \end{itemize}
        \end{itemize}

        Por tanto, tenemos que $[\bb{Q}(\alpha):\bb{Q}]$ es múltiplo de $2$ y de $3$, y menor o igual que $6$. Por tanto, $[\bb{Q}(\alpha):\bb{Q}] = 6$, y $\Irr(\alpha,\mathbb{Q})$ es de grado 6. Por tanto:
        \begin{equation*}
            \Irr(\alpha,\mathbb{Q}) = x^6 + 18x^4 - 4x^3 + 54x^2 - 90x - 23\in \bb{Q}[x]
        \end{equation*}
        \item $\Irr(\alpha,\mathbb{Q}(\sqrt{3}))$.
        
        Buscamos en $\bb{Q}(\sqrt{3})[x]$ un polinomio mónico del que $\alpha$ sea raíz. De la expresión de $\sqrt{3}$ en función de $\alpha$ obtenemos que:
        \begin{equation*}
            \alpha^3+9\alpha-2 = 3(\alpha+1)\sqrt{3} \Longrightarrow \alpha^3+(9-3\sqrt{3})\alpha-2-3\sqrt{3} = 0
        \end{equation*}

        No obstante, no es fácil ver si es irreducible o no. Por la primera torre descrita, tenemos que:
        \begin{equation*}
            [\bb{Q}(\alpha):\bb{Q}(\sqrt{3})] = \dfrac{[\bb{Q}(\alpha):\bb{Q}]}{[\bb{Q}(\sqrt{3}):\bb{Q}]} = \dfrac{6}{2} = 3
        \end{equation*}

        Por tanto, tenemos que el grado de $\Irr(\alpha,\mathbb{Q}(\sqrt{3})(\alpha))$ es 3, y como se tiene que $\bb{Q}(\sqrt{3})=\bb{Q}(\sqrt{3})(\alpha)$, tenemos que:
        \begin{equation*}
            \deg(\Irr(\alpha,\mathbb{Q}(\sqrt{3}))) = 3
        \end{equation*}

        Por tanto, se deduce que:
        \begin{equation*}
            \Irr(\alpha,\mathbb{Q}(\sqrt{3})) = x^3+(9-3\sqrt{3})x-2-3\sqrt{3}\in \bb{Q}(\sqrt{3})[x]
        \end{equation*}


        \item $\Irr(\sqrt[3]{2},\mathbb{Q}(\sqrt{3}))$.
        
        Buscamos en $\bb{Q}(\sqrt{3})[x]$ un polinomio mónico del que $\sqrt[3]{2}$ sea raíz. Consideramos:
        \begin{equation*}
            x^3-2\in \bb{Q}(\sqrt{3})[x]
        \end{equation*}

        No obstante, no es fácil ver si es irreducible o no.        
        Por la primera torre descrita, tenemos que:
        \begin{equation*}
            [\bb{Q}(\alpha):\bb{Q}(\sqrt{3})] = \dfrac{[\bb{Q}(\alpha):\bb{Q}]}{[\bb{Q}(\sqrt{3}):\bb{Q}]} = \dfrac{6}{2} = 3
        \end{equation*}

        Por tanto, tenemos que el grado de $\Irr(\sqrt[3]{2},\mathbb{Q}(\sqrt{3}))$ es $[\bb{Q}(\sqrt[3]{2},\sqrt{3}):\bb{Q}(\sqrt{3})] = 3$, y como se tiene que $\bb{Q}(\sqrt[3]{2},\sqrt{3})=\bb{Q}(\alpha)$, tenemos que:
        \begin{equation*}
            \deg(\Irr(\sqrt[3]{2},\mathbb{Q}(\sqrt{3}))) = 3
        \end{equation*}

        Por tanto, se deduce que:
        \begin{equation*}
            \Irr(\sqrt[3]{2},\mathbb{Q}(\sqrt{3})) = x^3-2\in \bb{Q}(\sqrt{3})[x]
        \end{equation*}
    \end{enumerate}
\end{ejercicio}
