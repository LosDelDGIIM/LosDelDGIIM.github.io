% Para cambiar al modo capítulo en el chapter que desees
\newcommand{\ejerciciosPorCapitulo}{%
  \counterwithout*{ejercicio}{section}%   Ya no se reinicia con cada \section
  \counterwithin*{ejercicio}{chapter}%   Se reinicia con cada \chapter
  \renewcommand{\theejercicio}{\thechapter.\arabic{ejercicio}}% Formato: ej. 2.1, 2.2...
}

% Para volver al comportamiento por defecto (por sección)
\newcommand{\ejerciciosPorSeccion}{%
  \counterwithout*{ejercicio}{chapter}%
  \counterwithin*{ejercicio}{section}%   Vuelve a reiniciarse con cada \section
  \renewcommand{\theejercicio}{%
    \ifnum\value{section}=0
      \arabic{ejercicio}%
    \else
      \thesection.\arabic{ejercicio}%
    \fi
  }%
}
