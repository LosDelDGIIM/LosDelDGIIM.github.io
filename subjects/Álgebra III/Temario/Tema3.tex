\section{Extensiones de Galois}

Del Capítulo anterior recordamos la Proposición~\ref{prop:comienzo_chap_2}, que nos servirá para comenzar este Capítulo:

\begin{center}
    Sea $F\leq K$ una extensión finita, entonces $|\Aut_F(K)| \leq [K:F]$.
\end{center}


Esta idea nos permite obtener grupos finitos a partir de extensiones de cuerpos finitas, mientras que la Proposición~\ref{prop:artin} nos permitirá el proceso contrario. Veamos un ejemplo de obtención de grupos finitos a partir de extensiones de cuerpos finitas.
\begin{ejemplo}
    Sabemos que $[\bb{Q}(\sqrt[3]{2}):\bb{Q}] = 3$, por lo que:
    \begin{equation*}
        \left|\Aut_{\bb{Q}}\left(\mathbb{Q}\left(\sqrt[3]{2}\right)\right)\right| = \left|\Aut\left(\mathbb{Q}\left(\sqrt[3]{2}\right)\right)\right| \leq 3
    \end{equation*}
    Y afirmamos que solo hay uno, ya que si observamos el diagrama:
    \begin{figure}[H]
        \centering
        \shorthandoff{""}
        \begin{tikzcd}
        \mathbb{Q} \arrow[r, "\iota"] \arrow[rd, "\iota"'] & {\mathbb{Q}\left(\sqrt[3]{2}\right)}                    \\
                                                           & {\mathbb{Q}\left(\sqrt[3]{2}\right)} \arrow[u, "\eta"']
        \end{tikzcd}
        \shorthandon{""}
    \end{figure}
    \noindent
    tenemos que raíces de $x^3-2$ en $\mathbb{Q}\left(\sqrt[3]{2}\right)$ solo hay 1. Sin embargo, en el ejemplo de la página~\pageref{ejemplo:aut_sqrt3_2} vimos que:
    \begin{equation*}
        \left|\Aut\left(\mathbb{Q}\left(\sqrt[3]{2},w\right)\right)\right| = 6
    \end{equation*}
    Por lo que la idea intuitiva es que faltan raíces en el cuerpo para poder tener todos los automorfismos.
\end{ejemplo}

\begin{notacion}
    Para no confundir la notación de ``subgrupo'' con la de ``subcuerpo'', siempre que tengamos $H$ un subgrupo de $G$ lo notaremos por $H < G$.
\end{notacion}


\begin{definicion}
    Sea $K$ un cuerpo y $G< \Aut(K)$ subgrupo, definimos el \underline{subcuerpo} \underline{fijo de $K$ bajo (la acción de) $G$} como el conjunto:
    \begin{equation*}
        K^G = \{a\in K : \sigma(a) = a \quad \forall \sigma\in G\}
    \end{equation*}
\end{definicion}

\begin{observacion}
    Sea $K$ un cuerpo y $G<\Aut(K)$, tenemos que:
    \begin{equation*}
        K^G \leq K
    \end{equation*}
    \begin{proof}
        Para probar que $K^G$ es un subcuerpo de $K$:
        \begin{itemize}
            \item Es claro que $1\in K^G$, ya que de hecho $\sigma(1) = 1\quad \forall \sigma\in \Aut(K)$.
            \item Si $a,b\in K^G$, tenemos entonces que $\sigma(a) = a, \sigma(b)=b\quad \forall \sigma\in G$, de donde:
                \begin{equation*}
                    \left.\begin{array}{r}
                        \sigma(a-b) = \sigma(a)-\sigma(b) = a-b \\
                        \sigma(ab) = \sigma(a)\sigma(b) = ab
                    \end{array}\right\} \quad\Longrightarrow\quad a-b,ab\in K^G
                \end{equation*}
            \item Finalmente, si $a\in K^G$ tenemos entonces que:
                \begin{equation*}
                    \sigma(a^{-1}) = \sigma(a)^{-1} = a^{-1} \quad\Longrightarrow\quad a^{-1}\in K^G
                \end{equation*}
        \end{itemize}
    \end{proof}
\end{observacion}
 
\begin{prop}[Artin]\label{prop:artin}
    Si $G$ es un subgrupo finito de $\Aut(K)$, entonces.
    \begin{equation*}
        \left[K:K^G\right] \leq |G|
    \end{equation*}
    \begin{proof}
        Sea $n=|G|$, será $G = \{\sigma_1, \ldots, \sigma_n\}$ y tomamos $m$ (con $m>n$) elementos de $K$, $\alpha_1,\ldots,\alpha_m \in K$, basta probar que estos son $K^G-$linealmente dependientes, para tener $[K:K^G]\leq n$. Para verlo, formamos la matriz:
        \begin{equation*}
            A = (\sigma_{j}(\alpha_i))_{i,j} = \left(\begin{array}{cccc}
                \sigma_1(\alpha_1) & \sigma_2(\alpha_1) & \cdots & \sigma_n(\alpha_1) \\
                \sigma_1(\alpha_2) & \sigma_2(\alpha_2) & \cdots & \sigma_n(\alpha_2) \\
                \vdots & \vdots & \ddots & \vdots \\
                \sigma_1(\alpha_m) & \sigma_2(\alpha_m) & \cdots & \sigma_n(\alpha_m) 
            \end{array}\right) \in  \cc{M}_{m\times n}(K)
        \end{equation*}
        cuyo rango es menor o igual que $n$, luego menor o igual que $m$, es decir, existe un vector
        \begin{equation*}
            0\neq v = (v_1, \ldots, v_m) \in K^m
        \end{equation*}
        tal que $vA = 0$. Ahora, de entre todos los vectores que cumplen dichas condiciones, tomamos como $v$ aquel con mínimo número de componentes no nulas y tal que alguna componente, digamos la $l-$ésima (con $1\leq l\leq m$), verifique que $v_l \in K^G$. Notemos que esto podemos conseguirlo siempre con $v_l = 1$, ya que como $v\neq 0$ ha de existir $1\leq l\leq m$ con $v_l\neq 0$, y podemos dividir todas las componentes del vector $v$ entre la $l-$ésima. Si escribimos la igualdad $vA = 0$, obtenemos las igualdades:
        \begin{equation*}
            \sum_{i=1}^m v_i \sigma_j(\alpha_i) = 0 \qquad \forall j \in \{1,\ldots,n\}
        \end{equation*}
        Y observamos que para obtener la dependencia lineal de los $\alpha_i$ falta ver que realmente los coeficientes $v_i$ están en $K^G$ (por ahora solo sabemos que están en $K$). Para ello, supuesto que $v_{l'}\notin K^G$, tendremos que $v_{l'} \neq \sigma_k(v_{l'})$ para cierto índice $k\in \{1,\ldots,n\}$. Tomamos ahora cualquier $\sigma\in G$ y definimos:
        \begin{equation*}
            \sigma(v) = (\sigma(v_1), \ldots, \sigma(v_n))
        \end{equation*}
        Si usamos esto para $\sigma_k$:
        \begin{equation*}
            \sigma_k(v) = (\sigma_k(v_1), \ldots, \sigma_k(v_m))
        \end{equation*}
        Podemos aplicar $\sigma_k$ a las igualdades anteriores, con lo que:
        \begin{equation*}
            \sum_{i=1}^m \sigma_k(v_i) \sigma_k(\sigma_j(\alpha_i)) = 0 \qquad \forall j \in \{1,\ldots,n\}
        \end{equation*}
        Observemos que:
        \begin{equation*}
            G = \{\sigma_1, \ldots, \sigma_n\} = \{\sigma_k\sigma_1, \ldots, \sigma_k\sigma_n\}
        \end{equation*}
        \begin{description}
            \item [$\supseteq )$] Como $G$ es un grupo tenemos que es cerrado para su operación.
            \item [$\subseteq )$] Tenemos a la derecha $n$ elementos distintos de $G$, y sabemos que son disintos porque si fuera $\sigma_k\sigma_i = \sigma_k\sigma_j$ para ciertos $i,j\in \{1,\ldots,n\}$ con $i\neq j$ tendríamos entonces que $\sigma_i = \sigma_j$, pero $G$ tenía $n$ elementos distintos.
        \end{description}
        Por lo que tenemos en realidad es:
        \begin{equation*}
            \sum_{i=1}^{m} \sigma_k(v_i)\sigma_j(\alpha_i) = 0 \qquad \forall j \in \{1,\ldots,n\}
        \end{equation*}

        o equivalentemente:
        \begin{equation*}
            \sigma_k(v)A = 0
        \end{equation*}
        Como $vA = 0$ y $\sigma_k(v)A=0$, tenemos que:
        \begin{equation*}
            (v-\sigma_k(v))A = 0, \qquad v-\sigma_k(v) \neq 0
        \end{equation*}
        ya que $v_{l'} \neq \sigma_k(v_{l'})$. Sin embargo, las componentes $l-$ésimas de los vectores eran iguales ($v_l = \sigma_k(v_l)$), por lo que hemos obtenido un vector $v-\sigma_k(v)$ que verifica que al multiplicarse por $A$ se obtiene cero y con al menos una componente no nula menos que $v$, \underline{contradicción}, que viene de suponer que $v_{l'}\notin K^G$, lo que nos dice que los coeficientes $v_i$ estaban en $K^G$. Si en la igualdad: 
        \begin{equation*}
            \sum_{i=1}^m v_i \sigma_j(\alpha_i) = 0 \qquad \forall j \in \{1,\ldots,n\}
        \end{equation*}
        tomamos aquel índice $j$ que verifica que $\sigma_j = Id_K$, tendremos entonces que:
        \begin{equation*}
            \sum_{i=1}^mv_i \alpha_i = 0, \qquad v_i \in K^G
        \end{equation*}
        lo que implica que $\alpha_1, \ldots, \alpha_m$ eran $K^G-$linealmente dependientes, por lo que:
        \begin{equation*}
            [K:K^G] \leq n = |G|
        \end{equation*}
    \end{proof}
\end{prop}

Queremos ahora ver como se comporta la correspondencia descrita entre extensiones finitas y grupos finitos.
\begin{lema}\label{lema:inclusiones}
    Para un cuerpo $K$, tenemos que:
    \begin{enumerate}
        \item Si $H< G$ son subgrupos de $\Aut(K)$, entonces $K^H \geq K^G$. % Hay menos cosas que comprobar
        \item Si $F\leq E$ son subcuerpos de $K$, entonces $\Aut_F(K)> \Aut_E(K)$.
        \item Si $G< \Aut(K)$, entonces $G< \Aut_{K^G}(K)$.
        \item Si $F\leq K$, entonces $F\leq F^{\Aut_F(K)}$.
    \end{enumerate}
    \begin{proof}
        Demostramos cada uno de los apartados de forma muy sencilla:
        \begin{enumerate}
            \item Sea $a\in K^G$, tenemos que $\sigma(a) = a\quad \forall \sigma\in G$, en particular para los automorfismos de $H$, por lo que $a\in K^H$.
            \item Sea $\sigma \in \Aut_E(K)$, tenemos entonces que $\sigma$ es $E-$lineal, pero como $F\leq E$, será $\sigma\in \Aut_F(K)$.
            \item Sea $\sigma\in G< \Aut(K)$, si tomamos $x\in K$ y $y\in K^G$, observamos que:
                \begin{equation*}
                    \sigma(y\cdot x) = \sigma(y)\cdot \sigma(x) = y\cdot \sigma(x)
                \end{equation*}
                Por lo que $\sigma\in \Aut_{K^G}(K)$.
            \item Sea $x\in F$ y $\sigma \in \Aut_F(K)$, entonces:
                \begin{equation*}
                    \sigma(x) = \sigma(x\cdot 1) = x\cdot \sigma(1) = x
                \end{equation*}
                Por lo que $x\in F^{\Aut_F(K)}$.
        \end{enumerate}
    \end{proof}
\end{lema}

\begin{teo}\label{teo:antes_teo_angular}
    Sea $K$ un cuerpo. Si $G$ es un subgrupo finito de $\Aut(K)$, entonces:
    \begin{equation*}
        [K:K^G] = |G| \qquad \text{y}\qquad G = \Aut_{K^G}(K)
    \end{equation*}
    \begin{proof}
        El Lema de Artin nos dice que $[K:K^G]\leq |G|$. Para la otra desigualdad, el Lema anterior nos dice que $G<\Aut_{K^G}(K)$ y la Proposición~\ref{prop:comienzo_chap_2} nos dice que $|\Aut_{K^G}(K)|\leq [K:K^G]$, por lo que:
        \begin{equation*}
            |G| \leq |\Aut_{K^G}(K)| \leq [K:K^G]
        \end{equation*}
        de donde $[K:K^G] = |G|$ y tendrá que ser $|G| = |\Aut_{K^G}(K)|$, por lo que finalmente $G = \Aut_{K^G}(K)$.
    \end{proof}
\end{teo}

\begin{ejemplo}\label{ejemplo:subcuerpos_fijos}
    Sea $K = \mathbb{Q}\left(\sqrt[3]{2},w\right)$ con $w$ una raíz cúbica primitiva de la unidad. Por el ejemplo de la página~\pageref{ejemplo:aut_sqrt3_2}, tenemos:
    \begin{equation*}
        \Aut(K) = \{\eta_{j,k} : j\in \{0,1,2\}, k \in \{1,2\}\}
        \qquad\text{donde}\qquad
        \begin{cases}
            \eta_{j,k}(\sqrt[3]{2}) = w^j \sqrt[3]{2} \\
            \eta_{j,k}(w) = w^k
        \end{cases}
    \end{equation*}

    Los subgrupos propios de $\Aut(K)$ (por el Teorema de Lagrange) son de orden 2 o 3, todos ellos cíclicos, por lo que tenemos que buscar elementos de orden 2 y 3. Calculamos el orden de cada elemento:
    \begin{align*}
        \sqrt[3]{2} &\stackrel{\eta_{0,1}}{\longmapsto} \sqrt[3]{2}\\
        w &\longmapsto w \\
        \\
        \sqrt[3]{2} &\stackrel{\eta_{0,2}}{\longmapsto} \sqrt[3]{2} \\
        w &\longmapsto w^2 \longmapsto w^4 = w \\
        \\
        \sqrt[3]{2} &\stackrel{\eta_{1,1}}{\longmapsto} w\sqrt[3]{2} \longmapsto w^2\sqrt[3]{2} \longmapsto w^3\sqrt[3]{2} = \sqrt[3]{2} \\
        w &\longmapsto w \\
        \\
        \sqrt[3]{2} &\stackrel{\eta_{1,2}}{\longmapsto} w\sqrt[3]{2} \longmapsto w^2 w\sqrt[3]{2} = \sqrt[3]{2} \\
        w &\longmapsto w^2 \longmapsto w \\
        \\
        \sqrt[3]{2} &\stackrel{\eta_{2,1}}{\longmapsto} w^2\sqrt[3]{2} \longmapsto w^4\sqrt[3]{2} = w\sqrt[3]{2} \longmapsto w^3\sqrt[3]{2} = \sqrt[3]{2} \\
        w &\longmapsto w \\
        \\
        \sqrt[3]{2} &\stackrel{\eta_{2,2}}{\longmapsto} w^2\sqrt[3]{2} \longmapsto w^6\sqrt[3]{2} = \sqrt[3]{2} \\
        w &\longmapsto w^2 \longmapsto w^4 = w
    \end{align*}

    Por tanto:
    \begin{equation*}
        O(\eta_{0,1}) = 1, \quad O(\eta_{0,2}) = O(\eta_{1,2}) = O(\eta_{2,2}) = 2, \quad O(\eta_{1,1}) = O(\eta_{2,1}) = 3
    \end{equation*}

    Como $\eta_{1,1}^2 = \eta_{2,1}$ y ambos son de orden 3, tenemos que los subgrupos propios son:
    \begin{equation*}
        \langle \eta_{1,1} \rangle  = \langle \eta_{2,1} \rangle \cong C_3, \qquad \langle \eta_{0,2} \rangle  \cong \langle \eta_{1,2} \rangle  \cong \langle \eta_{2,2} \rangle \cong C_2
    \end{equation*}

    Calculamos ahora los subcuerpos fijos de $K$ bajo la acción de estos subgrupos. \begin{itemize}
        \item $K^{\langle \id \rangle } = K$
        \item Buscamos ahora calcular $K^{\Aut(K)}$.
        
        En primer lugar, como $\bb{Q}$ es el cuerpo primo de $K$, tenemos que $\bb{Q}\leq K^{\Aut(K)}\leq K$. Por el Lema de la torre, tenemos que:
        \begin{equation*}
            [K:\mathbb{Q}] = [K:K^{\Aut(K)}]\cdot [K^{\Aut(K)}:\mathbb{Q}]
        \end{equation*}
        
        Por el Teorema~\ref{teo:antes_teo_angular}, tenemos que:
        \begin{equation*}
            [K:K^{\Aut(K)}] = |\Aut(K)| = 6
        \end{equation*}

        Además, sabemos que $[K:\mathbb{Q}] = 6$, por lo que $[K^{\Aut(K)}:\mathbb{Q}] = 1$, es decir, $K^{\Aut(K)} = \mathbb{Q}$.

        \item Buscamos ahora calcular $K^{\langle \eta_{1,1} \rangle }$.
        
        Como $\eta_{1,1}(w) = w$, tenemos que $w\in K^{\langle \eta_{1,1} \rangle }$, luego $\bb{Q}\leq \bb{Q}(w)\leq K^{\langle \eta_{1,1} \rangle }\leq K$. Por el Lema de la torre, tenemos que:
        \begin{equation*}
            [K:\bb{Q}] = [K:K^{\langle \eta_{1,1} \rangle }]\cdot [K^{\langle \eta_{1,1} \rangle }:\bb{Q}(w)]\cdot [\bb{Q}(w):\bb{Q}]
        \end{equation*}
        Por el Teorema~\ref{teo:antes_teo_angular},
        \begin{equation*}
            [K:K^{\langle \eta_{1,1} \rangle }] = |\langle \eta_{1,1} \rangle | = 3
        \end{equation*}
        Además, sabemos que $[\mathbb{Q}(w), \bb{Q}] = 2$, por lo que $[K^{\langle \eta_{1,1} \rangle }:\mathbb{Q}(w)] = 1$, es decir, $K^{\langle \eta_{1,1} \rangle } = \mathbb{Q}(w)$.

        \item Buscamos ahora calcular $K^{\langle \eta_{0,2} \rangle }$.
        
        Como $\eta_{0,2}(\sqrt[3]{2}) = \sqrt[3]{2}$, tenemos que $\sqrt[3]{2}\in K^{\langle \eta_{0,2} \rangle }$, luego consideramos la torre $\bb{Q}\leq \bb{Q}(\sqrt[3]{2})\leq K^{\langle \eta_{0,2} \rangle }\leq K$. Por el Lema de la torre, tenemos que:
        \begin{equation*}
            [K:\mathbb{Q}] = [K:K^{\langle \eta_{0,2} \rangle }]\cdot [K^{\langle \eta_{0,2} \rangle }:\mathbb{Q}(\sqrt[3]{2})]\cdot [\mathbb{Q}(\sqrt[3]{2}):\mathbb{Q}]
        \end{equation*}
        Por el Teorema~\ref{teo:antes_teo_angular},
        \begin{equation*}
            [K:K^{\langle \eta_{0,2} \rangle }] = |\langle \eta_{0,2} \rangle | = 2
        \end{equation*}
        Además, sabemos que $[\mathbb{Q}(\sqrt[3]{2}):\mathbb{Q}] = 3$, por lo que $[K^{\langle \eta_{0,2} \rangle }:\mathbb{Q}(\sqrt[3]{2})] = 1$, es decir, $K^{\langle \eta_{0,2} \rangle } = \mathbb{Q}(\sqrt[3]{2})$.

        \item Buscamos ahora calcular $K^{\langle \eta_{1,2} \rangle }$.
        
        Como $\eta_{1,2}(w^2\sqrt[3]{2}) = w^2\sqrt[3]{2}$, tenemos que $w^2\sqrt[3]{2}\in K^{\langle \eta_{1,2} \rangle }$, luego consideramos la torre $\bb{Q}\leq \bb{Q}(w^2\sqrt[3]{2})\leq K^{\langle \eta_{1,2} \rangle }\leq K$. Por el Lema de la torre, tenemos que:
        \begin{equation*}
            [K:\mathbb{Q}] = [K:K^{\langle \eta_{1,2} \rangle }]\cdot [K^{\langle \eta_{1,2} \rangle }:\mathbb{Q}(w^2\sqrt[3]{2})]\cdot [\mathbb{Q}(w^2\sqrt[3]{2}):\mathbb{Q}]
        \end{equation*}
        Por el Teorema~\ref{teo:antes_teo_angular},
        \begin{equation*}
            [K:K^{\langle \eta_{1,2} \rangle }] = |\langle \eta_{1,2} \rangle | = 2
        \end{equation*}
        Además, sabemos que $[\mathbb{Q}(w^2\sqrt[3]{2}):\mathbb{Q}] = 3$, por lo que $[K^{\langle \eta_{1,2} \rangle }:\mathbb{Q}(w^2\sqrt[3]{2})] = 1$, es decir, $K^{\langle \eta_{1,2} \rangle } = \mathbb{Q}(w^2\sqrt[3]{2})$.

        \item Buscamos ahora calcular $K^{\langle \eta_{2,2} \rangle }$.
        
        Como $\eta_{2,2}(w\sqrt[3]{2}) = w\sqrt[3]{2}$, tenemos que $w\sqrt[3]{2}\in K^{\langle \eta_{2,2} \rangle }$, luego consideramos la torre $\bb{Q}\leq \bb{Q}(w\sqrt[3]{2})\leq K^{\langle \eta_{2,2} \rangle }\leq K$. Por el Lema de la torre, tenemos que:
        \begin{equation*}
            [K:\mathbb{Q}] = [K:K^{\langle \eta_{2,2} \rangle }]\cdot [K^{\langle \eta_{2,2} \rangle }:\mathbb{Q}(w\sqrt[3]{2})]\cdot [\mathbb{Q}(w\sqrt[3]{2}):\mathbb{Q}]
        \end{equation*}
        Por el Teorema~\ref{teo:antes_teo_angular},
        \begin{equation*}
            [K:K^{\langle \eta_{2,2} \rangle }] = |\langle \eta_{2,2} \rangle | = 2
        \end{equation*}
        Además, sabemos que $[\mathbb{Q}(w\sqrt[3]{2}):\mathbb{Q}] = 3$, por lo que $[K^{\langle \eta_{2,2} \rangle }:\mathbb{Q}(w\sqrt[3]{2})] = 1$, es decir, $K^{\langle \eta_{2,2} \rangle } = \mathbb{Q}(w\sqrt[3]{2})$.
    \end{itemize}

    En el Teorema~\ref{teo:fundamental} veremos que todos los subcuerpos de $K$ son fijos por subgrupos de $\Aut(K)$, por lo que la anterior es una descripcion completa de los subcuerpos de $K$.
\end{ejemplo}

\begin{ejercicio}
    Sea $\alpha=w+\sqrt[3]{2}$, con $w$ una raíz cúbica primitiva de la unidad. Demostrar que $\alpha$ no pertenece a ninguno de los subcuerpos propios de $K$ descritos en el anterior ejemplo, en la página~\pageref{ejemplo:subcuerpos_fijos}.
    \begin{itemize}
        \item Para $K^{\Aut(K)} = \mathbb{Q}$.
        
        Consideramos $\sigma=\eta_{1,1}\in \Aut(K)$, tenemos que:
        \begin{equation*}
            \sigma(\alpha) = \eta_{1,1}(w+\sqrt[3]{2}) = \eta_{1,1}(w) + \eta_{1,1}(\sqrt[3]{2}) = w + w\sqrt[3]{2} =\alpha \iff \sqrt[3]{2} = 0
        \end{equation*}
        Por lo que $\alpha\notin K^{\Aut(K)}$.
        \item Para $K^{\langle \eta_{1,1} \rangle } = \mathbb{Q}(w)$.
        
        Como $\eta_{1,1}(\alpha) = w + w\sqrt[3]{2} \neq \alpha$, tenemos que $\alpha\notin K^{\langle \eta_{1,1} \rangle }$.
        \item Para $K^{\langle \eta_{0,2} \rangle } = \mathbb{Q}(\sqrt[3]{2})$.
        
        Como $\eta_{0,2}(\alpha) = w^2 + \sqrt[3]{2} = \alpha \iff w^2 = w\iff w=1$, tenemos que $\alpha\notin K^{\langle \eta_{0,2} \rangle }$.
        \item Para $K^{\langle \eta_{1,2} \rangle } = \mathbb{Q}(w^2\sqrt[3]{2})$

        Tenemos que:
        \begin{align*}
            \eta_{1,2}(\alpha) = w^2 + w\sqrt[3]{2} = \alpha &\iff w^2 + w\sqrt[3]{2} = w + \sqrt[3]{2} \iff (w^2-w) = (1-w)\sqrt[3]{2} \iff\\&\iff w(w-1) = (1-w)\sqrt[3]{2} \iff -w = \sqrt[3]{2} \iff w = -\sqrt[3]{2}
        \end{align*}

        Elevando al cubo, tendríamos que $1 = -2$, lo cual es imposible, por lo que $\alpha\notin K^{\langle \eta_{1,2} \rangle }$.
        \item Para $K^{\langle \eta_{2,2} \rangle } = \mathbb{Q}(w\sqrt[3]{2})$.
        
        Tenemos que:
        \begin{align*}
            \eta_{2,2}(\alpha) = w^2 + w^2\sqrt[3]{2} = \alpha &\iff w^2 + w^2\sqrt[3]{2} = w + \sqrt[3]{2} \iff \sqrt[3]{2}(w^2-1) = w-w^2 \iff\\&\iff \sqrt[3]{2} = \frac{w-w^2}{w^2-1} = \frac{w(1-w)}{(w-1)(w+1)} = \frac{-w}{w+1}
        \end{align*}

        Como $w$ es una raíz cúbica primitiva de la unidad, al dividir $x^3-1$ entre $x-1$ obtenemos que $w^2+w+1=0$, por lo que $w+1=-w^2$, y entonces:
        \begin{equation*}
            \sqrt[3]{2} = \frac{-w}{w+1} = \frac{-w}{-w^2} = \frac{1}{w} = w^2
        \end{equation*}

        Elevando al cubo, tendríamos que $2 = 1$, lo cual es imposible, por lo que $\alpha\notin K^{\langle \eta_{2,2} \rangle }$.
    \end{itemize}
\end{ejercicio}



\begin{definicion}[Polinomio separable]
    Sea $f\in F[x]$ con $\deg f \geq 1$, se dice que $f$ es separable si todas sus raíces (en un cuerpo de descomposición de $f$) son simples.
\end{definicion}

\begin{observacion}
    Las siguientes afirmaciones sobre $f\in F[x]$ son equivalentes:
    \begin{enumerate}
        \item $f$ es separable.
        \item $f$ tiene $\deg f$ raíces distintas en su cuerpo de descomposición.
        \item $\mcd (f,f') = 1$.
    \end{enumerate}
    \begin{proof}
        $1\Longleftrightarrow 2$ es evidente. Para $2\Longleftrightarrow 3$:
        \begin{description}
            \item [$\Longleftarrow )$] Si $\mcd (f,f') = 1$ tenemos por la identidad de Bezout que si $\alpha$ es una raíz de $f$ entonces no puede ser también raíz de $f'$, ya que entonces tendríamos $1=0$, por lo que las raíces de $f$ y $f'$ son distintas, luego todas las raíces de $f$ son simples.
            \item [$\Longrightarrow )$] Si fuera $\mcd(f,f')$ igual a un polinomio no constante, entonces $f$ y $f'$ compartirían las raíces de dicho polinomio, por lo que $f$ tendría raíces múltiples, con lo que no sería separable.
        \end{description}
    \end{proof}
\end{observacion}

\begin{ejemplo}
    Para mostrar la abundancia de polinomios separables así como la existencia de polinomios no separables:
    \begin{itemize}
        \item Si $F$ es un cuerpo con $\car(F)= 0$ y $f$ es irreducible, entonces $f$ es separable.

            Como $\car(F)=0$ y $\deg f\geq 1$, tenemos al ser $f$ no constante que $f'\neq 0$, y como $f$ es irreducible tendremos que $\mcd(f,f')=1$, de donde $f$ es separable.
        \item Sea $f=x^q-x\in \bb{F}_p[x]$, donde $q=p^n$, tenemos que $f$ es separable.

            Como $f'=qx^{q-1}-1 = -1\neq 0$, tenemos que $\mcd(f,f')=1$, por lo que $f$ es separable.
        \item Sea $\bb{F}_p(t)$ el cuerpo de fracciones del anillo de polinomios $\bb{F}_p[t]$, si consideramos el polinomio:
            \begin{equation*}
                f = x^p-t\in \bb{F}_p(t)[x]
            \end{equation*}
            tenemos que $f$ es irreducible (por Eisenstein para $t$) y que $f'=0$, con lo que $\mcd(f,f')=f\neq 1$, luego $f$ no es separable.
    \end{itemize}
\end{ejemplo}

\begin{definicion}[Extensión separable]
    Una extensión algebraica $F\leq K$ se dice separable si $\Irr(\alpha,F)$ es separable, para todo $\alpha\in K$.
\end{definicion}

\begin{observacion}
    Toda extensión algebraica en característica $0$ es separable.
\end{observacion}

\begin{definicion}[Extensión normal]
    Una extensión algebraica $F\leq K$ se dice normal si $\Irr(\alpha,F)$ se factoriza como producto de polinomios lineales en $K[x]$, para todo $\alpha\in K$.
\end{definicion}

\begin{ejemplo}
    Por ejemplo, la extensión $\mathbb{Q}\leq \mathbb{Q}(\sqrt[3]{2})$ no es normal pero sí es separable.\\

    Sabemos que $\Irr(\sqrt[3]{2},\mathbb{Q}) = x^3-2$, y que $x^3-2$ no se descompone en $\mathbb{Q}(\sqrt[3]{2})[x]$ como producto de polinomios lineales, ya que $\mathbb{Q}(\sqrt[3]{2})$ no contiene las raíces complejas de $x^3-2$. Sin embargo, como $\car(\mathbb{Q})=0$, tenemos que $x^3-2$ es separable, por lo que la extensión es separable.
\end{ejemplo}

\begin{observacion}
    Si $F\leq K$ es una extensión normal y $p\in F[x]$ es un polinomio mónico irreducible, entonces se verifica alguna de las dos siguientes afirmaciones:
    \begin{itemize}
        \item $p$ no tiene ninguna raíz en $K$.
        \item $p$ tiene todas sus raíces en $K$.
    \end{itemize}
    \begin{proof}
        Supongamos que $\alpha\in K$ es una raíz de $p$. En dicho caso, como $p$ es mónico e irreducible tenemos que $p = \Irr(\alpha,F)$, y como la extensión es normal tendremos que $p$ se descompone en $K[x]$ como producto de polinomios de grado uno, por lo que debe tener todas sus raíces en $K$.
    \end{proof}
\end{observacion}

\begin{teo}\label{teo:piedra_angular}
    Sea $F\leq K$ una extensión de cuerpos. Son equivalentes: 
    \begin{enumerate}[label=\roman*)]
        \item \label{item:teo_piedra_angular_i}
        $K$ es cuerpo de descomposición de un $f\in F[x]$ separable.
        \item $F\leq K$ es finita y $F = K^{\Aut_F(K)}$.
        \item $F = K^G$ para un subgrupo finito $G$ de $\Aut(K)$.
        \item $F\leq K$ es finita, normal y separable.
    \end{enumerate}
    \begin{proof}
        Veamos las equivalencias:
        \begin{description}
            \item [$i)\Longrightarrow  ii)$] Como $K$ es cuerpo de descomposición de cierto $f\in F[x]$, tenemos entonces que si $\alpha_1, \ldots, \alpha_s$ son las raíces de $f$ entonces:
                \begin{equation*}
                    K = F(\alpha_1, \ldots, \alpha_s)
                \end{equation*}
                Por lo que $F\leq K$ es finitamente generada. Si observamos ahora la demostración del Teorema~\ref{teo:finita_algebraica} observamos que solo usaba que los $\alpha_i$ eran algebraicos, por lo que podemos concluir que $F\leq K$ es finita.

                Sabemos ya que $F\leq F^{\Aut_F(K)}\leq K^{\Aut_F(K)}$. Como $F\leq K$ es finita tenemos entonces por la Proposición~\ref{prop:comienzo_chap_2} que $\Aut_F(K)$ es un subgrupo finito de $\Aut(K)$, de donde por el Teorema~\ref{teo:antes_teo_angular} tenemos que:
                \begin{equation*}
                    \left[K:K^{\Aut_F(K)}\right] = |\Aut_F(K)|
                \end{equation*}
                Finalmente, como $K$ es cuerpo de descomposición de $f\in F[x]$ separable, la Proposición~\ref{prop:cuerpo_desc_aut_f_lin} nos da la igualdad $(\ast)$:
                \begin{equation*}
                    \left[K:K^{\Aut_F(K)}\right] = |\Aut_F(K)| \AstIg [K:F]
                \end{equation*}
                y como tenemos $F\leq K^{\Aut_F(K)}\leq K$, el Lema de la Torre nos dice que $[K^{\Aut_F(K)}:F] = 1$, de donde $F = K^{\Aut_F(K)}$.
            \item [$ii)\Longrightarrow iii)$] Si la extensión es finita, tenemos entonces que $\Aut_F(K)$ es un subgrupo finito de $\Aut(K)$, con lo que tomando $G= \Aut_F(K)$ tenemos que $F=K^G$.
            \item [$iii)\Longrightarrow iv)$] Si $F=K^G$ con $G$ un subgrupo finito de $\Aut(K)$, la Proposición de Artin nos dice que:
                \begin{equation*}
                    [K:F] = [K:K^G] \leq |G|
                \end{equation*}
                por lo que $F\leq K$ es finita, luego también es algebraica.

                Sean $\alpha\in K$ y $h = \Irr(\alpha,F)\in F[x]$, tenemos que probar que $h$ se descompone en $K[x]$ como producto de polinomios lineales que no comparten raíces entre sí. Como $G$ actúa sobre $K$, podemos considerar la órbita de $\alpha$ (considerando todos sus elementos distintos):
                \begin{equation*}
                    Orb(\alpha) = \{\alpha_1, \ldots, \alpha_t\} \subseteq K
                \end{equation*}
                y podemos considerar el polinomio:
                \begin{equation*}
                    g = \prod_{i=1}^{t}(x-\alpha_i) = \sum_{j=0}^{t} a_j x^j \in K[x]
                \end{equation*}
                para ciertos $a_0,\ldots,a_t \in K$. Veamos que $a_j \in F$ para todo $j\in \{1,\ldots,t\}$, usando que $F=K^G$. Dado $\sigma\in G$:
                \begin{equation*}
                    \prod_{i=1}^{t} (x-\sigma(\alpha_i)) = g^\sigma = \sum_{j=0}^{t} \sigma(a_j) x^j
                \end{equation*}
                y vemos que $g = \prod\limits_{i=1}^{t}(x-\sigma(\alpha_i))$, puesto que al aplicar $\sigma$ sobre los elementos de la órbita los permuta, con lo que de la igualdad de la derecha deducimos que $\sigma(a_j) = a_j$, para todo $j\in \{1,\ldots,t\}$, con lo que tenemos tenemos $a_j\in F$ para todo $j \in \{1,\ldots,t\}$, luego $g\in F[x]$.

                Por una parte $g(\alpha) = 0$, puesto que $\alpha\in Orb(\alpha)$. Como $h=\Irr(\alpha,F)$, tenemos que $h$ divide a $g$.

                Por otra parte, cada $\alpha_i$ es raíz de $h$, ya que si $\alpha_i \in Orb(\alpha)$ tenemos que existe $\sigma\in G$ de forma que $\sigma(\alpha)=\alpha_i$, de donde (usando en $(\ast)$ que $h\in K^G[x]$):
                \begin{equation*}
                    h(\alpha_i) = h(\sigma(\alpha)) \AstIg \sigma(h(\alpha)) = \sigma(0) = 0
                \end{equation*}
                Como los elementos $\alpha_i$ son distintos, tenemos que $deg h \geq t$, pero como $h$ divide a $g$ y $g$ es un polinomio mónico de grado $t$, tenemos que $g=h$. Hemos probado que $\Irr(\alpha,F)$ se descompone en $K[x]$ como producto de polinomios de grado uno que no comparten raíces entre sí, para cada $\alpha\in K$. Es decir, hemos probado que $F\leq K$ es una extensión normal y separable.
            \item [$iv)\Longrightarrow i)$] Como $F\leq K$ es finita, tenemos entonces que existen $\alpha_1, \ldots, \alpha_s\in K$ algebraicos de forma que $K = F(\alpha_1, \ldots, \alpha_s)$. Podemos por tanto considerar $f_i = \Irr(\alpha_i, F)$, y tomamos como $f$ el producto de los $f_i$ eliminando repeticiones (es decir, multiplicamos todos los $f_i$ distintos). Como la extensión es normal y separable, cada uno de los $f_i$ se descompone como producto de polinomios de grado uno mónicos distintos, de donde $f$ es un polinomio separable, por lo que $K$ es un cuerpo de descomposición de $f$.
        \end{description}
    \end{proof}
\end{teo}

\begin{definicion}
    La órbita de $\alpha\in K$ bajo la acción de $G$ que ha aparecido en la demostración anterior, $Orb(\alpha)$ se llama el conjunto de conjugados de $\alpha$ bajo $G$.
\end{definicion}

Se trata de la generalización del concepto ``conjugado'' de un número complejo, pues basta considerar $\mathbb{R}\leq \mathbb{C}$ y $G=\{\id_{\mathbb{C}}, \sigma\}$, con $\sigma:\mathbb{C}\to\mathbb{C}$ la aplicación conjugación.\\

Debido a la equivalencia de todos estos apartados, surge la siguiente definición.
\begin{definicion}[Extensión de Galois]
    Una extensión $F\leq K$ se dice que es de Galois si es finita, normal y separable.
\end{definicion}

\begin{definicion}[Grupo de Galois]
    Sea $F\leq K$ una extensión de Galois. El grupo $\Aut_F(K)$ recibe el nombre de \underline{grupo de Galois de la extensión}.    
\end{definicion}

Notemos que el grupo de Galois de una extensión $F\leq K$ de Galois siempre es finito, porque $F\leq K$ es finita por ser de Galois y en dicho caso, por la Proposición~\ref{prop:comienzo_chap_2}, $|\Aut_F(K)|\leq [K:F]$.\\

El teorema anterior aporta herramientas útiles para estudiar las extensiones de Galois. En especial, el punto~\ref{item:teo_piedra_angular_i} nos da una forma práctica de comprobar que una extensión es de Galois.

\begin{ejemplo} Veamos algunos ejemplos de extensiones de Galois:\label{ejemplo:extensiones_de_galois}
    \begin{itemize}
        \item $\mathbb{Q}\leq \mathbb{Q}(\sqrt[3]{2},w)$. Como $\Irr(\sqrt[3]{2},\mathbb{Q}) = x^3-2$ es separable por tener $\car(\mathbb{Q})=0$, y $\mathbb{Q}(\sqrt[3]{2},w)$ es cuerpo de descomposición de $x^3-2\in \mathbb{Q}[x]$, tenemos que la extensión $\mathbb{Q}\leq \mathbb{Q}(\sqrt[3]{2},w)$ es de Galois.
        \item $\mathbb{F}_2\leq \mathbb{F}_{2^n}$. Como $\mathbb{F}_{2^n}$ es cuerpo de descomposición de $x^{2^n}-x\in \mathbb{F}_2[x]$, que es separable, tenemos que $\mathbb{F}_2\leq \mathbb{F}_{2^n}$ es de Galois.
        
        Buscamos calcular el grupo de Galois de $\mathbb{F}_2\leq \mathbb{F}_{2^n}$. Como $\bb{F}_2$ es el cuerpo primo, tenemos que:
        \begin{equation*}
            \Aut_{\bb{F}_2}(\bb{F}_{2^n}) = \Aut(\bb{F}_{2^n})
        \end{equation*}

        Al definir el automorfismo de Frobenius $\tau$ en la Definición~\ref{def:automorfismo_frobenius}, se vio que:
        \begin{equation*}
            \Aut(\bb{F}_{2^n}) = \langle \tau \rangle
        \end{equation*}
    \end{itemize}    
\end{ejemplo}

\begin{coro}
    En característica $0$, si $K$ es cuerpo de descomposición de un polinomio $f\in F[x]$, entonces $F\leq K$ es de Galois.
    \begin{proof}
        Consideramos la descomposición de $f$ en irreducibles:
        \begin{equation*}
            f = p_1^{n_1}\cdot \ldots \cdot p_t^{n_t}
        \end{equation*}
        con $p_i$ distintos. Obsevemos que $K$ es cuerpo de descomposición de $p_1\cdot \ldots\cdot p_t$. Cada uno de los $p_i$ es irreducible en $\car(F) = 0$, por lo que cada $p_i$ es separable. Como estos no pueden compartir raíces entre sí por ser irreducibles\footnote{Si $p_i$ tiene alguna raíz en común con $p_j$ entones $p_i$ dividiría a $p_j$, de donde deducimos que $p_i=p_j$ por ser ambos irreducibles, pero eran polinomios distintos.}, tendremos que $p_1\cdot \ldots\cdot p_t$ es separable. Por el Teorema anterior, la extensión es finita, normal y separable.
    \end{proof}
\end{coro}

\begin{coro}
    Si $F\leq K$ es de Galois y $F\leq E\leq K$ es una subextensión, entonces $E\leq K$ es de Galois.
    \begin{proof}
        Como $F\leq K$ es de Galois, entonces $K$ es cuerpo de descomposición de cierto $f\in F[x]$ separable, por lo que $K$ es cuerpo de descomposición de $f\in E[x]$, que sigue siendo separable.
    \end{proof}
\end{coro}

Notemos que, en general, $F\leq E$ no tiene por qué ser de Galois, como veremos en el siguiente ejemplo.

\begin{ejemplo}
    Si consideramos $\mathbb{Q}\leq \mathbb{Q}\left(\sqrt[3]{2}\right)$, tenemos una extensión finita y separable pero que no es de Galois, porque no es normal ya que $\Irr(\sqrt[3]{2},\mathbb{Q}) = x^3-2$ no se descompone como producto de polinomios lineales en $\mathbb{Q}\left(\sqrt[3]{2}\right)$, ya que $w\sqrt[3]{2}$ es una raíz del polinomio que no está en $\mathbb{Q}\left(\sqrt[3]{2}\right)$. Sin embargo, en el ejemplo de la página~\pageref{ejemplo:extensiones_de_galois} vimos que $\mathbb{Q}\leq \mathbb{Q}\left(\sqrt[3]{2},w\right)$ sí que es de Galois. En consecuencia, $\mathbb{Q}(\sqrt[3]{2})\leq \mathbb{Q}\left(\sqrt[3]{2},w\right)$ es de Galois.\\

    \noindent
    Sabemos que $\mathbb{Q}\leq \mathbb{Q}\left(\sqrt[3]{2}\right)$ no es normal porque $\Irr\left(\sqrt[3]{2},\mathbb{Q}\right) = x^3-2$ no se descompone como producto de polinomios lineales en $\mathbb{Q}\left(\sqrt[3]{2}\right)$, ya que $w\sqrt[3]{2}$ es una raíz del polinomio que no está en $\mathbb{Q}\left(\sqrt[3]{2}\right)$.
\end{ejemplo}

\begin{coro}
    Toda extensión de cuerpos finitos es de Galois con grupo de Galois cíclico.
    \begin{proof}
        Si tenemos una extensión $F\leq E$ de cuerpos finitos de característica $\car(F)=p$, tenemos entonces que:
        \begin{equation*}
            \bb{F}_p\leq F \leq E
        \end{equation*}
        con $\bb{F}_p\leq E$ de Galois, puesto que el polinomio $x^q-x \in \bb{F}_p[x]$ con $q=|E| = p^n$ es separable y $E$ es un cuerpo de descomposición suyo. Por el Corolario anterior, $F\leq E$ es de Galois. Por otro lado, como $\Aut(\bb{F}_{p^n}) = \langle \tau \rangle$ es cíclico, tenemos que $\Aut_F(E)$ es un subgrupo de $\Aut(\bb{F}_{p^n})$, y por tanto también es cíclico.
    \end{proof}
\end{coro}


\begin{ejemplo}
    Consideramos $\mathbb{Q}\leq E=\mathbb{Q}\left(\sqrt[3]{5},i\sqrt{5}\right)$, que es una extensión finita por ser algebraica y finitamente generada. Además, como $\car(\mathbb{Q})=0$, la extensión es separable. Veamos que no es normal, y por tanto no es de Galois.\\

    Si la extensión fuera normal, como $\sqrt[3]{5}\in E$, entonces $\Irr(\sqrt[3]{5},\mathbb{Q}) = x^3-5$ se descompondría en $E[x]$ como producto de polinomios lineales, por lo que todas sus raíces estarían en $E$. En particular, $w\sqrt[3]{5}\in E$, con $w$ una raíz cúbica primitiva de la unidad:
    \begin{equation*}
        w = e^{\frac{2\pi i}{3}} = \cos\left(\frac{2\pi}{3}\right) + i\sen\left(\frac{2\pi}{3}\right) = -\frac{1}{2} + i\frac{\sqrt{3}}{2}
    \end{equation*}

    Como $E$ es un cuerpo, $w\in E$, luego $i\sqrt{3}\in E$, y por tanto $\mathbb{Q}\left(i\sqrt{3},i\sqrt{5}\right)\leq E$. Por el Lema de la Torre, tenemos que:
    \begin{equation*}
        [E:\mathbb{Q}] = [E:\mathbb{Q}\left(i\sqrt{3},i\sqrt{5}\right)]\cdot \left[\mathbb{Q}\left(i\sqrt{3},i\sqrt{5}\right):\mathbb{Q}\right]
    \end{equation*}
    
    Buscamos calcular:
    \begin{equation*}
        \left[\mathbb{Q}\left(i\sqrt{3},i\sqrt{5}\right):\mathbb{Q}\right] = \left[\mathbb{Q}\left(i\sqrt{3},i\sqrt{5}\right):\mathbb{Q}\left(i\sqrt{3}\right)\right]\cdot \left[\mathbb{Q}\left(i\sqrt{3}\right):\mathbb{Q}\right]
    \end{equation*}
    Sabemos que $\left[\mathbb{Q}\left(i\sqrt{3}\right):\mathbb{Q}\right] = 2$, así como que $\left[\mathbb{Q}\left(i\sqrt{5},i\sqrt{3}\right):\mathbb{Q}\left(i\sqrt{3}\right)\right]\leq 2$:
    \begin{itemize}
        \item Si $\left[\mathbb{Q}\left(i\sqrt{5},i\sqrt{3}\right):\mathbb{Q}\left(i\sqrt{3}\right)\right]=1$, esto es porque $i\sqrt{5}\in \mathbb{Q}\left(i\sqrt{3}\right)$. En dicho caso, tendríamos que:
            \begin{equation*}
                i\sqrt{5} = a+bi\sqrt{3} \qquad a,b\in \mathbb{Q}
            \end{equation*}
            de donde $a=0$, con lo que $i\sqrt{5}=bi\sqrt{3}$, y elevando al cuadrado tendríamos que:
            \begin{equation*}
                -5 = -3b^2
            \end{equation*}
            de donde $b\in \mathbb{Q}$ es raíz de $3x^2-5$, pero:
            \begin{description}
                \item [Opción 1.] $3x^2-5$ es irreducible en $\mathbb{Q}[x]$ por Eisenstein (notemos que es primitivo). Por tanto, $3x^2-5$ no tiene raíces en $\mathbb{Q}$.
                \item [Opción 2.] Las posibles raíces racionales del polinomio enunciado son:
                    \begin{equation*}
                        1, -1, 5, -5, \frac{1}{3}, \frac{-1}{3}, \frac{5}{3}, \frac{-5}{3}
                    \end{equation*}
                    y ninguna es raíz.
            \end{description}

            Por tanto, $b\notin \mathbb{Q}$, con lo que $i\sqrt{5}\notin \mathbb{Q}\left(i\sqrt{3}\right)$, y por tanto: $$\left[\mathbb{Q}\left(i\sqrt{5},i\sqrt{3}\right):\mathbb{Q}\left(i\sqrt{3}\right)\right] = 2$$
    \end{itemize}

    Por tanto, $\left[\mathbb{Q}\left(i\sqrt{3},i\sqrt{5}\right):\mathbb{Q}\right] = 4$. Por otro lado, tenemos que:
    \begin{equation*}
        [E:\mathbb{Q}] = \left[\mathbb{Q}\left(\sqrt[3]{5},i\sqrt{5}\right):\mathbb{Q}(\sqrt[3]{5})\right]\cdot \left[\mathbb{Q}\left(\sqrt[3]{5}\right):\mathbb{Q}\right]
    \end{equation*}
    \begin{itemize}
        \item $\Irr(\sqrt[3]{5},\mathbb{Q}) = x^3-5$, que es irreducible por Eisenstein, luego $\left[\mathbb{Q}\left(\sqrt[3]{5}\right):\mathbb{Q}\right] = 3$.
        \item $\Irr(i\sqrt{5},\mathbb{Q}(\sqrt[3]{5})) = x^2+5$, que es irreducible por ser de grado $2$ y no tener raíces en $\bb{R}$ (y por tanto tampoco en $\mathbb{Q}(\sqrt[3]{5})$), por lo que $\left[\mathbb{Q}\left(\sqrt[3]{5},i\sqrt{5}\right):\mathbb{Q}(\sqrt[3]{5})\right] = 2$.
    \end{itemize}

    Por tanto, $[E:\mathbb{Q}] = 6$, y como $4$ no divide a $6$, tenemos una contradicción, que viene de suponer que la extensión era de Galois.
\end{ejemplo}

        \sqrt[3]{5}, w\sqrt[3]{5}, w^2\sqrt[3]{5}
    \end{equation*}
    con $w$ una raíz cúbica primitiva de la unidad:
    \begin{equation*}
        w = e^{\frac{2\pi i}{3}} = \cos\left(\frac{2\pi}{3}\right) + i\sen\left(\frac{2\pi}{3}\right) = -\frac{1}{2} + i\frac{\sqrt{3}}{2}
    \end{equation*}
    \begin{itemize}
        \item $\sqrt[3]{5}\in \bb{Q}(\sqrt[3]{5},i\sqrt{5})$ de manera trivial.
        \item Supuesto $w\sqrt[3]{5}\in \bb{Q}(\sqrt[3]{5},i\sqrt{5})$, entonces $w\in \bb{Q}(\sqrt[3]{5},i\sqrt{5})$, luego:
        \begin{equation*}
            i\sqrt{3} = 2w+1 \in \bb{Q}(\sqrt[3]{5},i\sqrt{5})
        \end{equation*}
        y por tanto $\bb{Q}(i\sqrt{3},i\sqrt{5})\leq \bb{Q}(\sqrt[3]{5},i\sqrt{5})$. Por el Lema de la Torre, tenemos que:
        \begin{equation*}
            [\bb{Q}(\sqrt[3]{5},i\sqrt{5}):\bb{Q}] = [\bb{Q}(\sqrt[3]{5},i\sqrt{5}):\bb{Q}(i\sqrt{3},i\sqrt{5})]\cdot [\bb{Q}(i\sqrt{3},i\sqrt{5}):\bb{Q}]
        \end{equation*}
        pero como vimos en el ejemplo anterior, $[\bb{Q}(i\sqrt{3},i\sqrt{5}):\bb{Q}] = 4$ y, además, $[\bb{Q}(\sqrt[3]{5},i\sqrt{5}):\bb{Q}] = 6$, y como $4$ no divide a $6$, tenemos una contradicción, de donde $w\sqrt[3]{5}\notin \bb{Q}(\sqrt[3]{5},i\sqrt{5})$.
        \item Como $w\sqrt[3]{5}\notin \bb{Q}(\sqrt[3]{5},i\sqrt{5})$, entonces $w\notin \bb{Q}(\sqrt[3]{5},i\sqrt{5})$, y por tanto $w^2=\nicefrac{1}{w}\notin \bb{Q}(\sqrt[3]{5},i\sqrt{5})$, con lo que $w^2\sqrt[3]{5}\notin \bb{Q}(\sqrt[3]{5},i\sqrt{5})$.
    \end{itemize}

    Notemos que esto prueba que la extensión $\bb{Q}\leq \bb{Q}(\sqrt[3]{5},i\sqrt{5})$ no es normal, y por tanto no es de Galois. Por tanto, $\Ex(\iota, \iota_1) = \{\eta_0\}$, con:
    \begin{equation*}
        \eta_0(\sqrt[3]{5}) = \sqrt[3]{5}
    \end{equation*}

    Por tanto, $\eta_0=\iota_3:\bb{Q}(\sqrt[3]{5})\hookrightarrow \bb{Q}(\sqrt[3]{5},i\sqrt{5})$ es la inclusión natural, y tenemos que:
    \begin{equation*}
        \Aut(\bb{Q}(\sqrt[3]{5},i\sqrt{5})) = \Ex(\iota_3, \iota_2)
    \end{equation*}

    Buscamos ahora $\Ex(\iota_3, \iota_2)$. Como $\Irr(i\sqrt{5},\bb{Q}(\sqrt[3]{5})) = x^2+5$ (que es irreducible en $\bb{Q}(\sqrt[3]{5})[x]$ por ser de grado $2$ y no tener raíces en $\bb{R}$, y por tanto tampoco en $\bb{Q}(\sqrt[3]{5})$), tenemos que $|\Ex(\iota_3, \iota_2)|$ es el número de raíces de $x^2+5$ en $\bb{Q}(\sqrt[3]{5},i\sqrt{5})$. Las raíces de $x^2+5$ son:
    \begin{equation*}
        \sqrt{5}\sqrt{-1} = \left\{i\sqrt{5}, -i\sqrt{5}\right\}
    \end{equation*}

    Como ambas raíces están en $\bb{Q}(\sqrt[3]{5},i\sqrt{5})$, tenemos que $|\Ex(\iota_3, \iota_2)| = 2$, y por tanto:
    \begin{equation*}
        \Aut(\bb{Q}(\sqrt[3]{5},i\sqrt{5})) = \Ex(\iota_3, \iota_2) = \{\eta_1, \eta_2\}
    \end{equation*}
    donde $\eta_1=\id$ y $\eta_2$ es el automorfismo que deja fijos a los elementos de $\bb{Q}(\sqrt[3]{5})$ y envía $i\sqrt{5}$ a $-i\sqrt{5}$:
    \begin{equation*}
        \eta_2(\sqrt[3]{5}) = \sqrt[3]{5}, \qquad \eta_2(i\sqrt{5}) = -i\sqrt{5}
    \end{equation*}

    Como $\eta_2^2 = \id$, tenemos que $\Aut(\bb{Q}(\sqrt[3]{5},i\sqrt{5}))$ es un grupo cíclico de orden $2$ generado por $\eta_2$:
    \begin{equation*}
        \Aut(\bb{Q}(\sqrt[3]{5},i\sqrt{5})) = \langle \eta_2 \rangle
    \end{equation*}

    Además, $\eta_2$ es la restricción de la conjugación compleja a $\bb{Q}(\sqrt[3]{5},i\sqrt{5})$.
\end{ejercicio}

\section{Teorema fundamental de la Teoría de Galois}

\begin{notacion}
    Notaremos:
    \begin{itemize}
        \item Si $F\leq K$ es una extensión y $F\leq E \leq K$ se dice que $E$ es una \underline{subextensión} de $F\leq K$. Denotamos al conjunto de todas ellas por $\Subex(F\leq K)$.
        \item Si $G$ es un grupo, llamamos $\Subgr(G)$ al conjunto de todos sus subgrupos.
        \item Si $H\in \Subgr(G)$, denotamos por $(G:H)$ al índice de $H$ en $G$.

            Recordemos que si $G$ es un grupo finito el Teorema de Lagrange nos decía que:
            \begin{equation*}
                |G| = (G:H)|H|
            \end{equation*}
    \end{itemize}
\end{notacion}

\begin{definicion}
    Sean $(A,\leq)$, $(B,\leq)$ dos conjuntos ordenados, un anti-isomorfismo de conjuntos ordenados es una aplicación biyectiva $f:A\to B$ de forma que:
    \begin{equation*}
        a\leq a' \Longleftrightarrow  f(a)\geq f(a')
    \end{equation*}
\end{definicion}

\begin{observacion}
    Si $G$ es un grupo y $F\leq K$ es una extensión de cuerpos, tenemos que $(\Subgr(G),<)$ y $(\Subex(F\leq K),\leq)$ son conjuntos ordenados, pues la relación indicada en cada conjunto es reflexiva, transitiva y antisimétrica.
\end{observacion}

\begin{teo} \label{teo:fundamental}
    Sea $F\leq K$ una extensión de Galois con grupo de Galois $G$. La aplicación 
    \Func{}{\Subgr(G)}{\Subex(F\leq K)}{H}{K^H}
    es un anti-isomorfismo de conjuntos ordenados, cuya isomorfismo inverso es 
    \Func{}{\Subex(F\leq K)}{\Subgr(G)}{E}{\Aut_E(K)}
    Si $H_1< H_2$ son subgrupos de $G$ y $E_2\leq E_1$ son las subextensiones de $F\leq K$ correspondientes a $H_1$ y $H_2$ por la anterior biyección, entonces:
    \begin{equation*}
        (H_2:H_1) = [E_1:E_2]
    \end{equation*}
    \begin{proof}
        Para ver que es un anti-isomorfismo, veamos que cada aplicación está bien definida y que una es la inversa de la otra:
        \begin{itemize}
            \item Sea $H\in \Subgr(G)$, tenemos entonces que $\{id_K\} < H < G$, y el Lema~\ref{lema:inclusiones} nos dice que:
                \begin{equation*}
                    K = K^{id_K} \geq K^H \geq K^G \AstIg F
                \end{equation*}
                donde en $(\ast)$ hemos usado que como $F\leq K$ es de Galois, tenemos el segundo punto del Teorema~\ref{teo:piedra_angular}. Por tanto, $K^H\in \Subex(F\leq K)$.
            \item Sea $E\in \Subex(F\leq K)$, tenemos entonces que $F\leq E \leq K$, y el Lema~\ref{lema:inclusiones} nos dice que:
                \begin{equation*}
                    G = \Aut_F(K) > \Aut_E(K) > \Aut_K(K) = \{id_K\}
                \end{equation*}
                de donde $\Aut_E(K)\in \Subgr(G)$.
            \item Sea $H\in \Subgr(G)$, si aplicamos las dos aplicaciones a $H$ obtenemos:
                \begin{equation*}
                    H \longmapsto K^H \longmapsto \Aut_{K^H}(K)
                \end{equation*}
                como $F\leq K$ es de Galois tenemos que es una extensión finita, por lo que $G$ es finito, y como $H$ es un subgrupo finito de $G$ podemos aplicar el Teorema~\ref{teo:antes_teo_angular}, obteniendo que $H = \Aut_{K^H}(K)$.
            \item Sea $E\in \Subex(F\leq K)$, si aplicamos las dos aplicaciones a $E$ obtenemos:
                \begin{equation*}
                    E\longmapsto \Aut_E(K) \longmapsto K^{\Aut_E(K)}
                \end{equation*}
                y como $F\leq K$ es de Galois deducimos que $E\leq K$ es también de Galois, pudiendo aplicar finalmente el segundo punto del Teorema~\ref{teo:piedra_angular}, obteniendo que $E = K^{\Aut_E(K)}$.
        \end{itemize}
        En consecuencia, la aplicación enunciada es un anti-isomorfismo de conjuntos ordenados.\\

        \noindent
        Para la segunda parte, si $H_1< H_2$ son subgrupos de $G$ y $E_2\leq E_1$ son las subextensiones de $F\leq K$ correspondientes a dichos subgrupos (es decir, $E_1 = K^{H_1}$ y $E_2 = K^{H_2}$), aplicando el Teorema~\ref{teo:antes_teo_angular} en $(\ast)$ y el Lema de la Torre en $(\ast\ast)$ tenemos entonces que:
        \begin{equation*}
            |H_2| \AstIg [K:E_2] \stackrel{(\ast\ast)}{=} [K:E_1] [E_1:E_2] \AstIg |H_1|[E_1:E_2]
        \end{equation*}

        de donde:
        \begin{equation*}
            [E_1:E_2] = \dfrac{|H_2|}{|H_1|} = (H_2:H_1)
        \end{equation*}
    \end{proof}
\end{teo}

\begin{definicion}[Conexión de Galois]
    La biyección del Teorema anterior recibe el nombre ``Conexión de Galois''.
\end{definicion}

\begin{ejemplo}
    Si consideramos la extensión de Galois $\mathbb{Q}\leq \mathbb{Q}\left(\sqrt[3]{2},w\right)$, vimos en el ejemplo de la página~\pageref{ejemplo:aut_sqrt3_2} que $\Aut\left(\mathbb{Q}\left(\sqrt[3]{2},w\right)\right)$ tenía 6 elementos, y en el ejemplo de la página~\pageref{ejemplo:subcuerpos_fijos} calculábamos $\Subgr(\Aut\left(\mathbb{Q}\left(\sqrt[3]{2},w\right)\right))$, obteniendo 6 subgrupos.\\ 

    Por la Conexión de Galois sabemos ahora que hay una biyección entre subgrupos de $\Aut\left(\mathbb{Q}\left(\sqrt[3]{2},w\right)\right)$ y subcuerpos $E$ de $\mathbb{Q}\left(\sqrt[3]{2},w\right)$ con $\mathbb{Q}\leq E\leq \mathbb{Q}\left(\sqrt[3]{2},w\right)$. Como $\mathbb{Q}$ es el subcuerpo primo de $\mathbb{Q}\left(\sqrt[3]{2},w\right)$, tenemos que hay una biyección entre subgrupos de $\Aut\left(\mathbb{Q}\left(\sqrt[3]{2},w\right)\right)$ y subcuerpos de $\mathbb{Q}\left(\sqrt[3]{2},w\right)$, como se vio en el ejemplo de la página~\pageref{ejemplo:subcuerpos_fijos}.
\end{ejemplo}

\begin{ejemplo}
    Sea $\bb{F}_q = \bb{F}_{p^n}$, nos preguntamos por los subcuerpos de dicho cuerpo. La extensión $\bb{F}_p\leq \bb{F}_{p^n}$ es de Galois por ser una extensión de cuerpos finitos, por lo que podemos tratar de usar la conexión de Galois. Más aún, habíamos visto que:
    \begin{equation*}
        \Aut(\bb{F}_{p^n}) = \Aut_{\bb{F}_p}(\bb{F}_{p^n}) = \langle \tau \rangle 
    \end{equation*}
    cíclico de orden $n$, con $\tau(\alpha)=\alpha^p$ el automorfismo de Frobenius. Los subgrupos están parametrizados por los divisores de $n$, con lo que:
    \begin{equation*}
        \Subgr(\Aut(\bb{F}_{p^n})) = \left\{\langle \tau^d \rangle : d \in Div(n) \right\}
    \end{equation*}
    Los subcuerpos de $\bb{F}_{p^n}$ son, por la conexión de Galois:
    \begin{equation*}
        \left\{\bb{F}_{p^n}^{\langle \tau^d \rangle }: d\in Div(n)\right\}
    \end{equation*}
    Fijado $d\in Div(n)$, tenemos que $\bb{F}_{p^n}^{\langle \tau^d \rangle }$ es un subcuerpo de $\bb{F}_{p^n}$, y por el Teorema~\ref{teo:fundamental}:
    \begin{equation*}
        \left[\bb{F}_{p^n}^{\langle \tau^d \rangle }:\bb{F}_p\right] = \left(\langle \tau \rangle : \langle \tau^d \rangle \right) = \dfrac{|\langle \tau \rangle|}{|\langle \tau^d \rangle|} = \dfrac{n}{\nicefrac{n}{d}} = d
    \end{equation*}
    Por tanto, como $|\bb{F}_p| = p$, tenemos que:
    \begin{equation*}
        \left|\bb{F}_{p^n}^{\langle \tau^d \rangle }\right| = p^d
    \end{equation*}
    
    
    Por tanto, cada cuerpo de $p^n$ elementos tiene un subcuerpo de cardinal $p^d$ con $d\in Div(n)$, y por el Teorema~\ref{teo:fundamental} sabemos que no hay más subcuerpos. Por ejemplo, un cuerpo de $64=2^6$ elementos tiene 4 subcuerpos (cada divisor de 6).
\end{ejemplo}

\begin{ejercicio}

    \begin{equation*}

    \end{equation*}
    Describir explícitamente, usando potencias de $a$, todos los subcuerpos de $\bb{F}_{16}$.\\

Por el ejemplo anterior, sabemos que los subcuerpos de $\bb{F}_{16}$ están parametrizados por los divisores de $4$, que son $1,2,4$. Por tanto, tenemos:
    \begin{itemize}
        \item Para $d=1$, tenemos que $\bb{F}_{16}^{\langle \tau \rangle}$ es un subcuerpo de cardinal $2^1=2$, y como $\bb{F}_2$ es el único cuerpo de cardinal $2$, luego:
            \begin{equation*}
                \bb{F}_{16}^{\langle \tau \rangle} = \bb{F}_2 = \left\{0,1\right\} = \left\{0,a^{15}\right\}
            \end{equation*}
        \item Para $d=2$, tenemos que $\bb{F}_{16}^{\langle \tau^2 \rangle}$ es un subcuerpo de cardinal $2^2=4$, y como $\bb{F}_4$ es el único cuerpo de cardinal $4$, luego:

            \end{equation*}
        \item Para $d=4$, tenemos que $\bb{F}_{16}^{\langle \tau^4 \rangle}$ es un subcuerpo de cardinal $2^4=16$, y como $\bb{F}_{16}$ es el único cuerpo de cardinal $16$, luego:

Vamos ahora a caracterizar que subextensiones de una extensión de Galois estan ligadas, por la correspondencia de Galois, con subgrupos normales del grupo de Galois.
\begin{lema}
    Sea $F\leq K$ de Galois con grupo de Galois $G$, sea $H\in \Subgr(G)$ y $E\in \Subex(F\leq K)$ su correspondiente mediante la conexión de Galois. Si $\sigma\in G$, entonces $\sigma H \sigma^{-1}$ y $\sigma(E)$ son correspondientes por la conexión de Galois.
    \begin{proof}
        De Álgebra II sabemos que si $H\in \Subgr(G)$ entonces para $\sigma\in G$ tenemos $\sigma H\sigma^{-1}\in \Subgr(G)$. Vemos también que $\sigma(E)$ sigue siendo una subextensión de $F\leq K$, por lo que la pregunta está bien planteada.

        \noindent
        Tenemos que $E = K^H$ y queremos probar que $\sigma(K^H) = K^{\sigma H \sigma^{-1}}$. Tendremos:
        \begin{align*}
            \alpha \in K^{\sigma H \sigma^{-1}} &\Longleftrightarrow \sigma\tau \sigma^{-1}(\alpha) = \alpha \quad \forall \tau \in H \Longleftrightarrow \tau\sigma^{-1}(\alpha) = \sigma^{-1}(\alpha) \quad \forall \tau \in H \\ &\Longleftrightarrow \sigma^{-1}(\alpha) \in K^H \Longleftrightarrow \alpha \in \sigma(K^H)
        \end{align*}
    \end{proof}
\end{lema}

\begin{teo}\label{teo:normal}
    Sea $F\leq K$ de Galois y $G$ su grupo de Galois, si $H\in \Subgr(G)$ y $E\in \Subex(F\leq K)$ es su correspondiente mediante la conexión de Galois entonces:
    \begin{equation*}
        H \text{\ es normal en\ } G \quad\Longleftrightarrow\quad F\leq E \text{\ es de Galois}
    \end{equation*}
    En cuyo caso, $\Aut_F(E)\cong G/H$.
    \begin{proof}
        Por doble implicación:
        \begin{description}
            \item [$\Longrightarrow )$] Si $H\lhd G$ tenemos entonces que:
                \begin{equation*}
                    \sigma(E) = \sigma(K^H) = K^{\sigma H\sigma^{-1}} = K^H = E \qquad \forall \sigma\in \Aut_F(K)
                \end{equation*}
                Por lo que podemos definir la aplicación $r:\Aut_F(K)\to \Aut_F(E)$ dada por:
                \begin{equation*}
                    r(\sigma) = \sigma\big|_E
                \end{equation*}
                y estará bien definida, pues $\sigma(E) = E$ para cada $\sigma\in \Aut_F(K)$. Es fácil ver que $r$ es un homomorfismo de grupos. Observamos ahora que:
                \begin{equation*}
                    \ker(r) = \left\{\sigma\in \Aut_F(K): \sigma\big|_E = \id_E\right\} = \Aut_E(K) = H
                \end{equation*}
                Aplicando el Primer Teorema de Isomorfía para grupos obtenemos que:
                \begin{equation*}
                    G/H\cong Im r
                \end{equation*}
                y si observamos ahora que:
                \begin{equation*}
                    [E:F] = (G:H) = |Im r| \leq |\Aut_F(E)| \leq [E:F]
                \end{equation*}
                Obtenemos que $Im r = \Aut_F(E)$, por lo que $r$ es sobreyectivo, tenemos ya que $G/H\cong \Aut_F(E)$. Para ver que $F\leq E$ es de Galois, trataremos de probar la tercera condición del Teorema~\ref{teo:piedra_angular}, de la que conocemos ya la inclusión $F\leq E^{\Aut_F(E)}$. Sea pues $\alpha \in E^{\Aut_F(E)}$, vemos que:
                \begin{equation*}
                    \alpha = r(\sigma)(\alpha) = \sigma\big|_E(\alpha) = \alpha \qquad \forall \sigma\in \Aut_F(K)
                \end{equation*}
                por lo que $\alpha \in K^{\Aut_F(K)} = F$, lo que nos da la otra inclusión, de donde $F\leq E$ es de Galois.
            \item [$\Longleftarrow )$] Si $F\leq E$ es de Galois tenemos entonces que $E$ es cuerpo de descomposición de cierto polinomio $f\in F[x]$ separable, que podemos suponer de la forma:
                \begin{equation*}
                    f = \prod_{i=1}^{n}(x-\alpha_i)
                \end{equation*}
                con $E = F(\alpha_1,\ldots,\alpha_n)$, ya que si $f$ no es mónico podemos dividir todos sus coeficientes entre el término líder, obteniendo un polinomio mónico con las mismas raíces. Sea $\sigma\in \Aut_F(K)$, por ser $f\in F[x]$ tenemos que:
                \begin{equation*}
                    f = f^{\sigma} = \prod_{i=1}^{n}(x-\sigma(\alpha_i))
                \end{equation*}
                por lo que será $\sigma(\alpha_i)=\alpha_j\in E$, en particular vemos que (usando una $F-$base de $E$ en función de $\alpha_i$) $\sigma(E) = E\quad \forall \sigma\in \Aut_F(K)$, por lo que:
                \begin{equation*}
                    K^H = E = \sigma(E) = \sigma(K^H) = K^{\sigma H\sigma^{-1}} \qquad \forall \sigma\in \Aut_F(K)
                \end{equation*}
                de donde $H =\sigma H \sigma^{-1}\quad \forall \sigma\in \Aut_F(K)$, por lo que $H\lhd \Aut_F(K)$.
        \end{description}
    \end{proof}
\end{teo}

\begin{ejemplo}
    Consideramos $f = x^4-2x^2-2\in \mathbb{Q}[x]$, y tomamos $K$ el cuerpo de descomposición de $f$. Se pide calcular o describir todos los subcuerpos de $K$.\\

    \noindent
    Observemos que $\mathbb{Q}\leq K$ es de Galois por ser $K$ cuerpo de descomposición de $f$ en característica $0$. Calculamos primero las raíces de $f$. Observamos que si $s$ es una de ellas, entonces $s^2$ es raíz de $x^2-2x-2$. Puesto que $\car(\bb{Q})\neq 2$:
    \begin{equation*}
        s^2 = \frac{2\pm \sqrt{12}}{2} = 1\pm \sqrt{3}
    \end{equation*}
    Obtenemos que las raíces de $f$ son $\alpha,-\alpha,\beta,-\beta$, donde:
    \begin{equation*}
        \alpha = \sqrt{\sqrt{3}+1}, \qquad \beta = i\sqrt{\sqrt{3}-1}
    \end{equation*}
    Sabemos en este momento que $K = \mathbb{Q}(\alpha,\beta)$. Si vemos que:
    \begin{equation*}
        \alpha\beta = i\sqrt{2}
    \end{equation*}
    Tenemos que $K = \mathbb{Q}(\alpha,\beta) = \mathbb{Q}(\alpha,\alpha\beta) = \mathbb{Q}\left(\sqrt{\sqrt{3}+1}, i\sqrt{2}\right)$. Como se piden calcular todos los subcuerpos de $K$ y $\bb{Q}\leq K$ es de Galois con $\bb{Q}$ el cuerpo primo de $K$, por la conexión de Galois sabemos que hay tantos subcuerpos como subgrupos de $\Aut_\mathbb{Q}(K)$. Por tanto, nos conviene calcular primero $\Aut_\mathbb{Q}(K)$. Sabemos que:
    \begin{equation*}
        |\Aut_\mathbb{Q}(K)| = [K:\mathbb{Q}]
    \end{equation*}

    Por el Lema de la Torre, tenemos que:
    \begin{equation*}
        [K:\mathbb{Q}] = \left[\mathbb{Q}\left(\alpha,i\sqrt{2}\right):\mathbb{Q}\right] = \left[\mathbb{Q}\left(\alpha,i\sqrt{2}\right):\mathbb{Q}(\alpha)\right]\left[\mathbb{Q}(\alpha):\mathbb{Q}\right]
    \end{equation*}
    \begin{itemize}
        \item Calculamos $[\mathbb{Q}(\alpha):\mathbb{Q}]$. Tenemos que $\alpha$ es raíz de $f$, y como $f$ es mónico e irreducible por Eisenstein para $p=2$, tenemos que $f = \Irr(\alpha,\mathbb{Q})$, de donde $[\mathbb{Q}(\alpha):\mathbb{Q}] = 4$.
        \item Por otra parte, como $i\sqrt{2}$ es raíz de $x^2+2$, tenemos que $[\mathbb{Q}\left(i\sqrt{2}\right):\mathbb{Q}]\leq 2$. Además, como $\mathbb{Q}(\alpha)\leq \mathbb{R}$, tenemos que $i\sqrt{2}\notin \mathbb{Q}(\alpha)$, de donde $[\mathbb{Q}\left(i\sqrt{2}\right):\mathbb{Q}]\geq 2$, y por tanto $[\mathbb{Q}\left(i\sqrt{2}\right):\mathbb{Q}] = 2$.
        
        Notemos que este argumento no podríamos haberlo hecho con $\beta$, ya que obtendríamos que $[\mathbb{Q}(\alpha,\beta):\mathbb{Q}(\alpha)] \in \{1,2,4\}$, y no podemos distinguir entre $2$ y $4$. Sabemos ya seguro que $f$ no es irreducible sobre $\mathbb{Q}(\alpha)$, pues si lo fuera tendríamos que $f=\Irr(\beta,\mathbb{Q}(\alpha))$, de donde sería $[\mathbb{Q}(\alpha,\beta):\mathbb{Q}(\alpha)] = 4$.
    \end{itemize}

    Por tanto, tenemos que:
    \begin{equation*}
        [K:\mathbb{Q}] = 2\cdot 4 = 8 = |\Aut_\mathbb{Q}(K)|
    \end{equation*}

\section{El Teorema Fundamental del Álgebra}
\noindent
Se tiene que $\mathbb{C}$ es el cuerpo de descomposición de $x^2+1\in \mathbb{R}[x]$, y como todas sus raíces son simples, tenemos que:
\begin{equation*}
    |\Aut_\mathbb{R}(\mathbb{C})| = [\mathbb{C}:\mathbb{R}] = 2
\end{equation*}

Notando por $\ol{\ \cdot\ }$ al automorfismo que conjuga cada número complejo, como este es un automorfismo de $\mathbb{C}$ que deja fijos a los reales, en particular es un automorfismo $\mathbb{R}-$lineal, luego $\ol{\ \cdot\ }\in \Aut_\mathbb{R}(\mathbb{C})$. Por lo que tenemos que:
\begin{equation*}
    \Aut_\mathbb{R}(\mathbb{C}) = \{\id, \ol{\ \cdot\ }\}
\end{equation*}

\noindent
Además, si $f\in \mathbb{C}[x]$, notaremos por $\ol{f}$ a $f^{\ol{\ \cdot\ }}$; es decir, $\ol{f}\in \mathbb{C}[x]$ será el polinomio obtenido de $f$ conjugando todos sus coeficientes.

\begin{teo}[Fundamental del Álgebra] 
    Si $f\in \mathbb{C}[x]$ no constante, entonces $f$ tiene todas sus raíces\footnote{O equivalentemente, $\mathbb{C}$ es cuerpo de descomposición de $f$.} en $\mathbb{C}$.
    \begin{proof}
        Sea $g=f\overline{f}\in \mathbb{R}[x]$, y consideramos $(x^2+1)g\in \mathbb{R}[x]$, que es no constante. Sea $K$ su cuerpo de descomposición, queremos probar que $K = \mathbb{C}$. Observemos que $K$ contiene a $\mathbb{C}$ por la Proposición de extensión:
        \begin{figure}[H]
            \centering
            \shorthandoff{""}
            \begin{tikzcd}
                \mathbb{R} \arrow[r] \arrow[rd] & K                    \\
                                                & \mathbb{C} \arrow[u]
            \end{tikzcd}
            \shorthandon{""}
        \end{figure}
        \noindent
        Además, las tres extensiones que aparecen son de Galois. El Lema de la Torre nos dice que 
        \begin{equation*}
            |\Aut_{\mathbb{R}}(K)| = [K:\mathbb{R}] = [K:\mathbb{C}][\mathbb{C}:\mathbb{R}] = [K:\mathbb{C}]\cdot 2
        \end{equation*}
        múltiplo de 2, por lo que $G=\Aut_\mathbb{R}(K)$ contiene un $2-$subgrupo de Sylow, $H$. La conexión de Galois nos dice que $[K^H:\mathbb{R}] = (G:H)$, con $(G:H)$ impar por ser $H$ un $2-$subgrupo de Sylow.\\

        \noindent
        Sea $\alpha\in K^H$, consideramos $\Irr(\alpha,\mathbb{R})$, que tiene grado impar por el Lema de la Torre (si su grado fuera par tendríamos entonces que $(G:H)$ también sería par, ya que $\mathbb{R}\leq K(\alpha)\leq K^H$), por lo que ha de tener una raíz real $\beta\in \mathbb{R}$, de donde $\Irr(\alpha,\mathbb{R})$ es divisible por $(x-\beta)$, pero como $\Irr(\alpha,\mathbb{R})$ es irreducible, este debe ser asociado a $x-\beta$, y los dos son mónicos, de donde $\Irr(\alpha,\mathbb{R}) = x-\beta$, luego $\beta = \alpha$ y tenemos por tanto que $[K^H:\mathbb{R}] = 1$, por lo que $G=H$, de donde $G$ es un $2-$grupo.\\

        \noindent
        Así, $\Aut_\mathbb{C}(K)$ es un $2-$grupo (por ser subgrupo de un $2-$grupo, $\Aut_\mathbb{R}(K)>\Aut_\mathbb{C}(K)$). Supongamos que $|\Aut_\mathbb{C}(K)|>1$. Como $\Aut_\mathbb{C}(K)$ es resoluble (por ser un $p-$grupo), tiene un subgrupo de índice $2$, $N$. Por la conexión de Galois, tenemos que $\mathbb{C}\leq K^N$ es una extensión de grado 2, por lo que $K^N = \mathbb{C}(\beta)$ para $\beta\in K^N$ con $\Irr(\beta,\mathbb{C})$ de grado 2. Si tratamos de obtener las raíces de este polinomio, obtenemos que $\beta\in \mathbb{C}$, porque cada número complejo tiene sus raíces cuadradas en $\mathbb{C}$, \underline{contradicción}, puesto que teníamos que $\beta\in K^N\setminus \mathbb{C}$, por lo que tenemos que $|\Aut_\mathbb{C}(K)| = 1$, de donde por la conexión de Galois, $K = \mathbb{C}$.
    \end{proof}
\end{teo}

\begin{coro}
    Si $f\in \mathbb{R}[x]$ es irreducible, entonces $\deg f \in \{1,2\}$.
    \begin{proof}
        Sea $f\in \mathbb{R}[x]$ irreducible con $\deg f\geq 2$, el Teorema Fundamental del Álgebra nos permite tomar $\alpha\in \mathbb{C}$ una raíz suya, de donde $\overline{\alpha}$ también será una raíz de $f$. Tenemos así que $(x-\alpha)(x-\overline{\alpha})$ divide a $f$, pero como $f$ es irreducible tiene que ser asociado a $(x-\alpha)(x-\overline{\alpha})$, por lo que $\deg f = 2$.
    \end{proof}
\end{coro}

\begin{ejercicio} 
    Sea $\overline{\mathbb{Q}}$ la clausura algebraica de $\mathbb{Q}$ en $\mathbb{C}$. Razonar que todo polinomio no constante $f\in \overline{\mathbb{Q}}[x]$ tiene todas sus raíces en $\overline{\mathbb{Q}}$.\\

    \noindent
    Sea $f\in \overline{\mathbb{Q}}[x]$ un polinomio no constante, $f$ será de la forma:
    \begin{equation*}
        f = \sum_{i=0}^{n}a_ix^i
    \end{equation*}
    con $a_0,a_1,\ldots,a_n\in \overline{\mathbb{Q}}$ todos estos algebraicos, por lo que la extensión $\mathbb{Q}\leq \mathbb{Q}(a_0,a_1,\ldots,a_n)$ es finitamente generada y algebraica, luego finita. Sea $\alpha$ una raíz de $f$, que por el Teorema Fundamental del Álgebra será $\alpha\in \mathbb{C}$, luego $\alpha$ es algebraico sobre $\mathbb{Q}(a_0,a_1,\ldots,a_n)$, por lo que tenemos la extensión:
    \begin{equation*}
        \mathbb{Q}\leq \mathbb{Q}(a_0,a_1,\ldots,a_n)
        \leq \mathbb{Q}(a_0,a_1,\ldots,a_n)(\alpha) = \mathbb{Q}(a_0,a_1,\ldots,a_n,\alpha)
    \end{equation*}
    La primera extensión hemos visto que es finita, y la segunda extensión es también finita con:
    \begin{equation*}
        [\mathbb{Q}(a_0,a_1,\ldots,a_n,\alpha):\mathbb{Q}(a_0,a_1,\ldots,a_n)] = \deg(\Irr(\alpha,\mathbb{Q}(a_0,a_1,\ldots,a_n))) \leq n
    \end{equation*}

    Por tanto, por el Lema de la Torre, la extensión $\mathbb{Q}\leq \mathbb{Q}(a_0,a_1,\ldots,a_n,\alpha)$ también es finita, luego es algebraica, de donde $\mathbb{Q}(a_0,a_1,\ldots,a_n,\alpha)\leq \overline{\mathbb{Q}}$, por lo que en particular $\alpha \in \overline{\mathbb{Q}}$.
\end{ejercicio}

\begin{ejercicio} % // TODO: UN EJERCICIO RESUELTO DE UN EXAMEN
    Tomemos $f=(x^3-2)(x^2-3)\in \mathbb{Q}[x]$ y su el cuerpo de descomposición de $f$ sobre $\mathbb{Q}$.
    \begin{enumerate}
        \item Decidir razonadamente si $i+\sqrt{3}\in K$.

            El cuerpo de descomposición es:
            \begin{equation*}
                K = \mathbb{Q}\left(\pm\sqrt{3}, \sqrt[3]{2},w\sqrt[3]{2},w^2\sqrt[3]{2}\right)
            \end{equation*}

            donde:
            \begin{equation*}
                w = -\frac{1}{2}+i\frac{\sqrt{3}}{2}
            \end{equation*}
            Tenemos entonces que $\sqrt{3}\in K$ así como que:
            \begin{equation*}
                w = \frac{w\sqrt[3]{2}}{\sqrt[3]{2}} \in K
            \end{equation*}
            Por lo que $i\sqrt{3}\in K$, de donde:
            \begin{equation*}
                i = \frac{i\sqrt{3}}{\sqrt{3}} \in K
            \end{equation*}
            Por tanto, tenemos que $i+\sqrt{3}\in K$.
        \item Calcular razonadamente $[K:\mathbb{Q}]$.
            Sabemos ya que:
            \begin{equation*}
                \mathbb{Q}\left(\sqrt{3},\sqrt[3]{2},i\right)\leq K
            \end{equation*}
            Y la otra inclusión la vemos viendo que todos los elementos que definen $K$ se expresan con operaciones con estos tres, con lo que:
            \begin{equation*}
                K = \mathbb{Q}\left(\sqrt{3},\sqrt[3]{2},i\right)
            \end{equation*}
            Con vistas al Lema de la Torre, vemos:
            \begin{equation*}
                \left[\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right):\mathbb{Q}\right] = \left[\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right):\mathbb{Q}\left(\sqrt{3}\right)\right] \left[\mathbb{Q}\left(\sqrt{3}\right):\mathbb{Q}\right]
            \end{equation*}
            Donde el segundo vale 3 por Eisenstein y el primero es menor o igual que 2, por lo que en total es menor o igual que 6. Si lo miramos de otra forma:
            \begin{equation*}
                \left[\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right):\mathbb{Q}\right] = \left[\mathbb{Q}\left(\sqrt[3]{2},\sqrt[3]{2}\right):\mathbb{Q}\left(\sqrt[3]{2}\right)\right] \left[\mathbb{Q}\left(\sqrt{3}\right):\mathbb{Q}\right]
            \end{equation*}
            Tenemos que el segundo vale 2 por Eisenstein y el primero es menor o igual que 3. En definitiva, tenemos que $\left[\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right):\mathbb{Q}\right] =6$. Ahora:
            \begin{equation*}
                \left[K:\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right)\right] = 2
            \end{equation*}
            Ya que $x^2+1$ es irreducible en $\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right)$, al no tener raíces en dicho cuerpo. En definitiva, el Lema de la Torre nos dice que:
            \begin{equation*}
                [K:\mathbb{Q}] = 12
            \end{equation*}
        \item Describir los elementos del grupo $\Aut(K)$.

            $F\leq K$ es de Galois por ser $K$ cuerpo de descomposición de $f$ con $\car(F) = 0$, por lo que:
            \begin{equation*}
                |\Aut(K)| = [K:F] = 12
            \end{equation*}
            Usamos varias veces la Proposición de Extensión. Calculamos primero las extensiones de la inclusión a $\mathbb{Q}(\sqrt{3})\stackrel{\eta_j}{\to} K$. Las mismas están en correspondencia biyectiva con las raíces de $\Irr\left(\sqrt{3},\mathbb{Q}\right) = x^2-3$ en $K$. A saber, son dos determinadas por:
            \begin{equation*}
                \eta_j(\sqrt{3}) = {(-1)}^{j}\sqrt{3} \qquad j \in \{0,1\}
            \end{equation*}
            Análogamente, con la misma Proposición tenemos que cada $\eta_j$ se extienden a $3$ homomorfismos $\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right)\stackrel{}{\to}K$, ya que las raíces de $x^3-2\in \mathbb{Q}\left(\sqrt{3}\right)[x]$ en $K$ son tres, a saber, $\sqrt[3]{2},w\sqrt[3]{2}, w^2\sqrt[3]{2}$. Obsérvese que $x^3-2\in \mathbb{Q}\left(\sqrt{3}\right)[x]$ es irreducible porque:
            \begin{equation*}
                3 = \left[\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right):\mathbb{Q}\left(\sqrt{3}\right)\right]
            \end{equation*}
            Así, obtengo $\eta_{j,k}:\mathbb{Q}\left(\sqrt{3},\sqrt[3]{2}\right)\to K$, determinados por:
            \begin{align*}
                \eta_{j,k}\left(\sqrt[3]{2}\right) &=  w^k \sqrt{2} \\
                \eta_{j,k}\left(\sqrt{3}\right) &= {(-1)}^{j}\sqrt{3} \qquad j\in \{0,1\},\quad k \in \{0,1,2\}
            \end{align*}
            De la misma manera, extendiendo cada $\eta_{j,k}$ según las raíces de $x^2+1$, obtengo que:
            \begin{equation*}
                \Aut(K) = \{\eta_{j,k,l} : j,l\in \{0,1\}, k\in \{0,1,2\}\}
            \end{equation*}
            donde cada uno de ellos está determinado por:
            \begin{align*}
                \eta_{j,k,l}\left(\sqrt{3}\right) &= {(-1)}^{j}\sqrt{3} \\
                \eta_{j,k,l}\left(\sqrt[3]{2}\right) &= w^k \sqrt[3]{2}\\
                \eta_{j,k.l}(i) &= {(-1)}^{l}i
            \end{align*}
        \item Describir los elementos de $\Aut_{\mathbb{Q}\left(i+\sqrt{3}\right)}(K)$ y decidir si es un subgrupo normal de $\Aut(K)$.

            Tenemos que el subgrupo enunciado es el correspondiente a la extensión
            \begin{equation*}
                \mathbb{Q}\leq \mathbb{Q}(i+\sqrt{3})
            \end{equation*}
            $\mathbb{Q}$ es correspondiente con $G$ y $\mathbb{Q}(i+\sqrt{3})$ es correspondiente con $H$. Como $[\mathbb{Q}\left(i+\sqrt{3}\right):\mathbb{Q}] = 4$, tenemos  por la conexión de Galois que $|H| = 3$, por lo que buscamos:
            \begin{equation*}
                i+\sqrt{3} = \eta_{j,k,l}(i+\sqrt{3}) = {(-1)}^{l}i + {(-1)}^{j}\sqrt{3}
            \end{equation*}
            si y solo si $l = 0 = j$, obtenemos 3 automorfismos, que son los únicos posibles para $H$. En definitiva:
            \begin{equation*}
                \Aut_{\mathbb{Q}\left(i+\sqrt{3}\right)}(K) = \{\eta_{0,k,0} : k\in \{0,1,2\}\}
            \end{equation*}
            Que es normal, porque la extensión $\mathbb{Q}\leq \mathbb{Q}(i+\sqrt{3})$ es de Galois. Veamos esto último, pues:
            \begin{equation*}
                \mathbb{Q}\left(i+\sqrt{3}\right) = \mathbb{Q}\left(i,\sqrt{3}\right)
            \end{equation*}
            por lo que $\mathbb{Q}(i+\sqrt{3})$ es cuerpo de descomposición de $(x^2+1)(x^2-3)$.
    \end{enumerate}
\end{ejercicio}

\begin{ejercicio} % // TODO: TMB DE EXAMEN, quizas es el que cae
    Sea $K$ cuerpo de descomposición de $f = (x^2+3)(x^3-3)$. 
    \begin{enumerate}
        \item Calcular todos los subcuerpos de $K$. 

            Las raíces de $f$ son:
            \begin{equation*}
                i\sqrt{3}, -i\sqrt{3}, \sqrt[3]{3}, w\sqrt[3]{3}, w^2\sqrt[3]{3}
            \end{equation*}
            donde:
            \begin{equation*}
                w = -\frac{1}{2}+i\frac{\sqrt{3}}{2}
            \end{equation*}
            Por lo que $K = \mathbb{Q}\left(i\sqrt{3}, -i\sqrt{3}, \sqrt[3]{3}, w\sqrt[3]{3}, w^2\sqrt[3]{3}\right)$. Sin embargo, veamos que:
            \begin{equation*}
                K = \mathbb{Q}\left(i\sqrt{3}, \sqrt[3]{3}\right)
            \end{equation*}
            \begin{description}
                \item [$\subseteq )$] Vemos que cada una de las raíces de $f$ podemos expresarla como el resultado de una cuenta por operaciones cerradas para cuerpos en función de elementos de $\mathbb{Q}$, $i\sqrt{3}$ y $\sqrt[3]{3}$, por lo que tenemos esta inclusión.
                \item [$\supseteq )$] Es obvia.
            \end{description}
            Por el Lema de la Torre tenemos que:
            \begin{equation*}
                [K:\mathbb{Q}] = \left[\mathbb{Q}\left(i\sqrt{3},\sqrt[3]{3}\right):\mathbb{Q}\right] = \left[\mathbb{Q}\left(i\sqrt{3},\sqrt[3]{3}\right):\mathbb{Q}\left(\sqrt[3]{3}\right)\right]   \left[\mathbb{Q}\left(\sqrt[3]{3}\right):\mathbb{Q}\right]
            \end{equation*}
            donde vemos fácilmente que:
            \begin{itemize}
                \item $\left[\mathbb{Q}\left(\sqrt[3]{3}\right):\mathbb{Q}\right]=3$, ya que $\Irr\left(\sqrt[3]{3},\mathbb{Q}\right) = x^3-3$, al ser un polinomio de grado 3 sin raíces en $\mathbb{Z}$ o bien por Eisenstein para $p=3$.
                \item $ \left[\mathbb{Q}\left(i\sqrt{3},\sqrt[3]{3}\right):\mathbb{Q}\left(\sqrt[3]{3}\right)\right] = 2$, ya que $\Irr\left(i\sqrt{3}:\mathbb{Q}\left(\sqrt[3]{3}\right)\right) = x^2+3$, por ser un polinomio de grado 2 sin raíces en $\mathbb{Q}\left(\sqrt[3]{3}\right)$ (sus raíces son complejas).
            \end{itemize}
            Por lo que $[K:\mathbb{Q}] = 6$. Como $\car(\mathbb{Q}) = 0$ y $K$ es cuerpo de descomposición de $f\in \mathbb{Q}[x]$ tenemos que la extensión $\mathbb{Q}\leq K$ es de Galois, por lo que:
            \begin{equation*}
                |\Aut(K)| = 6
            \end{equation*}
            Por lo que el grupo de Galois de la extensión será isomorfo a $C_6$ o a $D_3$.

            \noindent
            Calculamos los elementos de $\Aut(K)$, aplicando para ello dos veces la proposición de Extensión. Calculamos en primer lugar las extensiones de la inclusión a $\mathbb{Q}\left(\sqrt[3]{3}\right)\stackrel{\eta_j}{\to} K$, que están en correspondencia biyectiva con las raíces de:
            \begin{equation*}
                \Irr\left(\sqrt[3]{3},\mathbb{Q}\right) = x^3-3
            \end{equation*}
            a saber, $\sqrt[3]{3},w\sqrt[3]{3},w^2\sqrt[3]{3}$, por lo que las extensiones obtenidas son las determinadas por:
            \begin{equation*}
                \eta_j\left(\sqrt[3]{3}\right) = w^j \sqrt[3]{3} \qquad j \in \{0,1,2\}
            \end{equation*}
            Análogamente, para cada $\eta_j$ tenemos 2 extensiones suyas en automorfismos $K\to K$, puesto que estas están a su vez en correspondencia biyectiva con las raíces de:
            \begin{equation*}
                \Irr\left(i\sqrt{3},\mathbb{Q}\left(\sqrt[3]{3}\right)\right) = x^2+3
            \end{equation*}
            Que son $\pm i\sqrt{3}$, por lo que obtenemos en total 6 automorfismos, que son los determinados por:
            \begin{align*}
                \eta_{j,k}\left(\sqrt[3]{3}\right) &= w^j \sqrt[3]{3} \\
                \eta_{j,k}\left(i\sqrt{3}\right) &= {(-1)}^{k}i\sqrt{3} \qquad j\in \{0,1,2\}, k\in \{0,1\}
            \end{align*}
            De donde:
            \begin{equation*}
                \Aut(K) = \{\eta_{j,k}:j\in \{0,1,2\}, k\in \{0,1\}\}
            \end{equation*}
            Calculamos ahora los órdenes de cada uno de los elementos:
            \begin{itemize}
                \item Observamos que $\eta_{0,0} = i_K$, por lo que su orden es 1.
                \item Para $\eta_{1,0}$:
                    \begin{align*}
                        \sqrt[3]{3} &\longmapsto w\sqrt[3]{3} \longmapsto w^2\sqrt[3]{3} \longmapsto w^3\sqrt[3]{3}  = \sqrt[3]{3}\\
                        i\sqrt{3}&\longmapsto i\sqrt{3}
                    \end{align*}
                    Donde hemos usado que como $\eta_{1,0}\left(i\sqrt{3}\right) = i\sqrt{3}$ ha de ser por tanto $\eta_{1,0}(w) = w$, ya que:
                    \begin{equation*}
                        w = \frac{-1}{2}+\frac{i\sqrt{3}}{2}
                    \end{equation*}
                    Por lo que el orden de $\eta_{1,0}$ es 3.
                \item Para $\eta_{2,0}$:
                    \begin{align*}
                        \sqrt[3]{3}&\longmapsto w^2\sqrt[3]{3} \longmapsto w^4\sqrt[3]{3} = w\sqrt[3]{3} \longmapsto w^3\sqrt[3]{3} = \sqrt[3]{3} \\
                        i\sqrt{3} &\longmapsto i\sqrt{3}
                    \end{align*}
                    donde hemos vuelto a usar que como $\eta_{2,0}\left(i\sqrt{3}\right) = i\sqrt{3}$ entonces $\eta_{2,0}(w) = w$. En definitiva, el orden es 3.
                \item Para $\eta_{0,1}$:
                    \begin{align*}
                        \sqrt[3]{3}&\longmapsto \sqrt[3]{3} \\
                        i\sqrt{3} &\longmapsto -i\sqrt{3} \longmapsto i\sqrt{3}
                    \end{align*}
                    Por lo que su orden es 2.
                \item Para $\eta_{1,1}$:
                    \begin{align*}
                        \sqrt[3]{3}&\longmapsto w\sqrt[3]{3} = \left(\frac{-1}{2}+\frac{i\sqrt{3}}{2}\right)\sqrt[3]{3} \longmapsto \overline{w}w\sqrt[3]{3} = |w|^2\sqrt[3]{3} = \sqrt[3]{3} \\
                        i\sqrt{3} &\longmapsto -i\sqrt{3} \longmapsto i\sqrt{3}
                    \end{align*}
                    Por lo que su orden es 2.
