    \begin{ejercicio}
        Define el concepto de solución débil para el problema:
        \begin{equation}\label{eq:ec1}
            \begin{cases}
                u_t + (\omega \cdot \nabla_x) = 0 \quad& \omega_i \in L_{loc}^{\infty}(\left[0,T\right[ \times \R^N) \quad 1 \leq i \leq N \\
                u(0,x) = u_0 \quad& u_0 \in L_{loc}^1(\R^N)
            \end{cases}
        \end{equation}
        Partimos de que $\omega_i \in L_{loc}^{\infty}(\left[0,T\right[ \times \R^N) \quad 1 \leq i \leq N$, y $u_0 \in L_{loc}^1(\R^N)$. Así, una solución débil de \eqref{eq:ec1} es una función $u \in L^1_{loc}(\left[0, T \right[ \times \R^N)$ que verifica 
        $$- \int_{0}^{T} \int_{\R^N} \left[ \dfrac{\partial \varphi}{\partial t} + (\omega \cdot \nabla_x) \varphi \right] u \,dx\,dt = \int_{\R^N} u_0(x) \varphi(0,x)\,dx$$ 
        para todo $\varphi \in C_c^1(\left[0,T \right[ \times \R^N)$.
    \end{ejercicio}

    \begin{ejercicio}
        Resuelve el problema de valores iniciales unidimensional 
        \begin{equation}\label{eq:ec2}
            \begin{cases}
                \dfrac{\partial u}{\partial t} - tx \dfrac{\partial u}{\partial x} = 0 \\
                u_0(x) = x^2
            \end{cases}
        \end{equation}
        Sabemos por los dos teoremas vistos en clase correspondientes, que tanto la teoría clásica como la teoría débil nos proporcionan existencia y unicidad de soluciones de \eqref{eq:ec2}, y, en caso de verificarse todas las hipótesis, dicha solución única es
        $$u(t,x) = u_0(X(0;t,x))$$
        Veamos primero las hipótesis para obtener la solución por la vía clásica. En este caso, $\omega(t,x) = -tx$, y $u_0(x) = x^2$. Primero vemos que se verifican (H1)-(H4):
        \begin{itemize}
            \item[(H1)] Se cumple porque $\omega(t,x)$ es polinómica en las variables $t$ y $x$, en particular $\omega$ es continua en los dos argumentos.
            \item[(H2)] $\omega$ es localmente Lipschitz respecto de $x$, ya que $\omega \in C^1(\left[0,T \right[ \times \R)$, luego $C^1_x(\left[0,T \right[ \times \R)$, y sabemos por un resultado visto en Ecuaciones Diferenciales II que es l.l. resp. de $x$.
            Otra forma sería verlo por definición directamente: para cualesquiera $x,y \in \R$ se tiene que
            $$|\omega(t,x) - \omega(t,y)| = |t||x-y| \leq T|x-y|$$
            para cualquier entorno de $(t,x)$ y $(t,y)$ de la forma $I \times \R$, con $I \subseteq \left[0,T \right[$ un intervalo acotado. 
            \item[(H3)] Como $\omega(t,x) = -tx$, en particular, $$|\omega(t,x)| \leq |t||x| + 0 < T|x| + 0 = A|x| + B$$
            para todo $(t,x) \in \left[0, T \right[ \times \R$, por lo que el crecimiento es sublineal.
            \item[(H4)] La aplicación \Func{\dfrac{\partial \omega}{\partial x}}{\left[0, T \right[ \times \R}{\R}{(t,x)}{\dfrac{\partial \omega}{\partial x}(t,x) = -t} es polinómica en $t$, luego continua.
        \end{itemize}
        Como además es claro que $u_0 \in C^1(\R)$, el teorema mencionado al principio nos dice que el PVI \eqref{eq:ec2} admite una única solución, 
        $u \in C^1(\left[0, T \right[ \times \R)$, que viene dada por 
        $$u(t,x) = u_0(X(0;t,x))$$ 
        El sistema característico asociado a \eqref{eq:ec2} es 
        \begin{equation*}
            \begin{cases}
                \dfrac{\partial X}{\partial t}(t;s,x)
                =\omega(t,X(t;s,x)) = -tX(t;s,x)\\[6pt]
                X(s;s,x)=x
            \end{cases}\quad s \in \left[0, T \right[
        \end{equation*}
        La primera ecuación se resuelve por variables separadas (escribimos $X(t) \equiv X(t;s,x)$), suponiendo en todo momento que $X \neq 0$:
        $$\dfrac{dX}{dt} = X' = -tX \Longrightarrow \dfrac{dX}{X} = -t dt \Longrightarrow \ln|X(t)| = - \dfrac{t^2}{2} + C \Longrightarrow X(t) = Ke^{-t^2/2} \quad K \in \R$$
        Vemos que si $K = 0$ recuperamos la solución $X = 0$. Imponiendo la condición inicial $X(s) = x$:
        $$Ke^{-s^2/2} = X(s) = x \Longrightarrow K = xe^{s^2/2}$$
        luego
        $$X(t;s,x) = xe^{s^2/2}e^{-t^2/2} = xe^{\frac{s^2 - t^2}{2}}$$
        y como $X(0;t,x) = xe^{\frac{t^2}{2}}$, la única solución clásica de \eqref{eq:ec2} es
        $$\boxed{u(t,x) = u_0(X(0;t,x)) = \left( xe^{\frac{t^2}{2}} \right)^2 = x^2 e^{t^2}}$$
        Respecto a la vía débil, a raíz de (H4) vemos que no podemos aplicar el teorema correspondiente, ya que 
        $$\di_x(\omega) = \dfrac{\partial \omega}{\partial x}(t,x) = -t \neq 0 \text{ si } t > 0$$
    \end{ejercicio}

    \begin{ejercicio}
        Calcular las ondas de choque y de rarefacción del problema de Riemman asociado a la ecuación de Burgers con dato inicial:
        $$u_0(x) = \begin{cases}
            1 \quad& x<0 \\
            3 \quad& x>0
        \end{cases}$$
        Sabemos por teoría que las soluciones de la ecuación de Burgers (las trayectorias características) son rectas,
        donde la pendiente la determina la condición inicial $u(0,x) = u_0(x)$, o sea, $X(t;0,x) = x + u_0(x)t$. Así pues:
        $$X(t;0,x) = \begin{cases}
            x + t \quad& x < 0 \\
            x + 3t \quad& x > 0
        \end{cases}$$
        Una solución débil del problema de Riemann (que es el problema de Burgers con un dato inicial con salto) que conocemos es la siguiente:
        $$u_1(t,x) = \begin{cases}
            u^{-} \quad& x < \lm t \\
            u^{+} \quad& x > \lm t
        \end{cases}$$
        donde $u^{-} = 1$ y $u^{+} = 3$ se obtienen del dato inicial. Se ha visto en teoría que el único valor de $\lm$ para 
        que $u$ sea solución (débil) es, aplicando la condición de salto de Rankine-Hugoniot:
        $$\lm = \dfrac{1}{2}(u^{-} + u^{+}) = \dfrac{1}{2}(1+3) = 2$$
        por lo que ya tenemos una solución del problema de Riemann: 
        $$\boxed{u_1(t,x) = \begin{cases}
            1 \quad& x < 2t \\
            3 \quad& x > 2t
        \end{cases}}$$
        Como además el dato inicial $u_0(x)$ es creciente, sabemos por teoría que tendremos también ondas de rarefacción. Las características ya
        las hemos obtenido:
        $$X(t;0,x) = \begin{cases}
            x + t \quad& x < 0 \\
            x + 3t \quad& x > 0
        \end{cases}$$
        Obtenemos las rectas que delimitan la región de las ondas de rarefacción haciendo $x \to 0$ en ambos lados, y vemos que:
        $$\lim\limits_{x \to 0^{-}}X(t;0,x) = t$$
        $$\lim\limits_{x \to 0^{+}}X(t;0,x) = 3t$$
        Así, las fronteras de la región de rarefacción son $x = t$ y $x=3t$. Las tres regiones que tenemos son $x < t$, $t \leq x \leq 3t$ y $x > 3t$.
        Según se ha visto en teoría, las ondas de rarefacción pueden construirse como la solución autosemejante o autosimilar $u(t,x) = x/t$,
        y teniendo en cuenta los límites ya obtenidos de la región de rarefacción, una segunda solución (en este caso continua para $t > 0$)
        del problema de este ejercicio es:
        $$\boxed{u_2(t,x) = \begin{cases}
            1 \quad& x < t \\
            \dfrac{x}{t} \quad& t \leq x \leq 3t \quad t > 0\\
            3 \quad& x > 3t \\
        \end{cases}}$$
        En $t=0$ tenemos la condición inicial $u(0,x) = u_0(x)$.
    \end{ejercicio}
