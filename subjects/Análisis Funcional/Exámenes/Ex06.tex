    \begin{ejercicio}[3 puntos]~
        \begin{enumerate}[label=\alph*)]
            \item Enuncia la caracterización de las aplicaciones lineales y continuas entre espacios normados.
            
            Sean $X$ y $Y$ espacios normados y sea $T:X\to Y$ una aplicación lineal. Entonces, $T$ es continua si y solo si existe una constante $C>0$ tal que
            \begin{equation*}
                \|T(x)\|\leq C\|x\| \quad \forall x\in X.
            \end{equation*}
            \item Definición y caracterización de los isomorfismos topológicos entre espacios normados.
            
            Un isomorfismo topológico entre espacios normados $X$ y $Y$ es una aplicación lineal biyectiva $T:X\to Y$ tal que tanto $T$ como su inversa $T^{-1}:Y\to X$ son continuas. En otras palabras, una aplicación lineal $T$ es un isomorfismo topológico si y solo si es biyectiva y existen constantes $C_1, C_2 > 0$ tales que
            \begin{equation*}
                C_1\|x\| \leq \|T(x)\| \leq C_2\|x\| \quad \forall x\in X.
            \end{equation*}
            \item Enuncia el Teorema de Riesz.
            
            Si $X$ es un espacio normado y la esfera unidad:
            \begin{equation*}
                S = \{x\in X : \|x\|=1\}
            \end{equation*}
            es compacta, entonces $X$ es de dimensión finita.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}[4 puntos]
        Sea $X=C([0,1],\mathbb{R})$. Se dfine $T:X\to X$ mediante
        \begin{equation*}
            T(f)(x) = \int_{0}^{x^3} f(t)~dt  \qquad \forall x\in [0,1]
        \end{equation*}
        \begin{enumerate}[label=\alph*)]
            \item Prueba que $T$ es lineal, continua e inyectiva.
            
            En primer lugar, notemos que $T$ está bien definida, ya que para cada $f \in X$, por el Teorema Fundamental del Cálculo, su primitiva es continua. Como $T(f)$ es la composición de una primitiva de $f$ con la función continua $x \mapsto x^3$, se deduce que $T(f)$ es continua en $[0,1]$. Por lo tanto, $T(f) \in X$ para todo $f \in X$. Veamos que $T$ es lineal. Sean $f,g \in X$ y $\alpha \in \mathbb{R}$. Entonces:
            \begin{align*}
                T(\alpha f + g)(x) &= \int_{0}^{x^3} (\alpha f + g)(t)~dt 
                = \int_{0}^{x^3} (\alpha f(t) + g(t))~dt \\
                                    &= \alpha \int_{0}^{x^3} f(t)~dt + \int_{0}^{x^3} g(t)~dt \\
                                    &= \alpha T(f)(x) + T(g)(x)
            \end{align*}
            Por lo tanto, $T$ es lineal. Veamos que $T$ es continua. Para ello, consideremos la norma de $T(f)$:
            \begin{align*}
                \|T(f)\| &= \sup_{x\in [0,1]} |T(f)(x)| 
                = \sup_{x\in [0,1]} \left|\int_{0}^{x^3} f(t)~dt\right| 
                \leq \sup_{x\in [0,1]} \int_{0}^{x^3} |f(t)|~dt \\
                          &\leq \sup_{x\in [0,1]} \int_{0}^{x^3} \|f\|~dt 
                          = \sup_{x\in [0,1]} \|f\| \cdot x^3 
                          = \|f\| \cdot \sup_{x\in [0,1]} x^3 
                          = \|f\|
            \end{align*}
            Por lo tanto, $\|T(f)\| \leq \|f\|$, lo que implica que $T$ es continua. Ahora, veamos que $T$ es inyectiva. Supongamos que $T(f) = 0$. Entonces, para todo $x \in [0,1]$, tenemos:
            \begin{equation*}
                \int_{0}^{x^3} f(t)~dt = 0
            \end{equation*}
            Por el Teorema Fundamental del Cálculo, derivando ambos lados con respecto a $x$, obtenemos:
            \begin{equation*}
                f(x^3) \cdot 3x^2 = 0 \quad \forall x \in [0,1]
            \end{equation*}

            Para $x \in \left]0,1\right]$, tenemos $3x^2 \neq 0$, por lo que $f(x^3) = 0$. Como $x^3$ recorre el intervalo $[0,1]$ cuando $x$ recorre $[0,1]$, concluimos que:
            \begin{equation*}
                f(z) = 0 \quad \forall z \in \left]0,1\right]
            \end{equation*}
            Además, como $f$ es continua en $[0,1]$, tenemos que
            \begin{equation*}
                f(0) = \lim_{z \to 0^+} f(z) = 0
            \end{equation*}
            Por lo tanto, $f \equiv 0$, lo que demuestra que $T$ es inyectiva.
            \item Calcula la norma de $T$.
            
            Para calcular la norma de $T$, partimos de que $\|T\|\leq 1$ como se demostró en el apartado anterior. Ahora, consideremos la función constante $f(x) = 1$ para todo $x \in [0,1]$. Entonces, tenemos $\|f\| = 1$ y:
            \begin{equation*}
                \|T(f)\| = \sup_{x\in [0,1]} \left|\int_{0}^{x^3} 1~dt\right| = \sup_{x\in [0,1]} x^3 = 1
            \end{equation*}
            Por lo tanto, $\|T\| \geq 1$. Combinando esto con la desigualdad anterior, obtenemos que $\|T\| = 1$. Además, hemos demostrado que la norma de $T$ es alcanzada por la función constante $f(x) = 1$.
            \item Prueba que $T$ no es un isomorfismo topológico de $X$ sobre $T(X)$.
            
            Tenemos que $T$ es continua y, puesto que es inyectiva, es una biyección de $X$ sobre $T(X)$. Sin embargo, para que $T$ sea un isomorfismo topológico, necesitamos que $T^{-1}:T(X)\to X$ sea continua. Consideramos la sucesión de funciones $\{f_n\}$ en $X$ definida por:
            \begin{equation*}
                f_n(x) = x^n \quad \forall x \in [0,1]
            \end{equation*}
            Entonces, $\|f_n\| = 1$ para todo $n\in \bb{N}$. Ahora, consideremos $T(f_n)$:
            \begin{equation*}
                T(f_n)(x) = \int_{0}^{x^3} t^n~dt = \frac{x^{3(n+1)}}{n+1}
            \end{equation*}
            Por lo tanto:
            \begin{equation*}
                \|T(f_n)\| = \sup_{x\in [0,1]} \left|\frac{x^{3(n+1)}}{n+1}\right| = \frac{1}{n+1}
            \end{equation*}

            Recordamos que $T^{-1}$ es continua si y solo si existe una constante $C>0$ tal que:
            \begin{equation*}
                \|T^{-1}(g)\| \leq C\|g\| \quad \forall g \in T(X)
            \end{equation*}

            Empleando la biyectividad de $T$, tomando $g=T(f_n)$, obtenemos:
            \begin{equation*}
                1 = \|f_n\| = \|T^{-1}(T(f_n))\| \leq C\|T(f_n)\| = C\cdot \frac{1}{n+1}
            \end{equation*}

            Tomando el límite cuando $n\to \infty$, obtenemos:
            \begin{equation*}
                1 \leq C \cdot 0 = 0
            \end{equation*}
            lo cual es una contradicción. Por lo tanto, $T^{-1}$ no es continua, y concluimos que $T$ no es un isomorfismo topológico de $X$ sobre $T(X)$.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}[3 puntos]
        Sea $\{P_n\}_{n\in \mathbb{N}}$ una sucesión de polinomios de grado menor o igual que 20 tal que, para cada $k \in \{0,1,\ldots,20\}$ la sucesión $\{P_n^{k)}(0)\}_{n\in \mathbb{N}}$ es convergente. Prueba que $\{P_n\}_{n\in \mathbb{N}}$ converge uniformemente en $[-1,1]$.\\

        Consideremos el subespacio vectorial $X$ de polinomios de grado menor o igual que 20 definidos en $[-1,1]$, subespacio de $C[-1,1]$. Tenemos que:
        \begin{equation*}
            X=\cc{L}\{1,x,x^2,\ldots,x^{20}\}\Longrightarrow \dim(X)=21
        \end{equation*}
        Por tanto, $X$ es un espacio normado de dimensión finita. Tenemos que:
        \begin{itemize}
            \item $\left(X,\|\cdot\|_\infty\right)$ es un espacio normado.
            \item Veamos que, dado $P\in X$, se puede definir la norma
            \begin{equation*}
                \|P\|_* = \max_{k\in\{0,1,\ldots,20\}} |P^{(k)}(0)|
            \end{equation*}
            \begin{itemize}
                \item Si $\|P\|_*=0$, entonces:
                \begin{equation*}
                    |P^{(k)}(0)|=0 \Longrightarrow P^{(k)}(0)=0 \quad \forall k\in\{0,1,\ldots,20\}
                \end{equation*}
                Por tanto, por la fórmula de Taylor:
                \begin{equation*}
                    P(x) = \sum_{k=0}^{20} \frac{P^{(k)}(0)}{k!}x^k = 0\qquad \forall x\in[-1,1]
                \end{equation*}

                \item Para todo $\alpha\in\mathbb{R}$ y para todo $P\in X$ se tiene:
                \begin{align*}
                    \|\alpha P\|_* &= \max_{k\in\{0,1,\ldots,20\}} |(\alpha P)^{(k)}(0)| = \max_{k\in\{0,1,\ldots,20\}} |\alpha P^{(k)}(0)| =\\&= |\alpha| \max_{k\in\{0,1,\ldots,20\}} |P^{(k)}(0)| = |\alpha| \|P\|_*
                \end{align*}

                \item Para todo $P,Q\in X$ se tiene:
                \begin{align*}
                    \|P+Q\|_* &= \max_{k\in\{0,1,\ldots,20\}} |(P+Q)^{(k)}(0)| = \max_{k\in\{0,1,\ldots,20\}} |P^{(k)}(0) + Q^{(k)}(0)| \leq \\ &\leq \max_{k\in\{0,1,\ldots,20\}} |P^{(k)}(0)| + |Q^{(k)}(0)| \leq \\&\leq \max_{k\in\{0,1,\ldots,20\}} |P^{(k)}(0)| + \max_{k\in\{0,1,\ldots,20\}} |Q^{(k)}(0)| = \|P\|_* + \|Q\|_*
                \end{align*}
            \end{itemize}

            Por tanto, $\left(X,\|\cdot\|_*\right)$ es un espacio normado.
        \end{itemize}

        Por tanto, por el Corolario del Teorema de Hausdorff, las normas $\|\cdot\|_\infty$ y $\|\cdot\|_*$ son equivalentes.
        \begin{description}
            \item[Opción 1: Ver que $\{P_n\}$ es de Cauchy] 
            
            Buscamos en primer lugar ver que $\{P_n\}_{n\in \mathbb{N}}$ es una sucesión de Cauchy en $\left(X,\|\cdot\|_*\right)$. Dado $\varepsilon>0$, como para cada valor $k\in\{0,1,\ldots,20\}$ la sucesión $\{P_n^{(k)}(0)\}_{n\in \mathbb{N}}$ es convergente, en particular es de Cauchy, luego existe $N_k\in \mathbb{N}$ tal que:
            \begin{equation*}
                |P_n^{(k)}(0)-P_m^{(k)}(0)|<\varepsilon \quad \forall n,m\ge N_k
            \end{equation*}
            Tomando $N=\max\{N_0,N_1,\ldots,N_{20}\}$, tenemos que para todo $n,m\ge N$ y para todo $k\in\{0,1,\ldots,20\}$ se cumple
            \begin{equation*}
                |P_n^{(k)}(0)-P_m^{(k)}(0)|<\varepsilon
            \end{equation*}
            Por tanto, para todo $n,m\ge N$ se tiene:
            \begin{equation*}
                \|P_n-P_m\|_* = \max_{k\in\{0,1,\ldots,20\}} |P_n^{(k)}(0)-P_m^{(k)}(0)|<\varepsilon
            \end{equation*}
            Por tanto, $\{P_n\}_{n\in \mathbb{N}}$ es una sucesión de Cauchy en $\left(X,\|\cdot\|_*\right)$. Como $\left(X,\|\cdot\|_*\right)$ es un espacio normado de dimensión finita, es completo, por lo que $\{P_n\}_{n\in \mathbb{N}}$ converge en $\left(X,\|\cdot|_*\right)$ a un polinomio $P\in X$. Por tanto, puesto que ambas normas son equivalentes, $\{P_n\}_{n\in \mathbb{N}}$ converge a $P$ en $\left(X,\|\cdot\|_\infty\right)$, es decir,
            \begin{equation*}
                \lim_{n\to\infty} \|P_n-P\|_\infty = 0
            \end{equation*}

            \item[Opción 2: Ver que $\{P_n\}$ converge]
            
            Fijado $k\in\{0,1,\ldots,20\}$, como $\{P_n^{(k)}(0)\}_{n\in \mathbb{N}}$ es convergente, existe $c_k\in \mathbb{R}$ tal que:
            \begin{equation*}
                \lim_{n\to\infty} P_n^{(k)}(0) = c_k
            \end{equation*}

            Consideremos el polinomio $P\in X$ definido por:
            \begin{equation*}
                P(x) = \sum_{k=0}^{20} \frac{c_k}{k!} x^k
            \end{equation*}

            Por tanto, para cada $k\in\{0,1,\ldots,20\}$ se cumple:
            \begin{equation*}
                P^{(k)}(0) = c_k = \lim_{n\to\infty} P_n^{(k)}(0)
            \end{equation*}

            Por tanto, calculemos el polinomio diferencia $P_n-P$. Para todo $x\in[-1,1]$ se tiene:
            \begin{align*}
                (P_n-P)(x) &= \sum_{k=0}^{20} \frac{P_n^{(k)}(0)-c_k}{k!} x^k
            \end{align*}
            
            Veamos que $\{P_n\}_{n\in \mathbb{N}}$ converge a $P$ en $\left(X,\|\cdot\|_*\right)$. Como para cada $k\in\{0,1,\ldots,20\}$ se cumple que $\{P_n^{(k)}(0)\}_{n\in \mathbb{N}}$ converge a $c_k$, tenemos que:
            \begin{equation*}
                \lim_{n\to\infty} |P_n^{(k)}(0)-c_k| = 0 \quad \forall k\in\{0,1,\ldots,20\}
            \end{equation*}

            Por tanto, tomando el máximo sobre $k\in\{0,1,\ldots,20\}$, tenemos que:
            \begin{equation*}
                \lim_{n\to\infty} \|P_n-P\|_* = \lim_{n\to\infty} \max_{k\in\{0,1,\ldots,20\}} |P_n^{(k)}(0)-c_k| = 0
            \end{equation*}

            Por tanto, $\{P_n\}_{n\in \mathbb{N}}$ converge a $P$ en $\left(X,\|\cdot\|_*\right)$. Como ambas normas son equivalentes, $\{P_n\}_{n\in \mathbb{N}}$ converge a $P$ en $\left(X,\|\cdot\|_\infty\right)$, es decir,
            \begin{equation*}
                \lim_{n\to\infty} \|P_n-P\|_\infty = 0
            \end{equation*}
        \end{description}

        Como la convergencia en $\left(C[-1,1],\|\cdot\|_\infty\right)$ es equivalente a la convergencia uniforme, tenemos que $\{P_n\}_{n\in \mathbb{N}}$ converge uniformemente a $P$ en $[-1,1]$. Veamos este aspecto con más detalle. Dado $\varepsilon>0$, como $\left\{\|P_n-P\|_\infty\right\} \to 0$, existe $N\in \mathbb{N}$ tal que para todo $n\ge N$ se cumple:
        \begin{equation*}
            \|P_n-P\|_\infty < \varepsilon
        \end{equation*}
        
        Fijado $n\ge N$, tenemos que para todo $x\in[-1,1]$ se cumple:
        \begin{equation*}
            |P_n(x)-P(x)|\le \|P_n-P\|_\infty=\max_{x\in[-1,1]} |P_n(x)-P(x)|<\varepsilon
        \end{equation*}

        Por tanto, para cada $\varepsilon>0$ existe $N\in \mathbb{N}$ tal que para todo $n\ge N$ y para todo $x\in[-1,1]$ se cumple:
        \begin{equation*}
            |P_n(x)-P(x)|<\varepsilon
        \end{equation*}
        Por tanto, $\{P_n\}_{n\in \mathbb{N}}$ converge uniformemente a $P$ en $[-1,1]$.
    \end{ejercicio}
