    \begin{ejercicio}
        En $C[0, 1]$ definimos:
        \begin{equation*}
            \varphi(f) = \int_0^{\nicefrac{1}{2}} f(t)\,dt - \int_{\nicefrac{1}{2}}^1 f(t)\,dt\qquad \forall f \in C[0, 1]
        \end{equation*}

        Probar que $\varphi$ es un funcional lineal continuo y calcular su norma.\\

        Veamos en primer lugar que es lineal. Sean $f, g \in C[0, 1]$ y $\alpha, \beta \in \bb{K}$. Entonces:
        \begin{align*}
            \varphi(\alpha f + \beta g) &= \int_0^{\nicefrac{1}{2}} (\alpha f(t) + \beta g(t))\,dt - \int_{\nicefrac{1}{2}}^1 (\alpha f(t) + \beta g(t))\,dt =\\&= \alpha \int_0^{\nicefrac{1}{2}} f(t)\,dt + \beta \int_0^{\nicefrac{1}{2}} g(t)\,dt - \alpha \int_{\nicefrac{1}{2}}^1 f(t)\,dt - \beta \int_{\nicefrac{1}{2}}^1 g(t)\,dt = \alpha \varphi(f) + \beta \varphi(g)
        \end{align*}
        Por tanto, $\varphi$ es lineal. Veamos ahora que es continuo. Para ello, por comodidad, definimos:
        \Func{h}{[0, 1]}{\bb{K}}{t}{\begin{cases}
            1 & \text{si } t \in [0, \nicefrac{1}{2}]\\
            -1 & \text{si } t \in [\nicefrac{1}{2}, 1]
        \end{cases}}

        De esta forma, para todo $f \in C[0, 1]$, se tiene que:
        \begin{align*}
            \varphi(f) &= \int_0^{\nicefrac{1}{2}} f(t)\,dt - \int_{\nicefrac{1}{2}}^1 f(t)\,dt = \int_0^{\nicefrac{1}{2}} f(t)\,dt + \int_{\nicefrac{1}{2}}^1 -f(t)\,dt = \int_0^1 f(t) h(t)\,dt
        \end{align*}
        Veamos ahora entonces que $\varphi$ es continuo. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $|\varphi(f)| \leq M \|f\|_\infty$ para todo $f \in C[0, 1]$. En efecto, para todo $f \in C[0, 1]$, se tiene que:
        \begin{align*}
            |\varphi(f)| &= \left|\int_0^1 f(t) h(t)\,dt\right| \leq \int_0^1 |f(t) h(t)|\,dt = \int_0^1 |f(t)| \, dt \leq \int_0^1 \|f\|_\infty\,dt = \|f\|_\infty
        \end{align*}
        Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq 1$. Veamos ahora que $\|\varphi\| = 1$. 
        Por comodidad, para cada $n \in \bb{N}$, $n\geq 3$, notamos por $I_n$ a:
        \begin{equation*}
            I_n = \left[\frac{1}{2} - \frac{1}{n}, \frac{1}{2} + \frac{1}{n}\right]
        \end{equation*}
        
        Consideramos la función $f_n \in C[0, 1]$ definida por:
        \[f_n(t) = \begin{cases}
            1 & \text{si } t \in [0, \nicefrac{1}{2} - \nicefrac{1}{n}]\\
            \text{lineal} & \text{si } t \in I_n\\
            -1 & \text{si } t \in [\nicefrac{1}{2} + \nicefrac{1}{n}, 1]
        \end{cases}\]

        Notemos que no es necesario definir la función lineal en $I_n$, ya que al imponer que sea continua se puede deducir que es:
        \begin{equation*}
            f_n(t) = n\left(\frac{1}{2} - t\right) \quad \forall t \in I_n
        \end{equation*}

        Estas funciones se muestran en la Figura~\ref{fig:fn2}.
        \begin{figure}
            \centering
            \begin{tikzpicture}
                \begin{axis}[
                    axis lines = middle,

                    ymin = -1.3, ymax = 1.3,

                    }
                    \temp
                }

                \end{axis}
            \end{tikzpicture}
            \caption{Gráfica de la función $f_n$ para distintos valores de $n$.}
            \label{fig:fn2}
        \end{figure}

        Es directo que $f_n \in C[0, 1]$. Calculemos su norma:
        \begin{equation*}
            \|f_n\|_\infty = \max_{t \in [0, 1]} |f_n(t)| = 1
        \end{equation*}


        Calculemos ahora $|\varphi(f_n)|$:
        \begin{align*}
            |\varphi(f_n)| &= \left|\int_0^1 f_n(t) h(t)\,dt\right| = \left|\int_{[0, 1] \setminus I_n} f_n(t) h(t)\,dt + \int_{I_n} f_n(t) h(t)\,dt\right| =\\&= \left|\int_{[0, 1] \setminus I_n} 1 \,dt + \int_{I_n} f_n(t) h(t)\,dt\right| \geq 1-\lm(I_n)-\left|\int_{I_n} f_n(t) h(t)\,dt\right|\geq 1-2\lm(I_n) = 1 - \frac{4}{n}
        \end{align*}

        Por tanto, tenemos que:
        \begin{equation*}
            \|\varphi\|\geq 1-\frac{4}{n} \quad \forall n \in \bb{N}, n\geq 3
        \end{equation*}
        Tomando límite cuando $n \to \infty$, se tiene que:
        \begin{equation*}
            \|\varphi\| \geq 1
        \end{equation*}
        Por tanto, $\|\varphi\| = 1$.
    \end{ejercicio}


    \begin{ejercicio}[5 puntos]
        Probar el Teorema de Hausdorff de 1932 (junto con su lema previo): \textit{toda biyección lineal entre espacios normados de dimensión finita es un isomorfismo.}

        \begin{proof}
            Sea $T:X\to Y$ una biyección lineal entre espacios normados de dimensión finita. Como $\dim X=n\in \bb{N}$ finita, consideramos la biyección lineal de sus coordenadas; es decir:
            \Func{\Phi}{\bb{K}^n}{X}{(\alpha_1, \dots, \alpha_n)}{\sum_{i=1}^n \alpha_i x_i}
            donde $\{x_1, \dots, x_n\}$ es una base de $X$. De esta manera, $T\circ \Phi: \bb{K}^n \longrightarrow Y$ es una biyección lineal, y:
            \begin{equation*}
                T = (T\circ \Phi) \circ \Phi^{-1}
            \end{equation*}

            Demostramos ahora dos lemas previos que nos permitirán concluir que $T$ es un isomorfismo.
            \begin{lema}
                Fijado $n\in \bb{N}$, todo operador lineal $S:\bb{K}^n \longrightarrow Y$ con $Y$ un espacio normado es continuo.
            \end{lema}
            \begin{proof}
                Sea $S:\bb{K}^n \longrightarrow Y$ un operador lineal. Sea $\{e_1, \dots, e_n\}$ la base canónica de $\bb{K}^n$. Entonces, para todo $x\in \bb{K}^n$, se tiene que:
                \begin{equation*}
                    S(x) = S\left(\sum_{i=1}^n x(i) e_i\right) = \sum_{i=1}^n x(i) S(e_i)
                \end{equation*}

                Fijado $i\in \{1, \dots, n\}$, se tiene que la aplicación $x \longmapsto x(i)$ es continua. Como $S(e_i)$ es un vector fijo, se tiene que $\lm \mapsto \lm S(e_i)$ es continua por la continuidad de la multiplicación por un escalar. Por tanto, $x \longmapsto x(i) S(e_i)$ es continua. Como suma de funciones continuas, se tiene que $S$ es continua.
            \end{proof}
            \begin{lema}
                Fijado $n\in \bb{N}$, toda biyección lineal $S:\bb{K}^n \longrightarrow Y$ con $Y$ un espacio normado es un isomorfismo.
            \end{lema}
            \begin{proof}
                Sea $S:\bb{K}^n \longrightarrow Y$ una biyección lineal. Por el lema previo, $S$ es continua. Como $S$ es una biyección lineal, $S^{-1}:Y \longrightarrow \bb{K}^n$ es lineal, pero hemos de probar que es continua.\\

                Consideramos la esfera $A=\{x \in \bb{K}^n : \|x\| = 1\}$ para una norma que genere la topología usual en $\bb{K}^n$. Como $S$ es continua y $A$ es compacto, se tiene que $S(A)$ es compacto, luego la norma de $Y$ (que es continua) alcanza su mínimo en $S(A)$; es decir, $\exists y_0 \in S(A)$ tal que:
                \begin{equation*}
                    \|y_0\| = \min_{y \in S(A)} \|y\|
                \end{equation*}

                Veamos ahora que $S^{-1}$ es continua. Para cada $x\in \bb{K}^n$, lo escribimos como $x = \|x\|u$ con $u \in A$. Entonces:
                \begin{equation*}
                    \|S(x)\| = \|S(\|x\|u)\| = \|x\|\|S(u)\| \geq \|x\|\|y_0\|
                \end{equation*}
                
                Escribiendo $x=S^{-1}(y)$, se tiene que:
                \begin{equation*}
                    \|y\| \geq \|S^{-1}(y)\|\|y_0\| \implies \|S^{-1}(y)\| \leq \frac{1}{\|y_0\|}\|y\|
                \end{equation*}
                Por tanto, $S^{-1}$ es continua y $S$ es un isomorfismo.
            \end{proof}

            Por tanto, $T\circ \Phi: \bb{K}^n \longrightarrow Y$ y $\Phi: \bb{K}^n \longrightarrow X$ son isomorfismos, y como composición de isomorfismos, $T$ es un isomorfismo.
        \end{proof}
    \end{ejercicio}

    \begin{ejercicio}
        Justificar si el funcional $\varphi$ del Ejercicio 1 alcanza o no su norma.\\

        Supongamos que existe $f \in C[0, 1]$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = \|\varphi\| = 1$.    
        \begin{description}
            \item[Opción 1.] Entonces:
            \begin{equation*}
                1 = |\varphi(f)| = \left|\int_0^1 f(t) h(t)\,dt\right| \leq \int_0^1 |f(t) h(t)|\,dt = \int_0^1 |f(t)|\,dt \leq \int_0^1 \|f\|_\infty\,dt = 1
            \end{equation*}
            
            Por tanto, todas las desigualdades se convierten en igualdades.
            \begin{itemize}
                \item De la segunda desigualdad, se tiene:
                \begin{equation*}
                    \int_0^1 |f(t)|\,dt = \int_0^1 \|f\|_\infty\,dt \Longrightarrow
                    \int_0^1 \|f\|_\infty - |f(t)|\,dt = 0
                \end{equation*}

                Como el integrando es continuo y no negativo, se tiene que $\|f\|_\infty - |f(t)| = 0$ para todo $t \in [0, 1]$. Por tanto, $|f(t)| = \|f\|_\infty = 1$ para todo $t \in [0, 1]$.
                \item De la primera desigualdad, se tiene:
                \begin{equation*}
                    \left|\int_0^1 f(t) h(t)\,dt\right| = \int_0^1 |f(t) h(t)|\,dt
                \end{equation*}

                Consideramos la función signo definida en $\bb{C}$:
                \Func{\operatorname{sgn}}{\bb{C}}{\bb{C}}{z}{\begin{cases}
                    \dfrac{z}{|z|} & \text{si } z \neq 0\\
                    0 & \text{si } z = 0
                \end{cases}}
                
                Llamemos al valor de la integral:
                \begin{equation*}
                    w = \int_0^1 f(t) h(t)\,dt \in \mathbb{K}\setminus \{0\}
                \end{equation*}
                Como por hipótesis $|\varphi(f)| = |w| = 1 \neq 0$, definimos el escalar complejo unitario:
                \begin{equation*}
                    \theta = \operatorname{sgn}(w) = \frac{w}{|w|} = w \implies |\theta| = 1
                \end{equation*}
                Por las propiedades del conjugado complejo en la esfera unidad ($\theta \overline{\theta} = |\theta|^2 = 1$), se cumple que $\overline{\theta} w = \overline{\theta} \theta = 1 = |w|$. Multiplicando la integral por $\overline{\theta}$:
                \begin{equation*}
                    1 = |w| = \overline{\theta} \int_0^1 f(t) h(t)\,dt = \int_0^1 \overline{\theta} f(t) h(t)\,dt
                \end{equation*}
                Como el resultado es el número real $1$, su valor coincide con su parte real:
                \begin{equation*}
                    1 = \int_0^1 \Re\left( \overline{\theta} f(t) h(t) \right) dt
                \end{equation*}

                Por otra parte, usando que $|\overline{\theta}| = 1$, $|h(t)| = 1$ y que previamente probamos $|f(t)| = 1$ en todo $[0, 1]$, tenemos:
                \begin{equation*}
                    \int_0^1 \left| \overline{\theta} f(t) h(t) \right| dt = \int_0^1 1\,dt = 1
                \end{equation*}

                Restando ambas igualdades:
                \begin{equation*}
                    \int_0^1 \left( \left| \overline{\theta} f(t) h(t) \right| - \Re\left( \overline{\theta} f(t) h(t) \right) \right) dt = 1 - 1 = 0
                \end{equation*}

                Dado que para cualquier número complejo $z \in \mathbb{C}$ siempre se verifica $|z| \geq \Re(z)$, el integrando es una función no negativa casi por todas partes. Al tener integral nula, debe anularse casi por todas partes:
                \begin{equation*}
                    \Re\left( \overline{\theta} f(t) h(t) \right) = \left| \overline{\theta} f(t) h(t) \right| = 1 \quad \text{c.p.d.}
                \end{equation*}

                La igualdad entre la parte real y el módulo de un número complejo exige que su parte imaginaria sea nula y su parte real no negativa, lo que implica directamente:
                \begin{equation*}
                    \overline{\theta} f(t) h(t) = 1 \quad \text{c.p.d.}
                \end{equation*}

                Multiplicando a ambos lados por $\theta$ (ya que $\theta \overline{\theta} = 1$) y por $h(t)$ (ya que $h(t)^2 = 1$ en todo punto):
                \begin{equation*}
                    f(t) = \theta h(t) \quad \text{c.p.d.}
                \end{equation*}
            \end{itemize}

            Atendiendo a la definición de $h(t)$, esto significa que:
            \begin{itemize}
                \item Para todo $t \in [0, \nicefrac{1}{2}[$, se tiene $f(t) = \theta$.
                \item Para todo $t \in ]\nicefrac{1}{2}, 1]$, se tiene $f(t) = -\theta$.
            \end{itemize}

            Calculando los límites laterales al aproximarnos a $t = \nicefrac{1}{2}$:
            \begin{equation*}
                \lim_{t \to (\nicefrac{1}{2})^-} f(t) = \theta \quad \neq \quad -\theta = \lim_{t \to (\nicefrac{1}{2})^+} f(t)
            \end{equation*}
            donde se ha empleado que, puesto que $|\theta|=1$, entonces $\theta\neq 0$. Esto contradice que $f$ sea continua en $[0, 1]$. Por tanto, no existe ninguna función $f \in C[0, 1]$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = 1$, concluyendo que el funcional $\varphi$ no alcanza su norma.

            \item[Opción 2.] Multiplicando por el escalar $\lm = \frac{1}{\varphi(f)}$ (que cumple que $|\lm| = \frac{1}{|\varphi(f)|} = 1$), la función $g=\lm f$ cumple que $\|g\|_\infty = \|f\|_\infty = 1$ y $\varphi(g) = \lm \varphi(f) = 1$. Por tanto, sin pérdida de generalidad, podemos suponer que $\varphi(f) = 1$. Entonces, tenemos que:
            \begin{equation*}
                \varphi(f) = \int_0^{\nicefrac{1}{2}} f(t)\,dt - \int_{\nicefrac{1}{2}}^1 f(t)\,dt = 1
            \end{equation*}

            Tomando parte real, se tiene que:
            \begin{equation*}
                \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt - \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt = 1
            \end{equation*}

            Como $f$ es continua, $\Re(f): [0, 1] \longrightarrow \bb{R}$ es continua. Además, como $\|f\|_\infty = 1$, se tiene que $|\Re(f(t))| \leq |f(t)| \leq 1$ para todo $t \in [0, 1]$. Por tanto, tenemos que:
            \begin{equation*}
                -1 \leq \Re(f(t)) \leq 1 \quad \forall t \in [0, 1]
            \end{equation*}

            Consideramos ahora $\Re(f)(\nicefrac{1}{2})\in [-1, 1]$.
            \begin{itemize}
                \item Si $\Re(f)(\nicefrac{1}{2}) < 1$, entonces por continuidad $\exists \delta \in \bb{R}^+$ tal que $\Re(f)(t) < 1$ para todo $t \in [\nicefrac{1}{2} - \delta, \nicefrac{1}{2}]$ (conjunto de medida no nula). Por tanto, tenemos que:
                \begin{equation*}
                    \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt < \frac{1}{2}
                \end{equation*}

                Por otro lado, tenemos que:
                \begin{equation*}
                    \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt \geq -\frac{1}{2}
                \end{equation*}

                Uniendo ambas desigualdades, se tiene que:
                \begin{equation*}
                    1 = \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt - \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt < \frac{1}{2} - \left(-\frac{1}{2}\right) = 1
                \end{equation*}

                Por tanto, $\Re(f)(\nicefrac{1}{2}) < 1$ no es posible.
                \item Si $\Re(f)(\nicefrac{1}{2}) > -1$, entonces por continuidad $\exists \delta \in \bb{R}^+$ tal que $\Re(f)(t) > -1$ para todo $t \in [\nicefrac{1}{2}, \nicefrac{1}{2} + \delta]$. Por tanto, tenemos que:
                \begin{equation*}
                    \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt > -\frac{1}{2}
                \end{equation*}

                Por otro lado, tenemos que:
                \begin{equation*}
                    \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt \leq \frac{1}{2}
                \end{equation*}

                Uniendo ambas desigualdades, se tiene que:
                \begin{equation*}
                    1 = \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt - \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt < \frac{1}{2} - \left(-\frac{1}{2}\right) = 1
                \end{equation*}

                Por tanto, $\Re(f)(\nicefrac{1}{2}) > -1$ no es posible.
            \end{itemize}

            Por tanto, hemos llegado a que $\Re(f)(\nicefrac{1}{2})\geq 1$ y $\Re(f)(\nicefrac{1}{2})\leq -1$, lo cual es imposible. Por tanto, no existe $f \in C[0, 1]$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = \|\varphi\| = 1$. En otras palabras, $\varphi$ no alcanza su norma.
        \end{description}
    \end{ejercicio}


    \begin{ejercicio}
        Sea $X$ un espacio normado y $T:X\to Y$ un operador lineal y continuo. Demostrar que si $\dim(T(X)) < \infty$ y $\ker T$ es proximal en $X$, entonces $T$ alcanza su norma.

        \begin{proof}
            Si $T = 0$, su norma es $\|T\| = 0$ y se alcanza trivialmente en cualquier elemento de la esfera unidad de $X$. Supongamos en adelante que $T \neq 0$, lo que garantiza que $\|T\| > 0$. Nuestro objetivo consiste en encontrar un vector $u \in X$ con $\|u\| = 1$ tal que:
            \begin{equation*}
                \|T(u)\| = \|T\| = \sup_{\|x\| = 1} \|T(x)\| .
            \end{equation*}

            Para que una función continua alcance un máximo, la herramienta estándar es el Teorema de Weierstrass, el cual requiere compacidad. Sin embargo, si $X$ es de dimensión infinita, su esfera unidad $\{x \in X : \|x\| = 1\}$ nunca es compacta (por el Teorema de Riesz), de modo que no podemos buscar el vector $u$ directamente por optimización sobre dicha esfera.\\

            Para superar este obstáculo, observemos primero qué información resulta superflua para el operador $T$. Si tomamos cualquier vector $x \in X$ y le sumamos o restamos cualquier vector $y \in \ker T$, la linealidad de $T$ nos muestra que el núcleo no altera la imagen:
            \begin{equation*}
                T(x - y) = T(x) - T(y) = T(x) - 0 = T(x) .
            \end{equation*}
            Por tanto, aplicando la continuidad y acotación de $T$ a este vector depurado:
            \begin{equation*}
                \|T(x)\| = \|T(x - y)\| \leq \|T\| \|x - y\| \quad \forall y \in \ker T .
            \end{equation*}
            Como esta desigualdad es válida para todo $y \in \ker T$, sigue siéndolo si tomamos el ínfimo sobre todos los elementos del núcleo:
            \begin{equation*}
                \|T(x)\| \leq \|T\| \inf_{y \in \ker T} \|x - y\| = \|T\| \, d(x, \ker T) \quad \forall x \in X .
            \end{equation*}
            Esta cota clave nos revela que la imagen $\|T(x)\|$ no está condicionada por la norma propia de $x$, sino por su distancia mínima al subespacio $\ker T$, que es precisamente la norma del cociente $X/\ker T$. Esto justifica de forma natural trasladar el problema a dicho espacio, cuyo propósito es precisamente colapsar todas las direcciones redundantes identificando en una misma clase $[x] = x + \ker T$ a los vectores que tienen la misma imagen. Al ser $T$ continuo, $\ker T$ es un subespacio cerrado, por lo que el cociente $X/\ker T$ es un espacio normado dotado de la norma cociente usual:
            \begin{equation*}
                \|[x]\|_{X/\ker T} = d(x, \ker T).
            \end{equation*}

            Aquí interviene de manera decisiva la hipótesis de que la imagen es finita: por el primer teorema de isomorfía del álgebra lineal, el cociente es isomorfo a $T(X)$, con lo cual:
            \begin{equation*}
                \dim(X/\ker T) = \dim(T(X)) < \infty.
            \end{equation*}
            Al haber reducido el problema a un espacio de dimensión finita, recuperamos la propiedad de compacidad: por el Teorema de Hausdorff, la esfera unidad del cociente
            \begin{equation*}
                \mathbb{S}_{X/\ker T} = \{ [x] \in X/\ker T : d(x, \ker T) = 1 \}
            \end{equation*}
            es un conjunto cerrado y acotado, luego \textbf{compacto}. Consideramos ahora el operador lineal inducido:
            \Func{\widetilde{T}}{X/\ker T}{T(X)}{[x]}{T(x)}
            
            La función que a cada clase le asigna su norma $\|\widetilde{T}([x])\| = \|T(x)\|$ es una función real continua definida sobre el compacto $\mathbb{S}_{X/\ker T}$, por lo que el Teorema de Weierstrass garantiza que alcanza su máximo absoluto en alguna clase $[x_0] \in \mathbb{S}_{X/\ker T}$:
            \begin{equation*}
                \|T(x_0)\| = \|\widetilde{T}([x_0])\| = \sup_{[x] \in \mathbb{S}_{X/\ker T}} \|\widetilde{T}([x])\| = \sup_{d(x, \ker T) = 1} \|T(x)\|.
            \end{equation*}

            Comprobemos que este valor máximo coincide con la norma del operador, es decir, que $\|T(x_0)\| = \|T\|$.
            \begin{itemize}
                \item Por la cota obtenida anteriormente, para cualquier $x$ tal que $d(x, \ker T) = 1$ se cumple $\|T(x)\| \leq \|T\| \cdot 1 = \|T\|$. Tomando el supremo en la esfera del cociente:
                \begin{equation*}
                    \|T(x_0)\| \leq \|T\| .
                \end{equation*}
                \item Recíprocamente, sea $z \in X$ un vector cualquiera con $\|z\| = 1$. Como $0 \in \ker T$, la distancia al núcleo nunca supera la norma del vector: $$d(z, \ker T) \leq \|z - 0\| = 1$$
                Si $d(z, \ker T) > 0$, podemos normalizar la clase dividiendo entre su distancia para situarla en $\mathbb{S}_{X/\ker T}$:
                \begin{equation*}
                    \|T(z)\| = d(z, \ker T) \left\| T\left( \frac{z}{d(z, \ker T)} \right) \right\| \leq 1 \cdot \|T(x_0)\| = \|T(x_0)\|,
                \end{equation*}
                desigualdad que también es trivialmente cierta si $d(z, \ker T) = 0$ (pues en tal caso $z \in \ker T$ y $\|T(z)\| = 0$). Tomando el supremo sobre todos los vectores $z \in X$ con $\|z\| = 1$, obtenemos $\|T\| \leq \|T(x_0)\|$.
            \end{itemize}
            Deducimos así que, efectivamente:
            \begin{equation*}
                \|T(x_0)\| = \|T\| \quad \text{con } d(x_0, \ker T) = 1.
            \end{equation*}

            Llegados a este punto, hemos hallado una dirección óptima $[x_0]$, pero el representante $x_0 \in X$ aún no resuelve el ejercicio: aunque $d(x_0, \ker T) = 1$, su norma individual $\|x_0\|$ puede ser estrictamente mayor que $1$ (pues la distancia a un subespacio es un ínfimo de distancias, no la norma de cualquier vector de la clase). Es aquí donde interviene la hipótesis de que $\ker T$ es \textbf{proximal} en $X$: la proximalidad asegura que dicho ínfimo se alcanza en un punto concreto. Existe, por tanto, una mejor aproximación $y_0 \in \ker T$ para $x_0$ que cumple:
            \begin{equation*}
                \|x_0 - y_0\| = d(x_0, \ker T) = 1 .
            \end{equation*}

            Definimos finalmente el vector depurado:
            \begin{equation*}
                u = x_0 - y_0 \in X .
            \end{equation*}
            Comprobamos que $u$ satisface simultáneamente las dos condiciones requeridas. En primer lugar, su norma es exactamente $1$ por construcción. En segundo lugar, al ser $y_0 \in \ker T$, se tiene que $T(y_0) = 0$, de modo que:
            \begin{equation*}
                \|T(u)\| = \|T(x_0 - y_0)\| = \|T(x_0) - T(y_0)\| = \|T(x_0)\| = \|T\| .
            \end{equation*}
            En conclusión, existe $u \in X$ con $\|u\| = 1$ tal que $\|T(u)\| = \|T\|$, lo que demuestra que el operador $T$ alcanza su norma.
        \end{proof}

    \end{ejercicio}
