\begin{ejercicio}[2.5 puntos]\label{ej:1}
    Consideramos el siguiente subconjunto de $C[0,1]$:
    \begin{equation*}
        X=\left\{f \in C[0,1] : f(0) + f(1) = 0\right\}
    \end{equation*}
    Se define el siguiente funcional lineal:
    \Func{\varphi}{X}{\bb{K}}{f}{\int_{0}^{1} f(t)\,dt}

    Probar que $\varphi$ es continuo, y calcular su norma. ¿Se alcanza la norma?\\

    Veamos que $\varphi$ es continuo:
    \begin{align*}
        |\varphi(f)| = \left|\int_{0}^{1} f(t)\,dt\right| \leq \int_{0}^{1} |f(t)|\,dt \leq \int_{0}^{1} \|f\|_\infty\,dt = \|f\|_\infty
    \end{align*}

    Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq 1$. Veamos ahora que $\|\varphi\| = 1$. Para ello, para cada $n\in \bb{N}$, definimos los subconjuntos:
    \begin{align*}
        I_n^1 &= \left[0, \frac{1}{n}\right]\qquad\qquad
        I_n^2 = \left[1-\frac{1}{n}, 1\right]
    \end{align*}

    Definimos ahora la función $f_n \in C[0,1]$, mostradas en la Figura~\ref{fig:funciones_fn}, de la siguiente manera:
    \begin{equation*}
        f_n(t) = \begin{cases}
            0 & \text{si } t \in \{0, 1\}\\
            1 & \text{si } t \in [\nicefrac{1}{n}, 1 - \nicefrac{1}{n}]\\
            \text{lineal} & \text{si } t \in I_n^1 \cup I_n^2
        \end{cases}
    \end{equation*}
    \begin{figure}[h]
        \centering
        \begin{tikzpicture}
            \begin{axis}[
                axis lines = middle,
                xlabel = $t$,
                ylabel = {$f_n(t)$},
                ymin = -0.3, ymax = 1.3,
                xmin = 0, xmax = 1,
                xtick = {0, 0.5, 1},
                ytick = {-1, 0, 1},
                grid = both,
                width = 11cm,
                height = 6.5cm,
                legend pos = outer north east
            ]

            \foreach \n/\col in {3/blue, 5/red, 10/teal, 20/orange, 50/purple} {
                \edef\temp{
                    \noexpand\addplot[color=\col, thick] coordinates {
                        (0, 0)
                        (1/\n, 1)
                        (1 - 1/\n, 1)
                        (1, 0)
                    };
                    \noexpand\addlegendentry{$n = \n$}
                }
                \temp
            }

            \end{axis}
        \end{tikzpicture}
        \caption{Funciones $f_n$ para distintos valores de $n$.}
        \label{fig:funciones_fn}
    \end{figure}

    Tenemos que $f_n \in X$ y $\|f_n\|_\infty = 1$. Además, se tiene que:
    \begin{align*}
        |\varphi(f_n)| &= \left|\int_{0}^{1} f_n(t)\,dt\right| = \int_{0}^{1} f_n(t)\,dt = \int_{I_n^1} f_n(t)\,dt + \int_{\nicefrac{1}{n}}^{1 - \nicefrac{1}{n}} 1\,dt + \int_{I_n^2} f_n(t)\,dt
        \leq \\ &\leq 0+ \left(1 - \frac{2}{n}\right) + 0 = 1 - \frac{2}{n}
    \end{align*}

    Por tanto, se tiene que:
    \begin{equation*}
        \|\varphi\| = \sup\{|\varphi(f)| : f \in X, \|f\|_\infty = 1\} \geq |\varphi(f_n)| = 1 - \frac{2}{n} \quad \forall n \in \bb{N}
    \end{equation*}

    Tomando límite cuando $n \to \infty$, se tiene que:
    \begin{equation*}
        \|\varphi\| \geq 1
    \end{equation*}

    Por tanto, uniendo ambas desigualdades, se tiene que $\|\varphi\| = 1$. Veamos ahora que la norma no se alcanza. Supongamos que existe $f \in X$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = 1$. Entonces, se tendría que:
    \begin{align*}
        1 = |\varphi(f)| = \left|\int_{0}^{1} f(t)\,dt\right| \leq \int_{0}^{1} |f(t)|\,dt \leq \int_{0}^{1} \|f\|_\infty\,dt = 1
    \end{align*}

    Por tanto, todas las desigualdades son igualdades. En particular, se tiene que:
    \begin{equation*}
        \int_{0}^{1} \|f\|_\infty\, dt = \int_{0}^{1} |f(t)|\,dt \Longrightarrow \int_{0}^{1} \left(\|f\|_\infty - |f(t)|\right)\,dt = 0
    \end{equation*}

    Como $f$ es continua y $\|f\|_\infty - |f(t)| \geq 0$ para todo $t \in [0, 1]$, se tiene para todo $t \in [0, 1]$ que:
    \begin{equation*}
        \|f\|_\infty - |f(t)| = 0 \Longrightarrow |f(t)| = \|f\|_\infty = 1
    \end{equation*}

    Por otro lado, como $\varphi(t)\in \bb{K}$, se tiene que $\exists \theta \in \bb{R}$ tal que:
    \begin{equation*}
        \varphi(f) = \int_{0}^{1} f(t)\,dt = e^{i\theta} \left|\int_{0}^{1} f(t)\,dt\right| = e^{i\theta} \int_{0}^{1} |f(t)|\,dt
    \end{equation*}

    Multiplicando por $e^{-i\theta}$, se tiene que:
    \begin{equation*}
        \int_{0}^{1} e^{-i\theta} f(t)\,dt = \int_{0}^{1} |f(t)|\,dt
    \end{equation*}

    Tomando la parte real, se tiene que:
    \begin{equation*}
        \int_{0}^{1} \Re\left(e^{-i\theta} f(t)\right)\,dt = \int_{0}^{1} |f(t)|\,dt
        \Longrightarrow
        \int_{0}^{1} \left(|f(t)| - \Re\left(e^{-i\theta} f(t)\right)\right)\,dt = 0
    \end{equation*}

    Como $|e^{-i\theta} f(t)| = |f(t)|$ y la parte real de un número complejo es menor o igual que su módulo, se tiene que:
    \begin{equation*}
        |f(t)| - \Re\left(e^{-i\theta} f(t)\right) = 0 \Longrightarrow f(t) = e^{i\theta} |f(t)|= e^{i\theta} \quad \forall t \in [0, 1]
    \end{equation*}
    
    Como $f \in X$, se tiene que:
    \begin{equation*}
        f(0) + f(1) = e^{i\theta} + e^{i\theta} = 2 e^{i\theta} = 0
    \end{equation*}
    Lo cual es una contradicción, pues $|e^{i\theta}| = 1 \neq 0$. Por tanto, la norma no se alcanza.
\end{ejercicio}


\begin{ejercicio}[2.5 puntos]
    En $X=C[0,1]$ define el siguiente funcional lineal:
    \Func{\varphi}{X}{\bb{K}}{f}{\int_{0}^{1} t^5f(t^{13})\,dt}

    Probar que $\varphi$ es continuo, y calcular su norma. ¿Se alcanza la norma?\\

    Veamos que $\varphi$ es continuo:
    \begin{align*}
        |\varphi(f)| = \left|\int_{0}^{1} t^5 f(t^{13})\,dt\right| \leq \int_{0}^{1} t^5 |f(t^{13})|\,dt \leq \int_{0}^{1} t^5 \|f\|_\infty\,dt = \frac{1}{6} \|f\|_\infty
    \end{align*}

    Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq \nicefrac{1}{6}$. Veamos ahora que $\|\varphi\| = \nicefrac{1}{6}$. Para ello, consideramos la función $f_0 \in C[0,1]$ definida por:
    \begin{equation*}
        f_0(t) = 1 \quad \forall t \in [0, 1]
    \end{equation*}

    Tenemos que $\|f_0\|_\infty = 1$ y:
    \begin{align*}
        |\varphi(f_0)| = \left|\int_{0}^{1} t^5 f_0(t^{13})\,dt\right| = \int_{0}^{1} t^5\,dt = \frac{1}{6}
    \end{align*}

    Por tanto, se tiene que:
    \begin{equation*}
        \|\varphi\| = \sup\{|\varphi(f)| : f \in C[0,1], \|f\|_\infty = 1\} \geq |\varphi(f_0)| = \frac{1}{6}
    \end{equation*}

    Por tanto, uniendo ambas desigualdades, se tiene que $\|\varphi\| = \nicefrac{1}{6}$. Además, la norma se alcanza en $f_0$.
\end{ejercicio}


\begin{ejercicio}[2.5 puntos]
    En $X=C[0,1]$ define el siguiente funcional lineal:
    \Func{\varphi}{X}{\bb{K}}{f}{\int_{0}^{\nicefrac{1}{5}} f(t)\,dt - \int_{\nicefrac{1}{3}}^{1} f(t)\,dt}

    Probar que $\varphi$ es continuo, y calcular su norma. ¿Se alcanza la norma?\\

    Veamos que $\varphi$ es continuo:
    \begin{align*}
        |\varphi(f)| &= \left|\int_{0}^{\nicefrac{1}{5}} f(t)\,dt - \int_{\nicefrac{1}{3}}^{1} f(t)\,dt\right| \leq \left|\int_{0}^{\nicefrac{1}{5}} f(t)\,dt\right| + \left|\int_{\nicefrac{1}{3}}^{1} f(t)\,dt\right|
        \leq \int_{0}^{\nicefrac{1}{5}} |f(t)|\,dt + \int_{\nicefrac{1}{3}}^{1} |f(t)|\,dt \leq\\ &\leq \int_{0}^{\nicefrac{1}{5}} \|f\|_\infty\,dt + \int_{\nicefrac{1}{3}}^{1} \|f\|_\infty\,dt = \left(\frac{1}{5} - 0 + 1 - \frac{1}{3}\right) \|f\|_\infty = \frac{13}{15} \|f\|_\infty
    \end{align*}

    Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq \nicefrac{13}{15}$. Veamos ahora que $\|\varphi\| = \nicefrac{13}{15}$. Para ello, la función que buscamos debería valer $1$ en $[0, \nicefrac{1}{5}]$ y $-1$ en $[\nicefrac{1}{3}, 1]$. Como $\nicefrac{1}{5} < \nicefrac{1}{3}$, podemos definir la función $f_0 \in C[0,1]$ de la siguiente manera:
    \begin{equation*}
        f_0(t) = \begin{cases}
            1 & \text{si } t \in [0, \nicefrac{1}{5}]\\
            \text{lineal} & \text{si } t \in [\nicefrac{1}{5}, \nicefrac{1}{3}]\\
            -1 & \text{si } t \in [\nicefrac{1}{3}, 1]
        \end{cases}
    \end{equation*}

    Tenemos que $\|f_0\|_\infty = 1$ y:
    \begin{align*}
        \varphi(f_0) &= \int_{0}^{\nicefrac{1}{5}} f_0(t)\,dt - \int_{\nicefrac{1}{3}}^{1} f_0(t)\,dt = \int_{0}^{\nicefrac{1}{5}} 1\,dt - \int_{\nicefrac{1}{3}}^{1} (-1)\,dt = \frac{1}{5} + \left(1 - \frac{1}{3}\right) = \frac{13}{15}
    \end{align*}

    Por tanto, $|\varphi(f_0)| = \nicefrac{13}{15}$, luego $\|\varphi\| = \nicefrac{13}{15}$. Además, la norma se alcanza en $f_0$.
\end{ejercicio}

\begin{ejercicio}~
    \begin{enumerate}[label=(\alph*)]
        \item Todo espacio de Banach con base de Schauder es separable.
        
        Este resultado es cierto. De hecho, si $\{e_n\}_{n\in \bb{N}}$ es una base de Schauder en un espacio de Banach $X$, entonces el subespacio generado por $\{e_n\}_{n\in \bb{N}}$ es denso en $X$. Como este subespacio tiene dimensión numerable, se tiene que $X$ es separable.
        \item El espacio de las sucesiones acotadas, $l_\infty$, es un espacio métrico separable.
        
        Este resultado es falso. $\{e_n\}_{n\in \bb{N}}$ es una base de Schauder de $l_p$ para todo $1\leq p < \infty$, pero no lo es de $l_\infty$. De hecho, $l_\infty$ no es separable.

        % // TODO: Demostrado, pero fácil?
        \item $\{e_n\}_{n\in \bb{N}}$ es una base algebraica en $l_p$ para todo $1\leq p \leq \infty$ y $c_0$.
        
        No, $\{e_n\}_{n\in \bb{N}}$ es una base algebraica de las sucesiones de soporte finito $\bb{K}^{(\bb{N})}$, que es un subespacio propio de $l_p$ para todo $1\leq p \leq \infty$ y $c_0$. Por tanto, $\{e_n\}_{n\in \bb{N}}$ no es una base algebraica de $l_p$ ni de $c_0$. Notemos que, en el caso de las bases algebraicas, tan solo se consideran combinaciones lineales finitas de los vectores de la base, mientras que en el caso de las bases de Schauder se consideran combinaciones lineales infinitas.
        

        \item Un espacio normado $X$ es separable si y solo si tiene un conjunto numerable de vectores linealmente independientes cuya expansión lineal es densa en $X$.
        
        Este resultado es cierto, como se ha visto en clase.
    \end{enumerate}    
\end{ejercicio}

\begin{ejercicio}
    Sea $Y = \bb{K}^{(\bb{N})}$ (el espacio vectorial de las sucesiones casi nulas):
    \begin{enumerate}[label=(\alph*)]
        \item La norma de la suma y la norma del máximo son equivalentes en $Y$.
        
        Veamos que esto es falso. Consideremos, para cada $n\in \bb{N}$, la sucesión $x_n \in Y$ definida por:
        \begin{align*}
            x_n = \sum_{k=1}^{n} e_k
        \end{align*}

        Tenemos que $\|x_n\|_\infty = 1$ y $\|x_n\|_1 = n$. Por tanto, no existe una constante $C > 0$ tal que $\|x_n\|_\infty \leq C \|x_n\|_1$ para todo $n \in \bb{N}$. Por tanto, las normas no son equivalentes.
        \item La norma de la suma y la norma del máximo no son equivalentes en $Y$.
        
        En el apartado anterior hemos visto que esto es cierto.
        \item Cualesquiera dos normas definidas en $Y$ son equivalentes.
        
        Esto es falso. Si la dimensión de $\bb{K}^{(\bb{N})}$ fuese finita, entonces todas las normas serían equivalentes. Pero como $\bb{K}^{(\bb{N})}$ es de dimensión infinita, no todas las normas son equivalentes (como se ha visto).
        \item No se puede definir ninguna norma en $Y$.
        
        Esto es falso. Por ejemplo, la norma del máximo $\|\cdot\|_\infty$ es una norma en $Y$.
    \end{enumerate}
\end{ejercicio}

\begin{ejercicio}~
    \begin{enumerate}[label=(\alph*)]
        \item Un espacio normado es de Banach si y solo si toda serie absolutamente convergente es convergente.
        
        Este resultado es cierto.
        \item En un espacio normado, si $\sum\limits_{n=1}^{\infty} x_n$ es absolutamente convergente, entonces $\{x_n\}\to 0$.
        
        Si la serie es absolutamente convergente, entonces la serie de normas $\sum\limits_{n=1}^{\infty} \|x_n\|$ es convergente. Como el término general de una serie convergente tiende a cero, se tiene que $\|x_n\| \to 0$, y como la norma es continua, se tiene que $\{x_n\} \to 0$.
        \item Si $T: X \to Y$ es un operador lineal entre espacios de Banach y $\sum\limits_{n=1}^{\infty} x_n$ es convergente, entonces $\sum\limits_{n=1}^{\infty} T(x_n)$ es convergente.
        
        Esto es falso. Si $T$ no es continuo, no tiene por qué preservar la convergencia de las series.

        % // TODO: Contraejemplo
        \item Sustituir la norma de un espacio normado por otra norma equivalente puede cambiar las sucesiones de Cauchy.
        
        Esto es falso. Si $\|\cdot\|_1$ y $\|\cdot\|_2$ son normas equivalentes en un espacio normado $X$, entonces generan la misma topología, y por tanto las sucesiones de Cauchy son las mismas.
    \end{enumerate}    
\end{ejercicio}

\begin{ejercicio}~
    \begin{enumerate}[label=(\alph*)]
        \item Todo operador lineal que llega a un espacio normado de dimensión finita es continuo.
        
        Esto es falso. Si el dominio es de dimensión finita, entonces el operador es continuo. Si el codominio es de dimensión finita, entonces el operador es continuo si y solo si su núcleo es cerrado.

        \begin{equation*}
            \ker f \subsetneq \overline{\ker f}\subset X
        \end{equation*}

        De aquí, se tiene que $\ker f$ es un subespacio denso en $X$.

        \item Un operador lineal $T: X \to Y$ entre espacios de Banach es continuo si y solo si $\ker T$ es cerrado.
        
        Esto no es cierto. Si la dimensión del codominio es finita, entonces el operador es continuo si y solo si su núcleo es cerrado; pero si la dimensión del codominio es infinita, entonces el operador puede ser discontinuo y tener el núcleo cerrado.

\begin{ejercicio}~
    \begin{enumerate}[label=(\alph*)]
        \item Sea $\varphi$ una seminorma en un espacio vectorial $X$. Entonces, $\varphi(0)=0$ y $\varphi(x)\geq 0$ para todo $x\in X$.
        
        Esta es cierta. Las condiciones de una seminorma $\varphi$ son:
        \begin{enumerate}
            \item $\varphi(x+y) \leq \varphi(x) + \varphi(y)$ para todo $x, y \in X$ (subaditividad).
            \item $\varphi(\alpha x) = |\alpha| \varphi(x)$ para todo $\alpha \in \bb{K}$ y $x \in X$ (homogeneidad).
        \end{enumerate}

        De la segunda condición, tomando $\alpha = 0$ y $x \in X$, se tiene que:
        \begin{equation*}
            \varphi(0) = \varphi(0 \cdot x) = |0| \varphi(x) = 0
        \end{equation*}

        Además, de la primera condición:
        \begin{align*}
            0=\varphi(0) = \varphi(-x + x) \leq \varphi(-x) + \varphi(x) = |-1| \varphi(x) + \varphi(x) = 2\varphi(x) \Longrightarrow\\ \Longrightarrow \varphi(x) \geq 0\qquad \forall x \in X
        \end{align*}



        \item La distancia inducida por la norma $d(x,y)=\|x-y\|$ es invariante por traslaciones y homogénea por homotecias.
        
        Esta es cierta. Veamos que $d$ es invariante por traslaciones. Fijado $z \in X$, se tiene que:
        \begin{align*}
            d(x+z, y+z) &= \| (x+z) - (y+z) \| = \| x - y \| = d(x,y)\qquad \forall x, y \in X
        \end{align*}

        Veamos que $d$ es homogénea por homotecias. Fijado $\alpha \in \bb{K}$, se tiene que:
        \begin{align*}
            d(\alpha x, \alpha y) &= \| \alpha x - \alpha y \| = \| \alpha (x - y) \| = |\alpha| \| x - y \| = |\alpha| d(x,y)\qquad \forall x, y \in X
        \end{align*}
        \item En dimensión infinita, cualquier norma es continua pero no necesariamente uniformemente continua.
        
        Esta es falsa. En cualquier espacio normado, la norma es de hecho lipschitziana con constante 1, y por tanto uniformemente continua. En particular, la desigualdad triangular inversa nos da que:
        \begin{align*}
            |\|x\| - \|y\|| \leq \|x - y\|\qquad \forall x, y \in X
        \end{align*}
        \item Dos normas son equivalentes si y solo si existen constantes $c_1, c_2 > 0$ tales que $c_1\|x\|_1 \leq \|x\|_2 \leq c_2\|x\|_1$ para todo $x$.
        
        Esta es cierta. Es una importante caracterización de la equivalencia de normas en espacios normados.
    \end{enumerate}
\end{ejercicio}
