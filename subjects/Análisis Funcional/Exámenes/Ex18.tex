La parte Tipo Test del Examen puntúa 7.5 puntos. Puesto que hay 5 preguntas, cada una de ellas puntúa 1.5 puntos. Cada una de ellas tiene 4 respuestas posibles, de las cuales una, dos, tres o cuatro pueden ser correctas. La puntuación de cada pregunta se calcula de la siguiente manera:
\begin{itemize}
    \item Si se selecciona alguna respuesta incorrecta, la puntuación de la pregunta es 0.
    \item Si se seleccionan $k$ respuestas correctas y no se selecciona ninguna incorrecta, la puntuación de la pregunta es:
    \begin{equation*}
        \frac{k}{n} \cdot 1.5
    \end{equation*}
    donde $n$ es el número total de respuestas correctas de la pregunta.
\end{itemize}

\begin{ejercicio}[2.5 puntos]
    Consideramos el siguiente subconjunto de $C[0,1]$:
    \begin{equation*}
        X=\left\{f \in C[0,1] : f(0) + f(1) = 0\right\}
    \end{equation*}
    Se define el siguiente funcional lineal:
    \Func{\varphi}{X}{\bb{K}}{f}{\int_{0}^{1} f(t)\,dt}

    Probar que $\varphi$ es continuo, y calcular su norma. ¿Se alcanza la norma?\\

    Veamos que $\varphi$ es continuo:
    \begin{align*}
        |\varphi(f)| = \left|\int_{0}^{1} f(t)\,dt\right| \leq \int_{0}^{1} |f(t)|\,dt \leq \int_{0}^{1} \|f\|_\infty\,dt = \|f\|_\infty
    \end{align*}

    Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq 1$. Veamos ahora que $\|\varphi\| = 1$. Para ello, para cada $n\in \bb{N}$, definimos los subconjuntos:
    \begin{align*}
        I_n^1 &= \left[0, \frac{1}{n}\right]\qquad\qquad
        I_n^2 = \left[1-\frac{1}{n}, 1\right]
    \end{align*}

    Definimos ahora la función $f_n \in C[0,1]$, mostradas en la Figura~\ref{fig:funciones_fn}, de la siguiente manera:
    \begin{equation*}
        f_n(t) = \begin{cases}
            0 & \text{si } t \in \{0, 1\}\\
            1 & \text{si } t \in [\nicefrac{1}{n}, 1 - \nicefrac{1}{n}]\\
            \text{lineal} & \text{si } t \in I_n^1 \cup I_n^2
        \end{cases}
    \end{equation*}
    \begin{figure}[h]
        \centering
        \begin{tikzpicture}
            \begin{axis}[
                axis lines = middle,
                xlabel = $t$,
                ylabel = {$f_n(t)$},
                ymin = -0.3, ymax = 1.3,
                xmin = 0, xmax = 1,
                xtick = {0, 0.5, 1},
                ytick = {-1, 0, 1},
                grid = both,
                width = 11cm,
                height = 6.5cm,
                legend pos = outer north east
            ]

            \foreach \n/\col in {3/blue, 5/red, 10/teal, 20/orange, 50/purple} {
                \edef\temp{
                    \noexpand\addplot[color=\col, thick] coordinates {
                        (0, 0)
                        (1/\n, 1)
                        (1 - 1/\n, 1)
                        (1, 0)
                    };
                    \noexpand\addlegendentry{$n = \n$}
                }
                \temp
            }

            \end{axis}
        \end{tikzpicture}
        \caption{Funciones $f_n$ para distintos valores de $n$.}
        \label{fig:funciones_fn}
    \end{figure}

    Tenemos que $f_n \in X$ y $\|f_n\|_\infty = 1$. Además, se tiene que:
    \begin{align*}
        |\varphi(f_n)| &= \left|\int_{0}^{1} f_n(t)\,dt\right| = \int_{0}^{1} f_n(t)\,dt = \int_{I_n^1} f_n(t)\,dt + \int_{\nicefrac{1}{n}}^{1 - \nicefrac{1}{n}} 1\,dt + \int_{I_n^2} f_n(t)\,dt
        \leq \\ &\leq 0+ \left(1 - \frac{2}{n}\right) + 0 = 1 - \frac{2}{n}
    \end{align*}

    Por tanto, se tiene que:
    \begin{equation*}
        \|\varphi\| = \sup\{|\varphi(f)| : f \in X, \|f\|_\infty = 1\} \geq |\varphi(f_n)| = 1 - \frac{2}{n} \quad \forall n \in \bb{N}
    \end{equation*}

    Tomando límite cuando $n \to \infty$, se tiene que:
    \begin{equation*}
        \|\varphi\| \geq 1
    \end{equation*}

    Por tanto, uniendo ambas desigualdades, se tiene que $\|\varphi\| = 1$. Veamos ahora que la norma no se alcanza. Supongamos que existe $f \in X$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = 1$. Entonces, se tendría que:
    \begin{align*}
        1 = |\varphi(f)| = \left|\int_{0}^{1} f(t)\,dt\right| \leq \int_{0}^{1} |f(t)|\,dt \leq \int_{0}^{1} \|f\|_\infty\,dt = 1
    \end{align*}

    Por tanto, todas las desigualdades son igualdades. En particular, se tiene que:
    \begin{equation*}
        \int_{0}^{1} \|f\|_\infty\, dt = \int_{0}^{1} |f(t)|\,dt \Longrightarrow \int_{0}^{1} \left(\|f\|_\infty - |f(t)|\right)\,dt = 0
    \end{equation*}

    Como $f$ es continua y $\|f\|_\infty - |f(t)| \geq 0$ para todo $t \in [0, 1]$, se tiene para todo $t \in [0, 1]$ que:
    \begin{equation*}
        \|f\|_\infty - |f(t)| = 0 \Longrightarrow |f(t)| = \|f\|_\infty = 1
    \end{equation*}

    Por otro lado, como $\varphi(t)\in \bb{K}$, se tiene que $\exists \theta \in \bb{R}$ tal que:
    \begin{equation*}
        \varphi(f) = \int_{0}^{1} f(t)\,dt = e^{i\theta} \left|\int_{0}^{1} f(t)\,dt\right| = e^{i\theta} \int_{0}^{1} |f(t)|\,dt
    \end{equation*}

    Multiplicando por $e^{-i\theta}$, se tiene que:
    \begin{equation*}
        \int_{0}^{1} e^{-i\theta} f(t)\,dt = \int_{0}^{1} |f(t)|\,dt
    \end{equation*}

    Tomando la parte real, se tiene que:
    \begin{equation*}
        \int_{0}^{1} \Re\left(e^{-i\theta} f(t)\right)\,dt = \int_{0}^{1} |f(t)|\,dt
        \Longrightarrow
        \int_{0}^{1} \left(|f(t)| - \Re\left(e^{-i\theta} f(t)\right)\right)\,dt = 0
    \end{equation*}

    Como $|e^{-i\theta} f(t)| = |f(t)|$ y la parte real de un número complejo es menor o igual que su módulo, se tiene que:
    \begin{equation*}
        |f(t)| - \Re\left(e^{-i\theta} f(t)\right) = 0 \Longrightarrow f(t) = e^{i\theta} |f(t)|= e^{i\theta} \quad \forall t \in [0, 1]
    \end{equation*}
    
    Como $f \in X$, se tiene que:
    \begin{equation*}
        f(0) + f(1) = e^{i\theta} + e^{i\theta} = 2 e^{i\theta} = 0
    \end{equation*}
    Lo cual es una contradicción, pues $|e^{i\theta}| = 1 \neq 0$. Por tanto, la norma no se alcanza.
\end{ejercicio}


\begin{ejercicio}[2.5 puntos]
    En $C[0,1]$ define el siguiente funcional lineal:
    \Func{\varphi}{X}{\bb{K}}{f}{\int_{0}^{1} t^5f(t^{13})\,dt}

    Probar que $\varphi$ es continuo, y calcular su norma. ¿Se alcanza la norma?\\

    Veamos que $\varphi$ es continuo:
    \begin{align*}
        |\varphi(f)| = \left|\int_{0}^{1} t^5 f(t^{13})\,dt\right| \leq \int_{0}^{1} t^5 |f(t^{13})|\,dt \leq \int_{0}^{1} t^5 \|f\|_\infty\,dt = \frac{1}{6} \|f\|_\infty
    \end{align*}

    Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq \nicefrac{1}{6}$. Veamos ahora que $\|\varphi\| = \nicefrac{1}{6}$. Para ello, consideramos la función $f_0 \in C[0,1]$ definida por:
    \begin{equation*}
        f_0(t) = 1 \quad \forall t \in [0, 1]
    \end{equation*}

    Tenemos que $\|f_0\|_\infty = 1$ y:
    \begin{align*}
        |\varphi(f_0)| = \left|\int_{0}^{1} t^5 f_0(t^{13})\,dt\right| = \int_{0}^{1} t^5\,dt = \frac{1}{6}
    \end{align*}

    Por tanto, se tiene que:
    \begin{equation*}
        \|\varphi\| = \sup\{|\varphi(f)| : f \in C[0,1], \|f\|_\infty = 1\} \geq |\varphi(f_0)| = \frac{1}{6}
    \end{equation*}

    Por tanto, uniendo ambas desigualdades, se tiene que $\|\varphi\| = \nicefrac{1}{6}$. Además, la norma se alcanza en $f_0$.
\end{ejercicio}
