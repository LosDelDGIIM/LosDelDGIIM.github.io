        \begin{equation*}
            T(x)(k) = \frac{k}{k+1}x(k) \qquad \forall k\in \mathbb{N}, \quad \forall x\in c_0
        \end{equation*}
        \begin{enumerate}[label=\alph*)]

            \item \textbf{[3 puntos]} Demuestra que $\|T\| = 1$.

        \end{enumerate}

        \begin{enumerate}[label=\alph*)]

        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}[7 puntos]
        Se considera el operador lineal $T:c_0\to c_0$ definido por
        \begin{equation*}
            T(x)(k) = \frac{k}{k+1}x(k) \qquad \forall k\in \mathbb{N}, \quad \forall x\in c_0
        \end{equation*}
        \begin{enumerate}[label=\alph*)]
            \item \textbf{[3 puntos]} Demuestra que $T$ es continuo y $\|T\|\leq 1$.

                Recordamos que teníamos la norma:
                \begin{equation*}
                    \|x\| = \sup_{n\in \mathbb{N}}|x(n)|
                \end{equation*}
                
                Por lo que:
                \begin{equation*}
                    \|T(x)\| = \sup_{k\in \mathbb{N}}|T(x)(k)| = \sup_{k\in \mathbb{N}}\left|\frac{k}{k+1}x(k)\right| \leq \sup_{k\in \mathbb{N}}|x(k)| = \|x\|
                \end{equation*}
                de donde $\|T\|\leq 1$.
            \item \textbf{[3 puntos]} Demuestra que $\|T\| = 1$.
            
                Consideramos la sucesión de sucesiones canónicas $\{e_n\}_{n\in \mathbb{N}}$ de $c_0$, donde:
                \begin{equation*}
                    e_n(k) = \left\{\begin{array}{ll}
                        1 & \text{si\ } k = n \\
                        0 & \text{si\ } k \neq n
                    \end{array}\right.
                \end{equation*}
                Tenemos entonces que $\|e_n\| = 1 \quad \forall n\in \mathbb{N}$, así como que:
                \begin{equation*}
                    \|T(e_n)\| = \sup_{k\in \mathbb{N}}\left|\frac{k}{k+1}e_n(k)\right| = \sup_{k\in \mathbb{N}}\left|\frac{n}{n+1}\right| = \frac{n}{n+1}
                \end{equation*}
                
                Por tanto, para cada $n\in \mathbb{N}$ tenemos que:
                \begin{equation*}
                    \|T\| = \sup_{\|x\|=1}\|T(x)\| \geq \|T(e_n)\| = \frac{n}{n+1}
                \end{equation*}

                Tomando el límite cuando $n\to \infty$ obtenemos que $\|T\|\geq 1$, y como ya habíamos demostrado que $\|T\|\leq 1$, obtenemos que $\|T\| = 1$.
            \item \textbf{[1 punto]} Demuestra que $T$ no alcanza su norma.

                \begin{description}
                    \item[Opción 1] Supuesto que existe $x\in c_0$ de forma que $\|x\| =1$ y $\|T(x)\| = 1$, tenemos:
                    \begin{equation*}
                        1 = \|T(x)\| = \sup_{k\in \mathbb{N}}\left|\frac{k}{k+1}x(k)\right| = \sup_{k \in \mathbb{N}}\left(\frac{k}{k+1}|x(k)|\right)
                    \end{equation*}

                    Como $T(x)\in c_0$, tenemos que su norma del supremo es la norma del máximo, por lo que existe $k_0\in \mathbb{N}$ de forma que:
                    \begin{equation*}
                        1 = \sup_{k \in \mathbb{N}}\left(\frac{k}{k+1}|x(k)|\right) = \frac{k_0}{k_0+1}|x(k_0)|
                    \end{equation*}

                    Por tanto:
                    \begin{equation*}
                        |x(k_0)| = \frac{k_0+1}{k_0} > 1
                    \end{equation*}
                    lo que contradice que $\|x\| = 1$. Por tanto, no existe tal $x\in c_0$ y $T$ no alcanza su norma. 

                    \item[Opción 2] Supuesto que existe $x\in c_0$ de forma que $\|x\| =1$ y $\|T(x)\| = 1$, tenemos:
                    \begin{equation*}
                        1 = \|T(x)\| = \sup_{k\in \mathbb{N}}\left|\frac{k}{k+1}x(k)\right| = \sup_{k \in \mathbb{N}}\left(\frac{k}{k+1}|x(k)|\right)
                    \end{equation*}

                    Como $\{x\}\to 0$, para $\varepsilon = \nicefrac{1}{2}$ existe $m\in \mathbb{N}$ de forma que:
                    \begin{equation*}
                        |x(k)| < \frac{1}{2} \qquad \forall k\geq m
                    \end{equation*}
                    Por tanto:
                    \begin{align*}
                        1 &= \sup_{k \in \mathbb{N}}\left(\frac{k}{k+1}|x(k)|\right) \\
                        &= \max\left\{\sup_{k \in \{1,\ldots,m-1\}}\left(\frac{k}{k+1}|x(k)|\right), \sup_{k\geq m}\left(\frac{k}{k+1}|x(k)|\right)\right\} \\
                        &= \max\left\{\max_{k \in \{1,\ldots,m-1\}}\left(\frac{k}{k+1}|x(k)|\right), \sup_{k\geq m}\left(\frac{k}{k+1}|x(k)|\right)\right\}
                    \end{align*}
                    \begin{itemize}
                        \item Para $k\geq m$ tenemos que:
                        \begin{equation*}
                            \dfrac{k}{k+1}|x(k)| < |x(k)| < \frac{1}{2}
                            \Longrightarrow \sup_{k\geq m}\left(\frac{k}{k+1}|x(k)|\right) \leq \frac{1}{2}
                        \end{equation*}
                        \item Para $k\in \{1,\ldots,m-1\}$ tenemos que:
                        \begin{equation*}
                            \frac{k}{k+1}|x(k)| < |x(k)| \leq \|x\| = 1
                        \end{equation*}

                        Al tratarse de un conjunto finito, el supremo es el máximo, y por tanto:
                        \begin{equation*}
                            \max_{k \in \{1,\ldots,m-1\}}\left(\frac{k}{k+1}|x(k)|\right) < 1
                        \end{equation*}
                    \end{itemize}

                    Por tanto, hemos llegado a $1<1$, lo que es una contradicción. Por tanto, no existe tal $x\in c_0$ y $T$ no alcanza su norma.
                \end{description}
        \end{enumerate}
    \end{ejercicio}
