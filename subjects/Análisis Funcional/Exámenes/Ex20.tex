    \begin{ejercicio}
        Sean $X$ e $Y$ espacios de Banach y $T \colon X \longrightarrow Y$ un operador lineal, continuo y sobreyectivo. Prueba que existe $m > 0$ tal que para todo $y \in Y$ existe $x \in X$ con $T(x) = y$ y $\|x\| \leqslant m \|y\|$.
    \end{ejercicio}

    \begin{ejercicio}
        Sean $X$ e $Y$ espacios de Banach y sea $\{f_n\}$ una sucesión de elementos de $Y^*$ tal que $\bigcap\limits_{n=1}^\infty \ker f_n = \{0\}$. Demuestra que una aplicación lineal $T \colon X \longrightarrow Y$ es continua si y sólo si $f_n \circ T \in X^*$ para cada $n \in \bb{N}$.
    \end{ejercicio}

    \begin{ejercicio}
        Calcula el valor de $\alpha, \beta \in \bb{R}$ que minimizan la integral
        \[
            \int_{-1}^1 |e^x - \alpha - \beta x|^2 \, dx.
        \]
    \end{ejercicio}

    \begin{ejercicio}
        Desarrolla el siguiente tema: Lema de categoría de Baire, teorema de Banach-Steinhauss; aplicaciones.
    \end{ejercicio}

    \begin{ejercicio}[Extra]
        Sean $X$ e $Y$ espacios normados y $T \colon X \longrightarrow Y$ un operador lineal y continuo. Prueba que $T$ alcanza su norma si y sólo si $T^*$ alcanza su norma y la imagen por $T^*$ de alguno de los funcionales en los que $T^*$ alcanza su norma es un funcional que alcanza su norma en $X$. Da un ejemplo de un operador que no alcance su norma cuyo adjunto sí la alcance.
    \end{ejercicio}
