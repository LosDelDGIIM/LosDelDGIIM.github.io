        En el espacio de Banach $X = (C\left[0, 1\right], \|\cdot\|_\infty)$ se considera el funcional lineal $\varphi$ definido por
        \[
            \varphi(f) = \int_0^1 (2t - 1) f(t) \, dt \quad \forall f \in C\left[0, 1\right].
        \]
        \begin{enumerate}[label=\alph*)]
            \item Pruebe que $\varphi$ es continuo y calcule su norma.
            \item ¿Alcanza $\varphi$ su norma? Justifique su respuesta.
            \item ¿Existe $x^{**} \in X^{**}$ con $\|x^{**}\| = 1$ tal que $x^{**}(\varphi) = \|\varphi\|$? Justifique su respuesta.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}
        En el espacio de Banach $X = (C\left[0, 1\right], \|\cdot\|_\infty)$ se considera el funcional lineal $\varphi$ definido por
        \[
            \varphi(f) = \int_0^1 (2t - 1) f(t) \, dt \quad \forall f \in C\left[0, 1\right].
        \]
        \begin{enumerate}[label=\alph*)]
            \item Pruebe que $\varphi$ es continuo y calcule su norma.
            
            Veamos que $\varphi$ es continuo. Para ello:
            \begin{align*}
                |\varphi(f)| &= \left|\int_0^1 (2t - 1) f(t) \, dt\right| 
                \leq \int_0^1 |2t - 1| |f(t)| \, dt \\
                &\leq \|f\|_\infty \int_0^1 |2t - 1| \, dt
                = \|f\|_\infty \left[\int_0^{\nicefrac{1}{2}} (1 - 2t) \, dt + \int_{\nicefrac{1}{2}}^1 (2t - 1) \, dt\right] \\
                &= \|f\|_\infty \left[\left(t - t^2\right)\Big|_0^{\nicefrac{1}{2}} + \left(t^2 - t\right)\Big|_{\nicefrac{1}{2}}^1\right] = \|f\|_\infty \left[\left(\frac{1}{2} - \frac{1}{4}\right) + \left(1 - 1 - \left(\frac{1}{4} - \frac{1}{2}\right)\right)\right] \\
                &= \|f\|_\infty \left[\frac{1}{4} + \left(-\frac{1}{4} + \frac{1}{2}\right)\right] 
                = \|f\|_\infty \cdot \frac{1}{2}.
            \end{align*}

            Por lo tanto, $\varphi$ es continuo y $\|\varphi\| \leq \nicefrac{1}{2}$. Veamos ahora que $\|\varphi\| = \nicefrac{1}{2}$. La función teórica que deberíamos tomar es $f(t) = \operatorname{sign}(2t - 1)$, pero esta función no es continua. Sin embargo, podemos aproximarla por funciones continuas. Para cada $n \in \bb{N}$, definimos por comodidad:
            \begin{equation*}
                I_n = \left[\frac{1}{2} - \frac{1}{n}, \frac{1}{2} + \frac{1}{n}\right].
            \end{equation*}
            
            Definimos entonces la función continua:
            \begin{align*}
                f_n(t) = \begin{cases}
                    -1, & t \in [0, \nicefrac{1}{2} - \nicefrac{1}{n}] \\
                    1 & t \in [\nicefrac{1}{2} + \nicefrac{1}{n}, 1] \\
                    \text{lineal} & t \in I_n
                \end{cases}
            \end{align*}

            Notemos que no es necesario dar la expresión explícita de la parte lineal, ya que lo único que nos interesa es que $f_n$ sea continua y que $\|f_n\|_\infty = 1$. Estas funciones se muestran en la Figura~\ref{fig:funciones_fn}. 
            \begin{figure}[h]
                \centering
                \begin{tikzpicture}
                    \begin{axis}[
                        axis lines = middle,
                        xlabel = $t$,
                        ylabel = {$f_n(t)$},
                        ymin = -1.3, ymax = 1.3,
                        xmin = 0, xmax = 1,
                        xtick = {0, 0.5, 1},
                        ytick = {-1, 0, 1},
                        grid = both,
                        width = 11cm,
                        height = 6.5cm,
                        legend pos = south east,
                        legend cell align = left
                    ]

                    \foreach \n/\col in {3/blue, 5/red, 10/teal, 20/orange, 50/purple} {
                        \edef\temp{
                            \noexpand\addplot[color=\col, thick] coordinates {
                                (0, -1)
                                (0.5 - 1/\n, -1)
                                (0.5 + 1/\n, 1)
                                (1, 1)
                            };
                            \noexpand\addlegendentry{$n = \n$}
                        }
                        \temp
                    }

                    \end{axis}
                \end{tikzpicture}
                \caption{Gráfica de la función $f_n$ para distintos valores de $n$.}
                \label{fig:funciones_fn}
            \end{figure}

            Ahora, para cada $n \in \bb{N}$, tenemos que
            \begin{align*}
                \varphi(f_n) &= \int_0^1 (2t - 1) f_n(t) \, dt \\
                &= \int_0^1 (2t - 1)(f_n(t) - \operatorname{sign}(2t - 1)) \, dt + \int_0^1 (2t - 1) \operatorname{sign}(2t - 1) \, dt
                =\\&= \int_{I_n} (2t - 1)(f_n(t) - \operatorname{sign}(2t - 1)) \, dt + \int_0^1 |2t - 1| \, dt
            \end{align*}
            \begin{itemize}
                \item Por un lado, como hemos comprobado antes, se tiene:
                \begin{equation*}
                    \int_0^1 |2t - 1| \, dt = \frac{1}{2}.
                \end{equation*}
                \item Por otro lado, para todo $t \in I_n\subset [0, 1]$, se tiene que:
                \begin{align*}
                    |2t - 1| &\leq 1\\
                    |f_n(t) - \operatorname{sign}(2t - 1)| &\leq |f_n(t)| + |\operatorname{sign}(2t - 1)| \leq 1 + 1 = 2.
                \end{align*}

                Por lo tanto, se tiene que:
                \begin{align*}
                    \left|\int_{I_n} (2t - 1)(f_n(t) - \operatorname{sign}(2t - 1)) \, dt\right| & \leq \int_{I_n} |2t - 1| |f_n(t) - \operatorname{sign}(2t - 1)| \, dt \leq \\ & \leq 2 \int_{I_n} dt = 2 \lm(I_n) = 2 \cdot \frac{2}{n} = \frac{4}{n}.
                \end{align*}
            \end{itemize}

            Por lo tanto, para todo $n \in \bb{N}$, se tiene que:
            \begin{equation*}
                |\varphi(f_n)| \geq \frac{1}{2} - \frac{4}{n}.
            \end{equation*}

            Por tanto, se deduce que:
            \begin{equation*}
                \|\varphi\| \geq \frac{1}{2} - \frac{4}{n} \quad \forall n \in \bb{N}.
            \end{equation*}

            Tomando el límite cuando $n \to \infty$, se obtiene que:
            \begin{equation*}
                \|\varphi\| \geq \frac{1}{2}.
            \end{equation*}

            Combinando esto con la desigualdad anterior, se concluye que:
            \begin{equation*}
                \|\varphi\| = \frac{1}{2}.
            \end{equation*}



            \item ¿Alcanza $\varphi$ su norma? Justifique su respuesta.
            
            Veamos que $\varphi$ no alcanza su norma. Supongamos que existe $f_0 \in C\left[0, 1\right]$ con $\|f_0\|_\infty = 1$ tal que $|\varphi(f_0)| = \|\varphi\| = \nicefrac{1}{2}$. Entonces, se tendría que:
            \begin{align*}
                \frac{1}{2} = |\varphi(f_0)| &= \left|\int_0^1 (2t - 1) f_0(t) \,dt\right| \leq \int_0^1 |2t - 1| |f_0(t)| \, dt \leq \\ &\leq \int_0^1 |2t - 1| \|f_0\|_\infty \, dt = \int_0^1 |2t - 1| \, dt = \frac{1}{2}.
            \end{align*}

            Por lo tanto, se tendrían que cumplir todas las desigualdades anteriores como igualdades. En particular, se tendría que:
            \begin{itemize}
                \item Por un lado:
                \begin{equation*}
                    \int_0^1 |2t - 1| |f_0(t)| \, dt = \int_0^1 |2t - 1| \|f_0\|_\infty \, dt
                    \Longrightarrow
                    \int_0^1 |2t - 1| (\|f_0\|_\infty - |f_0(t)|) \, dt = 0.
                \end{equation*}

                Como se trata de una integral de una función continua y no negativa, se deduce que:
                \begin{equation*}
                    |2t - 1| (\|f_0\|_\infty - |f_0(t)|) = 0 \qquad \forall t \in [0, 1].
                \end{equation*}

                De esto se deduce que $|f_0(t)| = \|f_0\|_\infty = 1$ para todo $t \in [0, 1] \setminus \{\nicefrac{1}{2}\}$; y como $f_0$ y la norma son continuas, se deduce que:
                \begin{equation*}
                    |f_0(t)| = 1 \qquad \forall t \in [0, 1].
                \end{equation*}

                Notemos que, como $f_0$ puede tomar valores complejos, no se puede llegar a una contradicción directa.

                \item Por otro lado, se tendría que:
                \begin{equation*}
                    \left|\int_0^1 (2t - 1) f_0(t) \, dt\right| = \int_0^1 |2t - 1| |f_0(t)| \, dt
                \end{equation*}

                Como $f_0(t)$ puede ser complejo, el valor de la integral puede ser complejo. Por tanto $\exists \theta\in \bb{R}$ tal que:
                \begin{equation*}
                    \int_0^1 (2t - 1) f_0(t) \, dt = e^{i\theta} \left|\int_0^1 (2t - 1) f_0(t) \, dt\right| = e^{i\theta} \int_0^1 |2t - 1| |f_0(t)| \, dt
                \end{equation*}

                Por lo tanto, multiplicando por $e^{-i\theta}$, se deduce que:
                \begin{equation*}
                    \int_0^1 e^{-i\theta} (2t - 1) f_0(t) \, dt = \int_0^1 |2t - 1| |f_0(t)| \, dt
                \end{equation*}

                Tomando la parte real, se obtiene que:
                \begin{equation*}
                    \int_0^1 \Re\left(e^{-i\theta} (2t - 1) f_0(t)\right) \, dt = \int_0^1 |2t - 1| |f_0(t)| \, dt
                \end{equation*}

                Como $|e^{-i\theta}|=1$ y la parte real de un número complejo es menor o igual que su módulo, se deduce que:
                \begin{equation*}
                    \Re\left(e^{-i\theta} (2t - 1) f_0(t)\right) \leq |2t - 1| |f_0(t)| \qquad \forall t \in [0, 1].
                \end{equation*}

                Por lo tanto, se tendría que:
                \begin{equation*}
                    \int_0^1 \left(|2t - 1| |f_0(t)| - \Re\left(e^{-i\theta} (2t - 1) f_0(t)\right)\right) \, dt = 0
                \end{equation*}

                Como se trata de una integral de una función continua y no negativa, se deduce que:
                \begin{equation*}
                    |2t - 1| |f_0(t)| = \Re\left(e^{-i\theta} (2t - 1) f_0(t)\right) \qquad \forall t \in [0, 1].
                \end{equation*}

                De aquí, se deduce que $e^{-i\theta} (2t - 1) f_0(t)$ es un número real no negativo para todo $t \in [0, 1]$, luego:
                \begin{equation*}
                    e^{-i\theta} f_0(t) = \frac{|2t - 1| |f_0(t)|}{2t - 1}
                    \Longrightarrow
                    f_0(t) = e^{i\theta} \frac{|2t - 1|}{2t - 1} \qquad \forall t \in [0, 1] \setminus \{\nicefrac{1}{2}\}.
                \end{equation*}

                Estudiemos la continuidad en $t = \nicefrac{1}{2}$:
                \begin{align*}
                    \lim_{t \to \nicefrac{1}{2}^-} f_0(t) &= \lim_{t \to \nicefrac{1}{2}^-} e^{i\theta} \frac{|2t - 1|}{2t - 1} = \lim_{t \to \nicefrac{1}{2}^-} e^{i\theta} \frac{-(2t - 1)}{2t - 1} = -e^{i\theta},\\
                    \lim_{t \to \nicefrac{1}{2}^+} f_0(t) &= \lim_{t \to \nicefrac{1}{2}^+} e^{i\theta} \frac{|2t - 1|}{2t - 1} = \lim_{t \to \nicefrac{1}{2}^+} e^{i\theta} \frac{2t - 1}{2t - 1} = e^{i\theta}.
                \end{align*}

                Como $f_0$ es continua, se tiene que $e^{i\theta} = -e^{i\theta}$. No existe ningún número complejo con módulo 1 que cumpla esta igualdad, por lo que se llega a una contradicción. Por lo tanto, $\varphi$ no alcanza su norma.                
            \end{itemize}
            \item ¿Existe $x^{**} \in X^{**}$ con $\|x^{**}\| = 1$ tal que $x^{**}(\varphi) = \|\varphi\|$? Justifique su respuesta.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}
        Rodee las afirmaciones verdaderas (que pueden ser una, dos, tres o cuatro):
        \begin{enumerate}[label=\alph*)]
            \item Sea $X$ un subespacio cerrado de $\ell_{3000}$ de dimensión infinita ¿Puede ser $X$ isomorfo a $\ell_1$?
            \begin{enumerate}[label={a\arabic*)}]
                \item Si, porque $\ell_1 \subseteq \ell_{3000}$
                \item No, porque $\ell_1 \nsubseteq \ell_{3000}$
                \item No, porque $\ell_1$ es separable y $X$ no lo es
                \item No, porque $X^*$ es separable y $\ell_1^*$ no lo es
            \end{enumerate}
            \item Sea $\psi \colon \ell_2 \to \bb{R}$ el funcional lineal dado por $\psi(x) = \sum\limits_{n=1}^\infty \frac{x(n)}{n}$ para $x \in \ell_2$:
            \begin{enumerate}[label={b\arabic*)}]
                \item $\psi$ alcanza su norma porque $\ell_2$ es reflexivo
                \item $\psi$ alcanza su norma porque $\ell_2$ es un espacio de Hilbert
                \item $\psi$ no alcanza su norma
                \item $\psi$ no alcanza su norma porque $\ell_2$ no es completo
            \end{enumerate}
            \item Sea $Y = \bb{K}^{(\bb{N})}$ (el espacio vectorial de las sucesiones casi nulas):
            \begin{enumerate}[label={c\arabic*)}]
                \item la norma de la suma y la norma del máximo son equivalentes en $Y$
                \item la norma de la suma y la norma del máximo no son equivalentes en $Y$
                \item cualesquiera dos normas definidas en $Y$ son equivalentes
                \item no se puede definir ninguna norma en $Y$
            \end{enumerate}

            Veamos que $\|\cdot\|_1$ y $\|\cdot\|_\infty$ no son equivalentes en $Y$. Para ello, fijado $n \in \bb{N}$, consideremos la sucesión:
            \begin{equation*}
                x_n = \sum_{k=1}^n e_k \in Y,
            \end{equation*}
            donde $e_k$ es la sucesión que tiene un 1 en la posición $k$ y 0 en el resto de posiciones. Claramente $x_n \in Y$ y se tiene que:
            \begin{equation*}
                \|x_n\|_1 = n \quad \text{y} \quad \|x_n\|_\infty = 1.
            \end{equation*}

            Por tanto, si $\|\cdot\|_1$ y $\|\cdot\|_\infty$ fueran equivalentes, existiría una constante $C > 0$ tal que:
            \begin{equation*}
                n = \|x_n\|_1 \leq C \|x_n\|_\infty = C \quad \forall n \in \bb{N}.
            \end{equation*}

            Esto es imposible, por lo que se concluye que $\|\cdot\|_1$ y $\|\cdot\|_\infty$ no son equivalentes en $Y$. Por lo tanto, la única afirmación verdadera es la \textbf{c2)}.

            \item En un espacio normado $X$ tomamos un conjunto convexo y cerrado $C$ y un punto $x_0 \notin C$:
            \begin{enumerate}[label={d\arabic*)}]
                \item existe $f \in X^*$ tal que $\sup \operatorname{Re} f(C) \leq \operatorname{Re} f(x_0)$
                \item existe $f \in X^*$ tal que $\sup \operatorname{Re} f(C) < \operatorname{Re} f(x_0)$
                \item no se puede asegurar que $C$ se pueda separar de $x_0$ si el interior de $C$ es vacío
                \item existe un único $f \in X^*$ tal que $\sup \operatorname{Re} f(C) \leq \operatorname{Re} f(x_0)$
            \end{enumerate}
            \item Sea $Z$ un espacio normado de dimensión finita:
            \begin{enumerate}[label={e\arabic*)}]
                \item cualquier aplicación lineal que sale de $Z$ es abierta
                \item cualquier aplicación lineal que llega a $Z$ es continua
                \item cualquier aplicación lineal sobreyectiva que llega a $Z$ es abierta
                \item cualquier aplicación lineal que sale de $Z$ es continua
            \end{enumerate}

            Suponemos que todos los espacios vectoriales son normados, pues en necesario considerar una topología para poder hablar de continuidad y de aplicaciones abiertas. En el caso de la asignatura de Análisis Funcional, la topología empleada, salvo que se indique lo contrario, es la topología inducida por la norma.
            \begin{itemize}
                \item Consideramos el operador nulo:
                \Func{T}{Z}{X}{z}{0}

                Este operador es lineal y no es abierto, por lo que la afirmación \textbf{e1)} es falsa.

                \item Se ha visto en teoría que, si $X$ es un espacio normado de dimensión infinita, entonces existe un funcional lineal $f \in X^*$ que no es continuo. Por lo tanto, la afirmación \textbf{e2)} es falsa.
                
                Veamos cuál podría ser un ejemplo de un funcional lineal no continuo. Consideremos el espacio normado $c_{00}$, que sabemos es de dimensión infinita y tiene una base canónica $\{e_n\}_{n \in \bb{N}}$. Definimos el funcional lineal $f \colon c_{00} \to \bb{K}$ por:
                \begin{align*}
                    f(e_n) = n \quad \forall n \in \bb{N}.
                \end{align*}

                Este funcional es lineal, pero no es continuo pues, para cada $n \in \bb{N}$, se tiene que $\|e_n\|_\infty = 1$ y $|f(e_n)| = n$, por lo que no existe una constante $C > 0$ tal que $|f(x)| \leq C \|x\|_\infty$ para todo $x \in c_{00}$. Como $ \dim \bb{K}=1$ finita, se deduce que $f$ es un funcional lineal no continuo que llega a un espacio de dimensión finita. Por lo tanto, la afirmación \textbf{e2)} es falsa.
                
                \item Dado un operador lineal $T \colon X\to Z$ con $Z$ de dimensión finita sobreyectivo se puede descomponer (algebraicamente) como $T=\tilde{T}\circ q$, donde:
                \begin{itemize}
                    \item $q\colon X\to X/\ker(T)$ es la aplicación proyección en el cociente, que es lineal.
                    \item $\tilde{T}\colon X/\ker(T)\to Z$ es un isomorfismo (algebraico) entre espacios vectoriales de dimensión finita. Como $\dim(X/\ker(T))=\dim(Z)$ es finita, $\tilde{T}$ es continua y abierta.
                \end{itemize}

                Si $T$ fuese continuo, entonces $\ker(T)$ sería cerrado y, por lo tanto, se le podría dotar al cociente $X/\ker(T)$ de la norma del cociente, con lo que $q$ sería abierta y, por lo tanto, $T$ sería abierta. No obstante, si $T$ no es continuo, entonces $\ker(T)$ no es cerrado y no se le puede dotar al cociente $X/\ker(T)$ de la norma del cociente, por lo que es necesario emplear otro enfoque.

                \item Por un corolario del Teorema de Hausdorff, todo operador lineal entre espacios normados en el que el dominio es de dimensión finita es continuo. Por lo tanto, la afirmación \textbf{e4)} es verdadera.
            \end{itemize}
        \end{enumerate}
    \end{ejercicio}
