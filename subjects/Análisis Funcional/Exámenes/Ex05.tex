    \begin{ejercicio}[1.5 puntos]
        Sea $X$ el espacio normado de las funciones continuas en $[0,1]$ que se anulan en cero con la norma del máximo:
        \begin{equation*}
            X = \{f\in C[0,1] : f(0) = 0\}, \qquad \|f\|_\infty = \max\{|f(t)| : 0\leq t \leq 1\}
        \end{equation*}
        En $X$ se considera el funcional lineal $\varphi:X\to \mathbb{R}$ dado por:
        \begin{equation*}
            \varphi(f) = \int_{0}^{1} tf(t)~dt \qquad \forall f\in X
        \end{equation*}
        \begin{enumerate}[label=\alph*)]
            \item \textbf{[1 punto]} Calcula $\|\varphi\|$.
            
            Veamos en primer lugar que dicho funcional es continuo. Para ello, sea $f\in X$. Entonces:
            \begin{equation*}
                |\varphi(f)| = \left|\int_0^1 tf(t)~dt\right| \leq \int_0^1 |tf(t)|~dt \leq \int_0^1 t\|f\|_\infty~dt = \|f\|_\infty \int_0^1 t~dt = \frac{1}{2}\|f\|_\infty
            \end{equation*}
            Por lo tanto, $\|\varphi\| \leq \nicefrac{1}{2}$. Para ver que $\|\varphi\| = \nicefrac{1}{2}$, consideremos la sucesión de funciones $\{f_n\}$ mostrada en la Figura~\ref{fig:funciones}. Cada función $f_n$ está definida por:
            \begin{equation*}
                f_n(t) = \begin{cases}
                    nt & \text{si } t\in \left[0, \nicefrac{1}{n}\right], \\
                    1 & \text{si } t\in \left[\nicefrac{1}{n}, 1\right]
                \end{cases}
            \end{equation*}
            \begin{figure}[h]
                \centering
                \begin{tikzpicture}
                    \begin{axis}[
                        axis lines=middle,
                        xmin=0, xmax=1.1,
                        ymin=0, ymax=1.2,
                        xtick={0, 0.5, 1},
                        xticklabels={$0$, $\nicefrac{1}{2}$, $1$},
                        ytick={0, 1},
                        yticklabels={$0$, $1$},
                        width=10cm,
                        height=6cm,
                        legend pos=south east
                    ]
                        \foreach \n/\col in {2/blue!70, 4/red!70, 8/green!60!black} {
                            \edef\temp{
                                \noexpand\addplot[thick, color=\col] coordinates {
                                    (0, 0)
                                    (1/\n, 1)
                                    (1, 1)
                                };
                                \noexpand\addlegendentry{$n=\n$}

                                % \draw[dashed, color=\col] (axis cs:1/\n,0) -- (axis cs:1/\n,1);
                            }
                            \temp
                        }
                        \draw[dashed, gray] (axis cs:0,1) -- (axis cs:1,1);
                    \end{axis}
                \end{tikzpicture}
                \caption{Funciones $f_n$ para $n=2,4,8$.}
                \label{fig:funciones}                
            \end{figure}

            Se verifica que $\|f_n\|_\infty = 1$ para todo $n$. Además, se tiene:
            \begin{align*}
                \varphi(f_n) &= \int_0^{\nicefrac{1}{n}} t(nt)~dt + \int_{\nicefrac{1}{n}}^1 t(1)~dt = \int_0^{\nicefrac{1}{n}} n t^2~dt + \int_{\nicefrac{1}{n}}^1 t~dt = n\left[\frac{t^3}{3}\right]_0^{\nicefrac{1}{n}} + \left[\frac{t^2}{2}\right]_{\nicefrac{1}{n}}^1 =\\&= \frac{1}{3n^2} + \left(\frac{1}{2} - \frac{1}{2n^2}\right) = \frac{1}{2} - \frac{1}{6n^2}.
            \end{align*}

            Por lo tanto:
            \begin{equation*}
                \|\varphi\| \geq \dfrac{|\varphi(f_n)|}{\|f_n\|_\infty} = \frac{1}{2} - \frac{1}{6n^2}\qquad \forall n\geq 2
            \end{equation*}
            Tomando el límite cuando $n\to\infty$, obtenemos que $\|\varphi\| \geq \nicefrac{1}{2}$. Combinando esto con la desigualdad anterior, concluimos que $\|\varphi\| = \nicefrac{1}{2}$.
            \item \textbf{[0.5 puntos]} Prueba que $\varphi$ no alcanza su norma.\\
            
            Supongamos que existe $f\in X$ tal que $\|f\|_\infty = 1$ y $|\varphi(f)| = \nicefrac{1}{2}$. Entonces:
            \begin{equation*}
                \frac{1}{2} = |\varphi(f)| = \left|\int_0^1 tf(t)~dt\right| \leq \int_0^1 |tf(t)|~dt \leq \int_0^1 t\|f\|_\infty~dt = \frac{1}{2}
            \end{equation*}
            Por lo tanto, todas las desigualdades deben ser iguales. En particular,
            \begin{equation*}
                \int_0^1 |tf(t)|~dt = \int_0^1 t\|f\|_\infty~dt \implies \int_0^1 t\left(\|f\|_\infty - |f(t)|\right)~dt = 0
            \end{equation*}
            Como $t\geq 0$ y $\|f\|_\infty - |f(t)| \geq 0$ para todo $t\in [0,1]$, y haciendo uso de que $f$ es continua, se deduce que:
            \begin{equation*}
                t\left(\|f\|_\infty - |f(t)|\right) = 0 \qquad \forall t\in [0,1]
            \end{equation*}
            Por lo tanto, para todo $t\in \left]0,1\right]$, se tiene que $|f(t)| = \|f\|_\infty = 1$. Puesto que $f$ es continua con $f(0) = 0$, y el módulo de $f$ es una función continua, se tiene que:
            \begin{equation*}
                0 = |f(0)| = \left|\lim_{t\to 0^+} f(t)\right| = \lim_{t\to 0^+} |f(t)| = \lim_{t\to 0^+} 1 = 1
            \end{equation*}

            Esto es una contradicción, por lo que no existe tal función $f$ y $\varphi$ no alcanza su norma.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}[2 puntos]
        Sea $T:c_0\to l_2$ el operador lineal definido para todo $x\in c_0$ por
        \begin{equation*}
            [Tx](n) = \frac{x(n)}{n}\qquad \forall n\in \mathbb{N}
        \end{equation*}
        \begin{enumerate}[label=\alph*)]
            \item \textbf{[1 punto]} Calcula su norma y prueba que $T$ no la alcanza.
            
            Tenemos que:
            \begin{equation*}
                \|Tx\|_2^2 = \sum_{n=1}^{\infty} \left|\frac{x(n)}{n}\right|^2 \leq \sum_{n=1}^{\infty} \frac{\|x\|_\infty^2}{n^2} = \|x\|_\infty^2 \sum_{n=1}^{\infty} \frac{1}{n^2}
            \end{equation*}

            Como $2>1$, la serie armónica dada es convergente:
            \begin{equation*}
                \sum_{n=1}^{\infty} \frac{1}{n^2} = M<\infty
            \end{equation*}

            De hecho, se sabe que $M = \nicefrac{\pi^2}{6}$, aunque no es necesario para el cálculo de la norma. Por lo tanto:
            \begin{equation*}
                \|Tx\|_2^2 \leq M\|x\|_\infty^2 \implies \|Tx\|_2 \leq \sqrt{M}\|x\|_\infty
            \end{equation*}

            Por lo tanto, $T$ es continuo y $\|T\| \leq \sqrt{M}$. Para ver que $\|T\| = \sqrt{M}$, consideremos la sucesión $\{x_n\}$ en $c_{00}\subset c_0$ dada por:
            \begin{equation*}
                x_n(k) = \begin{cases}
                    1 & \text{si } k\leq n, \\
                    0 & \text{si } k>n
                \end{cases}
            \end{equation*}

            Entonces $\|x_n\|_\infty = 1$ y:
            \begin{equation*}
                \|Tx_n\|_2^2 = \sum_{k=1}^{\infty} \left|\frac{x_n(k)}{k}\right|^2 = \sum_{k=1}^{n} \frac{1}{k^2}
            \end{equation*}

            Por lo tanto:
            \begin{equation*}
                \|T\| \geq \frac{\|Tx_n\|_2}{\|x_n\|_\infty} = \sqrt{\sum_{k=1}^{n} \frac{1}{k^2}} \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Tomando el límite cuando $n\to\infty$, puesto que la raíz cuadrada es continua, obtenemos:
            \begin{equation*}
                \|T\| \geq \sqrt{\sum_{k=1}^{\infty} \frac{1}{k^2}} = \sqrt{M}
            \end{equation*}

            Combinando esto con la desigualdad anterior, concluimos que $\|T\| = \sqrt{M}$.\\

            Veamos ahora que $T$ no alcanza su norma. Supongamos que existe $x\in c_0$ tal que $\|x\|_\infty = 1$ y $\|Tx\|_2 = \sqrt{M}$. Entonces:
            \begin{equation*}
                \sqrt{M} = \|Tx\|_2 = \sqrt{\sum_{n=1}^{\infty} \left|\frac{x(n)}{n}\right|^2} \leq \sqrt{\sum_{n=1}^{\infty} \frac{\|x\|_\infty^2}{n^2}} = \sqrt{M}
            \end{equation*}

            Por tanto, todas las desigualdades deben ser iguales. En particular:
            \begin{equation*}
                \sum_{n=1}^{\infty} \left|\frac{x(n)}{n}\right|^2 = \sum_{n=1}^{\infty} \frac{\|x\|_\infty^2}{n^2} \implies |x(n)| = \|x\|_\infty = 1 \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Esto es una contradicción, ya que $x\in c_0$ y por tanto $\{x(n)\}$ debe tender a cero. Por lo tanto, $T$ no alcanza su norma.

            \item \textbf{[0.5 punos]} Prueba que $Y=T(c_0)$ es un subespacio denso en $l_2$ y que $T$ es una biyección lineal de $c_0$ sobre $Y$.
            
            Veamos en primer lugar que $T$ es una biyección lineal de $c_0$ sobre $Y$. La linealidad de $T$ es inmediata por la definición de $T$. Además, es claro que $T$ es sobreyectiva de $c_0$ sobre $Y=T(c_0)$ por definición de $Y$. Por último, veamos que $T$ es inyectiva. Supongamos que $T(x)=0$ para algún $x\in c_0$. Entonces:
            \begin{equation*}
                T(x)(n) = \frac{x(n)}{n} = 0 \qquad \forall n\in \mathbb{N}
            \end{equation*}
            Por lo tanto, $x(n) = 0$ para todo $n\in \mathbb{N}$, es decir, $x=0$. Por lo tanto, $T$ es inyectiva y por tanto una biyección lineal de $c_0$ sobre $Y$. Veamos ahora que $Y$ es denso en $l_2$. Sea $y\in l_2$, y queremos ver que $y\in \overline{Y}$, lo que equivale a ver que existe una sucesión $\{y_n\}\subset Y=T(c_0)$ tal que $\{y_n\}\to y$ en $l_2$. Consideremos la sucesión $\{x_n\}\subset c_{00}\subset c_0$ dada por:
            \begin{equation*}
                x_n(k) = \begin{cases}
                    k y(k) & \text{si } k\leq n, \\
                    0 & \text{si } k>n
                \end{cases}
            \end{equation*}
            
            Consideramos ahora la sucesión $\{y_n\}\subset Y$ dada por $y_n = T(x_n)$. Entonces:
            \begin{equation*}
                y_n(k) = T(x_n)(k) = \frac{x_n(k)}{k} = \begin{cases}
                    y(k) & \text{si } k\leq n, \\
                    0 & \text{si } k>n
                \end{cases}
            \end{equation*}

            Por lo tanto, $\{y_n\}$ es la sucesión de truncamientos de $y$. Veamos ahora que $\{y_n\}\to y$ en $l_2$. Para ello, consideremos:
            \begin{equation*}
                \|y_n - y\|_2^2 = \sum_{k=1}^{\infty} |y_n(k) - y(k)|^2 = \sum_{k=n+1}^{\infty} |y(k)|^2
            \end{equation*}

            Como $y\in l_2$, se tiene que $\sum\limits_{k=1}^{\infty} |y(k)|^2 < \infty$, y por tanto:
            \begin{equation*}
                \lim_{n\to\infty} \|y_n - y\|_2^2 = \lim_{n\to\infty} \sum_{k=n+1}^{\infty} |y(k)|^2 = 0
            \end{equation*}

            Esto prueba que $\{y_n\}\to y$ en $l_2$, y por tanto $y\in \overline{Y}$. Como $y\in l_2$ era arbitrario, se concluye que $\overline{Y} = l_2$, es decir, $Y$ es denso en $l_2$.

            \item \textbf{[0.5 puntos]} ¿Es continua la aplicación $T^{-1}:Y\to c_0$? ¿Es $Y$ cerrado en $l_2$?
            
            Veamos que $T^{-1}$ no es continua. Esta aplicación está definida por:
            \begin{equation*}
                T^{-1}(y)(n) = n y(n) \qquad \forall y\in Y, \forall n\in \mathbb{N}
            \end{equation*}

            Consideremos la sucesión $\{ne_n\}\subset c_{00}\subset c_0$, donde $e_n$ es la sucesión canónica de $l_2$. Entonces, $T(ne_n) = e_n \in Y$, luego consideramos la sucesión $\{e_n\}\subset Y$.
            \begin{align*}
                \|T^{-1}(e_n)\|_\infty = \|ne_n\|_\infty = n
            \end{align*}

            Por otro lado, $\|e_n\|_2 = 1$ para todo $n$. Por lo tanto, $\{e_n\}$ es una sucesión acotada en $Y$, pero $\{T^{-1}(e_n)\}$ no es acotada en $c_0$. Por lo tanto, $T^{-1}$ no es continua.\\
            
            Si $T^{-1}$ fuese continua, entonces $Y$ sería isomorfo a $c_0$, y por tanto $Y$ sería un espacio de Banach, luego $Y$ sería cerrado en $l_2$. Como $T^{-1}$ no es continua, no podemos concluir que nada de forma directa. Veamos que $Y$ no es cerrado en $l_2$. Como $Y$ es denso en $l_2$, entonces $Y$ es cerrado si y solo si $Y = l_2$. Sabemos que $Y\subset l_2$, pero veamos que $Y\neq l_2$. Consideremos la sucesión $y\in l_2$ dada por:
            \begin{equation*}
                y(n) = \frac{1}{n} \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Entonces $y\in l_2$, pero supongamos que $y\in Y$. Entonces existe $x\in c_0$ tal que $T(x) = y$, es decir:
            \begin{equation*}
                \frac{x(n)}{n} = \frac{1}{n} \implies x(n) = 1 \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Esto es una contradicción, ya que $x\in c_0$ y por tanto $\{x(n)\}$ debe tender a cero. Por lo tanto, $y\notin Y$, y por tanto $Y\neq l_2$. Como $Y$ es denso en $l_2$, se concluye que $Y$ no es cerrado en $l_2$.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}[1.5 puntos]
        Sean $X$ un espacio de Banach, $Y$ un espacio normado y $\cc{F}$ un subconjunto de $L(X,Y)$. Prueba que las siguientes afirmaciones son equivalentes:
        \begin{enumerate}[label=\alph*)]
            \item $\cc{F}$ está acotado.
            \item Para cada $x\in X$ y cada $g\in Y^\ast$ el conjunto $\{g(T(x)) : T\in \cc{F}\}$ está acotado.
        \end{enumerate}
        \begin{description}
            \item[$\Longrightarrow$)] Supongamos que $\cc{F}$ está acotado. Entonces, existe $M>0$ tal que:
            \begin{equation*}
                \|T\| \leq M \qquad \forall T\in \cc{F}
            \end{equation*}

            Sea $x\in X$ y $g\in Y^\ast$. Entonces, para cada $T\in \cc{F}$, por la continuidad de $g$ y la definición de la norma del funcional, se tiene:
            \begin{equation*}
                |g(T(x))| \leq \|g\|\cdot \|T(x)\|
            \end{equation*}

            Como $T\in \cc{F}$ es continuo, se tiene que:
            \begin{equation*}
                \|T(x)\| \leq \|T\|\cdot \|x\|
            \end{equation*}

            Como $\|T\| \leq M$, se concluye que:
            \begin{equation*}
                |g(T(x))| \leq \|g\|\cdot M \cdot \|x\|
            \end{equation*}

            Como $M, x, g$ son fijos, el conjunto $\{g(T(x)) : T\in \cc{F}\}$ está acotado.
        \end{description}
    \end{ejercicio}
