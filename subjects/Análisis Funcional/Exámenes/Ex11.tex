    \begin{ejercicio}
        En el espacio de Banach $C[0,1]$ se considera el subespacio
        \begin{equation*}
            X = \{f\in C[0,1] : f(0) = f(1)\}
        \end{equation*}
        en el que se define el funcional lineal $\varphi:X\to \bb{K}$ dado por
        \begin{equation*}
            \varphi(f) = \int_{0}^{1} t^8 f(t)~dt  \qquad \forall f\in X
        \end{equation*}
        \begin{enumerate}[label=(\alph*)]
            \item Probar que $X$ es completo.            
            \begin{description}
                \item[De manera rutinaria] Sea $\{f_n\}_{n\in \mathbb{N}}$ una sucesión de Cauchy en $X$. Como $C[0,1]$ es completo, existe $f\in C[0,1]$ tal que $\{f_n\}\to f$ empleando la norma del supremo. Faltaría por tanto ver que $f\in X$. Como se tiene que $f_n(0) = f_n(1)$ para todo $n\in \mathbb{N}$, tenemos que:
                \begin{equation*}
                    |f(0) - f(1)| = |f(0) - f_n(0) + f_n(1) - f(1)| \leq |f(0) - f_n(0)| + |f_n(1) - f(1)| \leq 2\|f - f_n\|
                \end{equation*}
                Tomando el límite cuando $n\to \infty$ obtenemos que $f(0) = f(1)$, por lo que $f\in X$ y $\{f_n\}\to f$ en $X$. Por tanto, $X$ es completo.

                \item[Otra manera] Queremos ver que $X$ es cerrado en $C[0,1]$. Consideramos el siguiente funcional lineal $\psi:C[0,1]\to \bb{K}$ dado por:
                \begin{equation*}
                    \psi(f) = f(0) - f(1) \qquad \forall f\in C[0,1]
                \end{equation*}
                Es fácil ver que $\psi$ es lineal y continuo, ya que:
                \begin{equation*}
                    |\psi(f)| = |f(0) - f(1)| \leq |f(0)| + |f(1)| \leq 2\|f\| \qquad \forall f\in C[0,1]
                \end{equation*}
                Por tanto, $\psi$ es continuo y $X = \ker(\psi)$, por lo que $X$ es cerrado en $C[0,1]$. Como $C[0,1]$ es completo, $X$ también lo es.
            \end{description}
            
            \item Probar que $\varphi$ es continuo y calcular su norma.            
            \begin{align*}
                |\varphi(f)| &= \left|\int_{0}^{1} t^8 f(t)~dt\right| \leq \int_{0}^{1} |t^8 f(t)|~dt = \int_{0}^{1} t^8 |f(t)|~dt \leq \int_{0}^{1} t^8 \|f\|~dt =\\&= \|f\|\int_{0}^{1} t^8~dt = \frac{1}{9}\|f\|
            \end{align*}

            Por tanto, $\varphi$ es continuo y $\|\varphi\|\leq \nicefrac{1}{9}$. Para ver que $\|\varphi\| = \nicefrac{1}{9}$ consideramos la función $f_0(t) = 1$ para todo $t\in [0,1]$. Tenemos que $f_0\in X$ y $\|f_0\| = 1$, así como que:
            \begin{equation*}
                |\varphi(f_0)| = \left|\int_{0}^{1} t^8~dt\right| = \frac{1}{9}
            \end{equation*}
            Por tanto, $\|\varphi\| \geq \nicefrac{1}{9}$, y como ya habíamos visto que $\|\varphi\|\leq \nicefrac{1}{9}$, obtenemos que $\|\varphi\| = \nicefrac{1}{9}$.
            \item ¿Alcanza $\varphi$ su norma?
            
            Ya hemos visto que, empleando la función $f_0(t) = 1$ para todo $t\in [0,1]$, se alcanza la norma de $\varphi$. Por tanto, $\varphi$ alcanza su norma.
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}
        En el espacio $C[0,1]$ se define
        \begin{equation*}
            F(x) = \int_{0}^{1} t^{15}x(t^{21})~dt 
        \end{equation*}
        para cada $x\in C[0,1]$. Prueba que $F$ es un funcional lineal y continuo y calcula su norma. ¿La alcanza?\\

        Claramente, $F: C[0,1]\to \bb{K}$ es un funcional. Veamos que es lineal. Para ello, sean $x,y\in C[0,1]$ y $\alpha\in \bb{K}$. Tenemos que:
        \begin{align*}
            F(\alpha x + y) &= \int_{0}^{1} t^{15}(\alpha x(t^{21}) + y(t^{21}))~dt = \int_{0}^{1} t^{15}\alpha x(t^{21})~dt + \int_{0}^{1} t^{15}y(t^{21})~dt =\\&= \alpha\int_{0}^{1} t^{15}x(t^{21})~dt + \int_{0}^{1} t^{15}y(t^{21})~dt = \alpha F(x) + F(y)
        \end{align*}
        Por tanto, $F$ es lineal. Veamos que es continuo. Para ello, sea $x\in C[0,1]$. Tenemos que:
        \begin{align*}
            |F(x)| &= \left|\int_{0}^{1} t^{15} x(t^{21})~dt\right| \leq \int_{0}^{1} |t^{15} x(t^{21})|~dt = \int_{0}^{1} t^{15} |x(t^{21})|~dt \leq \int_{0}^{1} t^{15} \|x\|~dt =\\&= \|x\|\int_{0}^{1} t^{15}~dt = \frac{1}{16}\|x\|
        \end{align*}
        Por tanto, $F$ es continuo y $\|F\|\leq \nicefrac{1}{16}$. Para ver que $\|F\| = \nicefrac{1}{16}$ consideramos la función $x_0(t) = 1$ para todo $t\in [0,1]$. Tenemos que $x_0\in C[0,1]$ y $\|x_0\| = 1$, así como que:
        \begin{equation*}
            |F(x_0)| = \left|\int_{0}^{1} t^{15}~dt\right| = \frac{1}{16}
        \end{equation*}
        Por tanto, $\|F\| \geq \nicefrac{1}{16}$, y como ya habíamos visto que $\|F\|\leq \nicefrac{1}{16}$, obtenemos que $\|F\| = \nicefrac{1}{16}$. Además, $F$ alcanza su norma, ya que $|F(x_0)| = \|F\|$.
    \end{ejercicio}

    \begin{ejercicio}
        Explica si las siguientes afirmaciones son ciertas o falsas. Cuando sean ciertas, indica el resultado de teoría que lo justifique o proporciona una prueba; cuando sean falsas, indica un contraejemplo.
        \begin{enumerate}
            \item Un espacio normado tiene dimensión finita si y sólo si todos sus subespacios vectoriales son cerrados.
            
            Veamos que es cierta.
            \begin{description}
                \item[$\Longrightarrow$)] Si $X$ es un espacio normado de dimensión finita, entonces por el Teorema de Hausdorff todo subespacio vectorial de $X$ es cerrado.
                
                \item[$\Longleftarrow$)] Supongamos que $X$ es un espacio normado de dimensión infinita, y buscamos un subespacio vectorial que no sea cerrado. Para ello, puesto que un funcional lineal $T: X\to \bb{K}$ es continuo si y sólo si su núcleo $\ker(T)$ es cerrado, buscamos un funcional $T: X\to \bb{K}$ lineal que no sea continuo, lo que nos afirmará que $\ker(T)$ no es cerrado.\\
                
                Como $X$ es de dimensión infinita, tenemos garantizada (se vió en la teoría) la existencia de un funcional lineal $T: X\to \bb{K}$ que no sea continuo. Por tanto, $\ker(T)$ no es cerrado, y hemos encontrado un subespacio vectorial de $X$ que no es cerrado. No obstante, veamos una construcción explícita de un funcional lineal que no sea continuo.\\
                
                Como $X$ es de dimensión infinita, consideramos su base algebraica $\cc{B}$, que será infinita pero no necesariamente numerable, y que consideramos normalizada. Consideramos un subconjunto numerable de $\cc{B}$, que podemos denotar por $\{u_n\}_{n\in \mathbb{N}}$, que es linealmente independiente. Definimos entonces el funcional lineal $T: X\to \bb{K}$ definido en su base por:
                \begin{align*}
                    T(u_n) &= n \qquad \forall n\in \mathbb{N}\\
                    T(u) &= 0 \qquad \forall u\in \cc{B}\setminus \{u_n\}_{n\in \mathbb{N}}
                \end{align*}

                Este funcional es lineal, pero no es continuo. Veámoslo:
                \begin{equation*}
                    \|T(u_n)\| = |T(u_n)| = n = n\|u_n\| \qquad \forall n\in \mathbb{N}
                \end{equation*}

                Por tanto, como esta igualdad es válida para todo $n\in \mathbb{N}$, no existe una constante $C>0$ tal que $\|T(x)\|\leq C\|x\|$ para todo $x\in X$, por lo que $T$ no es continuo. Por tanto, $\ker(T)$ no es cerrado, y hemos encontrado un subespacio vectorial de $X$ que no es cerrado.\\

                Por el recíproco, si todos los subespacios vectoriales de $X$ son cerrados, entonces $X$ tiene dimensión finita.
            \end{description}
            \item Si $X$ e $Y$ son espacios de Banach y la dimension de $Y$ es finita, entonces todo operador lineal de $X$ en $Y$ es continuo.
            
            Este resultado no es cierto. Dos resultados similares que sí son ciertos son:
            \begin{itemize}
                \item Si $X$ e $Y$ son espacios normados e $Y$ tiene dimensión finita, entonces un operador lineal de $X$ en $Y$ es continuo si y sólo si su núcleo es cerrado.
                \item Si $X$ e $Y$ son espacios normados y $X$ tiene dimensión finita, entonces todo operador lineal de $X$ en $Y$ es continuo.
            \end{itemize}

            No obstante, el resultado que se nos pide demostrar es falso, y buscamos un contraejemplo que, necesariamente, tendrá que tener $\dim(X) = \infty$ y, además, $\dim(Y) < \infty$. Consideramos $X = c_0\subset l_\infty$ e $Y=\bb{R}$. Partimos de la base algebraica de $c_{00}$, dada por $\{e_n\}_{n\in \mathbb{N}}$, y consideramos el funcional $T: c_{00}\to \bb{R}$ definido por:
            \begin{equation*}
                T(e_n) = n \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Este funcional es lineal, pero no es continuo. Veámoslo:
            \begin{equation*}
                \|T(e_n)\| = |T(e_n)| = n = n\|e_n\| \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Por tanto, como esta igualdad es válida para todo $n\in \mathbb{N}$, no existe una constante $C>0$ tal que $\|T(x)\|\leq C\|x\|$ para todo $x\in c_{00}$, por lo que $T$ no es continuo.\\

            Como $c_{00}$ no es de Banach, lo extendemos a $c_0$. Por el Lema de Zorn, se puede extender $\{e_n\}_{n\in \mathbb{N}}$ a una base algebraica de $c_0$, que denotamos por $\cc{B}$:
            \begin{equation*}
                \cc{B} = \{e_n\}_{n\in \mathbb{N}}\cup \cc{B}'\qquad \cc{B}'\subset c_0\qquad \cc{B}'\cap \{e_n\}_{n\in \mathbb{N}} = \emptyset
            \end{equation*}

            Definimos entonces el funcional lineal $T: c_0\to \bb{R}$ definido en su base por:
            \begin{align*}
                T(e_n) &= n \qquad \forall n\in \mathbb{N}\\
                T(u) &= 0 \qquad \forall u\in \cc{B}'
            \end{align*}

            Este funcional es lineal, pero no es continuo por la misma razón que antes. Por tanto, hemos encontrado un operador lineal de $c_0$ en $\bb{R}$ que no es continuo, y el resultado es falso.


            \item Sea $X$ un espacio de Banach. $X$ es separable si y sólo si todos sus subespacios cerrados son separables.
            \begin{description}
                \item[$\Longrightarrow$)] Si $X$ es separable, como $X$ es un espacio métrico entonces la propiedad de ser separable es hereditaria, por lo que todo subconjunto de $X$ es separable, y en particular todo subespacio cerrado de $X$ es separable.
                \item[$\Longleftarrow$)] Si todos los subespacios cerrados de $X$ son separables, entonces $X$ es separable, ya que $X$ es un subespacio cerrado de sí mismo.
            \end{description}
            \item El conjunto de todos los polinomios con coeficientes reales no es abierto ni es cerrado en $C[0,1]$.
            
            Veamos que no es abierto ni cerrado.
            \begin{itemize}
                \item Por el Teorema de Aproximación de Weierstrass, el conjunto de todos los polinomios con coeficientes reales es denso en $C[0,1]$. Como hay funciones continuas que no son polinomios, el conjunto de todos los polinomios con coeficientes reales no es cerrado. 
                
                \item Para ver que no es abierto, consideremos el polinomio $p_0(t) = 0$ para todo $t\in [0,1]$. Fijado $\varepsilon > 0$, consideremos la función:
                \Func{g_\varepsilon}{[0,1]}{\bb{R}}{t}{\veps\cdot \left|t-\frac{1}{2}\right|}

                Claramente, $g_\varepsilon\in C[0,1]$ y:
                \begin{equation*}
                    \|g_\varepsilon - p_0\| = \sup_{t\in [0,1]} |g_\varepsilon(t) - p_0(t)| = \sup_{t\in [0,1]} |g_\varepsilon(t)| = \sup_{t\in [0,1]} \veps\cdot \left|t-\frac{1}{2}\right| = \frac{\veps}{2} < \varepsilon
                \end{equation*}

                Por tanto, $g_\varepsilon\in B(p_0,\varepsilon)$. No obstante, $g_\varepsilon$ no es un polinomio pues no es diferenciable en $t = \nicefrac{1}{2}$. Por tanto, para todo $\varepsilon > 0$, existe $g_\varepsilon\in B(p_0,\varepsilon)$ tal que $g_\varepsilon$ no es un polinomio. Por tanto, el conjunto de todos los polinomios con coeficientes reales no es abierto.
            \end{itemize}
            \item El interior de un subconjunto compacto de un espacio de Banach es vacío.
            
            Si la dimensión del espacio de Banach fuese infinita, efectivamente el interior de un subconjunto compacto sería vacío. No obstante, si la dimensión del espacio de Banach fuese finita, el interior de un subconjunto compacto podría no ser vacío. Consideramos $\bb{R}^n$ con cualquiera de sus normas (pues todas son equivalentes). Consideramos su bola unidad cerrada, que es un subconjunto compacto de $\bb{R}^n$ por ser cerrado y acotado. Su interior es la bola unidad abierta, que no es vacío por contener al origen. Por tanto, el resultado es falso.
            \item El subespacio
                \begin{equation*}
                    X = \{f\in C[-1,1] : f(t) = f(-t) \quad \forall t\in [0,1]\}
                \end{equation*}
                considerado como subespacio del espacio $C[-1,1]$ con la norma del supremo es denso en $C[-1,1]$.\\

                Veamos que es falso. Fijado $t\in [0,1]$, consideramos el funcional:
                \Func{\Phi_t}{C[-1,1]}{\bb{K}}{f}{f(t) - f(-t)}

                Claramente, $\Phi_t$ es lineal, y además es continuo, ya que:
                \begin{equation*}
                    |\Phi_t(f)| = |f(t) - f(-t)| \leq |f(t)| + |f(-t)| \leq 2\|f\|
                \end{equation*}

                Por tanto, $\Phi_t$ es lineal y continuo. Además:
                \begin{align*}
                    \ker(\Phi_t) &= \{f\in C[-1,1] : \Phi_t(f) = 0\} = \{f\in C[-1,1] : f(t) - f(-t) = 0\} =\\&= \{f\in C[-1,1] : f(t) = f(-t)\}
                \end{align*}
                
                Por tanto:
                \begin{equation*}
                    X=\bigcap_{t\in [0,1]} \ker(\Phi_t)
                \end{equation*}

                Como la intersección arbitraria de cerrados es cerrado, $X$ es cerrado en $C[-1,1]$. Por tanto, puesto que $X\neq C[-1,1]$, $X$ no puede ser denso en $C[-1,1]$.
            \item En un espacio normado no completo, el único funcional lineal y continuo que alcanza su norma es el cero.\\
            
            Veamos que es falso. Consideremos el espacio normado $X = c_{00}$, que no es completo. Consideremos el funcional lineal $\varphi: c_{00}\to \bb{K}$ dado por:
            \begin{equation*}
                \varphi(x) = x(1) \qquad \forall x\in c_{00}
            \end{equation*}

            Veamos que $\varphi$ es lineal y continuo. Para ello, sea $x,y\in c_{00}$ y $\alpha\in \bb{K}$. Tenemos que:
            \begin{align*}
                \varphi(\alpha x + y) &= (\alpha x + y)(1) = \alpha x(1) + y(1) = \alpha \varphi(x) + \varphi(y)
            \end{align*}

            Por tanto, $\varphi$ es lineal. Además, para ver que es continuo, sea $x\in c_{00}$. Tenemos que:
            \begin{equation*}
                |\varphi(x)| = |x(1)| \leq \sup_{n\in \mathbb{N}} |x(n)| = \|x\|
            \end{equation*}
            Por tanto, $\varphi$ es continuo. Veamos que alcanza su norma. Consideremos $e_1\in c_{00}$, que es el elemento de la base canónica de $c_{00}$ definido por:
            \begin{equation*}
                e_1(n) = \begin{cases}
                    1 & \text{si } n = 1\\
                    0 & \text{si } n\neq 1
                \end{cases}
            \end{equation*}

            Claramente, $\|e_1\| = 1$, y además:
            \begin{equation*}
                |\varphi(e_1)| = |e_1(1)| = 1
            \end{equation*}

            Por tanto, $\varphi$ alcanza su norma, y hemos encontrado un funcional lineal y continuo que alcanza su norma en un espacio normado no completo que no es el cero. Por tanto, el resultado es falso.
            \item Dos hiperplanos cualesquiera de un espacio normado son isométricamente isomorfos.
            
            \begin{description}
                \item[Opción 1] Consideramos el espacio producto $\bb{R}^2\times \bb{R}$, donde:
                \begin{equation*}
                    \|((x,y), z)\| = \max\{\|(x,y)\|_2, |z|\}= \max\{\sqrt{x^2 + y^2}, |z|\} \qquad \forall (x,y,z)\in \bb{R}^2\times \bb{R}
                \end{equation*}

                Consideramos los hiperplanos:
                \begin{align*}
                    H_1 &= \{((x,y), z)\in \bb{R}^2\times \bb{R} : z = 0\}\\
                    H_2 &= \{((x,y), z)\in \bb{R}^2\times \bb{R} : y = 0\}
                \end{align*}

                Claramente, $H_1$ y $H_2$ son hiperplanos de $\bb{R}^2\times \bb{R}$.
                \begin{itemize}
                    \item Calculemos la norma en $H_1$. Sea $((x,y), z)\in H_1$. Tenemos que $z = 0$, por lo que:
                    \begin{equation*}
                        \|((x,y), z)\| = \max\{\sqrt{x^2 + y^2}, |z|\} = \max\{\sqrt{x^2 + y^2}, 0\} = \sqrt{x^2 + y^2}
                    \end{equation*}
                    Por tanto, $H_1$ es isométricamente isomorfo a $\left(\bb{R}^2, \|\cdot\|_2\right)$, con isomorfismo isométrico dado por:
                    \Func{\Phi_1}{H_1}{\bb{R}^2}{((x,y), 0)}{(x,y)}

                    \item Calculemos la norma en $H_2$. Sea $((x,y), z)\in H_2$. Tenemos que $y = 0$, por lo que:
                    \begin{equation*}
                        \|((x,y), z)\| = \max\{\sqrt{x^2 + y^2}, |z|\} = \max\{\sqrt{x^2 + 0^2}, |z|\} = \max\{|x|, |z|\}
                    \end{equation*}
                    Por tanto, $H_2$ es isométricamente isomorfo a $\left(\bb{R}^2, \|\cdot\|_\infty\right)$, con isomorfismo isométrico dado por:
                    \Func{\Phi_2}{H_2}{\bb{R}^2}{((x,0), z)}{(x,z)}
                \end{itemize}

                Como $\left(\bb{R}^2, \|\cdot\|_2\right)$ y $\left(\bb{R}^2, \|\cdot\|_\infty\right)$ no son isométricamente isomorfos, los hiperplanos $H_1$ y $H_2$ no son isométricamente isomorfos. Por tanto, el resultado es falso.
                
                \item[Opción 2] Consideremos el espacio $\left(\bb{R}^3, \|\cdot\|_\infty\right)$, cuya bola unitaria es el cubo de lado $2$ centrado en el origen. Consideremos los hiperplanos:
                \begin{align*}
                    H_1 &= \{(x,y,z)\in \bb{R}^3 : z = 0\}\\
                    H_2 &= \{(x,y,z)\in \bb{R}^3 : x+y+z = 0\}
                \end{align*}

                Claramente, $H_1$ y $H_2$ son hiperplanos de $\bb{R}^3$. Veamos que no son isométricamente isomorfos.
                \begin{itemize}
                    \item En $H_1$, su bola unitaria es el cuadrado de lado $2$ centrado en el origen, luego tiene 4 vértices.
                    \item En $H_2$, su bola unitaria es un hexágono regular, luego tiene 6 vértices.                    
                \end{itemize}

                Como el número de vértices de la bola unitaria es un invariante bajo isometrías, $H_1$ y $H_2$ no son isométricamente isomorfos. Por tanto, el resultado es falso.
            \end{description}

            \item Sea $X$ un espacio de Banach. Si todo hiperplano cerrado de $X$ es proximinal, entonces $X$ tiene dimensión finita.
            
            Si todo hiperplano cerrado de $X$ es proximinal, como todo hiperplano cerrado $H\subset X$ es de la forma $H = \ker(\varphi)$ para algún funcional lineal y continuo $\varphi\in X^*$, tenemos que $\ker(\varphi)$ es proximinal y, por tanto, $\varphi$ alcanza su norma. No obstante, no tenemos que $\dim X$ sea finita. Veamos que puede darse que la dimensión de $X$ sea infinita.\\

            Para ver un contraejemplo, sea el espacio de Banach $X = l_2 = \left(\bb{R}^\bb{N}, \|\cdot\|_2\right)$. Este espacio es de dimensión infinita, y veamos que todo funcional lineal y continuo $\varphi\in l_2^*$ alcanza su norma. Para ello, consideremos $\varphi\in l_2^*$. Como $l_2^*$ es isométricamente isomorfo a $l_2$, existe $y\in l_2$ tal que:
            \begin{equation*}
                \varphi(x) = \sum_{n=1}^{\infty} x(n)y(n) \qquad \forall x\in l_2
            \end{equation*}

            Consideramos ahora el vector $u\in l_2$ normalizado dado por:
            \begin{equation*}
                u(n) = \frac{y(n)}{\|y\|} \qquad \forall n\in \mathbb{N}
            \end{equation*}

            Claramente, $\|u\| = 1$, y además:
            \begin{align*}
                |\varphi(u)| &= \left|\sum_{n=1}^{\infty} u(n)y(n)\right| = \left|\sum_{n=1}^{\infty} \frac{y(n)}{\|y\|}y(n)\right| = \frac{1}{\|y\|}\left|\sum_{n=1}^{\infty} y(n)^2\right| = \frac{1}{\|y\|}\|y\|^2 = \|y\|
            \end{align*}

            Por tanto, $\varphi$ alcanza su norma, y como $\varphi$ era arbitrario, todo funcional lineal y continuo de $l_2$ alcanza su norma. Por tanto, todo hiperplano cerrado de $l_2$ es proximinal, y como $\dim(l_2) = \infty$, el resultado es falso.\\

            De hecho, en un ejercicio de las relaciones se probó que, para $1< p < \infty$, todo funcional lineal y continuo de $l_p$ alcanza su norma, y como $\dim(l_p) = \infty$, el resultado es falso para todo $1<p<\infty$.

            \item En el espacio vectorial $Z$ de los polinomios reales de grado menor o igual que $N\in \mathbb{N}$, consideramos la norma
                \begin{equation*}
                    \|p\| = \sup\{|p(t)| : t\in [0,1]\} \qquad \forall p\in Z
                \end{equation*}
                Entonces, para cada $k\in \mathbb{N}$, existe una constante $M_k>0$ tal que
                \begin{equation*}
                    |p(k)| \leq M_k\|p\| \qquad \forall p\in Z
                \end{equation*}
                ¿puede haber una constante $M$ que valga para todo $k\in \mathbb{N}$?\\

                La primera afirmación es verdadera. Para ver que es cierta, consideramos el funcional de evaluación en $k\in \mathbb{N}$:
                \Func{\delta_k}{Z}{\bb{K}}{p}{p(k)}

                Este funcional es lineal, pero su acotación no es trivial, pues:
                \begin{equation*}
                    |\delta_k(p)| = |p(k)|
                \end{equation*}

                Como $k\in \mathbb{N}$, no tenemos una cota inmediata. No obstante, como se tiene $\dim(Z) = N+1 < \infty$, tenemos que $\delta_k$ es inmediatamente continuo, y por tanto existe una constante $M_k>0$ (que desconocemos) tal que:
                \begin{equation*}
                    |\delta_k(p)| = |p(k)| \leq M_k\|p\| \qquad \forall p\in Z
                \end{equation*}

                No obstante, la segunda afirmación es falsa. Consideramos el polinomio dado por $p(t) = t$. Claramente, $p\in Z$ y $\|p\| = \sup\limits_{t\in [0,1]} |p(t)| = 1$. No obstante, para todo $k\in \mathbb{N}$, tenemos que:
                \begin{equation*}
                    |p(k)| = |k| = k
                \end{equation*}

                Por tanto, no existe una constante $M>0$ tal que $|p(k)|\leq M\|p\|$ para todo $k\in \mathbb{N}$, y la segunda afirmación es falsa.
        \end{enumerate}
    \end{ejercicio}
