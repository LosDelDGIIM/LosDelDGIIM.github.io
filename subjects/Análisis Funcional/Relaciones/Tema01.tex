\section{Espacios normados}

\subsection{Cuestiones básicas}

\begin{ejercicio}
Probar que dos normas en un mismo espacio vectorial son equivalentes si, y sólo si, dan lugar a los mismos conjuntos acotados.
\begin{description}
    \item[$\Longrightarrow$)] Sea $X$ un espacio vectorial y $\|\cdot\|_1$ y $\|\cdot\|_2$ dos normas equivalentes en $X$. Entonces existen constantes $C_1, C_2 \in \bb{R}^+$ tales que
    \[
    C_1 \|x\|_1 \leq \|x\|_2 \leq C_2 \|x\|_1 \quad \forall x \in X
    \]
    Sea $A \subset X$ un conjunto acotado en la norma $\|\cdot\|_1$. Entonces existe $M \in \bb{R}^+$ tal que $\|x\|_1 \leq M$ para todo $x \in A$. Por lo tanto, para todo $x \in A$ se tiene que
    \[
    \|x\|_2 \leq C_2 \|x\|_1 \leq C_2 M
    \]
    y, por tanto, $A$ es acotado en la norma $\|\cdot\|_2$, con cota $C_2 M$. De manera análoga se demuestra que si $A$ es acotado en la norma $\|\cdot\|_2$, entonces también lo es en la norma $\|\cdot\|_1$.

    \item[$\Longleftarrow$)] Sea $X$ un espacio vectorial y $\|\cdot\|_1$ y $\|\cdot\|_2$ dos normas en $X$ que dan lugar a los mismos conjuntos acotados.
    
    Fijado $x \in X \setminus \{0\}$, consideremos el conjunto $A = \left\{\dfrac{x}{\|x\|_1}\right\}$. Este conjunto es acotado en la norma $\|\cdot\|_1$, y por hipótesis también lo es en la norma $\|\cdot\|_2$. Por tanto, existe $C_2 \in \bb{R}^+$ tal que
    \[
    \left\|\frac{x}{\|x\|_1}\right\|_2 \leq C_2
    \]
    y, por tanto, $\|x\|_2 \leq C_2 \|x\|_1$. De manera análoga se demuestra que existe $C_1' \in \bb{R}^+$ tal que $\|x\|_1 \leq C_1' \|x\|_2$. Considerando $C_1 = \nicefrac{1}{C_1'}$, se obtiene que:
    \[
    C_1 \|x\|_1 \leq \|x\|_2 \leq C_2 \|x\|_1
    \]

    Puesto que $x \in X \setminus \{0\}$ es arbitrario, y para $x = 0$ se tiene que $\|0\|_1 = \|0\|_2 = 0$, se concluye que las normas $\|\cdot\|_1$ y $\|\cdot\|_2$ son equivalentes.
\end{description}
\end{ejercicio}

\begin{ejercicio}
Si $X$ es un espacio normado, para $x \in X$ y $r \in \bb{R}^+$, denotamos por $B(x, r)$ a la bola cerrada de centro $x$ y radio $r$. Para $x, y \in X$ y $r, \rho \in \bb{R}^+$, probar las siguientes equivalencias:
\begin{enumerate}[label={\alph*)}]
    \item $B(x, r) \cap B(y, \rho) \neq \emptyset \sii \|x - y\| \leq r + \rho$
    
    La situación está ilustrada en la Figura~\ref{fig:interseccion_bolas}.
    \begin{figure}[htbp]
        \centering
        \begin{tikzpicture}
            \draw[thick, red] (0,0) circle (2cm);
            \draw[thick, blue] (3,0) circle (1.5cm);

            % Puntos de interés
            \filldraw[red] (0,0) circle (2pt) node[anchor=north east] {$x$};
            \filldraw[blue] (3,0) circle (2pt) node[anchor=north west] {$y$};
            \filldraw[black] (1.75,0) circle (2pt) node[anchor=north] {$z$};

            % Etiquetas de los radios
            \draw[<->, dashed, red] (0,0) -- (0,2) node[midway, left] {$r$};
            \draw[<->, dashed, blue] (3,0) -- (3,1.5) node[midway, right] {$\rho$};

            % Etiqueta de las bolas
            \node at (-2.2,1.7) [red] {$B(x,r)$};
            \node at (4.8,1.2) [blue] {$B(y,\rho)$};
        \end{tikzpicture}
        \caption{Intersección no vacía de dos bolas cerradas.}
        \label{fig:interseccion_bolas}
    \end{figure}

    Consideramos la distancia inducida por la norma: $d(x,y)=\|x-y\|,~\forall x,y\in X$.
    \begin{description}
        \item[$\Longrightarrow)$]
        Sea $z \in B(x, r) \cap B(y, \rho)$. Por definición de bola cerrada, $d(x, z) \leq r$ y $d(y, z) \leq \rho$. Entonces, por la desigualdad triangular:
        \begin{equation*}
            \|x - y\| = d(x, y) \leq d(x, z) + d(z, y) \leq r + \rho
        \end{equation*}

        \item[$\Longleftarrow)$]
        Suponemos $\|x - y\| \leq r + \rho$, y buscamos $z \in B(x, r) \cap B(y, \rho)$.
        
        Si $x = y$, basta tomar $z = x$, ya que $\|x - x\| = 0 \leq r$ y $\|x - y\| = 0 \leq \rho$.
        
        Supongamos por tanto que $x \neq y$. Consideramos un punto del segmento que une $x$ e $y$, definido por $z := \lambda x + (1 - \lambda)y$ para $\lambda \in [0, 1]$. Estudiamos las condiciones para que pertenezca a ambas bolas:
        \begin{equation*}\begin{split}
            z \in B(x, r) &\iff \|z - x\| \leq r \iff \|(1 - \lambda)(y - x)\| = (1 - \lambda)\|x - y\| \leq r \\
            &\iff 1 - \frac{r}{\|x - y\|} \leq \lambda \\
            z \in B(y, \rho) &\iff \|z - y\| \leq \rho \iff \|\lambda(x - y)\| = \lambda \|x - y\| \leq \rho \\
            &\iff \lambda \leq \frac{\rho}{\|x - y\|}
        \end{split}\end{equation*}

        Por tanto, buscamos $\lambda \in [0, 1]$ que satisfaga:
        \begin{equation*}
            1 - \frac{r}{\|x - y\|} \leq \lambda \leq \frac{\rho}{\|x - y\|}
        \end{equation*}
        Dicho valor existe en $\bb{R}$ si y solo si el extremo inferior es menor o igual que el superior:
        \begin{equation*}
            1 - \frac{r}{\|x - y\|} \leq \frac{\rho}{\|x - y\|} \iff \|x - y\| - r \leq \rho \iff \|x - y\| \leq r + \rho
        \end{equation*}
        lo cual es cierto por hipótesis. Además, como $r, \rho > 0$, se tiene:
        $$1 - \frac{r}{\|x-y\|} < 1 \quad \text{y} \quad \frac{\rho}{\|x-y\|} > 0.$$ 
        Tomando, por ejemplo,
        \begin{equation*}
            \lambda = \max\left\{0, \, 1 - \frac{r}{\|x - y\|}\right\}
        \end{equation*}
        se comprueba que $\lambda \in [0, 1]$ y cumple ambas desigualdades. Por tanto, existe $z \in B(x, r) \cap B(y, \rho)$, concluyendo que la intersección no es vacía.
    \end{description}
    \item $B(x, r) \subset B(y, \rho) \sii \|x - y\| \leq \rho - r$
    
    La situación está ilustrada en la Figura~\ref{fig:inclusion_bolas}.
    \begin{figure}[htbp]
        \centering
        \begin{tikzpicture}
            % Bola mayor exterior B(y, rho)
            \draw[thick, blue] (0,0) circle (2cm);
            % Bola menor interior B(x, r)
            \draw[thick, red] (0.7,0) circle (0.8cm);

            % Puntos de interés
            \filldraw[blue] (0,0) circle (2pt) node[anchor=north east] {$y$};
            \filldraw[red] (0.7,0) circle (2pt) node[anchor=north] {$x$};
            \filldraw[black] (1.5,0) circle (2pt) node[anchor=north west] {$z$};

            % Etiquetas de los radios
            \draw[<->, dashed, blue] (0,0) -- (0,2) node[midway, left] {$\rho$};
            \draw[<->, dashed, red] (0.7,0) -- (0.7,0.8) node[midway, left] {$r$};

            % Semirrecta que ilustra la elección de z
            \draw[dotted, gray] (0,0) -- (2,0);

            % Etiqueta de las bolas
            \node at (-2.2,1.7) [blue] {$B(y,\rho)$};
            \node at (0.7,-1.1) [red] {$B(x,r)$};
        \end{tikzpicture}
        \caption{Inclusión de la bola $B(x,r)$ en la bola $B(y,\rho)$.}
        \label{fig:inclusion_bolas}
    \end{figure}

    \begin{description}
        \item[$\Longrightarrow)$]
        Supuesto $x = y$, tenemos que $B(x, r) \subset B(x, \rho)$ si y solo si $r \leq \rho$, lo cual equivale a $\|x - y\| = 0 \leq \rho - r$. Supongamos por tanto que $x \neq y$.

        Consideramos el punto:
        \begin{equation*}
            z = x + r \frac{x - y}{\|x - y\|}
        \end{equation*}
        Calculamos su distancia a $x$:
        \begin{equation*}
            \|z - x\| = \left\| r \frac{x - y}{\|x - y\|} \right\| = r \frac{\|x - y\|}{\|x - y\|} = r \leq r
        \end{equation*}
        Por tanto, $z \in B(x, r)$. Como por hipótesis $B(x, r) \subset B(y, \rho)$, se deduce que $z \in B(y, \rho)$, es decir, $\|z - y\| \leq \rho$.

        Calculamos explícitamente $\|z - y\|$:
        \begin{align*}
            \|z - y\| &= \left\| x - y + r \frac{x - y}{\|x - y\|} \right\| = \left\| (x - y)\left(1 + \frac{r}{\|x - y\|}\right) \right\| \\
            &= \left(1 + \frac{r}{\|x - y\|}\right)\|x - y\| = \|x - y\| + r
        \end{align*}
        Como $\|z - y\| \leq \rho$, obtenemos directamente:
        \begin{equation*}
            \|x - y\| + r \leq \rho \iff \|x - y\| \leq \rho - r
        \end{equation*}

        \item[$\Longleftarrow)$]
        Supongamos que $\|x - y\| \leq \rho - r$. Sea $z \in B(x, r)$ arbitrario, es decir, $\|z - x\| \leq r$. Veamos que $z \in B(y, \rho)$:
        \begin{equation*}
            \|z - y\| \leq \|z - x\| + \|x - y\| \leq r + (\rho - r) = \rho
        \end{equation*}
        Por tanto, $z \in B(y, \rho)$, demostrando la inclusión $B(x, r) \subset B(y, \rho)$.
    \end{description}
\end{enumerate}
\end{ejercicio}

\begin{ejercicio}
Sea $X$ un espacio de Banach y $\{B_n\}$ una sucesión de bolas cerradas en $X$, que sea decreciente, es decir, $B_{n+1} \subset B_n$ para todo $n \in \bb{N}$. Probar que $\bigcap\limits_{n=1}^\infty B_n \neq \emptyset$.\\

En vistas a emplear la propiedad de completitud de $X$, buscamos una sucesión de Cauchy $\{x_n\}$ con $x_n \in B_n$ para todo $n \in \bb{N}$. Sabiendo que $X$ es completo, la sucesión $\{x_n\}$ convergerá a un punto $x \in X$ que, por ser límite de una sucesión de puntos de $B_n$ conjunto cerrado, pertenecerá a $B_n$ para todo $n \in \bb{N}$, y por tanto a la intersección $\bigcap\limits_{n=1}^\infty B_n$. Formalicemos esta idea.

Fijado $n \in \bb{N}$, sea $B_n = B(x_n, r_n)$, con $x_n \in X$ y $r_n \in \bb{R}^+$. Como $\{B_n\}$ es decreciente, se tiene que $B_{n+1} \subset B_n$, y por lo demostrado en el ejercicio anterior, se deduce que $\|x_{n+1} - x_n\| \leq r_n - r_{n+1}$. De esta idea, deducimos que $\{r_n\}$ es una sucesión decreciente de números reales minorada por cero. Por la caracterización del ínfimo por sucesiones, $\{r_n\}$ converge a un número real $R\in \bb{R}^+_0$.
\begin{equation*}
    \{r_n\} \longrightarrow R \geq 0
\end{equation*}

Asímismo, y en vistas de demostrar que $\{x_n\}$ es de Cauchy, fijados $m, n \in \bb{N}$ con $m > n$, hemos de calcular $\|x_m - x_n\|$. Por la desigualdad triangular, tenemos que:
\begin{equation*}
    \|x_m - x_n\| \leq \sum_{k=n}^{m-1} \|x_{k+1} - x_k\| \leq \sum_{k=n}^{m-1} (r_k - r_{k+1}) = r_n - r_m
\end{equation*}

Como $\{r_n\}$ es convergente, en particular es de Cauchy, luego:
\begin{equation*}
    \forall \varepsilon > 0, \exists n_0 \in \bb{N} : m> n \geq n_0 \implies \|r_m - r_n\| < \varepsilon
\end{equation*}

Por tanto, fijado $\veps\in \bb{R}^+$, consideramos el valor de $n_0$ dado por ser $\{r_n\}$ de Cauchy. Para cualesquiera $m, n \in \bb{N}$ con $m > n \geq n_0$, se tiene que:
\begin{equation*}
    \|x_m - x_n\| \leq r_n - r_m < \varepsilon
\end{equation*}

Por tanto, $\{x_n\}$ es una sucesión de Cauchy, y como $X$ es completo, $\{x_n\}$ converge a un punto $x \in X$.
\begin{equation*}
    \{x_n\} \longrightarrow x \in X
\end{equation*}

Veamos ahora que $x \in \bigcap\limits_{n=1}^\infty B_n$. Fijado $n \in \bb{N}$, como $B_n$ es cerrado y $\{x_m\}_{m\geq n} \subset B_n$, se tiene que $x \in B_n$, puesto que el límite de una sucesión de puntos de un conjunto cerrado pertenece al conjunto. Por tanto, $x \in B_n$ para todo $n \in \bb{N}$, y por tanto $x \in \bigcap\limits_{n=1}^\infty B_n$.
\end{ejercicio}

\begin{ejercicio}
Probar que, si $X$ es un espacio normado, para cualesquiera $x, y \in X \setminus \{0\}$ se verifica la siguiente desigualdad:
\[
\left\| \frac{x}{\|x\|} - \frac{y}{\|y\|} \right\| \leq \frac{2\|x - y\|}{\max\{\|x\|, \|y\|\}}
\]

    Supongamos sin pérdida de generalidad que $\|y\| \leq \|x\|$, puesto que en caso contrario, intercambiando $x$ e $y$, se obtiene la misma desigualdad. Entonces, $\max\{\|x\|, \|y\|\} = \|x\|$. Sumamos y restamos $\dfrac{y}{\|x\|}$ en el primer miembro de la desigualdad:
    \begin{equation*}
        \left\| \frac{x}{\|x\|} - \frac{y}{\|y\|} \right\| = \left\| \frac{x}{\|x\|} - \frac{y}{\|x\|} + \frac{y}{\|x\|} - \frac{y}{\|y\|} \right\| \leq \left\| \frac{x - y}{\|x\|} \right\| + \left\| \dfrac{y}{\|x\|} - \dfrac{y}{\|y\|} \right\|
    \end{equation*}

    Respecto al primer sumando, por homogeneidad de la norma, se tiene que:
    \begin{equation*}
        \left\| \frac{x - y}{\|x\|} \right\| = \frac{\|x - y\|}{\|x\|}
    \end{equation*}

    Respecto al segundo sumando, factorizamos $y$ y aplicamos la desigualdad triangular inversa:
    \begin{equation*}
        \left\| \dfrac{y}{\|x\|} - \dfrac{y}{\|y\|} \right\| = \left\|y\right\| \left| \frac{1}{\|x\|} - \frac{1}{\|y\|} \right| = \|y\| \, \frac{|\|y\| - \|x\||}{\|x\| \, \|y\|} \leq \|y\| \,  \frac{\|x - y\|}{\|x\| \, \|y\|} = \frac{\|x - y\|}{\|x\|}
    \end{equation*}

    Por tanto, se concluye que:
    \begin{equation*}
        \left\| \frac{x}{\|x\|} - \frac{y}{\|y\|} \right\| \leq \frac{\|x - y\|}{\|x\|} + \frac{\|x - y\|}{\|x\|} = \frac{2\|x - y\|}{\|x\|} = \frac{2\|x - y\|}{\max\{\|x\|, \|y\|\}}
    \end{equation*}
\end{ejercicio}

\begin{ejercicio}
Sean $A$ y $B$ subconjuntos no vacíos de un espacio normado. Probar las siguientes afirmaciones:
\begin{enumerate}[label={\alph*)}]
    \item Si $A$ es abierto, entonces $A + B$ es abierto.
    
    Recordamos que:
    \begin{equation*}
        A + B = \{a + b : a \in A, b \in B\}
    \end{equation*}

    \begin{description}
        \item[Opción 1] Si $x \in A + B$, entonces existen $a \in A$ y $b \in B$ tales que $x = a + b$. Como $A$ es abierto, existe $r \in \bb{R}^+$ tal que $U(a, r) \subset A$. Consideremos la bola abierta $U(x, r)$, y tenemos que demostrar que $U(x, r) \subset A + B$. Sea $y \in U(x, r)$ arbitrario, y escribamos $y = (y -b) + b$. Veamos que $y - b \in U(a, r)$:
        \begin{equation*}
            \|y - b - a\| = \|y - (a + b)\| = \|y - x\| < r
        \end{equation*}
        Por tanto, $y - b \in U(a, r) \subset A$, y como $b \in B$, se puede concluir que $y = (y - b) + b \in A + B$. Por tanto, $U(x, r) \subset A + B$, y como $x \in A + B$ es arbitrario, se concluye que $A + B$ es abierto.
        
        \item[Opción 2] Tenemos que:
        \begin{equation*}
            A + B = \bigcup_{b \in B} (A + b)
        \end{equation*}

        Fijado $b \in B$, consideremos la siguiente aplicación:
        \Func{T_b}{X}{X}{x}{x + b}

        Esta aplicación, traslación por $b$, es un homeomorfismo en $X$, y por tanto $A + b=T_b(A)$ es abierto. Como $A + B$ es la unión de abiertos, se concluye que $A + B$ es abierto.
    \end{description}
    \item Si $A$ es compacto y $B$ es cerrado, entonces $A + B$ es cerrado.
    
    Consideramos una sucesión $\{x_n\}$ en $A + B$ que converge a un punto $x \in X$. Queremos ver que $x \in A + B$. Para cada $n \in \bb{N}$, existen $a_n \in A$ y $b_n \in B$ tales que:
    \begin{equation*}
        x_n = a_n + b_n
    \end{equation*}

    Como $A$ es compacto\footnote{Notemos que no estamos empleando el concepto general de compacidad, sino su equivalente para espacios métricos visto en Análisis Matemático I.}, existe una parcial $\{a_{\sigma(n)}\}$ de $\{a_n\}$ que converge a un punto $a \in A$. Por tanto, para cada $n\in \bb{N}$, tenemos que:
    \begin{equation*}
        b_{\sigma(n)} = x_{\sigma(n)} - a_{\sigma(n)} \in B
    \end{equation*}

    Además, como $\{x_n\}\to x$ y $\{a_{\sigma(n)}\}\to a$ y toda parcial de una sucesión convergente converge al mismo límite, se concluye que $\{b_{\sigma(n)}\}\to x - a$. Como $B$ es cerrado, se deduce que $x - a \in B$, y por tanto $x = a + (x - a) \in A + B$. Por tanto, $A + B$ es cerrado.
\end{enumerate}
Dar un ejemplo en el que $A$ y $B$ sean cerrados pero $A + B$ no sea cerrado.\\


Notemos que hemos de buscar un ejemplo en el que $A$ no sea compacto, ya que de lo contrario, por el resultado anterior, $A + B$ sería cerrado. En particular, podemos buscar un ejemplo. En $\bb{R}^n$, vimos que ser compacto es equivalente a ser cerrado y acotado, luego hemos de buscar dos conjuntos cerrados pero no acotados. Veamos dos ejemplos:
\begin{enumerate}
    \item Consideramos el espacio normado $\bb{R}$, y sean:
    \begin{equation*}
        A = \bb{Z} \quad \text{y} \quad B = \left\{n + \frac{1}{n+1} : n \in \bb{N}\right\}
    \end{equation*}

    Tenemos que $A$ y $B$ son cerrados, pero veamos que $A + B$ no es cerrado. Consideremos la sucesión $\{x_n\}$ definida por:
    \begin{equation*}
        x_n = -n + \left(n + \frac{1}{n+1}\right) = \frac{1}{n+1} \in A + B
    \end{equation*}
    La sucesión $\{x_n\}$ converge a $0$, y veamos que $0 \notin A + B$.
    \begin{itemize}
        \item Supongamos que $0 \in A + B$. Entonces, existen $a \in A=\bb{Z}$ y $b \in B$ tales que $0 = a + b$. Como $b\in B$, existe $n \in \bb{N}$ tal que $b = n + \frac{1}{n+1}$, luego:
        \begin{equation*}
            0 = a + b = a + n + \frac{1}{n+1} \implies -(a + n) = \frac{1}{n+1}
        \end{equation*}

        Como $a + n \in \bb{Z}$, se deduce que $-(a + n) \in \bb{Z}$. No obstante, tenemos que:
        \begin{equation*}
            0 < \frac{1}{n+1} < 1
        \end{equation*}
        lo cual es una contradicción, ya que ningún número entero puede ser igual a un número estrictamente positivo menor que $1$. Por tanto, $0 \notin A + B$.
    \end{itemize}
    Por tanto, $A + B$ no es cerrado.
    \item Consideramos el espacio normado $\bb{R}^2$, y sean:
    \begin{equation*}
        A = \bb{R}^+ \times \{0\} \quad \text{y} \quad B = \left\{(x, y) \in \bb{R}^2 : y = e^x\right\}
    \end{equation*}
    \begin{figure}
        \centering
        \begin{tikzpicture}
            \begin{axis}[
                axis lines = middle,
                xlabel = {$x$},
                ylabel = {$y$},
                xmin = -3, xmax = 3,
                ymin = -0.8, ymax = 5.5,
                legend pos = north west,       % Posición: north west, north east, south west, etc.
                legend cell align = {left},    % Alineación del texto en la leyenda
                samples = 100,                 % Calidad de la curva exponencial
                domain = -3:3
            ]
                % Conjunto A (eje o semieje horizontal)
                \addplot[thick, red, domain=-5:5] {0};
                \addlegendentry{Conjunto $A$}

                % Conjunto B (curva exponencial)
                \addplot[thick, blue, domain=-3:3] {exp(x)};
                \addlegendentry{Conjunto $B$}
                
            \end{axis}
        \end{tikzpicture}
        \caption{Conjuntos $A$ y $B$ en $\mathbb{R}^2$.}
        \label{fig:conjuntos_A_B}
    \end{figure}
    Los conjuntos $A$ y $B$ están mostrados en la Figura~\ref{fig:conjuntos_A_B}. El conjunto $A$ es trivialmente cerrado, y el conjunto $B$ es cerrado porque es la gráfica de una función continua. No obstante, tenemos que $A + B=\bb{R} \times \bb{R}^+$, que no es cerrado, ya que su frontera $\bb{R} \times \{0\}$ no está incluida en el conjunto.
\end{enumerate}
\end{ejercicio}

\begin{ejercicio}
En el espacio vectorial $C^1\left[0, 1\right]$ formado por todas las funciones de clase $C^1$ en el intervalo $\left[0, 1\right]$, con valores en $\bb{K}$, se define
\[
\|f\| = |f(0)| + \max\left\{ |f'(t)| : t \in \left[0, 1\right] \right\} \quad \forall f \in C^1\left[0, 1\right]
\]
Probar que de esta forma se obtiene una norma completa en $C^1\left[0, 1\right]$.\\

    En primer lugar, veamos que $\|\cdot\|$ está bien definida. Para cada $f\in C^1\left[0, 1\right]$, $f'$ es continua en $\left[0, 1\right]$, y por el teorema de Weierstrass, alcanza su máximo en dicho intervalo. Por tanto, $\max\left\{ |f'(t)| : t \in \left[0, 1\right] \right\}$ está bien definido y es finito. Por tanto, $\|f\|\in \bb{R}^+_0$ para todo $f \in C^1\left[0, 1\right]$, luego $\|\cdot\|$ está bien definida. Por simplicidad, empleamos la norma del supremo en $C\left[0, 1\right]$ como se ha visto en teoría:
    \begin{equation*}
        \|f\|_\infty := \max\left\{ |f(t)| : t \in \left[0, 1\right] \right\} \quad \forall f \in C\left[0, 1\right]
    \end{equation*}

    De esta forma, podemos escribir la norma como:
    \begin{equation*}
        \|f\| = |f(0)| + \|f'\|_\infty \quad \forall f \in C^1\left[0, 1\right]
    \end{equation*}

    Veamos ahora que $\|\cdot\|$ es una norma. Para ello, hemos de verificar las tres propiedades de una norma:
    \begin{enumerate}
        \item $\|f\| = 0 \iff f = 0$.
        
        Si $f = 0$, entonces $f(0) = 0$ y $f' = 0$, luego $\|f\| = 0$. Supongamos ahora que $\|f\| = 0$. Entonces, $|f(0)| = 0$ y $\|f'\|_\infty = 0$. Por ser $|\cdot|$ y $\|\cdot\|_\infty$ normas, se deduce que $f(0) = 0$ y $f' = 0$. De $f' = 0$ se deduce que $f$ es constante, y como $f(0) = 0$, se concluye que $f = 0$. Por tanto, $\|f\| = 0 \iff f = 0$.
        \item $\|\alpha f\| = |\alpha| \, \|f\|$ para todo $\alpha \in \bb{K}$ y todo $f \in C^1\left[0, 1\right]$.
        \begin{align*}
            \|\alpha f\| &= |\alpha f(0)| + \|(\alpha f)'\|_\infty \AstIg |\alpha| \, |f(0)| + \| \alpha f'\|_\infty \stackrel{(\ast\ast)}{=} |\alpha| \, |f(0)| + |\alpha| \, \|f'\|_\infty =\\&= |\alpha| (|f(0)| + \|f'\|_\infty) = |\alpha| \, \|f\|
        \end{align*}
        donde en $(\ast)$ hemos empleado tanto la desigualdad triangular en $\bb{K}$ como la linealidad de la derivada, y en $(\ast\ast)$ hemos empleado la homogeneidad de la norma del supremo.
        \item $\|f + g\| \leq \|f\| + \|g\|$ para todo $f, g \in C^1\left[0, 1\right]$.
        \begin{align*}
            \|f + g\| &= |(f + g)(0)| + \|(f + g)'|_\infty \leq |f(0)| + |g(0)| + \|f' + g'\|_\infty \leq\\&\leq |f(0)| + |g(0)| + \|f'\|_\infty + \|g'\|_\infty =\\&= (|f(0)| + \|f'\|_\infty) + (|g(0)| + \|g'\|_\infty) = \|f\| + \|g\|
        \end{align*}
        donde en la primera desigualdad hemos empleado tanto la desigualdad triangular en $\bb{K}$ como la derivada de la suma, y en la segunda desigualdad hemos empleado la desigualdad triangular para la norma del supremo.
    \end{enumerate}

    Por tanto, $\|\cdot\|$ es una norma en $C^1\left[0, 1\right]$. Veamos ahora que $C^1\left[0, 1\right]$ es completo con esta norma. Sea $\{f_n\}$ una sucesión de Cauchy en $C^1\left[0, 1\right]$. Entonces, para todo $\veps > 0$, existe $n_0 \in \bb{N}$ tal que:
    \begin{equation*}
        m, n \geq n_0 \implies \|f_n - f_m\| = |f_n(0) - f_m(0)| + \|f_n' - f_m'\|_\infty < \veps
    \end{equation*}

    En particular, $\{f_n(0)\}$ es una sucesión de Cauchy en $\bb{K}$, y como $\bb{K}$ es completo, existe $a \in \bb{K}$ tal que:
    \begin{equation*}
        \{f_n(0)\} \longrightarrow a \in \bb{K}
    \end{equation*}

    Además, también se tiene que $\{f_n'\}$ es una sucesión de Cauchy en $C\left[0, 1\right]$ con la norma del supremo, y como $C\left[0, 1\right]$ es completo, existe $g \in C\left[0, 1\right]$ tal que:
    \begin{equation*}
        \{f_n'\} \longrightarrow g \in C\left[0, 1\right]
    \end{equation*}

    Para demostrar la completitud, en primer lugar definimos nuestro candidato a límite $f:\left[0, 1\right] \to \bb{K}$, que ha de cumplir que $f(0) = a$ y $f' = g$. Para ello, definimos:
    \begin{equation*}
        f(x) = a + \int_{0}^x g(s)\,ds \quad \forall x \in \left[0, 1\right]
    \end{equation*}

    Por el teorema fundamental del cálculo, $f\in C^1\left[0, 1\right]$ y $f' = g$. Además, se tiene que:
    \begin{equation*}
        f(0) = a + \int_{0}^0 g(s)\,ds = a
    \end{equation*}

    Para demostrar la completitud, hemos de ver que $\{f_n\} \to f$ en $C^1\left[0, 1\right]$. Para ello, hemos de demostrar que:
    \begin{equation*}
        \left\{\|f_n - f\|\right\} = \left\{|f_n(0) - f(0)| + \|f_n' - f'\|_\infty\right\} = \left\{|f_n(0) - a| + \|f_n' - g\|_\infty\right\} \longrightarrow 0
    \end{equation*}

    Como $\{f_n(0)\} \to a$ en $\bb{K}$ y $\{f_n'\} \to g$ en $C\left[0, 1\right]$, se tiene que $\left\{|f_n(0)-a|\right\} \to 0$ y $\{\|f_n' - g\|_\infty\} \to 0$, y por tanto $\{f_n\} \to f$ en $C^1\left[0, 1\right]$. Por tanto, $C^1\left[0, 1\right]$ es completo con la norma definida.
\end{ejercicio}

\begin{ejercicio}
En el espacio $\bb{K}^{\bb{N}}$ de todas las sucesiones de escalares, se considera el subespacio definido por
\[
X = \left\{ x \in \bb{K}^{\bb{N}} : \sum_{n=1}^\infty \frac{|x(n)|}{n} < \infty \right\}
\]
y se define una norma en $X$ escribiendo
\[
\|x\| = \sum_{n=1}^\infty \frac{|x(n)|}{n} \quad \forall x \in X
\]
Probar que con esta norma, $X$ es un espacio de Banach.\\

En primer lugar, aunque no lo pide explícitamente el enunciado, veamos que $\|\cdot\|$ es una norma. Para ello, hemos de verificar las tres propiedades de una norma:
\begin{enumerate}
    \item $\|x\| = 0 \iff x = 0$.
    
    Si $x = 0$, entonces $x(n) = 0$ para todo $n \in \bb{N}$, y por tanto $\|x\| = 0$. Supongamos ahora que $\|x\| = 0$. Entonces, $\sum\limits_{n=1}^\infty \frac{|x(n)|}{n} = 0$, y como todos los sumandos son no negativos, se deduce que $|x(n)| = 0$ para todo $n \in \bb{N}$, y por tanto $x(n) = 0$ para todo $n \in \bb{N}$, luego $x = 0$.

    \item $\|\alpha x\| = |\alpha| \, \|x\|$ para todo $\alpha \in \bb{K}$ y todo $x \in X$.
    \begin{equation*}
        \|\alpha x\| = \sum_{n=1}^\infty \frac{|\alpha x(n)|}{n} = \sum_{n=1}^\infty \frac{|\alpha| \, |x(n)|}{n} = |\alpha| \sum_{n=1}^\infty \frac{|x(n)|}{n} = |\alpha| \, \|x\|
    \end{equation*}

    \item $\|x + y\| \leq \|x\| + \|y\|$ para todo $x, y \in X$.
    \begin{equation*}
        \|x + y\| = \sum_{n=1}^\infty \frac{|x(n) + y(n)|}{n} \leq \sum_{n=1}^\infty \frac{|x(n)| + |y(n)|}{n} = \sum_{n=1}^\infty \frac{|x(n)|}{n} + \sum_{n=1}^\infty \frac{|y(n)|}{n} = \|x\| + \|y\|
    \end{equation*}
\end{enumerate}

Por tanto, $\|\cdot\|$ es una norma en $X$. Veamos ahora que $X$ es completo con esta norma. Aunque podría hacerse de forma directa, vemos que la norma es muy similar a la dada en $\ell_1$, luego veamos que $X$ es isomorfo a $\ell_1$. Consideramos la aplicación:
\Func{\Phi}{X}{\ell_1}{x}{\Phi(x)}
donde la sucesión $\Phi(x)$ está definida por:
\Func{\Phi(x)}{\bb{N}}{\bb{K}}{n}{\frac{x(n)}{n}}

Veamos que $\Phi$ es un isomorfismo lineal.
\begin{itemize}
    \item En primer lugar, veamos que $\Phi$ está bien definida; es decir, que dado $x \in X$, se tiene que $\Phi(x) \in \ell_1$. Como $x \in X$, se tiene que:
    \begin{equation*}
        \sum_{n=1}^\infty \frac{|x(n)|}{n} < \infty
    \end{equation*}
    Por tanto, $\Phi(x) \in \ell_1$.
    \item Veamos que $\Phi$ es lineal. Sean $x, y \in X$ y $\alpha, \beta \in \bb{K}$. Entonces, para todo $n \in \bb{N}$, se tiene que:
    \begin{equation*}
        \Phi(\alpha x + \beta y)(n) = \frac{(\alpha x + \beta y)(n)}{n} = \frac{\alpha x(n) + \beta y(n)}{n} = \alpha \frac{x(n)}{n} + \beta \frac{y(n)}{n} = \alpha \Phi(x)(n) + \beta \Phi(y)(n)
    \end{equation*}
    Por tanto, $\Phi(\alpha x + \beta y) = \alpha \Phi(x) + \beta \Phi(y)$, y se concluye que $\Phi$ es lineal.
    \item Veamos que $\Phi$ es biyectiva. En primer lugar, veamos que es inyectiva. Supongamos que $\Phi(x) = \Phi(y)$ para algunos $x, y \in X$. Entonces, para todo $n \in \bb{N}$, se tiene que:
    \begin{equation*}
        \Phi(x)(n) = \Phi(y)(n) \implies \frac{x(n)}{n} = \frac{y(n)}{n} \implies x(n) = y(n)
    \end{equation*}
    Por tanto, $x = y$, y se concluye que $\Phi$ es inyectiva. Ahora, veamos que es sobreyectiva. Sea $z \in \ell_1$. Definimos $x \in X$ por:
    \begin{equation*}
        x(n) = n z(n) \quad \forall n \in \bb{N}
    \end{equation*}
    Puesto que $z \in \ell_1$, se tiene que $x\in X$. Además, para todo $n \in \bb{N}$, se tiene que:
    \begin{equation*}
        \Phi(x)(n) = \frac{x(n)}{n} = \frac{n z(n)}{n} = z(n)
    \end{equation*}
    Por tanto, $\Phi(x) = z$, y se concluye que $\Phi$ es sobreyectiva. Por tanto, $\Phi$ es biyectiva.
    \item Por el momento, tenemos que $\Phi$ es un isomorfismo lineal entre $X$ y $\ell_1$. Veamos que es un isomorfismo isométrico; es decir, que preserva la norma. Para ello, sea $x \in X$. Entonces, se tiene que:
    \begin{equation*}
        \|\Phi(x)\|_{1} = \sum_{n=1}^\infty |\Phi(x)(n)| = \sum_{n=1}^\infty \left|\frac{x(n)}{n}\right| = \sum_{n=1}^\infty \frac{|x(n)|}{n} = \|x\|_X
    \end{equation*}
    Por tanto, $\Phi$ es un isomorfismo isométrico entre $X$ y $\ell_1$.
\end{itemize}

Por tanto, $X$ es isométrico a $\ell_1$. Veamos ahora que $\ell_1$ es completo.
\begin{itemize}
    \item Sea $\{z_n\}$ una sucesión de Cauchy en $\ell_1$. Entonces, para todo $\veps > 0$, existe $n_0 \in \bb{N}$ tal que:
    \begin{equation*}
        m, n \geq n_0 \implies \|z_n - z_m\|_1 = \sum_{k=1}^\infty |z_n(k) - z_m(k)| < \veps
    \end{equation*}
    Fijado $k \in \bb{N}$, se tiene que $|z_n(k)-z_m(k)| \leq \|z_n - z_m\|_1$ para todo $m, n \geq n_0$, y por tanto $\{z_n(k)\}$ es una sucesión de Cauchy en $\bb{K}$. Como $\bb{K}$ es completo, existe $z(k) \in \bb{K}$ tal que $\{z_n(k)\} \to z(k)$. Definimos la sucesión $z \in \bb{K}^{\bb{N}}$ por:
    \begin{equation*}
        z(n) = \lim_{m \to \infty} z_m(n) \quad \forall n \in \bb{N}
    \end{equation*}

    La demostración concluirá mostrando que $z \in \ell_1$ y que $\{z_n\} \to z$ en $\ell_1$. Para ello, consideramos la sucesión de sumas parciales:
    \begin{equation*}
        S_n = \sum_{k=1}^n |z(k)| \quad \forall n \in \bb{N}
    \end{equation*}
    Veamos que $\{S_n\}$ es acotada. Para ello, sea $\veps>0$. Entonces, existe $n_0 \in \bb{N}$ tal que:
    \begin{equation*}
        m, n \geq n_0 \implies \|z_n - z_m\|_1 = \sum_{k=1}^\infty |z_n(k) - z_m(k)| < \veps
    \end{equation*}
    Fijado $n \geq n_0$, tenemos que:
    \begin{equation*}
        \sum_{k=1}^\infty |z_n(k) - z(k)| = \lim_{m \to \infty} \sum_{k=1}^\infty |z_n(k) - z_m(k)| \leq \veps
    \end{equation*}
    Por tanto, $\sum\limits_{k=1}^\infty |z(k)| \leq \sum\limits_{k=1}^\infty |z_n(k) - z(k)| + \sum\limits_{k=1}^\infty |z_n(k)| \leq \veps + \|z_n\|_1$, luego $\sum\limits_{k=1}^\infty |z(k)|$ está acotada. Por tanto, $z \in \ell_1$. Además, se tiene que:
    \begin{equation*}
        \|z_n - z\|_1 = \sum_{k=1}^\infty |z_n(k) - z(k)| \leq \veps \quad \forall n \geq n_0
    \end{equation*}
    Por tanto, $\{z_n\} \to z$ en $\ell_1$. Por tanto, $\ell_1$ es completo.
\end{itemize}

Como $X$ es isométrico a $\ell_1$ y $\ell_1$ es completo, se concluye que $X$ es completo con la norma definida.
\end{ejercicio}

\begin{ejercicio}
Dar un ejemplo de una sucesión $\{x_n\}$ en $\ell_1$:
\begin{enumerate}
    \item que converja a cero en $\ell_\infty$ pero no esté acotada en $\ell_2$.
    
    En primer lugar, notemos que $\ell_1 \subset \ell_2 \subset \ell_\infty$. Buscamos una sucesión $\{x_n\}$ en $\ell_1$ que converja a cero en $\ell_\infty$, es decir, que $\|x_n\|_\infty \to 0$, pero que no esté acotada en $\ell_2$.
    \begin{itemize}
        \item Como $\{x_n\}$ está en $\ell_1$, ha de tenerse que:
        \begin{equation*}
            \|x_n\|_1 = \sum_{k=1}^\infty |x_n(k)| < \infty \quad \forall n \in \bb{N}
        \end{equation*}
        
        Esto se logrará de forma sencilla si consideramos sucesiones de soporte finito, es decir, sucesiones que son cero a partir de cierto índice. De esta forma, la suma infinita se reduce a una suma finita, y por tanto es finita.

        \item Como $\{x_n\}$ ha de converger a cero en $\ell_\infty$, ha de tenerse  que:
        \begin{equation*}
            \|x_n\|_\infty = \sup_{k \in \bb{N}} |x_n(k)| \to 0
        \end{equation*}

        Una forma sencilla de lograr esto es emplear sucesiones constantes, de forma que el supremo se reduce al valor constante, y podemos hacer que este valor constante tienda a cero.
        \item Como $\{x_n\}$ no ha de estar acotada en $\ell_2$, ha de tenerse que:
        \begin{equation*}
            \|x_n\|_2 = \left(\sum_{k=1}^\infty |x_n(k)|^2\right)^{\nicefrac{1}{2}} \to \infty
        \end{equation*}
    \end{itemize}

    Consideramos la sucesión $\{x_n\}$ definida por:
    \begin{equation*}
        x_n(k) = \begin{cases}
            \nicefrac{1}{n^\alpha} & \text{si } k \leq n\\
            0 & \text{si } k > n
        \end{cases}
    \end{equation*}
    \begin{itemize}
        \item Al tratarse de una sucesión de soporte finito, se tiene que $\{x_n\} \in \ell_1$ para todo $n \in \bb{N}$.
        \item Fijado $n \in \bb{N}$, tenemos que:
        \begin{equation*}
            \|x_n\|_\infty = \sup_{k \in \bb{N}} |x_n(k)| = \sup_{k \in \bb{N}} \frac{1}{n^\alpha} = \frac{1}{n^\alpha}
        \end{equation*}

        Por tanto, $\|x_n\|_\infty \to 0$ si y solo si $\alpha > 0$.
        \item Fijado $n \in \bb{N}$, tenemos que:
        \begin{equation*}
            \|x_n\|_2 = \left(\sum_{k=1}^\infty |x_n(k)|^2\right)^{\nicefrac{1}{2}} = \left(\sum_{k=1}^n \frac{1}{n^{2\alpha}}\right)^{\nicefrac{1}{2}} = \left(\frac{n}{n^{2\alpha}}\right)^{\nicefrac{1}{2}} = \frac{1}{n^{\alpha - \nicefrac{1}{2}}}
        \end{equation*}
        Por tanto, $\|x_n\|_2 \to \infty$ si y solo si $\alpha-\nicefrac{1}{2} < 0 \iff \alpha < \nicefrac{1}{2}$.
    \end{itemize}

    Por tanto, para que $\{x_n\}$ cumpla las condiciones del enunciado, basta con tomar $\alpha\in \bb{R}$ tal que:
    \begin{equation*}
        0 < \alpha < \nicefrac{1}{2}
    \end{equation*}



    \item que converja a cero en $\ell_2$ pero no esté acotada en $\ell_1$.
    
    De forma análoga al apartado anterior, consideramos la sucesión $\{x_n\}$ definida por:
    \begin{equation*}
        x_n(k) = \begin{cases}
            \nicefrac{1}{n^\alpha} & \text{si } k \leq n\\
            0 & \text{si } k > n
        \end{cases}
    \end{equation*}
    \begin{itemize}
        \item Al tratarse de una sucesión de soporte finito, se tiene que $\{x_n\} \in \ell_1$ para todo $n \in \bb{N}$.
        \item Fijado $n \in \bb{N}$, tenemos que:
        \begin{equation*}
            \|x_n\|_2 = \left(\sum_{k=1}^n |x_n(k)|^2\right)^{\nicefrac{1}{2}} = \left(\sum_{k=1}^n \frac{1}{n^{2\alpha}}\right)^{\nicefrac{1}{2}} = \left(\frac{n}{n^{2\alpha}}\right)^{\nicefrac{1}{2}} = \frac{1}{n^{\alpha - \nicefrac{1}{2}}}
        \end{equation*}
        Por tanto, $\|x_n\|_2 \to 0$ si y solo si $\alpha-\nicefrac{1}{2} > 0 \iff \alpha > \nicefrac{1}{2}$.
        \item Fijado $n \in \bb{N}$, tenemos que:
        \begin{equation*}
            \|x_n\|_1 = \sum_{k=1}^n |x_n(k)| = \sum_{k=1}^n \frac{1}{n^\alpha} = \frac{n}{n^\alpha} = \frac{1}{n^{\alpha - 1}}
        \end{equation*}
        Por tanto, $\|x_n\|_1 \to \infty$ si y solo si $\alpha-1 < 0 \iff \alpha < 1$.
    \end{itemize}

    Por tanto, para que $\{x_n\}$ cumpla las condiciones del enunciado, basta con tomar $\alpha\in \bb{R}$ tal que:
    \begin{equation*}

    \end{equation*}
\end{enumerate}

\begin{ejercicio}
Para cada $n \in \bb{N}$ y cada $t \in \left[0, 1\right]$ se define $f_n(t) = t^n - t^{n+1}$ y $g_n(t) = t^n - t^{2n}$. Estudiar la convergencia de las sucesiones $\{f_n\}$ y $\{g_n\}$:
\begin{enumerate}
    \item en el espacio de Banach $C\left[0, 1\right]$.
    \begin{itemize}
        \item En el caso de $\{f_n\}$, buscamos el límite de la sucesión $\{f_n\}$ en $C\left[0, 1\right]$. Para obtener un candidato a límite, estudiamos la convergencia puntual de $\{f_n\}$. Fijado $t \in \left[0, 1\right]$, se tiene que:
        \begin{equation*}
            \lim_{n \to \infty} f_n(t) = \lim_{n \to \infty} (t^n - t^{n+1}) = \lim_{n \to \infty} t^n (1 - t) = 0 \cdot (1 - t) = 0
        \end{equation*}

        Por tanto, $\{f_n\}$ converge puntualmente a la función nula $0(t) = 0$ para todo $t \in \left[0, 1\right]$. Por tanto, el candidato a límite es la función nula. Para ver si converge en $C\left[0, 1\right]$, hemos de ver si:
        \begin{equation*}
            \lim_{n \to \infty} \|f_n - 0\|_\infty = 0
        \end{equation*}

        Tenemos que:
        \begin{equation*}
            \|f_n-0\|_\infty = \max_{t \in \left[0, 1\right]} |f_n(t)| = \max_{t \in \left[0, 1\right]} |t^n - t^{n+1}|
        \end{equation*}
        Para encontrar el punto crítico, derivamos $f_n(t)$:
        \begin{equation*}
            f_n'(t) = n t^{n-1} - (n+1) t^n = t^{n-1} (n - (n+1)t) = 0\iff \begin{cases}
                t = 0\\
                n - (n+1)t = 0 \implies t = \frac{n}{n+1}
            \end{cases}
        \end{equation*}

        Hemos de ver si dicho punto crítico es un máximo. Para ello, estudiamos el signo de $f_n'(t)$:
        \begin{itemize}
            \item Para $t \in \left]0, \frac{n}{n+1}\right[$, se tiene que $f_n'(t) > 0$, luego $f_n(t)$ es creciente en este intervalo.
            \item Para $t \in \left]\frac{n}{n+1}, 1\right]$, se tiene que $f_n'(t) < 0$, luego $f_n(t)$ es decreciente en este intervalo.
        \end{itemize}

        Por tanto, $t = \frac{n}{n+1}$ es un máximo relativo. Veamos el máximo absoluto:
        \begin{align*}
            f_n(0) = f_n(1) &= 0
            \\f_n\left(\frac{n}{n+1}\right) &= \left(\frac{n}{n+1}\right)^n - \left(\frac{n}{n+1}\right)^{n+1} = \left(\frac{n}{n+1}\right)^n \left(1 - \frac{n}{n+1}\right) =\\&= \left(\frac{n}{n+1}\right)^n \cdot \frac{1}{n+1}>0
        \end{align*}

        Por tanto, el máximo absoluto se alcanza en $t = \frac{n}{n+1}$, y tenemos que:
        \begin{equation*}
            \|f_n-0\|_\infty = f_n\left(\frac{n}{n+1}\right) = \left(\frac{n}{n+1}\right)^n \cdot \frac{1}{n+1}
        \end{equation*}

        Por tanto, tenemos que:
        \begin{equation*}
            \lim_{n \to \infty} \|f_n-0\|_\infty = \lim_{n \to \infty} \left(\frac{n}{n+1}\right)^n \cdot \frac{1}{n+1} = \dfrac{1}{e} \cdot 0 = 0
        \end{equation*}

        Por tanto, $\{f_n\}$ converge a $0$ en $C\left[0, 1\right]$.
        \begin{observacion}
            Notemos que, gracias a la estructura de espacio normado, vemos las funciones como elementos de un espacio vectorial, y por tanto tan solo se habla de convergencia en norma. Es cierto que, en $\ell_\infty([0, 1])$, la convergencia en norma equivale a la convergencia uniforme visto desde el punto de vista de funciones, pero no es necesario hablar de convergencia uniforme para demostrar que $\{f_n\}$ converge a $0$ en $C\left[0, 1\right]$.
        \end{observacion}

        \item En el caso de $\{g_n\}$, calculamos el límite puntual para obtener un candidato a límite. Fijado $t \in \left[0, 1\right]$, se tiene que:
        \begin{equation*}
            \lim_{n \to \infty} g_n(t) = \lim_{n \to \infty} (t^n - t^{2n}) = \lim_{n \to \infty} t^n (1 - t^n) = 0 \cdot (1 - 0) = 0
        \end{equation*}

        Por tanto, $\{g_n\}$ converge puntualmente a la función nula $0(t) = 0$ para todo $t \in \left[0, 1\right]$. Por tanto, el candidato a límite es la función nula. Para ver si converge en $C\left[0, 1\right]$, hemos de ver si:
        \begin{equation*}
            \lim_{n \to \infty} \|g_n - 0\|_\infty = 0
        \end{equation*}
        
        Tenemos que:
        \begin{equation*}
            \|g_n-0\|_\infty = \max_{t \in \left[0, 1\right]} |g_n(t)| = \max_{t \in \left[0, 1\right]} |t^n - t^{2n}|
        \end{equation*}

        Para encontrar el punto crítico, derivamos $g_n(t)$:
        \begin{equation*}
            g_n'(t) = n t^{n-1} - 2n t^{2n-1} = n t^{n-1} (1 - 2 t^n) = 0\iff \begin{cases}
                t = 0\\
                1 - 2 t^n = 0 \implies t = \left(\frac{1}{2}\right)^{\nicefrac{1}{n}}
            \end{cases}
        \end{equation*}

        Hemos de ver si dicho punto crítico es un máximo. Para ello, estudiamos el signo de $g_n'(t)$:
        \begin{itemize}
            \item Para $t \in \left]0, \left(\frac{1}{2}\right)^{\nicefrac{1}{n}}\right[$, se tiene que $g_n'(t) > 0$, luego $g_n(t)$ es creciente en este intervalo.
            \item Para $t \in \left]\left(\frac{1}{2}\right)^{\nicefrac{1}{n}}, 1\right]$, se tiene que $g_n'(t) < 0$, luego $g_n(t)$ es decreciente en este intervalo.
        \end{itemize}

        Por tanto, $t = \left(\frac{1}{2}\right)^{\nicefrac{1}{n}}$ es un máximo relativo. Veamos el máximo absoluto:
        \begin{align*}
            g_n(0) = g_n(1) &= 0
            \\g_n\left(\left(\frac{1}{2}\right)^{\nicefrac{1}{n}}\right) &= \left(\frac{1}{2}\right)^{\nicefrac{1}{n} \cdot n} - \left(\frac{1}{2}\right)^{\nicefrac{1}{n} \cdot 2n} = \frac{1}{2} - \frac{1}{4} = \frac{1}{4} > 0
        \end{align*}

        Por tanto, el máximo absoluto se alcanza en $t = \left(\frac{1}{2}\right)^{\nicefrac{1}{n}}$, y tenemos que:
        \begin{equation*}
            \|g_n-0\|_\infty = g_n\left(\left(\frac{1}{2}\right)^{\nicefrac{1}{n}}\right) = \frac{1}{4}
        \end{equation*}

        Por tanto, tomando límite:
        \begin{equation*}
            \lim_{n \to \infty} \|g_n-0\|_\infty = \lim_{n \to \infty} \frac{1}{4} = \frac{1}{4} \neq 0
        \end{equation*}

        Por tanto, $\{g_n\}$ no converge a $0$ en $C\left[0, 1\right]$. El lector podría pensar que $\{g_n\}$ converge a otra función $g$ en $C\left[0, 1\right]$, pero no es así, puesto que la convergencia en norma implica la convergencia puntual, ya que, fijado $t \in \left[0, 1\right]$, se tiene que:
        \begin{equation*}
            |g_n(t) - g(t)|\leq \|g_n - g\|_\infty \to 0
        \end{equation*}
        Por tanto, si $\{g_n\}$ converge a $g$ en $C\left[0, 1\right]$, entonces $\{g_n\}$ converge puntualmente a $g$. Como $\{g_n\}$ converge puntualmente a $0$ pero ya se ha visto que no converge a $0$ en $C\left[0, 1\right]$, se concluye que $\{g_n\}$ no converge en $C\left[0, 1\right]$.
    \end{itemize}
    \item en el espacio de Banach $L_p$, con $1 \leq p < \infty$.
    % // TODO: Ver forma con T. Convergencia Dominada y T. Convergencia Monótona. Si no, preguntarle. Usar h_n(t) = t^n.
    
    Como se vio en el apartado anterior, $\{f_n\}$ y $\{g_n\}$ convergen puntualmente a $0$ en $\left[0, 1\right]$. En este caso la convergencia en norma no implica la convergencia puntual, por lo que no sería sencillo demostrar que no convergen en $L_p$. Sin embargo, veamos que sí convergen en $L_p$. Para ello, basta ver que $\|f_n\|_p \to 0$ y $\|g_n\|_p \to 0$.
    \begin{itemize}
        \item El caso de $\{f_n\}$, es más sencillo. Puesto que $C([0, 1])$ es isométricamente isomorfo a $L_\infty$, tenemos que:
        \begin{equation*}
            \|f_n - 0\|_\infty = \left(\frac{n}{n+1}\right)^n \cdot \frac{1}{n+1}
        \end{equation*}
        donde, en este caso, la norma $\|\cdot\|_\infty$ es la norma de $L_\infty$ (aunque su valor sea el mismo que la norma de $C([0, 1])$ por ser estos isométricamente isomorfos). Por otro lado, de la relación entre los espacios de Lebesgue, tenemos que $f_n-0 \in L_p$, y:
        \begin{equation*}
            0\leq \|f_n - 0\|_p \leq \|f_n - 0\|_\infty = \left(\frac{n}{n+1}\right)^n \cdot \frac{1}{n+1}
        \end{equation*}

        Tomando límite cuando $n \to \infty$ por el Lema del Sandwich, se concluye que $\|f_n - 0\|_p \to 0$, y por tanto $\{f_n\}$ converge a $0$ en $L_p$.
        \item El caso de $\{g_n\}$, es más complicado, puesto que no se puede aplicar el Lema del Sandwich (al ser la norma de la diferencia $\nicefrac{1}{4}\neq 0$). Por tanto, hemos de calcular la norma $\|g_n\|_p$, aunque también usaremos el Lema del Sandwich. Tenemos que, para todo $t\in [0, 1]$ y todo $n \in \bb{N}$:
        \begin{align*}
            0\leq |g_n(t)|^p = |t^n - t^{2n}|^p = (t^n - t^{2n})^p \leq t^{np}
        \end{align*}

        Por tanto, por ser el operador integral estrictamente creciente, se tiene que:
        \begin{align*}
            0\leq \int_0^1 |g_n(t)|^p dt \leq \int_0^1 t^{np} dt = \left[\frac{t^{np+1}}{np+1}\right]_0^1 = \frac{1}{np+1}
        \end{align*}

        Por tanto, tomando raíz $p$-ésima:
        \begin{align*}
            0\leq \|g_n\|_p = \left(\int_0^1 |g_n(t)|^p dt\right)^{\nicefrac{1}{p}} \leq \left(\frac{1}{np+1}\right)^{\nicefrac{1}{p}}
        \end{align*}

        Por tanto, tomando límite cuando $n \to \infty$ por el Lema del Sandwich, se concluye que $\|g_n\|_p \to 0$, y por tanto $\{g_n\}$ converge a $0$ en $L_p$.
    \end{itemize}


\end{enumerate}
\end{ejercicio}

\begin{ejercicio}
Se consideran los conjuntos
\[
U = \{x \in c_0 : |x(n)| < 1 \;\; \forall n \in \bb{N}\} \quad \text{y} \quad V = \{x \in \ell_\infty : |x(n)| < 1 \;\; \forall n \in \bb{N}\}
\]
Probar que $U$ es abierto en $c_0$ pero $V$ no es abierto en $\ell_\infty$.\\

Veamos que $U$ es abierto en $c_0$. Para ello, sea $x \in U$, y buscamos un $r\in \bb{R}^+$ tal que $B(x, r) \subset U$; es decir, buscamos $r\in \bb{R}^+$ tal que, para todo $y \in c_0$ con $\|x-y\|_\infty < r$, se tenga que $y \in U$. De la definición de $U$, buscamos $r\in \bb{R}^+$ tal que, para todo $y \in c_0$ con $\|x-y\|_\infty < r$, se tenga que $|y(n)| < 1$ para todo $n \in \bb{N}$. En vez de dar de manera directa el valor de $r\in \bb{R}^+$, vamos a ver qué condiciones ha de cumplir $r$ para que se cumpla lo anterior.
\begin{itemize}
    \item  Sea $y \in c_0$ con $\|x-y\|_\infty < r$. Buscamos acotar $|y(n)|$ para todo $n \in \bb{N}$, pero solo tenemos acotado $|x(n)-y(n)|$ y $|x(n)|$ para todo $n \in \bb{N}$. Por tanto, vamos a acotar $|y(n)|$ empleando la desigualdad triangular:
    \begin{equation*}
        |y(n)| = |y(n) - x(n) + x(n)| \leq |y(n) - x(n)| + |x(n)| < r + |x(n)| \quad \forall n \in \bb{N}
    \end{equation*}

    Por tanto, buscamos $r\in \bb{R}^+$ tal que $r + |x(n)| \leq 1$ para todo $n \in \bb{N}$; o equivalente, buscamos $r\in \bb{R}^+$ tal que $r \leq 1 - |x(n)|$ para todo $n \in \bb{N}$. Como $0$ es un minorante de $1 - |x(n)|$ para todo $n \in \bb{N}$, podemos tomar:
    \begin{equation*}
        r = \inf\{1 - |x(n)| : n \in \bb{N}\} = 1 - \sup\{|x(n)| : n \in \bb{N}\} = 1 - \|x\|_\infty
    \end{equation*}

    Veamos que $\|x\|_\infty < 1$. Como $x\in U\subset c_0$ y en $c_0$ la norma del supremo es la del máximo, tenemos que $\exists n_0 \in \bb{N}$ tal que $|x(n_0)| = \|x\|_\infty$. Como $x \in U$, se tiene que $|x(n_0)| < 1$, luego $\|x\|_\infty < 1$. Por tanto, $r = 1 - \|x\|_\infty > 0$.
\end{itemize}

Veamos ahora la demostración rigurosa de que $U$ es abierto en $c_0$. Sea $x \in U$, y tomamos $r = 1 - \|x\|_\infty > 0$. Sea $y \in c_0$ tal que $\|x-y\|_\infty < r$. Entonces, para todo $n \in \bb{N}$, se tiene que:
\begin{equation*}
    |y(n)| = |y(n) - x(n) + x(n)| \leq |y(n) - x(n)| + |x(n)| < r + |x(n)| \leq r + \|x\|_\infty = 1-\|x\|_\infty + \|x\|_\infty = 1
\end{equation*}
Por tanto, $y \in U$, y se concluye que $B(x, r) \subset U$. Por tanto, $U$ es abierto en $c_0$.
\begin{itemize}
    \item Como aspecto adicional, veamos que $U=B(0, 1)$ con la norma del máximo (puesto que, en $c_0$, la norma del supremo es la del máximo). En efecto, tenemos que:
    \begin{description}
        \item[$\subset$)] Sea $x \in U$. Entonces, $|x(n)| < 1$ para todo $n \in \bb{N}$, luego $$\|x\|_\infty = \max\{|x(n)| : n \in \bb{N}\} < 1$$
        Por tanto, $x \in B(0, 1)$. Notemos que es clave que $x \in c_0$, ya que, de lo contrario, no se podría garantizar que la norma del supremo sea la del máximo, y por tanto tendríamos que el supremo podría ser $1$.
        \item[$\supset$)] Sea $x \in B(0, 1)$. Entonces, $\|x\|_\infty < 1$, luego $|x(n)| \leq \|x\|_\infty < 1$ para todo $n \in \bb{N}$, luego $x \in U$.
    \end{description}
\end{itemize}

Veamos ahora que $V$ no es abierto en $\ell_\infty$. Para ello, veremos que $l_\infty \setminus V$ no es cerrado. Buscamos una sucesión $\{x_n\}$ en $\ell_\infty \setminus V$ que converja a un elemento $x\in V$. Sea el punto límite
\begin{equation*}
    x(k) = 1-\frac{1}{k} \quad \forall k \in \bb{N}
\end{equation*}

Como $\sup\{|x(k)| : k \in \bb{N}\} = 1< \infty$, se tiene que $x \in \ell_\infty$. Además, como $|x(k)| < 1$ para todo $k \in \bb{N}$, se tiene que $x \in V$. Por tanto, $x$ es un candidato a límite. Para definir la sucesión que converjerá a $x$, buscamos que sea muy similar a excepción de un elemento, que será el que haga que la sucesión no esté en $V$. Sea la sucesión $\{x_n\}$ definida por:
\begin{equation*}
    x_n(k) = \begin{cases}
        1 & \text{si } k = n\\
        x(k) & \text{si } k \neq n
    \end{cases}
\end{equation*}

Tenemos que $\sup\{|x_n(k)| : k \in \bb{N}\} = 1 < \infty$, luego $x_n \in \ell_\infty$. Además, como $|x_n(n)| = 1$, se tiene que $x_n \notin V$. Veamos que $\{x_n\}$ converge a $x$ en $\ell_\infty$.
\begin{equation*}
    \|x_n - x\|_\infty = \sup_{k \in \bb{N}} |x_n(k) - x(k)| = \sup_{k \in \bb{N}} \left|\begin{cases}
        1 - x(n) & \text{si } k = n\\
        0 & \text{si } k \neq n
    \end{cases}\right| = |1 - x(n)| = \left|1 - \left(1 - \frac{1}{n}\right)\right| = \frac{1}{n}
\end{equation*}

Por tanto, $\|x_n - x\|_\infty \to 0$, luego $\{x_n\}$ converge a $x$ en $\ell_\infty$. Por tanto, $\ell_\infty \setminus V$ no es cerrado, y por tanto $V$ no es abierto en $\ell_\infty$.
\end{ejercicio}

\begin{ejercicio}
Sean $X$ e $Y$ espacios normados y $T: X \longrightarrow Y$ un operador lineal. Supongamos que, para toda sucesión $\{x_n\}$ de vectores de $X$ tal que $\{x_n\} \to 0$, se tiene que la sucesión $\{T(x_n)\}$ está acotada. Probar que $T$ es continuo.
\begin{description}
    \item[Opción 1] Demostramos el contrarrecíproco. Supongamos que $T$ no es continuo. Entonces, por la caracterización de la continuidad de los operadores lineales, para todo $n \in \bb{N}$, existe $v_n \in X$ tal que:
    \begin{equation*}
        \|T(v_n)\| > n \|v_n\|
    \end{equation*}
    En particular, tenemos que $v_n \neq 0$. Construimos la sucesión $\{x_n\}$ definida por:
    \begin{equation*}
        x_n = \frac{v_n}{\sqrt{n}\|v_n\|}
    \end{equation*}

    Veamos que $\{x_n\} \to 0$. En efecto, para todo $n \in \bb{N}$, se tiene que:
    \begin{equation*}
        \|x_n\| = \left\|\frac{v_n}{\sqrt{n}\|v_n\|}\right\| = \frac{1}{\sqrt{n}} \to 0
    \end{equation*}

    Veamos que $\{T(x_n)\}$ no está acotada. En efecto, para todo $n \in \bb{N}$, se tiene que:
    \begin{equation*}
        \|T(x_n)\| = \left\|T\left(\frac{v_n}{\sqrt{n}\|v_n\|}\right)\right\| = \frac{1}{\sqrt{n}\|v_n\|} \|T(v_n)\| > \frac{1}{\sqrt{n}\|v_n\|} \cdot n \|v_n\| = \dfrac{n}{\sqrt{n}} = \sqrt{n}
    \end{equation*}
    Por tanto, $\{T(x_n)\}$ no está acotada. Queda demostrado el contrarrecíproco, y por tanto, $T$ es continuo.
    
    \item[Opción 2] Por la caracterización de la continuidad de los operadores lineales, basta con ver que $T$ es continuo en $0$. Sea $\{x_n\}$ una sucesión de vectores de $X$ tal que $\{x_n\} \to 0$, y queremos ver que $\{T(x_n)\} \to 0$. Consideramos la sucesión $\{y_n\}$ definida por:
    \begin{equation*}
        y_n = \begin{cases}
            \dfrac{x_n}{\|x_n\|} \cdot \sqrt{\|x_n\|} & \text{si } x_n \neq 0\\
            0 & \text{si } x_n = 0
        \end{cases}
    \end{equation*}

    Veamos que $\{y_n\} \to 0$. Si $x_n = 0$, entonces $y_n = 0$ y $\|y_n\| = 0$. Si $x_n \neq 0$, entonces:
    \begin{equation*}
        \|y_n\| = \left\|\frac{x_n}{\|x_n\|} \cdot \sqrt{\|x_n\|}\right\| = \sqrt{\|x_n\|} \to 0
    \end{equation*}
    donde lo último se cumple porque $\{x_n\} \to 0$ y la norma y la raíz cuadrada son funciones continuas.
    Por tanto, se tiene que $\{y_n\} \to 0$. Por hipótesis, la sucesión $\{T(y_n)\}$ está acotada; es decir, existe $M \in \bb{R}^+$ tal que $\|T(y_n)\| \leq M$ para todo $n \in \bb{N}$. Veamos que $\{T(x_n)\} \to 0$. En efecto, si $x_n = 0$, entonces $T(x_n) = 0$ y $\|T(x_n)\| = 0$. Si $x_n \neq 0$, entonces:
    \begin{align*}
        \|T(x_n)\| &= \left\|T\left(\frac{x_n}{\|x_n\|} \cdot \|x_n\|\right)\right\| = \left\|T\left(\frac{x_n}{\|x_n\|} \cdot \sqrt{\|x_n\|}\right)\right\| \cdot \sqrt{\|x_n\|} =\\&= \|T(y_n)\| \cdot \sqrt{\|x_n\|} \leq M \cdot \sqrt{\|x_n\|}\to 0
    \end{align*}

    Por tanto, $\{T(x_n)\} \to 0$, y por tanto $T$ es continuo.
\end{description}
\end{ejercicio}

\begin{ejercicio}
En cada uno de los siguientes casos, probar que la aplicación $f \mapsto T(f)$ es un operador lineal continuo de $C\left[0, 1\right]$ en sí mismo, y calcular su norma:
\begin{enumerate}
    \item $\left[T(f)\right](t) = \int_{0}^t f(s)\,ds \quad \forall t \in \left[0, 1\right], \; \forall f \in C\left[0, 1\right]$
    
    En primer lugar, hemos de ver que está bien definida; es decir, que se tiene $T(f) \in C\left[0, 1\right]$ para todo $f \in C\left[0, 1\right]$. En efecto, como $f \in C\left[0, 1\right]$, se tiene que $f$ es integrable en $\left[0, 1\right]$, y por tanto $\int_{0}^t f(s)\,ds$ está bien definida para todo $t \in \left[0, 1\right]$. Además, por el Teorema Fundamental del Cálculo, se tiene que $T(f)$ es continua en $\left[0, 1\right]$. Por tanto, $T(f) \in C\left[0, 1\right]$ para todo $f \in C\left[0, 1\right]$, y por tanto $T$ está bien definida.

    Veamos ahora que $T$ es lineal. Sean $f, g \in C\left[0, 1\right]$ y $\alpha, \beta \in \bb{K}$. Entonces, para todo $t \in \left[0, 1\right]$, se tiene que:
    \begin{align*}
        \left[T(\alpha f + \beta g)\right](t) &= \int_{0}^t (\alpha f(s) + \beta g(s))\,ds = \alpha \int_{0}^t f(s)\,ds + \beta \int_{0}^t g(s)\,ds =\\&= \alpha \left[T(f)\right](t) + \beta \left[T(g)\right](t)
    \end{align*}
    Por tanto, $T$ es lineal.

    Veamos ahora que $T$ es continuo. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T(f)\|_\infty \leq M \|f\|_\infty$ para todo $f \in C\left[0, 1\right]$. En efecto, para todo $f \in C\left[0, 1\right]$, se tiene que:
    \begin{align*}
        \|T(f)\|_\infty &= \max_{t \in \left[0, 1\right]} \left|\int_{0}^t f(s)\,ds\right| \leq \max_{t \in \left[0, 1\right]} \int_{0}^t |f(s)|\,ds \leq \max_{t \in \left[0, 1\right]} \int_{0}^t \|f\|_\infty\,ds = \max_{t \in \left[0, 1\right]} \|f\|_\infty t = \|f\|_\infty
    \end{align*}

    Por tanto, $T$ es continuo y $\|T\| \leq 1$.

    Para ver que $\|T\| = 1$, aplicamos la caracterización de la norma de un operador lineal continuo:
    \begin{equation*}
        \|T\| = \sup_{f \in C\left[0, 1\right], f \neq 0} \frac{\|T(f)\|_\infty}{\|f\|_\infty}
    \end{equation*}
    
    
    Buscamos por tanto $f \in C\left[0, 1\right]$ tal que $\|T(f)\|_\infty = \|f\|_\infty$. En efecto, si tomamos $f(t) = 1$ para todo $t \in \left[0, 1\right]$, se tiene que:
    \begin{equation*}
        \|T(f)\|_\infty = \max_{t \in \left[0, 1\right]} \left|\int_{0}^t 1\,ds\right| = \max_{t \in \left[0, 1\right]} t = 1 = \|f\|_\infty
    \end{equation*}
    Por tanto, $\|T\| \geq 1$, y por tanto $\|T\| = 1$. Notemos además que, en este caso, la norma se alcanza.
    \item $\left[T(f)\right](t) = t^2 f(0) \quad \forall t \in \left[0, 1\right], \; \forall f \in C\left[0, 1\right]$

    \begin{align*}
        \left[T(\alpha f + \beta g)\right](t) &= t^2 (\alpha f(0) + \beta g(0)) = \alpha t^2 f(0) + \beta t^2 g(0) = \alpha \left[T(f)\right](t) + \beta \left[T(g)\right](t)
    \end{align*}
    Por tanto, $T$ es lineal.

    \begin{align*}
        \|T(f)\|_\infty &= \max_{t \in \left[0, 1\right]} |t^2 f(0)| = \max_{t \in \left[0, 1\right]} t^2 |f(0)| = |f(0)| \leq \|f\|_\infty
    \end{align*}
    Por tanto, $T$ es continuo y $\|T\| \leq 1$.

    Por tanto, $\|T\| = 1$ y la norma se alcanza.

    \begin{align*}
        \left[T(\alpha f + \beta g)\right](t) &= (\alpha f + \beta g)(t^2) = \alpha f(t^2) + \beta g(t^2) = \alpha \left[T(f)\right](t) + \beta \left[T(g)\right](t)
    \end{align*}
    Por tanto, $T$ es lineal.

    Veamos ahora que $T$ es continuo. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T(f)\|_\infty \leq M \|f\|_\infty$ para todo $f \in C\left[0, 1\right]$. En efecto, para todo $f \in C\left[0, 1\right]$, se tiene que:
    \begin{align*}
        \|T(f)\|_\infty &= \max_{t \in \left[0, 1\right]} |f(t^2)| = \|f\|_\infty
    \end{align*}
    Por tanto, $T$ es continuo y $\|T\|\leq 1$.

        \|T(f)\|_\infty = \max_{t \in \left[0, 1\right]} |f(t^2)| = \max_{t \in \left[0, 1\right]} 1 = 1 = \|f\|_\infty
    \end{equation*}
    Por tanto, $\|T\| = 1$ y la norma se alcanza.

\begin{ejercicio}
Fijada $g \in L_\infty\left[0, 1\right]$, probar que, para $1 \leq p \leq \infty$, la aplicación $f \mapsto fg$ es un operador lineal continuo de $L_p\left[0, 1\right]$ en sí mismo, y calcular su norma.\\

Como $g \in L_\infty\left[0, 1\right]$, se tiene que $g\in L_p\left[0, 1\right]$ para todo $1 \leq p \leq \infty$. Por tanto, para todo $f \in L_p\left[0, 1\right]$, se tiene que $fg \in L_p\left[0, 1\right]$, y por tanto la aplicación está bien definida. Además, es lineal, ya que para todo $f_1, f_2 \in L_p\left[0, 1\right]$ y todo $\alpha, \beta \in \bb{K}$, se tiene que:
\begin{equation*}
    [T(\alpha f_1 + \beta f_2)](t) = (\alpha f_1 + \beta f_2)(t) g(t) = \alpha f_1(t) g(t) + \beta f_2(t) g(t) = \alpha [T(f_1)](t) + \beta [T(f_2)](t)
\end{equation*}

Veamos ahora que $T$ es continuo. Distinguimos casos:
\begin{itemize}
    \item Caso $p < \infty$. Tenemos que:
    \begin{align*}
        \|T(f)\|_p^p &= \int_0^1 |f(t) g(t)|^p dt
        = \int_0^1 |f(t)|^p |g(t)|^p dt \leq \int_0^1 |f(t)|^p \|g\|_\infty^p dt =\\&= \|g\|_\infty^p \int_0^1 |f(t)|^p dt = \|g\|_\infty^p \|f\|_p^p
    \end{align*}
    Por tanto, tomando raíz $p$-ésima, se tiene que:
    \begin{equation*}
        \|T(f)\|_p \leq \|g\|_\infty \|f\|_p
    \end{equation*}
    Por tanto, $T$ es continuo y $\|T\| \leq \|g\|_\infty$.
    \item Caso $p = \infty$. Tenemos que:
    \begin{align*}
        \|T(f)\|_\infty &= \esssup_{t \in [0, 1]} |f(t) g(t)| \leq \esssup_{t \in [0, 1]} |f(t)|\|g\|_\infty = \|f\|_\infty \|g\|_\infty
    \end{align*}
    Por tanto, $T$ es continuo y $\|T\| \leq \|g\|_\infty$.
\end{itemize}

En cualquier caso, $T$ es continuo y $\|T\| \leq \|g\|_\infty$. Veamos ahora que $\|T\| = \|g\|_\infty$. Para ello, basta con ver que existe $f \in L_p\left[0, 1\right]$ tal que $\|T(f)\|_p = \|g\|_\infty \|f\|_p$.
\begin{itemize}
    \item Si $\|g\|_\infty = 0$, entonces $g = 0$ c.p.d., y por tanto $T(f) = fg = 0$ c.p.d. para todo $f \in L_p\left[0, 1\right]$. Por tanto, $\|T(f)\|_p = 0 = \|g\|_\infty \|f\|_p$ para todo $f \in L_p\left[0, 1\right]$.
    \item Si $\|g\|_\infty > 0$ y $p<\infty$, para cada $\veps\in \bb{R}^+$ tal que $0<\veps < \|g\|_\infty$, por la definición de supremo esencial existe un conjunto medible $C_\veps \subset [0, 1]$ tal que:
    \begin{equation*}
        |g(t)| > \|g\|_\infty - \veps \quad \forall t \in C_\veps
    \end{equation*}

    Con el objetivo de conseguir la igualdad, buscamos un $f \in L_p\left[0, 1\right]$ tal que el cociente siguiente sea lo grande posible (sabiendo que está acotado por $\|g\|_\infty$):
    \begin{equation*}
        \frac{\|T(f)\|_p}{\|f\|_p} = \frac{\|fg\|_p}{\|f\|_p}
        = \dfrac{\displaystyle\left(\int_{0}^1 |f(t) g(t)|^p dt\right)^{\nicefrac{1}{p}}}{\displaystyle\left(\int_{0}^1 |f(t)|^p dt\right)^{\nicefrac{1}{p}}}
        = \left(\frac{\displaystyle\int_{0}^1 |f(t) g(t)|^p dt}{\displaystyle\int_{0}^1 |f(t)|^p dt}\right)^{\nicefrac{1}{p}}\leq \|g\|_\infty
    \end{equation*}

    Con el objetivo de maximizar el cociente, buscamos que $f$ tan solo tome valores distintos de cero cuando $g$ sea lo más grande posible, es decir, cuando $|g(t)| > \|g\|_\infty - \veps$. Por tanto, buscamos que $f$ sea distinto de cero solo en el conjunto $C_\veps$. Por tanto, empleando la función característica de $C_\veps$, tenemos:
    \Func{f_{\veps}=\chi_{C_\veps}}{[0, 1]}{\bb{K}}{t}{\chi_{C_\veps}(t)}

    Calculemos dicha norma:
    \begin{equation*}
        \|\chi_{C_\veps}\|_p = \left(\int_{0}^1 |\chi_{C_\veps}(t)|^p dt\right)^{\nicefrac{1}{p}} = \left(\int_{C_\veps} 1 dt\right)^{\nicefrac{1}{p}} = \left(\lm(C_\veps)\right)^{\nicefrac{1}{p}}
    \end{equation*}

    Buscamos ahora $\|T(f_\veps)\|_p$:
    \begin{align*}
        \|T(f_\veps)\|_p^p &= \int_{0}^1 |f_\veps(t) g(t)|^p dt = \int_{0}^1 \left|\chi_{C_\veps}(t) g(t)\right|^p dt = \int_{0}^1 |\chi_{C_\veps}(t)|^p |g(t)|^p dt =\\&=  \int_{C_\veps} |g(t)|^p dt \geq \int_{C_\veps} (\|g\|_\infty - \veps)^p dt = (\|g\|_\infty - \veps)^p \lm(C_\veps)
    \end{align*}

    Tomando raíz $p$-ésima, se tiene que:
    \begin{equation*}
        \|T(f_\veps)\|_p \geq (\|g\|_\infty - \veps) \left(\lm(C_\veps)\right)^{\nicefrac{1}{p}}
        = (\|g\|_\infty - \veps) \|\chi_{C_\veps}\|_p = (\|g\|_\infty - \veps) \|f_\veps\|_p
    \end{equation*}

    Por tanto, tenemos que:
    \begin{equation*}
        \|g\|_\infty - \veps \leq \|T\| \leq \|g\|_\infty
    \end{equation*}
    Tomando límite cuando $\veps \to 0$, por el Lema del Sandwich, se tiene que:
    \begin{equation*}
        \|T\| = \|g\|_\infty
    \end{equation*}

    \item Si $\|g\|_\infty > 0$ y $p=\infty$, consideramos la función $f(t) = 1$ para todo $t \in [0, 1]$. Entonces, $\|f\|_\infty = 1$, y tenemos que:
    \begin{equation*}
        \|T(f)\|_\infty = \esssup_{t \in [0, 1]} |f(t) g(t)| = \esssup_{t \in [0, 1]} |g(t)| = \|g\|_\infty
        = \|g\|_\infty \|f\|_\infty
    \end{equation*}
    Por tanto, $\|T\| = \|g\|_\infty$.
\end{itemize}
\end{ejercicio}

\begin{ejercicio}
Sea $\{e_n\}$ la base de vectores unidad de $c_0$. Probar que, para $f \in c_0^*$, las siguientes afirmaciones son equivalentes:
\begin{enumerate}
    \item La función $|f|$ tiene máximo en la esfera unidad de $c_0$.
    \item Existe $m \in \bb{N}$ tal que $f(e_n) = 0$ para todo $n \in \bb{N}$ con $n \geq m$.
\end{enumerate}
Dar un ejemplo en el que no se verifiquen estas afirmaciones.
\begin{proof}
    Dado $f\in c_0^*$, por ser este isométricamente isomorfo a $\ell_1$, existe una sucesión $\{y_n\} \in \ell_1$ tal que:
    \begin{equation*}
        f(x) = \sum_{n=1}^{\infty} x_n y_n \quad \forall x \in c_0
    \end{equation*}

    Evaluando en los vectores de la base, se tiene que:
    \begin{equation*}
        f(e_n) = \sum_{k=1}^{\infty} (e_n)_k y_k = y_n \quad \forall n \in \bb{N}
    \end{equation*}

    Por tanto, la sucesión $\{y_n\}$ es precisamente la sucesión de valores de $f$ en los vectores de la base, de forma que:
    \begin{equation*}
        f(x) = \sum_{n=1}^{\infty} x_n f(e_n) \quad \forall x \in c_0
    \end{equation*}

    \begin{description}

        \item[$(1) \implies (2)$] Supongamos que $|f|$ tiene máximo en la esfera unidad de $c_0$; es decir, existe $x_0 \in c_0$ con $\|x_0\|_\infty = 1$ tal que:
        \begin{equation*}
            |f(x_0)| = \sup\{|f(x)| : x \in c_0, \|x\|_\infty = 1\} = \|f\|
        \end{equation*}

        Como $x_0 \in c_0$ y $\{e_n\}$ es una base de Schauder de $c_0$, se tiene que:
        \begin{equation*}
            x_0 = \sum_{n=1}^{\infty} x_0(n) e_n
        \end{equation*}

        Por tanto, $|x_0(n)|\leq \|x_0\|_\infty = 1$ para todo $n \in \bb{N}$. Además, como $x_0 \in c_0$, se tiene que $\{x_0\}\to 0$. Por tanto, tomando $\veps=\nicefrac{1}{2}$, existe $m \in \bb{N}$ tal que:
        \begin{equation*}
            |x_0(n)| \leq \nicefrac{1}{2} \quad \forall n \in \bb{N}, n \geq m
        \end{equation*}

        Por tanto, tenemos que:
        \begin{align*}
            \|f\| &= |f(x_0)| = \left|\sum_{n=1}^{\infty} x_0(n) f(e_n)\right| \leq \sum_{n=1}^{\infty} |x_0(n)| |f(e_n)|
            =\\&= \sum_{n=1}^{m-1} |x_0(n)| |f(e_n)| + \sum_{n=m}^{\infty} |x_0(n)| |f(e_n)|
        \end{align*}
        \begin{itemize}
            \item Para la primera suma, tenemos que $|x_0(n)| \leq 1$ para todo $n\in \bb{N}$, y por tanto:
            \begin{equation*}
                \sum_{n=1}^{m-1} |x_0(n)| |f(e_n)| \leq \sum_{n=1}^{m-1} |f(e_n)|
            \end{equation*}
            \item Para la segunda suma, tenemos que $|x_0(n)| \leq \nicefrac{1}{2}$ para todo $n \geq m$, y por tanto:
            \begin{equation*}
                \sum_{n=m}^{\infty} |x_0(n)| |f(e_n)| \leq \frac{1}{2} \sum_{n=m}^{\infty} |f(e_n)|
            \end{equation*}
            
            Por reducción al absurdo, supongamos que $f(e_n) \neq 0$ para algún $n \geq m$. Entonces, se tendría que:
            \begin{equation*}
                \sum_{n=m}^{\infty} |f(e_n)| > 0
            \end{equation*}
            Por tanto, se tendría que:
            \begin{equation*}
                \sum_{n=m}^{\infty} |x_0(n)| |f(e_n)| < \sum_{n=m}^{\infty} |f(e_n)|
            \end{equation*}
        \end{itemize}

        Uniendo ambas desigualdades, se tiene que:
        \begin{align*}
            \|f\| &< \sum_{n=1}^{m-1} |f(e_n)| + \sum_{n=m}^{\infty} |f(e_n)| = \sum_{n=1}^{\infty} |f(e_n)| = \|f\|
        \end{align*}

        Esto es una contradicción, y por tanto $f(e_n) = 0$ para todo $n \in \bb{N}$ con $n \geq m$.

        \item[$(2) \implies (1)$] Supongamos que existe $m \in \bb{N}$ tal que $f(e_n) = 0$ para todo $n \in \bb{N}$ con $n \geq m$. Entonces, para todo $x \in c_0$, se tiene que:
        \begin{equation*}
            f(x) = \sum_{k=1}^{\infty} x_k f(e_k) = \sum_{k=1}^{m-1} x_k f(e_k)
        \end{equation*}

        Calculamos la norma de $f$:
        \begin{align*}
            \|f\| &= \|f(e_n)\|_1 = \sum_{k=1}^{\infty} |f(e_k)| = \sum_{k=1}^{m-1} |f(e_k)|
        \end{align*}

        Buscamos acotar $|f(x)|$ para todo $x \in c_0$ con $\|x\|_\infty = 1$. En efecto, para todo $x \in c_0$ con $\|x\|_\infty = 1$, se tiene que:
        \begin{align*}
            |f(x)| &= \left|\sum_{k=1}^{m-1} x_k f(e_k)\right| \leq \sum_{k=1}^{m-1} |x_k| |f(e_k)| \leq \sum_{k=1}^{m-1} |f(e_k)| = \|f\|
        \end{align*}

        Por tanto, $|f|$ tiene supremo en la esfera unidad de $c_0$. Con el objetivo de ver que alcanza su máximo, buscamos un $x \in c_0$ con $\|x\|_\infty = 1$ tal que:
        \begin{itemize}
            \item $\sum\limits_{k=1}^{m-1} |x_k| |f(e_k)| = \sum\limits_{k=1}^{m-1} |f(e_k)|$. Para ello, basta con tomar $|x_k| = 1$ para todo $k = 1, \ldots, m-1$.
            \item $\left|\sum\limits_{k=1}^{m-1} x_k f(e_k)\right| = \sum\limits_{k=1}^{m-1} |x_k| |f(e_k)|$. Para ello, basta con que $x_kf(e_k) = |x_k f(e_k)|$ para todo $k = 1, \ldots, m-1$.
        \end{itemize}

        Uniendo ambas condiciones, una primera aproximación de $x$ que cumple con ellas es:
        \begin{equation*}
            x_k = \dfrac{|f(e_k)||x_k|}{f(e_k)} = \dfrac{|f(e_k)|}{f(e_k)} \quad \forall k\in \bb{N}
        \end{equation*}

        No obstante, esto tiene dos problemas. Por un lado, podría ser $f(e_k)=0$, y por otro lado no tenemos asegurado que $x \in c_0$. Consideramos por tanto:
        \begin{equation*}
            x(k) = \begin{cases}
                \dfrac{|f(e_k)|}{f(e_k)} & \text{si } f(e_k) \neq 0 \text{ y } k < m\\
                0 & \text{en otro caso}
            \end{cases}
        \end{equation*}

        Es directo ver que $x \in c_0$ y $\|x\|_\infty = 1$. Además, se tiene que:
        \begin{align*}
            |f(x)| &= \left|\sum_{k=1}^{m-1} x_k f(e_k)\right| = \left|\sum_{k=1}^{m-1} \dfrac{|f(e_k)|}{f(e_k)} f(e_k)\right| = \sum_{k=1}^{m-1} |f(e_k)| = \|f\|
        \end{align*}

        Por tanto, $|f|$ alcanza su máximo en la esfera unidad de $c_0$.
    \end{description}

\end{proof}
\end{ejercicio}

\begin{ejercicio}
Encontrar un operador lineal acotado de $\ell_2$ en otro espacio de Banach que no alcance su norma (esto es, que el supremo que define su norma no sea un máximo).\\

Consideramos el operador lineal $T: \ell_2 \longrightarrow \ell_2$ definido por:
    \begin{equation*}
        T(x)(n) = \dfrac{n}{n+1} x(n) \quad \forall n \in \bb{N}, \; \forall x \in \ell_2
    \end{equation*}

    Es directo ver que $T$ es lineal. Veamos que es acotado. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T(x)\|_2 \leq M \|x\|_2$ para todo $x \in \ell_2$. En efecto, para todo $x \in \ell_2$, se tiene que:
    \begin{align*}
        \|T(x)\|_2^2 &= \sum_{n=1}^{\infty} |T(x)(n)|^2 = \sum_{n=1}^{\infty} \left|\dfrac{n}{n+1} x(n)\right|^2 = \sum_{n=1}^{\infty} \left(\dfrac{n}{n+1}\right)^2 |x(n)|^2 \leq \sum_{n=1}^{\infty} |x(n)|^2 = \|x\|_2^2
    \end{align*}
    Por tanto, $T$ es acotado y $\|T\| \leq 1$. Veamos ahora que $\|T\| = 1$. Para ello, si tomamos $x=e_n$ para algún $n \in \bb{N}$, se tiene que:
    \begin{equation*}
        \|T(e_n)\|_2 = \left\|\dfrac{n}{n+1} e_n\right\|_2 = \dfrac{n}{n+1} \|e_n\|_2
    \end{equation*}

    Por tanto, por definición de supremo, se tiene que:
    \begin{equation*}
        \|T\|\geq \dfrac{n}{n+1} \quad \forall n \in \bb{N}
    \end{equation*}
    Tomando límite cuando $n \to \infty$, se tiene que:
    \begin{equation*}
        \|T\| \geq 1
    \end{equation*}
    Por tanto, $\|T\| = 1$. No obstante, supongamos que existe $x \in \ell_2$ con $\|x\|_2 = 1$ tal que $\|T(x)\|_2 = 1$. Entonces, se tendría que:
    \begin{align*}
        1 = \|T(x)\|_2^2 &= \sum_{n=1}^{\infty} |T(x)(n)|^2 = \sum_{n=1}^{\infty} \left|\dfrac{n}{n+1} x(n)\right|^2 = \sum_{n=1}^{\infty} \left(\dfrac{n}{n+1}\right)^2 |x(n)|^2 
        \leq\\&\leq
        \sum_{n=1}^{\infty} |x(n)|^2 = \|x\|_2^2 = 1
    \end{align*}
    Esto implica que la desigualdad es una igualdad, y al tratarse de series convergentes, se tiene que:
    \begin{equation*}
        \sum_{n=1}^{\infty} |x(n)|^2 - \sum_{n=1}^{\infty} \left(\dfrac{n}{n+1}\right)^2 |x(n)|^2 = \sum_{n=1}^{\infty} \left(1 - \left(\dfrac{n}{n+1}\right)^2\right) |x(n)|^2 = 0
    \end{equation*}

    Por tanto, al ser la serie de términos no negativos que suma $0$, se tiene que cada término es $0$, es decir:
    \begin{itemize}
        \item $|x(n)| = 0$ para todo $n \in \bb{N}$. En este caso, $x = 0$, lo que contradice que $\|x\|_2 = 1$.
        \item $\dfrac{n}{n+1} = 1$ para todo $n \in \bb{N}$. Esto es imposible.
    \end{itemize}

    Por tanto, no existe $x \in \ell_2$ con $\|x\|_2 = 1$ tal que $\|T(x)\|_2 = 1$. Por tanto, $T$ no alcanza su norma.
\end{ejercicio}

        \varphi(f) &= \int_0^{\nicefrac{1}{2}} f(t)\,dt - \int_{\nicefrac{1}{2}}^1 f(t)\,dt = \int_0^{\nicefrac{1}{2}} f(t)\,dt + \int_{\nicefrac{1}{2}}^1 -f(t)\,dt = \int_0^1 f(t) h(t)\,dt
    \end{align*}
    Veamos ahora entonces que $\varphi$ es continuo. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $|\varphi(f)| \leq M \|f\|_\infty$ para todo $f \in C[0, 1]$. En efecto, para todo $f \in C[0, 1]$, se tiene que:
    \begin{align*}
        |\varphi(f)| &= \left|\int_0^1 f(t) h(t)\,dt\right| \leq \int_0^1 |f(t) h(t)|\,dt = \int_0^1 |f(t)| \, dt \leq \int_0^1 \|f\|_\infty\,dt = \|f\|_\infty
    \end{align*}
    Por tanto, $\varphi$ es continuo y $\|\varphi\| \leq 1$. Veamos ahora que $\|\varphi\| = 1$. Consideramos para cada $n \in \bb{N}$, $n\geq 3$, la función $f_n \in C[0, 1]$ definida por:
    \[f_n(t) = \begin{cases}
        1 & \text{si } t \in [0, \nicefrac{1}{2} - \nicefrac{1}{n}]\\
        n\left(\nicefrac{1}{2} - t\right) & \text{si } t \in [\nicefrac{1}{2} - \nicefrac{1}{n}, \nicefrac{1}{2}]\\
        -n\left(t - \nicefrac{1}{2}\right) & \text{si } t \in [\nicefrac{1}{2}, \nicefrac{1}{2} + \nicefrac{1}{n}]\\
        -1 & \text{si } t \in [\nicefrac{1}{2} + \nicefrac{1}{n}, 1]
    \end{cases}\]

    Estas funciones se muestran en la Figura~\ref{fig:fn2}.

    Es directo que $f_n \in C[0, 1]$. Calculemos su norma:
    \begin{equation*}
        \|f_n\|_\infty = \max_{t \in [0, 1]} |f_n(t)| = 1
    \end{equation*}

    Por comodidad, notamos por $I_n$ a:
    \begin{equation*}
        I_n = \left[\frac{1}{2} - \frac{1}{n}, \frac{1}{2} + \frac{1}{n}\right]
    \end{equation*}

    Calculemos ahora $|\varphi(f_n)|$:
    \begin{align*}
        |\varphi(f_n)| &= \left|\int_0^1 f_n(t) h(t)\,dt\right| = \left|\int_{[0, 1] \setminus I_n} f_n(t) h(t)\,dt + \int_{I_n} f_n(t) h(t)\,dt\right| =\\&= \left|\int_{[0, 1] \setminus I_n} 1 \,dt + \int_{I_n} f_n(t) h(t)\,dt\right| \geq 1-\lm(I_n)-\left|\int_{I_n} f_n(t) h(t)\,dt\right|\geq 1-2\lm(I_n) = 1 - \frac{4}{n}
    \end{align*}

    Por tanto, tenemos que:
    \begin{equation*}
        \|\varphi\|\geq 1-\frac{4}{n} \quad \forall n \in \bb{N}, n\geq 3
    \end{equation*}
    Tomando límite cuando $n \to \infty$, se tiene que:
    \begin{equation*}
        \|\varphi\| \geq 1
    \end{equation*}
    Por tanto, $\|\varphi\| = 1$.\\

    Veamos ahora si se alcanza la norma. Para ello, supongamos que existe $f \in C[0, 1]$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = \|\varphi\| = 1$. Multiplicando por el escalar $\lm = \frac{1}{\varphi(f)}$ (que cumple que $|\lm| = \frac{1}{|\varphi(f)|} = 1$), la función $g=\lm f$ cumple que $\|g\|_\infty = \|f\|_\infty = 1$ y $\varphi(g) = \lm \varphi(f) = 1$. Por tanto, sin pérdida de generalidad, podemos suponer que $\varphi(f) = 1$. Entonces, tenemos que:
    \begin{equation*}
        \varphi(f) = \int_0^{\nicefrac{1}{2}} f(t)\,dt - \int_{\nicefrac{1}{2}}^1 f(t)\,dt = 1
    \end{equation*}

    Tomando parte real, se tiene que:
    \begin{equation*}
        \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt - \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt = 1
    \end{equation*}

    Como $f$ es continua, $\Re(f): [0, 1] \longrightarrow \bb{R}$ es continua. Además, como $\|f\|_\infty = 1$, se tiene que $|\Re(f(t))| \leq |f(t)| \leq 1$ para todo $t \in [0, 1]$. Por tanto, tenemos que:
    \begin{equation*}
        -1 \leq \Re(f(t)) \leq 1 \quad \forall t \in [0, 1]
    \end{equation*}

    Consideramos ahora $\Re(f)(\nicefrac{1}{2})\in [-1, 1]$.
    \begin{itemize}
        \item Si $\Re(f)(\nicefrac{1}{2}) < 1$, entonces por continuidad $\exists \delta \in \bb{R}^+$ tal que $\Re(f)(t) < 1$ para todo $t \in [\nicefrac{1}{2} - \delta, \nicefrac{1}{2}]$ (conjunto de medida no nula). Por tanto, tenemos que:
        \begin{equation*}
            \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt < \frac{1}{2}
        \end{equation*}

        Por otro lado, tenemos que:
        \begin{equation*}
            \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt \geq -\frac{1}{2}
        \end{equation*}

        Uniendo ambas desigualdades, se tiene que:
        \begin{equation*}
            1 = \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt - \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt < \frac{1}{2} - \left(-\frac{1}{2}\right) = 1
        \end{equation*}

        Por tanto, $\Re(f)(\nicefrac{1}{2}) < 1$ no es posible.
        \item Si $\Re(f)(\nicefrac{1}{2}) > -1$, entonces por continuidad $\exists \delta \in \bb{R}^+$ tal que $\Re(f)(t) > -1$ para todo $t \in [\nicefrac{1}{2}, \nicefrac{1}{2} + \delta]$. Por tanto, tenemos que:
        \begin{equation*}
            \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt > -\frac{1}{2}
        \end{equation*}

        Por otro lado, tenemos que:
        \begin{equation*}
            \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt \leq \frac{1}{2}
        \end{equation*}

        Uniendo ambas desigualdades, se tiene que:
        \begin{equation*}
            1 = \int_0^{\nicefrac{1}{2}} \Re(f(t))\,dt - \int_{\nicefrac{1}{2}}^1 \Re(f(t))\,dt < \frac{1}{2} - \left(-\frac{1}{2}\right) = 1
        \end{equation*}

        Por tanto, $\Re(f)(\nicefrac{1}{2}) > -1$ no es posible.
    \end{itemize}

    Por tanto, hemos llegado a que $\Re(f)(\nicefrac{1}{2})\geq 1$ y $\Re(f)(\nicefrac{1}{2})\leq -1$, lo cual es imposible. Por tanto, no existe $f \in C[0, 1]$ con $\|f\|_\infty = 1$ tal que $|\varphi(f)| = \|\varphi\| = 1$. En otras palabras, $\varphi$ no alcanza su norma.

\end{ejercicio}

\newpage\subsection{Operaciones con espacios normados}

\begin{ejercicio}
Dado un espacio normado $X$, sean $f, g \in X^* \setminus \{0\}$. Probar que los espacios normados $\ker f$ y $\ker g$, ambos con la norma inducida por $X$, son isomorfos. Dar un ejemplo en el que dichos espacios no sean isométricamente isomorfos.\\

    \begin{description}
        \item[Opción 1: Sin dar el isomorfismo]~\\
        
        Si $\ker f = \ker g$, entonces $\ker f \cong \ker g$. Por tanto, podemos suponer que se tiene $\ker f \neq \ker g$. Buscamos un subespacio $M$ cerrado de $X$ de forma que se tenga $X = \ker f \oplus M = \ker g \oplus M$. Si encontramos un subespacio así, entonces:
    \begin{equation*}
        \ker f \cong X / M \cong \ker g
    \end{equation*}

    En primer lugar, buscamos $u\in X$ tal que $f(u)\neq 0$ y $g(u)\neq 0$. Por reducción al absurdo, supongamos que no existe tal $u$. Entonces, para todo $u \in X$, se tendría que $f(u) = 0$ o $g(u) = 0$. En particular, se tendría:
    \begin{equation*}
        X= \ker f \cup \ker g
    \end{equation*}
    \begin{itemize}
        \item Supongamos $\ker f \subseteq \ker g$. Entonces $X= \ker g$, lo que contradice que $g \neq 0$.
        \item Supongamos $\ker g \subseteq \ker f$. Entonces $X= \ker f$, lo que contradice que $f \neq 0$.
        \item Supongamos que $\ker f \not\subseteq \ker g$ y $\ker g \not\subseteq \ker f$. Entonces, existen $u_1 \in \ker f \setminus \ker g$ y $u_2 \in \ker g \setminus \ker f$. Consideramos $u = u_1 + u_2$. Entonces, se tiene que:
        \begin{align*}
            f(u) &= f(u_1 + u_2) = f(u_1) + f(u_2) = 0 + f(u_2) \neq 0\\
            g(u) &= g(u_1 + u_2) = g(u_1) + g(u_2) = g(u_1) + 0 \neq 0
        \end{align*}

        Por tanto, $u\in X\setminus (\ker f \cup \ker g)$, lo que contradice nuestra hipótesis.
    \end{itemize}

    En conclusión, existe $u \in X$ tal que $f(u)\neq 0$ y $g(u)\neq 0$. Para todo $x\in X$, buscamos una descomposición de $x$ en un elemento de $\ker f$ y otro elemento:
    \begin{equation*}
        x = x - \dfrac{f(x)}{f(u)} u + \dfrac{f(x)}{f(u)} u\qquad \forall x \in X
    \end{equation*}
    \begin{itemize}
        \item $x - \dfrac{f(x)}{f(u)} u \in \ker f$.
        \item $\dfrac{f(x)}{f(u)} u \in \cc{L}(u)$.
    \end{itemize}

    Por tanto, cabe preguntarse si $X = \ker f \oplus \cc{L}(u)$. Aunque demostrar que se trata de una suma directa algebraica es sencillo, puesto que hemos de demostrar que se trata de una suma directa topológica, empleamos directamente la proyección lineal. Consideramos la aplicación siguiente:
    \Func{P}{X}{X}{x}{f(x)\cdot \dfrac{u}{f(u)}}

    La aplicación $p$ es lineal. Veamos que se trata de una proyección sobre $X$. En efecto, para todo $x \in X$, se tiene que:
    \begin{align*}
        P(P(x)) &= P\left(f(x)\cdot \dfrac{u}{f(u)}\right) = f\left(f(x)\cdot \dfrac{u}{f(u)}\right)\cdot \dfrac{u}{f(u)} = f(x)\cdot \dfrac{f(u)}{f(u)}\cdot \dfrac{u}{f(u)} =\\&= f(x)\cdot \dfrac{u}{f(u)} = P(x)
    \end{align*}

    Por tanto, $P$ es una proyección sobre $X$. Además, se tiene que:
    \begin{align*}
        \ker P &= \left\{x \in X : f(x)\cdot \dfrac{u}{f(u)} = 0\right\} = \ker f\\
        P(X) &= \left\{f(x)\cdot \dfrac{u}{f(u)} : x \in X\right\}\subset \cc{L}(u)\\
        P(u) &= u\Longrightarrow \cc{L}(u) \subset P(X)
    \end{align*}

    Por tanto, $P(X) = \cc{L}(u)$. Esto nos confirma que:
    $$X = \ker P \oplus P(X) = \ker f \oplus \cc{L}(u)$$
    
    Veamos ahora que $P$ es continua. Para ello, basta con ver que existe $M' \in \bb{R}^+$ tal que $\|P(x)\| \leq M' \|x\|$ para todo $x \in X$. En efecto, para todo $x \in X$, se tiene que:
    \begin{align*}
        \|P(x)\| &= \left\|f(x)\cdot \dfrac{u}{f(u)}\right\| = |f(x)| \cdot \left\|\dfrac{u}{f(u)}\right\| \leq \|f\| \cdot \|x\| \cdot \left\|\dfrac{u}{f(u)}\right\| = \|f\| \cdot \left\|\dfrac{u}{f(u)}\right\| \cdot \|x\|
    \end{align*}

    Por tanto, $P$ es continua. Para concluir que se da una suma directa topológica, basta con ver que $\ker P$ y $P(X)$ son cerrados. En efecto, $\ker P$ es cerrado por ser $P$ continua, y $P(X)=\ker(\id_X - P)$ es cerrado por ser $\id_X - P$ continua. Por tanto, $X = \ker f \oplus \cc{L}(u)$ es una suma directa topológica. De forma análoga, se demuestra que $X = \ker g \oplus \cc{L}(u)$ es una suma directa topológica. Por tanto, tenemos que:
    \begin{equation*}
        \ker f \cong X / \cc{L}(u) \cong \ker g
    \end{equation*}

    \item[Opción 2: Dando el isomorfismo]~\\
    Sea $u \in X$ tal que $f(u) \neq 0$ y $g(u) \neq 0$ (que existe por lo demostrado en la opción anterior). Consideramos la aplicación:
    \Func{T}{\ker f}{\ker g}{x}{x - \dfrac{g(x)}{g(u)} u}

    Veamos en primer lugar que $T$ está bien definida. Para ello, basta con ver que $T(x) \in \ker g$ para todo $x \in \ker f$. En efecto, para todo $x \in \ker f$, se tiene que:
    \begin{align*}
        g(T(x)) &= g\left(x - \dfrac{g(x)}{g(u)} u\right) = g(x) - \dfrac{g(x)}{g(u)} g(u) = g(x) - g(x) = 0
    \end{align*}

    Por tanto, $T(x) \in \ker g$ para todo $x \in \ker f$, y por tanto $T$ está bien definida. Como es combinación lineal de aplicaciones lineales, $T$ es lineal. Veamos ahora que $T$ es continua. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T(x)\| \leq M \|x\|$ para todo $x \in \ker f$. En efecto, para todo $x \in \ker f$, se tiene que:
    \begin{align*}
        \|T(x)\| &= \left\|x - \dfrac{g(x)}{g(u)} u\right\| \leq \|x\| + \left|\dfrac{g(x)}{g(u)}\right| \cdot \|u\| \leq \|x\| + \dfrac{\|g\| \cdot \|x\|}{|g(u)|} \cdot \|u\| =\\&= \left(1 + \dfrac{\|g\| \cdot \|u\|}{|g(u)|}\right) \cdot \|x\|
    \end{align*}

    Por tanto, $T$ es continua. Veamos ahora que tiene inversa continua. Consideramos la aplicación:
    \Func{S}{\ker g}{\ker f}{y}{y - \dfrac{f(y)}{f(u)} u}

    De forma análoga a lo que hemos hecho con $T$, se demuestra que $S$ está bien definida, es lineal y continua. Veamos ahora que $S$ es la inversa de $T$. En efecto, para todo $x \in \ker f$, se tiene que:
    \begin{align*}
        S(T(x)) &= S\left(x - \dfrac{g(x)}{g(u)} u\right) = \left(x - \dfrac{g(x)}{g(u)} u\right) - \dfrac{f\left(x - \dfrac{g(x)}{g(u)} u\right)}{f(u)} u
        =\\&= x - \dfrac{g(x)}{g(u)} u - \dfrac{f(x) - \dfrac{g(x)}{g(u)} f(u)}{f(u)} u
        = x - \dfrac{g(x)}{g(u)} u - \dfrac{0 - \dfrac{g(x)}{g(u)} f(u)}{f(u)} u
        =\\&= x - \dfrac{g(x)}{g(u)} u + \dfrac{g(x)}{g(u)} u = x = \id_{\ker f}(x)
    \end{align*}



    Para todo $y \in \ker g$, de manera análoga, se tiene que $T(S(y)) = y = \id_{\ker g}(y)$. Por tanto, $S$ es la inversa de $T$, y por tanto $T$ es un isomorfismo entre $\ker f$ y $\ker g$.
    \end{description}

    Veamos ahora un ejemplo en el que $\ker f$ y $\ker g$ no sean isométricamente isomorfos. Consideramos $X = \bb{R}^2\times \bb{R}$ con la norma del máximo, donde en $\bb{R}^2$ se considera la norma euclídea y en $\bb{R}$ la norma del máximo. En $X$, consideramos los funcionales lineales continuos:
    \begin{align*}
        f(x, y, z) &= y\\
        g(x, y, z) &= z
    \end{align*}

    Entonces, se tiene que:
    \begin{align*}
        \ker f &= \left\{(x, y, z) : y = 0\right\} = \left\{(x, 0, z) : x, z \in \bb{R}\right\}\\
        \ker g &= \left\{(x, y, z) : z = 0\right\} = \left\{(x, y, 0) : x, y \in \bb{R}\right\}
    \end{align*}

    Ahora, veamos que $\ker f$ y $\ker g$ no son isométricamente isomorfos.
    \begin{itemize}
        \item Para todo $(x, 0, z) \in \ker f$, se tiene que:
        \begin{equation*}
            \|(x, 0, z)\| = \max\left\{\sqrt{x^2 + 0^2}, |z|\right\} = \max\left\{|x|, |z|\right\}
        \end{equation*}
        Por tanto, la norma inducida en $\ker f$ es la norma del máximo en $\bb{R}^2$, y por tanto $\ker f$ es isométricamente isomorfo a $\ell_\infty^2$.
        \item Para todo $(x, y, 0) \in \ker g$, se tiene que:
        \begin{equation*}
            \|(x, y, 0)\| = \max\left\{\sqrt{x^2 + y^2}, |0|\right\} = \sqrt{x^2 + y^2}
        \end{equation*}
        Por tanto, la norma inducida en $\ker g$ es la norma euclídea en $\bb{R}^2$, y por tanto $\ker g$ es isométricamente isomorfo a $\ell_2^2$.
    \end{itemize}
    
    Veamos ahora que $\ell_\infty^2$ y $\ell_2^2$ no son isométricamente isomorfos. Supongamos que existe un isomorfismo lineal continuo $T: \ell_\infty^2 \longrightarrow \ell_2^2$ que es isométrico. Entonces, consideramos el segmento $J=[(1, 0), (1, 1)] \subset \ell_\infty^2$.
    \begin{align*}
        J=[(1, 0), (1, 1)] &= \left\{(1, t) : t \in [0, 1]\right\} = \left\{(1-t)(1, 0) + t(1, 1) : t \in [0, 1]\right\}\\
        T(J) &= \left\{T(1-t)(1, 0) + t(1, 1) : t \in [0, 1]\right\} = [T(1, 0), T(1, 1)]\\
        \|J(t)\|_\infty &= \|(1, t)\|_\infty = 1 \quad \forall t \in [0, 1]
    \end{align*}

    Por tanto, $J$ es un segmento de la esfera unidad de $\ell_\infty^2$. Como $T$ es isométrico, $T(J)$ es un segmento de la esfera unidad de $\ell_2^2$. Sin embargo, la esfera unidad de $\ell_2^2$ es un círculo, y no contiene segmentos. Por tanto, hemos llegado a una contradicción, y por tanto $\ell_\infty^2$ y $\ell_2^2$ no son isométricamente isomorfos. En consecuencia, $\ker f$ y $\ker g$ no son isométricamente isomorfos.
\end{ejercicio}

\begin{ejercicio}
Si $X$ e $Y$ son espacios normados y $Z = X \times Y$ es el espacio normado producto, probar que $Z^*$ es isomorfo al espacio de Banach producto dado por $X^* \times Y^*$.\\

Definimos las aplicaciones de inclusión:
\Func{i_X}{X}{Z}{x}{(x, 0)}
\Func{i_Y}{Y}{Z}{y}{(0, y)}

En los espacios producto aún no fijamos la norma, pues se hará cuando nos convenga. Dado $h\in Z^*$, definimos $h_X = h \circ i_X$ y $h_Y = h \circ i_Y$. Es claro que $h_X,h_Y$ son funcionales lineales, pero veamos que son continuos:
\begin{align*}
    |h_X(x)| &= |h(i_X(x))| = |h(x, 0)| \leq \|h\| \cdot \|(x, 0)\| = \|h\| \cdot \|x\|\\
    |h_Y(y)| &= |h(i_Y(y))| = |h(0, y)| \leq \|h\| \cdot \|(0, y)\| = \|h\| \cdot \|y\|
\end{align*}
Notemos que en las desigualdades anteriores aún no hemos fijado la norma del espacio producto $Z$. Por tanto, $h_X \in X^*$ y $h_Y \in Y^*$. Definimos:
\Func{T}{Z^*}{X^* \times Y^*}{h}{(h_X, h_Y)}

Como cada una de las componentes es lineal, $T$ es lineal. Para la inversa, dado $(f, g) \in X^* \times Y^*$, definimos $h: Z \longrightarrow \bb{K}$ por:
\begin{equation*}
    h(x, y) = f(x) + g(y)
\end{equation*}
Es claro que $h$ es lineal. Veamos que es continua. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $|h(x, y)| \leq M \|(x, y)\|$ para todo $(x, y) \in Z$. En efecto, para todo $(x, y) \in Z$, se tiene que:
\begin{align*}
    |h(x, y)| &= |f(x) + g(y)| \leq |f(x)| + |g(y)| \leq \|f\| \cdot \|x\| + \|g\| \cdot \|y\| \leq \\ &\leq \max\{\|f\|, \|g\|\} \cdot (\|x\| + \|y\|) = \max\{\|f\|, \|g\|\} \cdot \|(x, y)\|_1
\end{align*}

En este punto, hemos de fijar la norma del espacio producto $Z=X \times Y$. Si fijamos la norma $\|(x, y)\|_1 = \|x\| + \|y\|$, entonces $h$ es continua. Por tanto, $h \in Z^*$. Definimos:
\Func{S}{X^* \times Y^*}{Z^*}{(f, g)}{h}

Es claro que $S$ es lineal. Veamos que $S$ es la inversa de $T$. En efecto, para todo $h \in Z^*$, se tiene que:
\begin{align*}
    S(T(h))(x, y) &= S(h_X, h_Y)(x, y) = h_X(x) + h_Y(y) = h(i_X(x)) + h(i_Y(y)) = h(x, 0) + h(0, y) =\\&= h(x, y) \Longrightarrow S(T(h)) = h \Longrightarrow S \circ T = \id_{Z^*}
\end{align*}

Para todo $(f, g) \in X^* \times Y^*$, se tiene que:
\begin{align*}
    \begin{cases}
        T(S(f, g))_X(x) &= S(f, g)(i_X(x)) = f(x) + g(0) = f(x) \Longrightarrow T(S(f, g))_X = f\\
        T(S(f, g))_Y(y) &= S(f, g)(i_Y(y)) = f(0) + g(y) = g(y) \Longrightarrow T(S(f, g))_Y = g
    \end{cases}
\end{align*}

Por tanto:
$$T(S(f, g)) = (f, g) \Longrightarrow T \circ S = \id_{X^* \times Y^*}$$

Por tanto $T$ es una aplicación lineal biyectiva. Veamos que $T$ es continua. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T(h)\| \leq M \|h\|$ para todo $h \in Z^*$. En efecto, para todo $h \in Z^*$, se tiene que:
\begin{align*}
    \|T(h)\|_\infty &= \|(h_X, h_Y)\|_\infty = \max\{\|h_X\|, \|h_Y\|\} \leq \|h\|
\end{align*}
donde la última desigualdad se sigue de las desigualdades empleadas para demostrar la continuidad de $h_X$ y $h_Y$. En este punto, hemos de fijar la norma del espacio producto $X^* \times Y^*$. Si fijamos la norma $\|(f, g)\| = \max\{\|f\|, \|g\|\}$, entonces $T$ es continua. Veamos ahora que $S$ es continua. Para ello, basta con ver que existe $M' \in \bb{R}^+$ tal que $\|S(f, g)\| \leq M' \|(f, g)\|$ para todo $(f, g) \in X^* \times Y^*$. En efecto, para todo $(f, g) \in X^* \times Y^*$, se tiene que:
\begin{align*}
    \|S(f, g)\| = \|h\|
\end{align*}

Acotamos ahora $\|h\|$. Para ello, basta con ver que existe $M'' \in \bb{R}^+$ tal que $|h(x, y)| \leq M' \|(x, y)\|$ para todo $(x, y) \in Z$. En efecto, para todo $(x, y) \in Z$, se tiene que:
\begin{align*}
    |h(x, y)| &\leq \max\{\|f\|, \|g\|\} \cdot (\|x\| + \|y\|) = \|(f, g)\|_\infty \cdot \|(x, y)\|_1
\end{align*}

Por tanto, $\|S(f, g)\| = \|h\| \leq \|(f, g)\|_\infty$, y por tanto $S$ es continua. En conclusión, $T$ es un isomorfismo entre $Z^*$ y $X^* \times Y^*$.
\end{ejercicio}

\begin{ejercicio}
En el espacio normado cociente $\ell_\infty / c_0$, calcular la norma cociente $\|x + c_0\|$, para cada $x \in \ell_\infty$.\\

Tenemos que:
\begin{align*}
    \|x + c_0\| &= \inf\limits_{y \in c_0} \|x - y\|_\infty
    = \inf\limits_{y \in c_0} \sup\limits_{n \in \bb{N}} |x(n) - y(n)|
\end{align*}

Por un lado, para todo $y \in c_0$ y $k\in \bb{N}$, se tiene que:
\begin{align*}
    \|x - y\|_\infty = \sup\limits_{n \in \bb{N}} |x(n) - y(n)| \geq |x(k) - y(k)|\geq |x(k)| - |y(k)|
    \Longrightarrow \|x - y\|_\infty + |y(k)| \geq |x(k)|
\end{align*}

Fijamos ahora $n_0 \in \bb{N}$, y tomamos el supremo sobre $k\geq n_0$. Entonces, para todo $y \in c_0$, se tiene que:
\begin{align*}
    \|x - y\|_\infty + \sup\limits_{k \geq n_0} |y(k)| \geq \sup\limits_{k \geq n_0} |x(k)|
\end{align*}

Como $y \in c_0$, se tiene que $\lim\limits_{k \to \infty} |y(k)| = 0$. Por tanto, tomando el límite cuando $n_0 \to \infty$, se tiene que:
\begin{align*}
    \|x - y\|_\infty \geq \lim_{n_0 \to \infty} \sup\limits_{k \geq n_0} |x(k)| =: \limsup\limits_{k \to \infty} |x(k)|
\end{align*}

Como esto es cierto para todo $y \in c_0$, se tiene que:
\begin{align*}
    \|x + c_0\| = \inf\limits_{y \in c_0} \|x - y\|_\infty \geq \limsup\limits_{k \to \infty} |x(k)|
\end{align*}~\\

Veamos ahora que $\|x + c_0\| \leq \limsup\limits_{k \to \infty} |x(k)|$. Para ello, para cada $n\in \bb{N}$, definimos $y_n \in c_0$ por:
\begin{equation*}
    y_n(k) = \begin{cases}
        x(k) & \text{si } k \leq n\\
        0 & \text{si } k > n
    \end{cases}
\end{equation*}

Como $y_n$ es de soporte finito, se tiene que $y_n \in c_0$. Además, se tiene que:
\begin{align*}
    x-y_n(k) = \begin{cases}
        0 & \text{si } k \leq n\\
        x(k) & \text{si } k > n
    \end{cases}
\end{align*}

Por tanto, se tiene que:
\begin{align*}
    \|x - y_n\|_\infty = \sup\limits_{k > n} |x(k)|
\end{align*}

Como $y_n \in c_0$, se tiene que:
\begin{align*}
    \|x + c_0\| = \inf\limits_{y \in c_0} \|x - y\|_\infty \leq \|x - y_n\|_\infty = \sup\limits_{k > n} |x(k)|
\end{align*}

Tomando ínfimo sobre $n \in \bb{N}$, se tiene que:
\begin{align*}
    \|x + c_0\| \leq \inf\limits_{n \in \bb{N}} \sup\limits_{k > n} |x(k)|= \limsup\limits_{k \to \infty} |x(k)|
\end{align*}

En conclusión, se tiene que:
\begin{align*}
    \|x + c_0\| = \limsup\limits_{k \to \infty} |x(k)|
\end{align*}

\end{ejercicio}

\begin{ejercicio}
En el espacio de Banach $C\left[0, 1\right]$ se considera el subespacio
\[
M = \left\{g \in C\left[0, 1\right] : g(t) = 0 \;\; \forall t \in \left[0, \nicefrac{1}{2}\right]\right\}
\]
Probar que $M$ es cerrado en $C\left[0, 1\right]$, y calcular la norma cociente $\|f + M\|$ para cada $f \in C\left[0, 1\right]$.\\

Veamos primero que $M$ es cerrado. Para ello, hay dos alternativas.
\begin{description}
    \item[Caracterización de cerrados por sucesiones]
    
    Sea $\{f_n\}$ una sucesión en $M$ que converge a $f \in C[0, 1]$. Entonces, para todo $t \in [0, \nicefrac{1}{2}]$, se tiene que:
    \begin{align*}
        |f(t) - 0| = |f(t) - f_n(t)| \leq \|f - f_n\|_\infty \longrightarrow 0
    \end{align*}

    Por tanto, $f(t) = 0$ para todo $t \in [0, \nicefrac{1}{2}]$, y por tanto $f \in M$. En consecuencia, $M$ es cerrado.
    \item[Aplicando el funcional de evaluación]
    
    Para cada $t \in [0, 1]$, consideramos el funcional de evaluación:
    \Func{\delta_t}{C[0, 1]}{\bb{K}}{f}{f(t)}

    Tenemos que:
    \begin{align*}
        M = \bigcap\limits_{t \in [0, \nicefrac{1}{2}]} \ker \delta_t
    \end{align*}

    Como cada $\ker \delta_t$ es cerrado por ser el núcleo de un funcional lineal continuo, y como la intersección de cerrados es cerrado, se tiene que $M$ es cerrado.

    \item[Definiendo un funcional lineal continuo]
    
    Consideramos el funcional lineal:
    \Func{T}{C[0, 1]}{C[0, \nicefrac{1}{2}]}{f}{f|_{[0, \nicefrac{1}{2}]}}

    Veamos que $T$ es continua. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T(f)\|_\infty \leq M \|f\|_\infty$ para todo $f \in C[0, 1]$. En efecto, para todo $f \in C[0, 1]$, se tiene que:
    \begin{align*}
        \|T(f)\|_\infty = \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t)| \leq \sup\limits_{t \in [0, 1]} |f(t)| = \|f\|_\infty
    \end{align*}

    Por tanto, $T$ es continua. Además, se tiene que:
    \begin{align*}
        M = \ker T
    \end{align*}

    Por tanto, $M$ es cerrado por ser el núcleo de un funcional lineal continuo.
    \begin{observacion}
        Esto nos puede hacer pensar que la norma en el cociente por $M$ es $\|T(f)\|_{\infty}$.
    \end{observacion}
\end{description}

Por tanto, como es cerrado, podemos considerar el espacio cociente $C[0, 1] / M$. Para calcular la norma cociente, tenemos que:
\begin{align*}
    \|f + M\| &= \inf\limits_{g \in M} \|f - g\|_\infty = \inf\limits_{g \in M} \sup\limits_{t \in [0, 1]} |f(t) - g(t)|
    \geq \inf\limits_{g \in M} \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t) - g(t)| =\\&= \inf\limits_{g \in M} \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t)| = \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t)|
\end{align*}

Por tanto, tenemos una cota inferior para la norma cociente. Para ver que se trata de una cota superior, basta con encontrar $g_0\in M$ tal que:
\begin{equation*}
    \|f - g_0\|_\infty \leq \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t)|
\end{equation*}

Como $g_0\in M$, se tiene que $g_0(t) = 0$ para todo $t \in [0, \nicefrac{1}{2}]$. Para $t \in [\nicefrac{1}{2}, 1]$, queremos que su diferencia sea lo menor posible. Podríamos pensar en tomar la función $g_0(t) = f(t)$ para todo $t \in [\nicefrac{1}{2}, 1]$, pero esto no nos garantiza que $g_0$ sea continua. Para ello, consideramos:
\Func{g_0}{[0, 1]}{\bb{K}}{t}{\begin{cases}
    0 & \text{si } t \in [0, \nicefrac{1}{2}]\\
    f(t)- f(\nicefrac{1}{2}) & \text{si } t \in [\nicefrac{1}{2}, 1]
\end{cases}}

Tenemos que $g_0\in M$, y además:
\begin{itemize}
    \item Si $t \in [0, \nicefrac{1}{2}]$, se tiene que:
    \begin{align*}
        |f(t) - g_0(t)| = |f(t) - 0| = |f(t)|
    \end{align*}
    \item Si $t \in [\nicefrac{1}{2}, 1]$, se tiene que:
    \begin{align*}
        |f(t) - g_0(t)| = |f(t) - (f(t) - f(\nicefrac{1}{2}))| = |f(\nicefrac{1}{2})|
    \end{align*}
\end{itemize}

En consecuencia, se tiene que:
\begin{align*}
    \|f - g_0\|_\infty = \sup\limits_{t \in [0, 1]} |f(t) - g_0(t)| = \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t)|
\end{align*}

Tomando el ínfimo sobre $g \in M$, se tiene que:
\begin{align*}
    \|f + M\| = \inf\limits_{g \in M} \|f - g\|_\infty \leq \|f - g_0\|_\infty = \sup\limits_{t \in [0, \nicefrac{1}{2}]} |f(t)|
\end{align*}

\end{ejercicio}

\begin{ejercicio}
Para $1 \leq p < \infty$, se consideran los subespacios de $L_p$ definidos por
\[
M = \left\{f \in L_p : f(t) = 0 \text{ c.p.d. } t \in \left[0, \nicefrac{1}{2}\right]\right\}
\]
\[
Z = \left\{f \in L_p : f(t) = 0 \text{ c.p.d. } t \in \left[\nicefrac{1}{2}, 1\right]\right\}
\]
Probar que $L_p = M \oplus Z$, suma topológica directa. Deducir que $L_p$ es isomorfo al espacio de Banach producto $L_p \times L_p$.\\

    Puesto que se nos pide directamente que la suma sea topológica, buscamos la proyección lineal. Consideramos la aplicación:
    \Func{P}{L_p}{L_p}{f}{f\cdot \chi_{[0, \nicefrac{1}{2}]}}

    Veamos en primer lugar que $P$ está bien definida. Para ello, basta con ver que $P(f) \in L_p$ para todo $f \in L_p$. En efecto, para todo $f \in L_p$, se tiene que:
    \begin{align*}
        \int_{0}^1 |P(f)(t)|^p\,dt = \int_{0}^1 |f(t)\cdot \chi_{[0, \nicefrac{1}{2}]}(t)|^p\,dt = \int_{0}^{\nicefrac{1}{2}} |f(t)|^p\,dt \leq \int_{0}^{1} |f(t)|^p\,dt < \infty
    \end{align*}

    Por tanto, $P(f) \in L_p$ para todo $f \in L_p$, y por tanto $P$ está bien definida. Además, $P$ es lineal. Veamos ahora que es una proyección. Para ello, basta con ver que $P^2 = P$. En efecto, para todo $f \in L_p$, se tiene que:
    \begin{align*}
        P^2(f) = P(P(f)) = P(f\cdot \chi_{[0, \nicefrac{1}{2}]}) = f\cdot \chi_{[0, \nicefrac{1}{2}]}\cdot \chi_{[0, \nicefrac{1}{2}]} = f\cdot \chi_{[0, \nicefrac{1}{2}]} = P(f)
    \end{align*}

    Por tanto, $P$ es una proyección. Veamos ahora que $P$ es continua. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|P(f)\|_p \leq M \|f\|_p$ para todo $f \in L_p$. En efecto, para todo $f \in L_p$, se tiene que:
    \begin{align*}
        \|P(f)\|_p^p = \int_{0}^1 |P(f)(t)|^p\,dt = \int_{0}^{\nicefrac{1}{2}} |f(t)|^p\,dt \leq \int_{0}^{1} |f(t)|^p\,dt = \|f\|_p^p
    \end{align*}

    Por tanto, $\|P(f)\|_p \leq \|f\|_p$ para todo $f \in L_p$, y por tanto $P$ es continua. Por tanto, $L_p = \ker P \oplus P(L_p)$ es una suma directa topológica. Veamos ahora el valor de $\ker P$ y $P(L_p)$. Por un lado, se tiene que:
    \begin{align*}
        \ker P &= \left\{f \in L_p : P(f) = 0\right\} = \left\{f \in L_p : f(t)\cdot \chi_{[0, \nicefrac{1}{2}]}(t) = 0 \text{ c.p.d. } t \in [0, 1]\right\} =\\&= \left\{f \in L_p : f(t) = 0 \text{ c.p.d. } t \in [0, \nicefrac{1}{2}]\right\} = M
    \end{align*}

    Veamos ahora que $P(L_p) = Z$.
    \begin{description}
        \item[$\subset$)] Sea $f \in P(L_p)$. Entonces, existe $g \in L_p$ tal que $f = P(g) = g\cdot \chi_{[0, \nicefrac{1}{2}]}$. Por tanto, para todo $t \in [\nicefrac{1}{2}, 1]$, se tiene que:
        \begin{align*}
            f(t) = g(t)\cdot \chi_{[0, \nicefrac{1}{2}]}(t) = g(t)\cdot 0 = 0
        \end{align*}
        Por tanto, $f \in Z$, y por tanto $P(L_p) \subset Z$.
        \item[$\supset$)] Sea $f \in Z$. Entonces, para casi todo $t \in [0, 1]$, se tiene que:
        \begin{align*}
            f(t) = f(t)\cdot \chi_{[0, \nicefrac{1}{2}]}(t) = P(f)(t)
        \end{align*}
        Por tanto, $f = P(f) \in P(L_p)$, y por tanto $Z \subset P(L_p)$. 
    \end{description}

    Por tanto, $P(L_p) = Z$. Como $M=\ker P$ y $Z=\ker(\id - P)$ son cerrados por ser núcleos de operadores lineales continuos, se tiene que $L_p = M \oplus Z$ es una suma directa topológica. En consecuencia, $L_p\cong M \times Z$. Falta ver que $M \cong L_p$ y $Z \cong L_p$. Para ello, consideramos las aplicaciones:
    \Func{T_Z}{Z}{L_p}{f}{\tilde{f}}
    \Func{T_M}{M}{L_p}{f}{\ol{f}}
    \Func{\tilde{f}}{[0, 1]}{\bb{K}}{t}{f(\nicefrac{t}{2})}
    \Func{\ol{f}}{[0, 1]}{\bb{K}}{t}{f\left(\frac{1+t}{2}\right)}

    Es claro que $T_Z$ y $T_M$ son lineales. Veamos que son continuas. Para ello, basta con ver que existe $M \in \bb{R}^+$ tal que $\|T_Z(f)\|_p \leq M \|f\|_p$ para todo $f \in Z$, y de manera análoga para $T_M$. En efecto, para todo $f \in Z$, se tiene que:
    \begin{align*}
        \|T_Z(f)\|_p^p = \int_{0}^1 |\tilde{f}(t)|^p\,dt = \int_{0}^1 |f(\nicefrac{t}{2})|^p\,dt = 2 \int_{0}^{\nicefrac{1}{2}} |f(s)|^p\,ds = 2 \int_{0}^{1} |f(s)|^p\,ds = 2 \|f\|_p^p
    \end{align*}

    Por tanto, $T_Z$ es continua. De manera análoga, para todo $f \in M$, se tiene que:
    \begin{align*}
        \|T_M(f)\|_p^p = \int_{0}^1 |\ol{f}(t)|^p\,dt = \int_{0}^1 \left|f\left(\frac{1+t}{2}\right)\right|^p\,dt = 2 \int_{\nicefrac{1}{2}}^{1} |f(s)|^p\,ds = 2 \int_{0}^{1} |f(s)|^p\,ds = 2 \|f\|_p^p
    \end{align*}
    Por tanto, $T_M$ es continua. Veamos ahora que son biyectivas. De la igualdad de las normas, se deduce que $T_Z$ y $T_M$ son inyectivas. Veamos ahora que son sobreyectivas. Sea $g \in L_p$. Entonces, definimos $f \in Z$ por:
    \Func{f}{[0, 1]}{\bb{K}}{t}{g(2t)\cdot \chi_{[0, \nicefrac{1}{2}]}(t)}
    Es claro que $f \in Z$, y además:
    \begin{align*}
        T_Z(f)(t) = \tilde{f}(t) = f\left(\frac{t}{2}\right) = g\left(2 \cdot \frac{t}{2}\right)\cdot \chi_{[0, \nicefrac{1}{2}]}\left(\frac{t}{2}\right) = g(t)\cdot \chi_{[0, 1]}(t) = g(t) \Longrightarrow T_Z(f) = g
    \end{align*}

    Por tanto, $T_Z$ es sobreyectiva. De manera análoga, sea $g \in L_p$. Entonces, definimos $f \in M$ por:
    \Func{f}{[0, 1]}{\bb{K}}{t}{g(2t-1)\cdot \chi_{[\nicefrac{1}{2}, 1]}(t)}
    Es claro que $f \in M$, y además:
    \begin{align*}
        T_M(f)(t) &= \ol{f}(t) = f\left(\frac{1+t}{2}\right) = g\left(2 \cdot \frac{1+t}{2}-1\right)\cdot \chi_{[\nicefrac{1}{2}, 1]}\left(\frac{1+t}{2}\right) =\\&= g(t)\cdot \chi_{[0, 1]}(t) = g(t) \Longrightarrow T_M(f) = g
    \end{align*}

    Por tanto, $T_M$ es sobreyectiva. En consecuencia, $T_Z$ y $T_M$ son biyecciones y, de la igualdad de las normas, se demuestra que sus inversas son continuas, luego son isomorfismos entre $Z$ y $L_p$, y entre $M$ y $L_p$, respectivamente. En conclusión, se tiene que:
    \begin{align*}
        L_p \cong M \times Z \cong L_p \times L_p
    \end{align*}
\end{ejercicio}

\begin{ejercicio}
Sean $X$ e $Y$ espacios normados y supongamos que $Y$ tiene dimensión finita. Probar que todo operador lineal sobreyectivo $T: X \longrightarrow Y$ es una aplicación abierta.

\begin{description}
    \item[Si $T$ fuese continua:] Entonces, $\ker T=T^{-1}(\{0\})$ es cerrado, y por tanto $X/\ker T$ es un espacio normado. Por tanto, empleamos la factorización canónica $T = \tilde{T} \circ \pi$, donde $\pi: X \longrightarrow X/\ker T$ es la proyección canónica y $\tilde{T}: X/\ker T \longrightarrow Y$ es el operador lineal biyectivo inducido por $T$:
    \begin{figure}[h]
        \centering
        % https://tikzcd.yichuanshen.de/#N4Igdg9gJgpgziAXAbVABwnAlgFyxMJZABgBpiBdUkANwEMAbAVxiRAA0QBfU9TXfIRQBGclVqMWbAJrdeIDNjwEiZYePrNWiEAB1dAMwBOdAMbB2XYPoDWMI8AAqXLt3EwoAc3hFQxiAC2SGQgOBBIohJabI5yfkaBwdRhSABM1JpSOvpoWCDUDHQARjAMAAr8ykIgRlieABY4cSD+QYjpoeGIkZnaeroA7liweAywTq4FxaUVSoJstQ1NXBRcQA
        \begin{tikzcd}

        \frac{X}{\ker{T}} \arrow[ru, "\wt{T}"'] &  
        \end{tikzcd}
    \end{figure}

    Como $\wt{T}$ es biyectiva y $Y$ tiene dimensión finita, se tiene que $X/\ker T$ también tiene dimensión finita. Por tanto, $\wt{T}$ es una aplicación lineal biyectiva entre espacios normados de dimensión finita, y por el Teorema de Hausdorff, $\wt{T}$ es un isomorfismo (en particular, es abierta). Por tanto, como $\pi$ es abierta, se tiene que $T = \wt{T} \circ \pi$ es una aplicación abierta.\\

    Notemos que, como $T$ no es necesariamente continua, no podemos considerar el espacio cociente $X/\ker T$, por lo que hemos de emplear otro enfoque.
\end{description}

Queremos aplicar que un operador lineal $H: X \longrightarrow Y$ entre espacios normados es abierto si y solo si $H(U_X)$ es un entorno de $0$ en $Y$.
\begin{description}
    \item[$\Longrightarrow$)] Como $H(0)=0$ por linealidad y $0\in U_X$, se tiene que $0 \in H(U_X)$. Además, como $H$ es abierta, se tiene que $H(U_X)$ es abierto en $Y$, y por tanto es un entorno de $0$ en $Y$.
    \item[$\Longleftarrow$)] Sea $U\subset X$ abierto. Queremos ver que $H(U)$ es abierto en $Y$. Fijado $y\in H(U)$, existe $x \in U$ tal que $H(x) = y$. Como $U$ es abierto, existe $\varepsilon > 0$ tal que $x + \varepsilon U_X \subset U$. Por linealidad de $H$, se tiene que:
    \begin{align*}
        H(x+\varepsilon U_X) = H(x) + \varepsilon H(U_X) \subset H(U)
        \Longrightarrow y+\varepsilon H(U_X) \subset H(U)
    \end{align*}

    Como $H(U_X)$ es un entorno de $0$ en $Y$, se tiene que $\varepsilon H(U_X)$ es un entorno de $0$ en $Y$, y por tanto $y+\varepsilon H(U_X)$ es un entorno de $y$ en $Y$. En consecuencia, $H(U)$ es abierto en $Y$.
\end{description}

Por tanto, para probar que $T$ es abierta, basta con ver que $T(U_X)$ es un entorno de $0$ en $Y$. Como cada subconjunto de un espacio vectorial tiene complemento algebráico, sea $N\subset X$ el complemento algebráico de $\ker T$. Entonces,
$$X = \ker T \oplus N$$

Veamos que $T_{|N}$ es una biyección (que será lineal por ser la restricción de un operador lineal).
\begin{itemize}
    \item Sea $v\in N$ tal que $T(v) = 0$. Entonces, $v \in \ker T \cap N$, y por ser $N$ un complemento algebráico de $\ker T$, se tiene que $v = \ker T \cap N = \{0\}$, y por tanto $v=0$. En consecuencia, $T_{|N}$ es inyectiva.
    \item Sea $y \in Y$. Como $T$ es sobreyectiva, existe $x \in X$ tal que $T(x) = y$. Como $X = \ker T \oplus N$, existe $v \in N$ y $w \in \ker T$ tales que $x = v + w$. Por linealidad de $T$, se tiene que:
    \begin{align*}
        T(x) = T(v + w) = T(v) + T(w) = T(v) + 0 = T(v) \Longrightarrow y = T(v)
    \end{align*}

    En consecuencia, $T_{|N}$ es sobreyectiva.
\end{itemize}

Por tanto, $T_{|N}$ es una biyección lineal entre espacios normados. Como $Y$ tiene dimensión finita, se tiene que $N$ también tiene dimensión finita, y por el Teorema de Hausdorff, $T_{|N}$ es un isomorfismo (en particular, es abierta). Por tanto, $T_{|N}(U_N)$ es un entorno de $0$ en $Y$, donde $U_N = U_X \cap N$. Como $U_N \subset U_X$, se tiene que:
\begin{align*}
    T_{|N}(U_N) = T(U_N) \subset T(U_X)
\end{align*}

Como $T_{|N}(U_N)$ es un entorno de $0$ en $Y$, se tiene que $T(U_X)$ es un entorno de $0$ en $Y$. En consecuencia, $T$ es una aplicación abierta.
\end{ejercicio}

\begin{ejercicio}

Probar que existen $a, b \in \bb{K}$ tales que $f(t) = a\,e^t + b\,e^{-t}$ para todo $t \in \left[0, 1\right]$.\\

        M = \cc{L}\{e^t, e^{-t}\} = \left\{a\,e^t + b\,e^{-t} : a, b \in \bb{K}\right\}
    \end{equation*}

    Como $\dim M\leq 2<\infty$, por el corolario del Teorema de Hausdorff, se tiene que $M$ es cerrado en $X$. Como $\{f_n\}\subset M$ y $\{f_n\}\to f$ en norma, por ser $M$ cerrado, se tiene que $f \in M$.

    \item[Opción 2:] Buscamos convertir la condición del enunciado en una distancia. Tenemos que:
    \begin{equation*}
        0\leq d(f, M)\leq \|f - f_n\|_4 = \left(\int_{0}^1 |f(t) - a_n e^t - b_n e^{-t}|^4\,dt\right)^{\nicefrac{1}{4}}
    \end{equation*}

    Tomando límite cuando $n \to \infty$, por el Lema del Sandwich, se tiene que:
    \begin{align*}
        d(f, M) = 0 \Longrightarrow f \in \ol{M} = M
    \end{align*}
\end{description}


En cualquier caso, como $f\in M$, existen $a, b \in \bb{K}$ tales que $f(t) = a\,e^t + b\,e^{-t}$ para todo $t \in [0, 1]$.
\end{ejercicio}

\begin{ejercicio}
Probar que, para cada $N \in \bb{N}$, existe una constante $C_N \in \bb{R}^+$ verificando que, si $P$ es un polinomio de grado $N$ con coeficientes complejos, se tiene:
\[
\sum_{k=0}^N |P(k)| \leq C_N \int_{0}^1 |P(t)|\,dt
\]

    Sea $X = \cc{P}_{\leq N}(\bb{K})$, el espacio vectorial de los polinomios de grado a lo sumo $N$ con coeficientes complejos. Sabemos que $X=\cc{L}\{1, t, t^2, \ldots, t^N\}$, y por tanto $\dim X = N+1 < \infty$. Consideramos ahora:
    \begin{align*}
        \|P\|_1 = \sum_{k=0}^N |P(k)|
    \end{align*}
    Demostremos que se trata de una norma:
    \begin{itemize}
        \item $\|P\|_1 = 0 \Longrightarrow P(k) = 0$ para todo $k=0, 1, \ldots, N$, luego $P$ tiene grado a lo sumo $N$ y tiene al menos $N+1$ raíces distintas, luego $P=0$. 
        \item Para todo $\alpha \in \bb{K}$ y todo $P \in X$, se tiene que:
        \begin{align*}
            \|\alpha P\|_1 = \sum_{k=0}^N |\alpha P(k)| = |\alpha| \sum_{k=0}^N |P(k)| = |\alpha| \|P\|_1
        \end{align*}
        \item Para todo $P, Q \in X$, se tiene que:
        \begin{align*}
            \|P + Q\|_1 &= \sum_{k=0}^N |(P + Q)(k)| = \sum_{k=0}^N |P(k) + Q(k)| \leq \sum_{k=0}^N |P(k)| + |Q(k)| =\\&= \sum_{k=0}^N |P(k)| + \sum_{k=0}^N |Q(k)| = \|P\|_1 + \|Q\|_1
        \end{align*}
    \end{itemize}

    Consideramos ahora:
    \begin{align*}
        \|P\|_2 = \int_{0}^1 |P(t)|\,dt
    \end{align*}
    Demostremos que se trata de una norma:
    \begin{itemize}
        \item $\|P\|_2 = 0 \Longrightarrow |P(t)| = 0$ para casi todo $t \in [0, 1]$, y por ser $P$ continua, se tiene que $P(t) = 0$ para todo $t \in [0, 1]$. Por tener $P$ grado a lo sumo $N$, se tiene que $P=0$.
        \item Para todo $\alpha \in \bb{K}$ y todo $P \in X$, se tiene que:
        \begin{align*}
            \|\alpha P\|_2 = \int_{0}^1 |\alpha P(t)|\,dt = |\alpha| \int_{0}^1 |P(t)|\,dt = |\alpha| \|P\|_2
        \end{align*}
        \item Para todo $P, Q \in X$, se tiene que:
        \begin{align*}
            \|P + Q\|_2 &= \int_{0}^1 |(P + Q)(t)|\,dt = \int_{0}^1 |P(t) + Q(t)|\,dt \leq \int_{0}^1 |P(t)| + |Q(t)|\,dt =\\&= \int_{0}^1 |P(t)|\,dt + \int_{0}^1 |Q(t)|\,dt = \|P\|_2 + \|Q\|_2
        \end{align*}
    \end{itemize}

    Por tanto, $\|\cdot\|_1$ y $\|\cdot\|_2$ son normas en $X$. Como $\dim X < \infty$, por el Teorema de Hausdorff, se tiene que $\|\cdot\|_1$ y $\|\cdot\|_2$ son equivalentes. En particular, existe $C_N \in \bb{R}^+$ tal que:
    \begin{align*}
        \|P\|_1 \leq C_N \|P\|_2 \Longrightarrow \sum_{k=0}^N |P(k)| \leq C_N \int_{0}^1 |P(t)|\,dt
    \end{align*}
\end{ejercicio}

\begin{ejercicio}
Sea $X$ un espacio normado y $M$ un subespacio de dimensión finita de $X$, tal que $M \neq X$. Probar que existe $x \in X$ tal que $\|x\| = d(x, M) = 1$. Deducir que, si $X$ tiene dimensión infinita, existe una sucesión $\{x_n\}$ en la esfera unidad de $X$, tal que $\|x_n - x_m\| \geq 1$ para cualesquiera $n, m \in \bb{N}$ con $m \neq n$.\\

Comenzamos probando que existe $x \in X$ tal que $\|x\| = d(x, M) = 1$.
\begin{description}
    \item[Vía constructiva:] Como $M \neq X$, existe $x_0 \in X \setminus M$. Como $M$ es de dimensión finita, es cerrado, y por tanto $d(x_0, M) > 0$. Además, como $M$ es un subespacio de dimensión finita, se tiene que $M$ es proximinal, y por tanto existe $y_0 \in M$ tal que:
    \begin{align*}
        d(x_0, M) = \|x_0 - y_0\|> 0
    \end{align*}

    Definimos ahora:
    \begin{align*}
        x = \frac{x_0 - y_0}{\|x_0 - y_0\|} \Longrightarrow \|x\| = 1
    \end{align*}

    Veamos que $d(x, M) = 1$.
    \begin{description}
        \item[$\leq$)] Como $0 \in M$, se tiene que:
        \begin{align*}
            d(x, M) = \inf\limits_{y \in M} \|x - y\| \leq \|x - 0\| = \|x\| = 1
        \end{align*}
        \item[$\geq$)] Sea $y \in M$. Entonces:
        \begin{align*}
            \|x - y\| &= \left\|\frac{x_0 - y_0}{\|x_0 - y_0\|} - y\right\| = \frac{1}{\|x_0 - y_0\|}\left\|(x_0 - y_0) - \|x_0 - y_0\|y\right\|
            =\\&= \frac{1}{\|x_0 - y_0\|}\left\|x_0 - (y_0 + \|x_0 - y_0\|y)\right\|
        \end{align*}

        Como $y_0,y \in M$, se tiene que $y_0 + \|x_0 - y_0\|y \in M$, y por tanto:
        \begin{align*}
            \|x - y\| \geq \frac{1}{\|x_0 - y_0\|} d(x_0, M) = \frac{\|x_0 - y_0\|}{\|x_0 - y_0\|} = 1
        \end{align*}
    \end{description}


    \item[Vía el cociente:] Consideramos el espacio cociente $X/M$, que podemos dotar de la norma cociente por ser $M$ cerrado al ser de dimensión finita. Como $M \neq X$, se tiene que $X/M \neq \{0\}$, luego su esfera unidad es no vacía. Sea $x + M \in X/M$ tal que $\|x + M\| = 1$. Entonces, por definición de la norma cociente, se tiene que:
    \begin{align*}
        1 = \|x + M\| = \inf\limits_{y \in M} \|x - y\| = d(x, M)
    \end{align*}

    Como $M$ es proximinal, existe $y_0 \in M$ tal que:
    \begin{align*}
        1 = \|x + M\| = \|x - y_0\|
    \end{align*}

    Sea por tanto $x_0 = x - y_0$. Entonces, $\|x_0\| = 1$ y, puesto que $y_0 \in M$, se tiene que:
    \begin{align*}
        1 = \|x+M\| = \|x_0 + M\| = d(x_0, M)
    \end{align*}

\end{description}

Buscamos ahora una sucesión $\{x_n\}$ en la esfera unidad de $X$, tal que se tenga $\|x_n - x_m\| \geq 1$ para cualesquiera $n, m \in \bb{N}$ con $m \neq n$. Para ello, definimos por inducción la sucesión $\{x_n\}$:
\begin{itemize}
    \item Sea $x_1 \in X$ tal que $\|x_1\| = d(x_1, \{0\}) = 1$, que existe por lo que hemos probado antes.
    \item Sea $n \in \bb{N}$ y supongamos que existen $x_1, \ldots, x_n \in X$ tales que $\|x_i\| = 1$ y $\|x_i - x_j\| \geq 1$ para cualesquiera $i, j \in \{1, \ldots, n\}$ con $i \neq j$. Consideramos el subespacio de dimensión finita
    \begin{align*}
        M_n = \cc{L}\{x_1, \ldots, x_n\} \subset X
    \end{align*}
    Como $\dim M_n \leq n < \infty$ y $M_n \neq X$ por ser $X$ de dimensión infinita, se tiene que existe $x_{n+1} \in X$ tal que $\|x_{n+1}\| = d(x_{n+1}, M_n) = 1$. En particular, para todo $i \in \{1, \ldots, n\}$, se tiene que:
    \begin{align*}
        \|x_{n+1} - x_i\| \geq d(x_{n+1}, M_n) = 1
    \end{align*}
\end{itemize}

Por tanto, por inducción, existe una sucesión $\{x_n\}$ en la esfera unidad de $X$, tal que $\|x_n - x_m\| \geq 1$ para cualesquiera $n, m \in \bb{N}$ con $m \neq n$. Notemos que esta sucesión no puede tener ninguna subsucesión convergente, por lo que la esfera unidad de $X$ no es compacta.
\end{ejercicio}

\begin{ejercicio}
En el espacio de Banach $C\left[0, 1\right]$ se considera el subespacio
\[
M = \left\{f \in C\left[0, 1\right] : \int_{0}^1 f(t)\,dt = 0\right\}
\]
Probar que $M$ es proximinal en $C\left[0, 1\right]$.

\begin{description}
    \item[Proximinalidad de los hiperplanos cerrados:]
    
    Consideramos el funcional integral $I\in C[0, 1]^*\setminus \{0\}$ definido por:
    \Func{I}{C[0, 1]}{\bb{K}}{f}{\int_{0}^1 f(t)\,dt}
    
    Como $M=\ker I$, $M$ es proximinal en $C[0, 1]$ si y solo si $|I|$ tiene máximo en la esfera unidad de $C[0, 1]$. Para cualquier $f\in C[0, 1]$ con $\|f\|_\infty = 1$, por la continuidad de $I$ con $\|I\|=1$, se tiene que:
    \begin{align*}
        |I(f)| \leq \|I\| \|f\|_\infty = 1
    \end{align*}

    Para la función $f_0(t) = 1$ para todo $t \in [0, 1]$, se tiene que $\|f_0\|_\infty = 1$ y:
    \begin{align*}
        |I(f_0)| = \left|\int_{0}^1 1\,dt\right| = 1
    \end{align*}
    Por lo tanto, $|I|$ tiene máximo en la esfera unidad de $C[0, 1]$, y así $M$ es proximinal en $C[0, 1]$.


    \item[Método Constructivo:] En este enfoque, buscamos aplicar de manera directa la definición de conjunto proximinal; es decir, dado $f\in C[0,1]$, buscamos $f_0\in M$ tal que $d(f,M)=\|f-f_0\|_\infty$. Consideramos la función:
    \Func{f_0}{[0, 1]}{\bb{K}}{t}{f(t) - \int_{0}^1 f(s)\,ds}

    Por la linealidad de la integral, se tiene que:
    \begin{align*}
        \int_{0}^1 f_0(t)\,dt = \int_{0}^1 f(t)\,dt - \int_{0}^1 \int_{0}^1 f(s)\,ds\,dt = \int_{0}^1 f(t)\,dt - \int_{0}^1 f(s)\,ds = 0
    \end{align*}
    Por tanto, $f_0 \in M$. Además, para todo $g \in M$, se tiene que:
    \begin{align*}
        \|f - f_0\|_\infty &= \left\|f - \left(f - \int_{0}^1 f(s)\,ds\right)\right\|_\infty = \left\|\int_{0}^1 f(s)\,ds\right\|_\infty = \left|\int_{0}^1 f(s)\,ds\right| =\\&= |I(f)| \AstIg |I(f)-0| = |I(f)-I(g)| = |I(f-g)| \leq\\&\leq  \|I\| \|f-g\|_\infty = \|f-g\|_\infty
    \end{align*}
    donde en $(\ast)$ hemos usado que $g \in M = \ker I$. Por lo tanto, $f_0$ es un elemento de $M$ que minimiza la distancia de $f$ a $M$, y así $M$ es proximinal en $C[0, 1]$.
\end{description}
\end{ejercicio}
