\section{Teorema de Hahn-Banach}

\begin{ejercicio}
    Sean $\varphi, \psi$ dos seminormas en un espacio vectorial $X$, y consideramos $f: X \longrightarrow \bb{K}$ un funcional lineal verificando que
    \[
        |f(x)| \leq \varphi(x) + \psi(x) \quad \forall x \in X
    \]
    Probar que existen dos funcionales lineales $g, h: X \longrightarrow \bb{K}$ tales que:
    \[
        f(x) = g(x) + h(x), \quad |g(x)| \leq \varphi(x), \quad |h(x)| \leq \psi(x) \quad \forall x \in X
    \]
    \begin{observacion}
        En el espacio vectorial $X \times X$ considérese la seminorma $p$ definida por $p(x_1, x_2) = \varphi(x_1) + \psi(x_2)$ y en el subespacio $\Delta = \{(x, x) : x \in X\}$ considérese el funcional lineal $h$ definido por $h(x, x) = f(x)$.
    \end{observacion}

    Consideramos el espacio vectorial $X \times X$, y la seminorma $p$ definida por:
    \Func{p}{X}{\bb{R}^+_0}{(x_1, x_2)}{\varphi(x_1) + \psi(x_2)}

    Comprobemos que $p$ es una seminorma. En primer lugar, $p$ está bien definida. Además:
    \begin{itemize}
        \item \ul{Subaditividad:} Dado $(x_1, x_2), (y_1, y_2) \in X \times X$, se tiene que:
        \begin{align*}
            p((x_1, x_2) + (y_1, y_2)) &= p(x_1 + y_1, x_2 + y_2) \\
            &= \varphi(x_1 + y_1) + \psi(x_2 + y_2) \\
            &\leq \varphi(x_1) + \varphi(y_1) + \psi(x_2) + \psi(y_2) \\
            &= p(x_1, x_2) + p(y_1, y_2)
        \end{align*}
        \item \ul{Homogeneidad:} Dado $(x_1, x_2) \in X \times X$ y $\lambda \in \bb{K}$, se tiene que:
        \begin{align*}
            p(\lambda(x_1, x_2)) &= p(\lambda x_1, \lambda x_2)
            = \varphi(\lambda x_1) + \psi(\lambda x_2)
            = |\lambda| \varphi(x_1) + |\lambda| \psi(x_2)
            = |\lambda| p(x_1, x_2)
        \end{align*}
    \end{itemize}

    Por tanto, $p$ es una seminorma en $X \times X$. Consideremos ahora el conjunto $\Delta = \{(x, x) : x \in X\} \subseteq X \times X$. Comprobemos que $\Delta$ es un subespacio vectorial de $X \times X$. Dado $(x, x), (y, y) \in \Delta$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        (x, x) + (y, y) &= (x + y, x + y) \in \Delta \\
        \lambda(x, x) &= (\lambda x, \lambda x) \in \Delta
    \end{align*}

    Por tanto, $\Delta$ es un subespacio vectorial de $X \times X$. Definimos ahora el funcional lineal $h: \Delta \longrightarrow \bb{K}$ por:
    \Func{h}{\Delta}{\bb{K}}{(x, x)}{f(x)}

    Comprobemos que $h$ es lineal. Dado $(x, x), (y, y) \in \Delta$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        h((x, x) + (y, y)) &= h(x + y, x + y) = f(x + y) = f(x) + f(y) = h(x, x) + h(y, y) \\
        h(\lambda(x, x)) &= h(\lambda x, \lambda x) = f(\lambda x) = \lambda f(x) = \lambda h(x, x)
    \end{align*}

    Por tanto, $h$ es un funcional lineal en $\Delta$. Además, dado $(x, x) \in \Delta$, se tiene que:
    \begin{align*}
        \Re h(x, x) &= \Re f(x) \leq |f(x)| \leq \varphi(x) + \psi(x) = p(x, x)
    \end{align*}

    Por tanto, $h$ es un funcional lineal acotado en $\Delta$ con respecto a la seminorma $p$. Aplicando el Teorema de Hahn-Banach, existe un funcional lineal $H: X \times X \longrightarrow \bb{K}$ que extiende a $h$ y verifica que:
    \[
        |H(x_1, x_2)| \leq p(x_1, x_2) = \varphi(x_1) + \psi(x_2) \quad \forall (x_1, x_2) \in X \times X
    \]


    Definimos ahora los funcionales lineales $g, k: X \longrightarrow \bb{K}$ por:
    \Func{g}{X}{\bb{K}}{x}{H(x, 0)}
    \Func{k}{X}{\bb{K}}{x}{H(0, x)}

    Los funcionales $g$ y $k$ son lineales por ser la restricción de un funcional lineal. Además, dado $x \in X$, se tiene que:
    \begin{align*}
        g(x) + k(x) &= H(x, 0) + H(0, x) = H(x, x) \AstIg h(x, x) = f(x)
    \end{align*}
    donde en $(\ast)$ se ha empleado que $H$ extiende a $h$. Además, dado $x \in X$, se tiene que:
    \begin{align*}
        |g(x)| &= |H(x, 0)| \leq p(x, 0) = \varphi(x) + \psi(0) = \varphi(x) \\
        |k(x)| &= |H(0, x)| \leq p(0, x) = \varphi(0) + \psi(x) = \psi(x)
    \end{align*}
\end{ejercicio}

\begin{ejercicio}
    Sean $X$ un espacio normado real y $M$ un subespacio vectorial de $X$. Probar que si $f, g \in M^*$ verifican que $|f(m)| + |g(m)| \leq \|m\|$ para todo $m \in M$, entonces existen $h, k \in X^*$ que extienden a $f$ y $g$ respectivamente y verifican
    \[
        |h(x)| + |k(x)| \leq \|x\| \quad \forall x \in X
    \]
    \begin{observacion}
        Considérense $f + g$ y $f - g$.
    \end{observacion}

    Podríamos intentar aplicar el Teorema de Hahn-Banach a los funcionales lineales $f, g \in M^*$, pero no podemos acotar $|f(m)|$ y $|g(m)|$ por $\|m\|$, sino que solo podemos acotar $|f(m)| + |g(m)|$ por $\|m\|$. Por tanto, no podemos aplicar directamente el Teorema de Hahn-Banach a $f$ y $g$. Sin embargo, podemos considerar los funcionales lineales $f + g$ y $f - g$, que sí podemos acotar por $\|m\|$.\\

    Consideremos los funcionales lineales $\varphi_1, \varphi_2: M \longrightarrow \bb{R}$ definidos por:
    \begin{equation*}
        \varphi_1 = f + g, \quad \varphi_2 = f - g
    \end{equation*}

    Dado $m \in M$, se tiene que:
    \begin{align*}
        \Re \varphi_1(m) &\leq |\varphi_1(m)| = |f(m) + g(m)| \leq |f(m)| + |g(m)| \leq \|m\| \\
        \Re \varphi_2(m) &\leq |\varphi_2(m)| = |f(m) - g(m)| \leq |f(m)| + |g(m)| \leq \|m\|
    \end{align*}

    Por tanto, $\varphi_1$ y $\varphi_2$ son funcionales lineales acotados en $M$ con respecto a la norma de $X$. Aplicando el Teorema de Hahn-Banach, existen funcionales lineales $\Phi_1, \Phi_2: X \longrightarrow \bb{R}$ que extienden a $\varphi_1$ y $\varphi_2$ respectivamente y verifican que:
    \begin{equation*}
        |\Phi_1(x)| \leq \|x\|, \quad |\Phi_2(x)| \leq \|x\| \quad \forall x \in X
    \end{equation*}

    Notemos que las extensiones $\Phi_1$ y $\Phi_2$ son lineales y acotadas, por lo que son continuas con norma $\|\Phi_1\| \leq 1$ y $\|\Phi_2\| \leq 1$. En vistas de obtener los valores de $h$ y $k$, definimos el sistema siguiente (se refleja este proceso para que el lector pueda ver cómo se construyen las extensiones):
    \begin{equation*}
        \begin{cases}
            \Phi_1 &= h + k \\
            \Phi_2 &= h - k
        \end{cases}
    \end{equation*}

    Resolviendo el sistema, obtenemos:
    \begin{equation*}
        h = \frac{\Phi_1 + \Phi_2}{2}, \quad k = \frac{\Phi_1 - \Phi_2}{2}
    \end{equation*}

    Por tanto, una vez se han obtenido las extensiones $\Phi_1$ y $\Phi_2$, podemos definir los funcionales lineales $h, k: X \longrightarrow \bb{R}$ por:
    \begin{equation*}
        h(x) = \frac{\Phi_1(x) + \Phi_2(x)}{2}, \quad k(x) = \frac{\Phi_1(x) - \Phi_2(x)}{2}
    \end{equation*}

    Es claro que $h,k\in X^*$, y que extienden a $f$ y $g$ respectivamente. Por último, hemos de comprobar que $|h(x)| + |k(x)| \leq \|x\|$ para todo $x \in X$. Dado $x \in X$, una primera idea sería aplicar la desigualdad triangular, pero no nos sirve, pues nos daría:
    \begin{align*}
        |h(x)| + |k(x)| &\leq \frac{|\Phi_1(x)| + |\Phi_2(x)|}{2} + \frac{|\Phi_1(x)| + |\Phi_2(x)|}{2} = |\Phi_1(x)| + |\Phi_2(x)| \leq \|x\| + \|x\| = 2\|x\|
    \end{align*}

    Como vemos, nos da un factor de 2 que no nos sirve. Emplearemos la siguiente desigualdad, válida para todo $a, b \in \bb{R}$ (aún no habíamos empleado que $\bb{K} = \bb{R}$):
    \begin{equation*}
        |a| + |b| = \max\{|a + b|, |a - b|\}
    \end{equation*}

    Por tanto, dado $x \in X$, se tiene que:
    \begin{align*}
        |h(x)| + |k(x)| &= \max\{|h(x) + k(x)|, |h(x) - k(x)|\} = \max\left\{\left|\Phi_1(x)\right|, \left|\Phi_2(x)\right|\right\} \leq \|x\|
    \end{align*}
\end{ejercicio}

\begin{ejercicio}
    Sea $X$ un espacio normado complejo, y $f: \bb{C} \longrightarrow X$ una función entera, es decir, derivable en todo punto de $\bb{C}$. Probar que, si $f$ está acotada, entonces $f$ es constante. (Se supone conocido el caso $X = \bb{C}$, conocido como el Teorema de Liouville).\\

    Por reducción al absurdo, supongamos que $f$ no es constante. Entonces, existen $z_1, z_2 \in \bb{C}$ tales que $f(z_1) \neq f(z_2)$. Por el Teorema de Hahn-Banach, existe un funcional lineal y continuo $x^* \in X^*$ tal que:
    \[
        x^*(f(z_1) - f(z_2)) = \|f(z_1) - f(z_2)\|
    \]

    Como $f(z_1) \neq f(z_2)$, se tiene que:
    \begin{align*}
        \|f(z_1) - f(z_2)\| \neq 0\Longrightarrow
        x^*(f(z_1) - f(z_2)) \neq 0\Longrightarrow
        x^*(f(z_1)) \neq x^*(f(z_2))
    \end{align*}
    donde en la última implicación se ha empleado la linealidad de $x^*$. Por tanto, $x^* \circ f: \bb{C} \longrightarrow \bb{C}$ no es constante.\\
    
    Si probamos que $x^* \circ f: \bb{C} \longrightarrow \bb{C}$ es una función entera y acotada, obtendremos una contradicción con el Teorema de Liouville (el cual afirma que toda función entera y acotada es constante); contradicción que nos permitirá concluir que $f$ es constante.
    \begin{itemize}
        \item Comprobemos que $x^* \circ f$ es entera. Dado $z_0 \in \bb{C}$, se tiene para todo $z \in \bb{C}\setminus \{z_0\}$ que:
        \begin{align*}
            \dfrac{(x^* \circ f)(z) - (x^* \circ f)(z_0)}{z - z_0} &= \dfrac{x^*(f(z)) - x^*(f(z_0))}{z - z_0} = x^*\left(\dfrac{f(z) - f(z_0)}{z - z_0}\right)
        \end{align*}

        Como $f$ es entera, se tiene que:
        \begin{align*}
            \exists \lim_{z \to z_0} \dfrac{f(z) - f(z_0)}{z - z_0} = f'(z_0)
        \end{align*}

        Además, como $x^*$ continua, se tiene que:
        \begin{align*}
            \exists \lim_{z \to z_0} x^*\left(\dfrac{f(z) - f(z_0)}{z - z_0}\right) = x^*\left(\lim_{z \to z_0} \dfrac{f(z) - f(z_0)}{z - z_0}\right) = x^*(f'(z_0))
        \end{align*}

        Por tanto, $x^* \circ f$ es entera y $(x^* \circ f)'(z_0) = x^*(f'(z_0))$.

        \item Comprobemos que $x^* \circ f$ es acotada. Como $f$ es acotada, existe $M > 0$ tal que $\|f(z)\| \leq M$ para todo $z \in \bb{C}$. Además, como $x^*$ es continua, consideramos $\|x^*\| \in \bb{R}^+_0$. Por tanto, dado $z \in \bb{C}$, se tiene que:
        \begin{align*}
            |(x^* \circ f)(z)| = |x^*(f(z))| \leq \|x^*\| \cdot \|f(z)\| \leq \|x^*\| \cdot M\qquad \forall z \in \bb{C}
        \end{align*}
        Por tanto, $x^* \circ f$ es acotada.
    \end{itemize}

    Como $x^* \circ f$ es entera y acotada; y anteriormente hemos probado que $x^* \circ f$ no es constante, obtenemos una contradicción con el Teorema de Liouville. La contradicción proviene de suponer que $f$ no es constante, por lo que concluimos que $f$ es constante.
\end{ejercicio}

\begin{ejercicio}
Sea $X$ un espacio de Banach y $Z$ un subespacio cerrado de $X^*$, tal que $Z \neq X^*$. Supongamos que $Z$ separa los puntos de $X$, es decir,
\[
x \in X, \; x^*(x) = 0 \;\; \forall x^* \in Z \implies x = 0
\]
Probar que $X$ no es reflexivo.\\

    Recordemos en primer lugar qué significa que una familia de funcionales $Z \subseteq X^*$ \textit{separe los puntos} de $X$. Dados dos puntos distintos $x, y \in X$ ($x \neq y$), decimos que un funcional $f \in X^*$ separa a $x$ de $y$ si los distingue mediante sus imágenes, esto es, si $f(x) \neq f(y)$. Por la linealidad de los funcionales de $Z$, se tiene que $f(x) \neq f(y)$ si y solo si $f(x - y) \neq 0$. Así, pedir que para cada par de puntos distintos $x \neq y$ exista $f \in Z$ con $f(x) \neq f(y)$ equivale a exigir que, para cada vector no nulo $v = x - y \in X \setminus \{0\}$, exista $f \in Z$ tal que $f(v) \neq 0$. Tomando el contrarrecíproco, obtenemos precisamente la caracterización del enunciado:
    \[
        x \in X, \quad x^*(x) = 0 \;\; \forall x^* \in Z \implies x = 0.
    \]

    Continuamos ahora con el ejercicio en sí. Hay dos opciones.
    \begin{description}
        \item[Opción 1.] Como $Z \neq X^*$, podemos fijar un funcional $y^* \in X^* \setminus Z$. Dado que $Z$ es un subespacio cerrado de $X^*$, existe un funcional $x^{**} \in Z^\perp \subseteq X^{**}$ tal que $\|x^{**}\| = 1$ y $x^{**}(y^*) = d(y^*, Z) > 0$.

        Supongamos ahora, por reducción al absurdo, que $X$ es reflexivo. Entonces, la inyección canónica $J : X \longrightarrow X^{**}$ de $X$ en su bidual es sobreyectiva, por lo que existe $x \in X$ tal que $x^{**} = J(x)$. Por tanto, dado cualquier $x^* \in Z$, se tiene que:
        \begin{equation*}
            x^*(x) = (J(x))(x^*) = x^{**}(x^*) \AstIg 0,
        \end{equation*}
        donde en $(\ast)$ se ha empleado que $x^{**} \in Z^\perp$. Así pues, $x^*(x) = 0$ para todo $x^* \in Z$. Como $Z$ separa los puntos de $X$, deducimos que $x = 0$. En consecuencia,
        \[
            x^{**} = J(x) = J(0) = 0,
        \]
        lo cual entra en contradicción con que $\|x^{**}\| = 1$. Concluimos así que $X$ no es reflexivo.
        
        \item[Opción 2.] Supongamos que $X$ es reflexivo. Entonces, la inyección canónica $J_X : X \longrightarrow X^{**}$ de $X$ en su bidual es sobreyectiva. Buscamos calcular $Z^\perp\cap J_X(X)$. Sea $x^{**} \in Z^\perp\cap J_X(X)$. Entonces, existe $x \in X$ tal que $x^{**} = J_X(x)$. Por tanto, dado cualquier $x^* \in Z$, se tiene que:
        \begin{equation*}
            x^*(x) = (J_X(x))(x^*) = x^{**}(x^*) \AstIg 0
        \end{equation*}
        donde en $(\ast)$ se ha empleado que $x^{**} \in Z^\perp$. Así pues, $x^*(x) = 0$ para todo $x^* \in Z$. Como $Z$ separa los puntos de $X$, deducimos que $x = 0$. En consecuencia:
        \[
            x^{**} = J_X(x) = J_X(0) = 0
        \]

        Por tanto, $Z^\perp\cap J_X(X) = \{0\}$. Como $X· $ es reflexivo, $J_X(X) = X^{**}$, por lo que $Z^\perp\cap J_X(X) = Z^\perp\cap X^{**} = Z^\perp$. Así pues, $Z^\perp = \{0\}$, luego concluimos que $Z$ es denso y, por ser cerrado, $Z = X^*$. Esto contradice la hipótesis de que $Z \neq X^*$. Concluimos así que $X$ no es reflexivo.
    \end{description}
\end{ejercicio}

\begin{ejercicio}
    Probar que un subespacio de $c_0$, con la norma inducida por $c_0$, no puede ser isomorfo a $\ell_1$.\\

    Demostramos anteriormente el siguiente Lema.
    \begin{lema}
        Sean $A,B$ dos espacios topológicos y $f:A\to B$ continua. Si $A$ es separable, entonces $f(A)$ es separable.
        \begin{proof}
            Supongamos que $A$ es separable. Entonces, existe un subconjunto denso y numerable $D \subseteq A$:
            \begin{equation*}
                D = \{d_n : n \in \bb{N}\} \subseteq A
            \end{equation*}

            Como $f$ es una aplicación, se tiene que $f(D) \subseteq B$ es numerable. Veamos que $f(D)$ es denso en $f(A)$:
            \begin{equation*}
                f(A) = f\left(\overline{D}\right) \subseteq \overline{f(D)}\subseteq \overline{f(A)}
            \end{equation*}

            Por tanto, $\overline{f(D)} = f(A)$ (donde la clausura se toma en $f(A)$), y por tanto $f(D)$ es denso en $f(A)$. Concluimos así que $f(A)$ es separable.
        \end{proof}
    \end{lema}

    Sea ahora $M$ un subespacio de $c_0$. Tenemos que:
    \begin{equation*}
        M^* \cong c_0^*/M^\perp
    \end{equation*}

    En primer lugar, sabemos que $\ell_1 \cong c_0^*$, $c_0^*$ es separable. Además, como la aplicación cociente
    \begin{equation*}
        q: c_0^* \longrightarrow c_0^*/M^\perp
    \end{equation*}
    es continua y sobreyectiva, se tiene que $c_0^*/M^\perp$ es separable. Por tanto, $M^*$ es separable.\\

    Supongamos ahora que $M \cong \ell_1$. Entonces, $M^* \cong \ell_1^* \cong \ell_\infty$. Como $\ell_\infty$ no es separable, obtenemos una contradicción. Concluimos así que $M$ no puede ser isomorfo a $\ell_1$.
\end{ejercicio}

\begin{ejercicio}
Sean $X$ un espacio normado y $M$ un subespacio vectorial de $X$. Probar que para cada operador $T \in L(M, \ell_\infty)$, existe $S \in L(X, \ell_\infty)$ tal que $S|_M = T$ y $\|S\| = \|T\|$.\\

    El Teorema de Extensión de Hahn-Banach nos permite extender funcionales, pero no operadores. No obstante, fijado $n\in \bb{N}$, podemos considerar el funcional lineal que proyecta sobre la $n$-ésima coordenada, y aplicar el Teorema de Extensión de Hahn-Banach a dicho funcional. Consideramos:
    \Func{\pi_n}{\ell_\infty}{\bb{K}}{x}{x(n)}

    Tenemos que $\pi_n \in \ell_\infty^{\#}$. Veamos que es continuo:
    \begin{equation*}
        |\pi_n(x)| = |x(n)| \leq \sup_{k \in \bb{N}} |x(k)| = \|x\|_\infty\qquad \forall x \in \ell_\infty
    \end{equation*}

    Por tanto, $\pi_n\in \ell_\infty^*$. Además, considerando $e_n \in \ell_\infty$ se tiene que $\|e_n\|_\infty = 1$ y $\pi_n(e_n) = 1$, por lo que $\|\pi_n\| = 1$, y por tanto $\pi_n \in \ell_\infty^*$ con $\|\pi_n\| = 1$.\\

    Dado $T \in L(M, \ell_\infty)$, consideremos el funcional lineal siguiente:
    \begin{align*}
        T_n = \pi_n \circ T: M \longrightarrow \bb{K}
    \end{align*}

    Veamos que $T_n \in M^*$:
    \begin{equation*}
        |T_n(m)| = |\pi_n(T(m))| \leq \|T(m)\|_\infty \leq \|T\| \cdot \|m\|\qquad \forall m \in M
    \end{equation*}

    Por tanto, $T_n \in M^*$ con $\|T_n\| \leq \|T\|$. Aplicando el Teorema de Extensión de Hahn-Banach (pues $T_n \in M^*$), existe $S_n \in X^*$ que extiende a $T_n$ y verifica que $\|S_n\| = \|T_n\| \leq \|T\|$.\\

    Emplear dicha extensión para cada $n \in \bb{N}$ nos permite definir el siguiente operador:
    \Func{S}{X}{\ell_\infty}{x}{\{S_n(x)\}_{n \in \bb{N}}}

    Es necesario probar que $S$ está bien definido, que es lineal, que extiende a $T$ y que es continuo con $\|S\| = \|T\|$.\\

    Veamos en primer lugar que $S$ está bien definido. Dado $x \in X$, se tiene que:
    \begin{equation*}
        \|S_n(x)\| \leq \|S_n\| \cdot \|x\| = \|T_n\| \cdot \|x\| \leq \|T\| \cdot \|x\|\Longrightarrow \sup_{n \in \bb{N}} |S_n(x)| \leq \|T\| \cdot \|x\| < +\infty
    \end{equation*}

    Por tanto, $S(x) = \{S_n(x)\}_{n \in \bb{N}} \in \ell_\infty$, y por tanto $S$ está bien definido. Veamos que $S$ es lineal. Dado $x, y \in X$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        S(x + y) &= \{S_n(x + y)\}_{n \in \bb{N}} = \{S_n(x) + S_n(y)\}_{n \in \bb{N}} = \{S_n(x)\}_{n \in \bb{N}} + \{S_n(y)\}_{n \in \bb{N}} = S(x) + S(y) \\
        S(\lambda x) &= \{S_n(\lambda x)\}_{n \in \bb{N}} = \{\lambda S_n(x)\}_{n \in \bb{N}} = \lambda \{S_n(x)\}_{n \in \bb{N}} = \lambda S(x)
    \end{align*}

    Por tanto, $S$ es lineal. Veamos que $S$ extiende a $T$. Dado $m \in M$, se tiene que:
    \begin{equation*}
        S(m) = \{S_n(m)\}_{n \in \bb{N}} = \{T_n(m)\}_{n \in \bb{N}} = \{\pi_n(T(m))\}_{n \in \bb{N}} = T(m)
    \end{equation*}

    Por tanto, $S|_M = T$. Veamos ahora que $S$ es continuo. Dado $x \in X$, se tiene que:
    \begin{equation*}
        \|S(x)\|_\infty = \sup_{n \in \bb{N}} |S_n(x)| \leq \sup_{n \in \bb{N}} \|S_n\| \cdot \|x\| = \sup_{n \in \bb{N}} \|T_n\| \cdot \|x\| \leq \|T\| \cdot \|x\|
    \end{equation*}

    Por tanto, $S \in L(X, \ell_\infty)$ con $\|S\| \leq \|T\|$. Veamos que, de hecho, $\|S\| = \|T\|$. Dado $m \in M$, se tiene que:
    \begin{equation*}
        \|S\| = \sup_{x \in X, \|x\| \leq 1} \|S(x)\|_\infty \geq \sup_{m \in M, \|m\| \leq 1} \|S(m)\|_\infty = \sup_{m \in M, \|m\| \leq 1} \|T(m)\|_\infty = \|T\|
    \end{equation*}

    Por tanto, $\|S\| = \|T\|$. De esta manera, hemos construido un operador $S \in L(X, \ell_\infty)$ que extiende a $T$ y verifica que $\|S\| = \|T\|$.
\end{ejercicio}

\begin{ejercicio}
    Sean $X$ un espacio normado, $M$ un subespacio cerrado y $u \in X$. Probar que existe $f \in X^*$ tal que $|f(x)| \leq \mathrm{dist}(x, M)$ para todo $x \in X$ y, además, $f(u) = \mathrm{dist}(u, M)$.\\

    Distinguimos dos casos. Si $u\in M$, consideramos el funcional nulo $f\cong 0\in X^*$. En tal caso, se tiene:
    \begin{align*}
        |f(x)| &= 0\leq \mathrm{dist}(x, M) \quad \forall x \in X\\
        f(u) &= 0 = \mathrm{dist}(u, M)
    \end{align*}

    Supongamos ahora que $u \notin M$. Como $M$ es cerrado, existe $f\in M^\perp$ con $\|f\| = 1$ y $f(u) = \mathrm{dist}(u, M)$. En tal caso, para cada $m\in M$, se tiene que:
    \begin{align*}
        |f(x)| &= |f(x) - f(m)| = |f(x - m)| \leq \|f\| \cdot \|x - m\| = \|x - m\|
    \end{align*}

    Tomando el ínfimo sobre todos los $m \in M$, obtenemos que:
    \begin{align*}
        |f(x)| \leq \inf_{m \in M} \|x - m\| = \mathrm{dist}(x, M) \quad \forall x \in X
    \end{align*}
\end{ejercicio}

\begin{ejercicio}
    Fijado $n \in \bb{N}$, probar que existe un funcional lineal y continuo $f\in C\left[0, 1\right]^*$ verificando que $f(p) = p'(0)$ para todo polinomio $p$ de grado menor o igual que $n$. ¿Existe un funcional lineal y continuo $f$ en $C\left[0, 1\right]$ tal que $f(p) = p'(0)$ para todo polinomio $p$?\\

    Consideramos $\cc{P}_{\leq n} = \{p \in \cc{P} : \deg(p) \leq n\}$, que es un subespacio vectorial de $C[0, 1]$. Como $\dim \cc{P}_{\leq n} = n + 1 < +\infty$, se tiene que $\cc{P}_{\leq n}$ es un espacio de dimensión finita. Consideramos el funcional lineal $g: \cc{P}_{\leq n} \longrightarrow \bb{K}$ definido por:
    \Func{g}{\cc{P}_{\leq n}}{\bb{K}}{p}{p'(0)}

    Veamos que $g$ es lineal. Dado $p, q \in \cc{P}_{\leq n}$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        g(p + q) &= (p + q)'(0) = p'(0) + q'(0) = g(p) + g(q) \\
        g(\lambda p) &= (\lambda p)'(0) = \lambda p'(0) = \lambda g(p)
    \end{align*}

    Por tanto, $g$ es lineal. Además, como $\cc{P}_{\leq n}$ es un espacio de dimensión finita, todos los funcionales lineales en $\cc{P}_{\leq n}$ son continuos. Por tanto, $g \in \cc{P}_{\leq n}^*$. Por el Teorema de Extensión de Hahn-Banach, existe un funcional lineal y continuo $f \in C[0, 1]^*$ que extiende a $g$, es decir, $f|_{\cc{P}_{\leq n}} = g$. Por tanto:
    \[
        f(p) = g(p) = p'(0) \quad \forall p \in \cc{P}_{\leq n}
    \]

    Por tanto, queda probado lo pedido. Respecto a la segunda parte del enunciado, supongamos que existe un funcional lineal y continuo $f \in C[0, 1]^*$ tal que se cumpla $f(p) = p'(0)$ para todo polinomio $p$. Como buscamos probar que $f$ no es continuo, basta con encontrar una sucesión de polinomios $\{p_n\}_{n \in \bb{N}}$ tal que $\|p_n\|_\infty$ esté acotada y $|f(p_n)|$ no lo esté. Como $f(p) = p'(0)$ para todo polinomio $p$, basta con encontrar una sucesión de polinomios $\{p_n\}_{n \in \bb{N}}$ tal que $\|p_n\|_\infty$ esté acotada y $|p_n'(0)|$ no lo esté.
    \begin{itemize}
        \item Como buscamos que $|p_n'(0)|$ no esté acotada, imponemos que $p_n'(0) = n$ para cada $n \in \bb{N}$.
        \item Como buscamos que $\|p_n\|_\infty$ esté acotada, imponemos que $\|p_n\|_\infty = 1$ para cada $n \in \bb{N}$.
    \end{itemize}
    
    Consideramos por tanto la sucesión $\{p_n\}_{n \in \bb{N}}\subset C[0, 1]$ definida por:
    \begin{align*}
        p_n(t) &= (1 - t)^n\qquad \forall t \in [0, 1]
    \end{align*}

    Para cada $n\in \bb{N}$, se tiene que:
    \begin{align*}
        p_n'(t) &= -n(1 - t)^{n - 1}\qquad \forall t \in [0, 1]\\
        p_n'(0) &= -n\\
        \|p_n\|_\infty &= \sup_{t \in [0, 1]} |p_n(t)| = \sup_{t \in [0, 1]} |(1 - t)^n| = 1
    \end{align*}

    Como $f$ es continuo, existe $M > 0$ tal que:
    \begin{equation*}
        |f(p)| \leq M \|p\|_\infty \quad \forall p \in C[0, 1]
    \end{equation*}

    En particular, para cada $n \in \bb{N}$, se tiene que:
    \begin{align*}
        |f(p_n)| = |p_n'(0)| = n \leq M \|p_n\|_\infty = M \quad \forall n \in \bb{N}
    \end{align*}

    Como $n$ es arbitrario, obtenemos una contradicción. Por tanto, no existe un funcional lineal y continuo $f \in C[0, 1]^*$ tal que $f(p) = p'(0)$ para todo polinomio $p$.
\end{ejercicio}

\begin{ejercicio}[Límites de Banach]
Para cada $n \in \bb{N}$ consideramos $f_n: \ell_\infty \longrightarrow \bb{K}$ el funcional definido por
\[
f_n(x) = \frac{x(1) + x(2) + \dots + x(n)}{n} \quad (x \in \ell_\infty)
\]
Sea $M = \{x \in \ell_\infty : \{f_n(x)\} \text{ converge}\}$ y definamos $f(x) = \lim\limits_n f_n(x)$ ($x \in M$).
\begin{enumerate}
    \item Probar que $f_n \in \ell_\infty^*$ y que $\|f_n\| = 1$ para todo $n \in \bb{N}$.
    
    Tenemos que:
    \begin{equation*}
        f_n(x) = \dfrac{1}{n} \sum_{k = 1}^n x(k)
    \end{equation*}

    Comprobemos en primer lugar que $f_n$ es lineal. Dado $x, y \in \ell_\infty$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        f_n(x + y) &= \dfrac{1}{n} \sum_{k = 1}^n (x(k) + y(k)) = \dfrac{1}{n} \sum_{k = 1}^n x(k) + \dfrac{1}{n} \sum_{k = 1}^n y(k) = f_n(x) + f_n(y) \\
        f_n(\lambda x) &= \dfrac{1}{n} \sum_{k = 1}^n \lambda x(k) = \lambda \dfrac{1}{n} \sum_{k = 1}^n x(k) = \lambda f_n(x)
    \end{align*}

    Por tanto, $f_n$ es lineal. Veamos ahora que $f_n$ es continuo. Dado $x \in \ell_\infty$, se tiene que:
    \begin{align*}
        |f_n(x)| &= \left|\dfrac{1}{n} \sum_{k = 1}^n x(k)\right| \leq \dfrac{1}{n} \sum_{k = 1}^n |x(k)| \leq \dfrac{1}{n} \sum_{k = 1}^n \|x\|_\infty = \dfrac{1}{n} \cdot n \cdot \|x\|_\infty = \|x\|_\infty
    \end{align*}
    Por tanto, $f_n$ es continuo y $\|f_n\|\leq 1$. Consideremos ahora los vectores unitarios $\{e_k\}_{k \in \bb{N}}$ de $\ell_\infty$, donde $e_k$ es la sucesión que tiene un 1 en la $k$-ésima coordenada y 0 en las demás. Consideramos:
    \begin{equation*}
        x = \sum_{k = 1}^n e_k\in \ell_\infty
    \end{equation*}

    Es claro que $\|x\|_\infty = 1$. Además, se tiene que:
    \begin{align*}
        f_n(x) &= \dfrac{1}{n} \sum_{k = 1}^n x(k) = \dfrac{1}{n} \sum_{k = 1}^n 1 = \dfrac{1}{n} \cdot n = 1
    \end{align*}

    Por tanto, $\|f_n\| \geq |f_n(x)| = 1$. En vistas de que anteriormente habíamos probado que $\|f_n\| \leq 1$, obtenemos que $\|f_n\| = 1$.
    \item Probar que $M$ es un subespacio vectorial de $\ell_\infty$ que contiene al espacio $c$ de las sucesiones convergentes.
    
    Comprobemos que $M$ es un subespacio vectorial de $\ell_\infty$. Dado $x, y \in M$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        f_n(x + y) &= \dfrac{1}{n} \sum_{k = 1}^n (x(k) + y(k)) = \dfrac{1}{n} \sum_{k = 1}^n x(k) + \dfrac{1}{n} \sum_{k = 1}^n y(k) = f_n(x) + f_n(y) \\
        f_n(\lambda x) &= \dfrac{1}{n} \sum_{k = 1}^n \lambda x(k) = \lambda \dfrac{1}{n} \sum_{k = 1}^n x(k) = \lambda f_n(x)
    \end{align*}

    Como $x, y \in M$, se tiene que $\{f_n(x)\}$ y $\{f_n(y)\}$ convergen. Por tanto, $\{f_n(x + y)\} = \{f_n(x) + f_n(y)\}$ también converge, y $\{f_n(\lambda x)\} = \{\lambda f_n(x)\}$ también converge. Por tanto, $x + y \in M$ y $\lambda x \in M$. Por tanto, $M$ es un subespacio vectorial de $\ell_\infty$.\\

    Sea $x \in c$. Si $\bb{K}=\bb{C}$, entonces $\{\Re x_n\}$ y $\{\Im x_n\}$ convergen, luego podemos reducirnos al caso real. Queremos ver que $\{f_n(x)\}$ converge. En vistas de aplicar el Criterio de Stolz, tenemos que:
    \begin{equation*}
        \left\{\dfrac{\sum\limits_{k = 1}^{n+1} x(k) - \sum\limits_{k = 1}^n x(k)}{(n + 1) - n}\right\} = \{x(n + 1)\}
    \end{equation*}

    Como $\{x(n + 1)\}$ converge, aplicando el Criterio de Stolz se tiene que:
    \begin{equation*}
        \left\{\dfrac{\sum\limits_{k = 1}^n x(k)}{n}\right\} = \{f_n(x)\} \text{ converge}
    \end{equation*}

    De hecho, se tiene que:
    \begin{equation*}
        \lim_{n \to \infty} f_n(x) = \lim_{n \to \infty} x(n)
    \end{equation*}

    Por tanto, $x \in M$. Como $x \in c$ es arbitrario, se tiene que $c \subseteq M$.

    \item Probar que $f \in M^*$ con $\|f\| = 1$ y que $f(x) = \lim \{x_n\}$ para todo $x \in c$.
    
    Dado $x \in M$, se tiene que $\{f_n(x)\}$ converge, y por tanto $f(x) = \lim\limits_n f_n(x)$ está bien definido. Veamos que $f$ es lineal. Dado $x, y \in M$ y $\lambda \in \bb{K}$, se tiene que:
    \begin{align*}
        f(x + y) &= \lim_{n \to \infty} f_n(x + y) = \lim_{n \to \infty} (f_n(x) + f_n(y)) = \lim_{n \to \infty} f_n(x) + \lim_{n \to \infty} f_n(y) = f(x) + f(y) \\
        f(\lambda x) &= \lim_{n \to \infty} f_n(\lambda x) = \lim_{n \to \infty} (\lambda f_n(x)) = \lambda \lim_{n \to \infty} f_n(x) = \lambda f(x)
    \end{align*}

    Por tanto, $f$ es lineal. Veamos que $f$ es continuo. Dado $x \in M$, se tiene que:
    \begin{align*}
        |f(x)| &= \left|\lim_{n \to \infty} f_n(x)\right| = \lim_{n \to \infty} |f_n(x)| \leq \lim_{n \to \infty} \|f_n\| \cdot \|x\| = \lim_{n \to \infty} \|x\| = \|x\|
    \end{align*}

    Por tanto, $f$ es continuo y $\|f\| \leq 1$. Consideremos ahora la sucesión constantemente igual a 1, es decir, $x = (1, 1, 1, \dots)$, que trivialmente es convergente y por tanto $x \in c \subseteq M$, con $\|x\| = 1$. Se tiene que:
    \begin{align*}
        f(x) &= \lim_{n \to \infty} f_n(x) = \lim_{n \to \infty} x(n) = 1
    \end{align*}

    Por tanto, $\|f\| \geq |f(x)| = 1$. En vistas de que anteriormente habíamos probado que $\|f\| \leq 1$, obtenemos que $\|f\| = 1$.
    \item Sea $\tau(x) = (x(2), x(3), \dots)$ para todo $x \in \ell_\infty$. Probar que se cumple que $x - \tau(x) \in \ker(f) \subseteq M$ para todo $x \in \ell_\infty$.\\
    
    Fijamos $x \in \ell_\infty$. Se tiene que:
    \begin{align*}
        x-\tau(x) &= (x(1), x(2), x(3), \dots) - (x(2), x(3), x(4), \dots) =\\&= (x(1) - x(2), x(2) - x(3), x(3) - x(4), \dots)
    \end{align*}

    Consideremos la sucesión $\{f_n(x - \tau(x))\}$, y queremos ver que converge. Se tiene que:
    \begin{align*}
        f_n(x - \tau(x)) &= \dfrac{1}{n} \sum_{k = 1}^n (x(k) - x(k + 1)) =\\&= \dfrac{1}{n} \left(\sum_{k = 1}^n x(k) - \sum_{k = 1}^n x(k + 1)\right) = \dfrac{1}{n} \left(\sum_{k = 1}^n x(k) - \sum_{k = 2}^{n + 1} x(k)\right) \\
        &= \dfrac{1}{n} \left(x(1) - x(n + 1)\right) = \dfrac{x(1) - x(n + 1)}{n}
    \end{align*}

    Como $x \in \ell_\infty$, para todo $n \in \bb{N}$ se tiene que:
    \begin{align*}
        \left|f_n(x - \tau(x))\right| = \left|\dfrac{x(1) - x(n + 1)}{n}\right| \leq \dfrac{|x(1)| + |x(n + 1)|}{n} \leq \dfrac{2\|x\|_\infty}{n}
    \end{align*}

    Por tanto, tomando límite cuando $n \to \infty$, obtenemos que:
    \begin{align*}
        \lim_{n \to \infty} f_n(x - \tau(x)) = 0
    \end{align*}

    Por tanto, $x - \tau(x) \in M$ y $f(x - \tau(x)) = 0$; es decir, $x - \tau(x) \in \ker(f)$. Como $x \in \ell_\infty$ es arbitrario, obtenemos que $x - \tau(x) \in \ker(f) \subseteq M$ para todo $x \in \ell_\infty$.


    \item Deducir que existe $L \in \ell_\infty^*$, extensión de $f$, tal que $\|L\| = 1$ y $L(x) = L(\tau^n(x))$ para todo $x \in \ell_\infty$ y para todo $n \in \bb{N}$.
    
    Como $M$ es un subespacio vectorial de $\ell_\infty$ y $f \in M^*$, aplicando el Teorema de Extensión de Hahn-Banach obtenemos que existe $L \in \ell_\infty^*$ que extiende a $f$, es decir, $L|_M = f$, y verifica que $\|L\| = \|f\| = 1$. Para ver que $L(x) = L(\tau^n(x))$, hacemos inducción sobre $n$.
    \begin{itemize}
        \item \ul{Caso base: $n = 1$}. Para todo $x \in \ell_\infty$, se tiene que:
        \begin{align*}
            L(x) - L(\tau(x)) &= L(x - \tau(x)) = f(x - \tau(x)) = 0
        \end{align*}
        donde se ha empleado que $x - \tau(x) \in M$ y que $L|_M = f$. Por tanto, $L(x) = L(\tau(x))$.
        \item \ul{Paso inductivo}. Supongamos que $L(x) = L(\tau^n(x))$ para algún $n \in \bb{N}$. Entonces, por el caso base, se tiene que:
        \begin{align*}
            L(x) = L(\tau(x))
        \end{align*}

        Aplicando la hipótesis inductiva a $\tau(x)\in \ell_\infty$, obtenemos que:
        \begin{align*}
            L(\tau(x)) = L(\tau^{n + 1}(x))
        \end{align*}

        Por tanto, $L(x) = L(\tau^{n + 1}(x))$.
    \end{itemize}

    Por el principio de inducción, se tiene que $L(x) = L(\tau^n(x))$ para todo $n \in \bb{N}$ y para todo $x \in \ell_\infty$.
    \item Probar que $L(0, \nicefrac{1}{2}, 0, \nicefrac{1}{2}, \dots) = \nicefrac{1}{4}$. ¿Cuánto vale $L(1, 0, 0, 1, 0, 0, 1, \dots)$?
    
    Puesto que ninguna de las sucesiones $(0, \nicefrac{1}{2}, 0, \nicefrac{1}{2}, \dots)$ y $(1, 0, 0, 1, 0, 0, 1, \dots)$ es convergente, no podemos aplicar directamente la definición de $f$. Sin embargo, podemos usar la propiedad de que $L(x) = L(\tau^n(x))$ para todo $n \in \bb{N}$ y para todo $x \in \ell_\infty$. Consideramos la sucesión $x = (0, \nicefrac{1}{2}, 0, \nicefrac{1}{2}, \dots)$. Aplicando $\tau$, obtenemos:
    \begin{align*}
        \tau(x) &= (\nicefrac{1}{2}, 0, \nicefrac{1}{2}, 0, \dots)
    \end{align*}

    Por tanto:
    \begin{align*}
        2L(x) &= L(x) + L(\tau(x)) = L(x + \tau(x)) = L((0, \nicefrac{1}{2}, 0, \nicefrac{1}{2}, \dots) + (\nicefrac{1}{2}, 0, \nicefrac{1}{2}, 0, \dots)) =\\&= L((\nicefrac{1}{2}, \nicefrac{1}{2}, \nicefrac{1}{2}, \nicefrac{1}{2}, \dots)) = f((\nicefrac{1}{2}, \nicefrac{1}{2}, \nicefrac{1}{2}, \nicefrac{1}{2}, \dots)) =\\&= \lim_{n \to \infty} f_n((\nicefrac{1}{2}, \nicefrac{1}{2}, \nicefrac{1}{2}, \nicefrac{1}{2}, \dots))
        = \lim_{n \to \infty} \dfrac{1}{n} \sum_{k = 1}^n \frac{1}{2} = \lim_{n \to \infty} \dfrac{1}{n} \cdot n \cdot \frac{1}{2} = \frac{1}{2}
    \end{align*}
    Por tanto, $L(x) = \nicefrac{1}{4}$.
    
    Ahora consideramos la sucesión $y = (1, 0, 0, 1, 0, 0, 1, \dots)$. Aplicando $\tau$, obtenemos:
    \begin{align*}
        \tau(y) &= (0, 0, 1, 0, 0, 1, 0, \dots)\\
        \tau^2(y) &= (0, 1, 0, 0, 1, 0, 0, \dots)
    \end{align*}

    Por tanto:
    \begin{align*}
        3L(y) &= L(y) + L(\tau(y)) + L(\tau^2(y)) = L(y + \tau(y) + \tau^2(y)) = L((1, 1, 1, 1, 1, 1, \dots)) =\\&= f((1, 1, 1, 1, 1, 1, \dots)) = \lim_{n \to \infty} f_n((1, 1, 1, 1, 1, 1, \dots)) =\\&= \lim_{n \to \infty} \dfrac{1}{n} \sum_{k = 1}^n 1 = \lim_{n \to \infty} \dfrac{1}{n} \cdot n \cdot 1 = 1
    \end{align*}
    Por tanto, $L(y) = \nicefrac{1}{3}$.
\end{enumerate}
\end{ejercicio}

\begin{ejercicio}
    Sean $A$ y $B$ subconjuntos no vacíos, abiertos, convexos y disjuntos de un espacio normado $X$. Probar que existen $f \in X^*$ con $\|f\| = 1$ y $\alpha \in \bb{R}$ tales que $\Re\,f(a) < \alpha < \Re\,f(b)$ para todo $a \in A$ y $b \in B$. ¿Es siempre posible encontrar $f$ tal que $\sup \Re\,f(A) < \inf \Re\,f(B)$?\\

    Por el Teorema de Separación en espacios normados, puesto que $A$ es abierto, existen $g \in X^*\setminus\{0\}$ y $\beta \in \bb{R}$ tales que:
    \begin{equation*}
        \Re\,g(a) < \beta \leq \Re\,g(b) \quad \forall a \in A, \forall b \in B
    \end{equation*}

    Consideremos ahora $f = \nicefrac{g}{\|g\|}$. Entonces, $\|f\| = 1$ y se tiene que:
    \begin{equation*}
        \Re\,f(a) = \Re\,\left(\dfrac{g(a)}{\|g\|}\right) = \dfrac{\Re\,g(a)}{\|g\|} < \dfrac{\beta}{\|g\|} = \alpha \leq \dfrac{\Re\,g(b)}{\|g\|} = \Re\,f(b) \quad \forall a \in A, \forall b \in B
    \end{equation*}

    Solo resta probar que $\alpha < \Re f(b)$ para todo $b \in B$.
    \begin{description}
        \item[Opción 1:] Supongamos, por reducción al absurdo, que existe algún $b_0 \in B$ tal que $\Re f(b_0) = \alpha$. Como $B$ es abierto, existe $r > 0$ tal que $B(b_0, r) \subseteq B$. Además, como $\|f\| = 1 \neq 0$, existe un vector $x_0 \in X$ con $\Re f(x_0) > 0$. Tomando $t > 0$ suficientemente pequeño para que $t\|x_0\| < r$, el punto $b_0 - t x_0$ pertenece a $B(b_0, r) \subseteq B$ (pues $\|(b_0 - t x_0) - b_0\| = t\|x_0\| < r$), pero verifica:
        \[
            \Re f(b_0 - t x_0) = \Re f(b_0) - t\,\Re f(x_0) = \alpha - t\,\Re f(x_0) < \alpha,
        \]
        lo cual contradice que $\alpha \leqslant \Re f(b)$ para todo $b \in B$. La contradicción surge de suponer que existe $b_0 \in B$ tal que $\Re f(b_0) = \alpha$. Por tanto, $\alpha < \Re f(b)$ para todo $b \in B$.

        \item[Opción 2:] Supongamos (ahora lo demostraremos) que $\Re f$ es una aplicación abierta. Entonces, como $B$ es abierto, $\Re f(B)$ es un subconjunto abierto de $\bb{R}$. Supongamos que $\exists b_0 \in B$ tal que $\Re f(b_0) = \alpha$; es decir, $\alpha \in \Re f(B)$. Como $\Re f(B)$ es abierto, existe $\delta > 0$ tal que:
        \begin{equation*}
            \left]\alpha - \delta, \alpha + \delta\right[ \subseteq \Re f(B)
        \end{equation*}

        En particular, $\alpha - \delta \in \Re f(B)$, y por tanto $\exists b_1 \in B$ tal que $\Re f(b_1) = \alpha - \delta < \alpha$, lo cual contradice la desigualdad $\alpha \leq \Re f(b)$ para todo $b \in B$. La contradicción surge de suponer que existe $b_0 \in B$ tal que $\Re f(b_0) = \alpha$. Por tanto, $\alpha < \Re f(b)$ para todo $b \in B$.\\

        Veamos ahora que $u := \Re f : X \to \bb{R}$ es una aplicación abierta. Observemos en primer lugar que $u$ es $\bb{R}$-lineal, continua y no nula: en efecto, si $u(x) = 0$ para todo $x \in X$, entonces (tanto si $\bb{K} = \bb{R}$ como si $\bb{K} = \bb{C}$, en cuyo caso $f(x) = u(x) - i\,u(ix)$) se tendría $f = 0$, lo que contradice $\|f\| = 1$.
        
        Veamos ahora que $u$ es sobreyectiva. Como $u$ es continua y no nula, se tiene que $Im(u)$ es un subespacio vectorial real de $\bb{R}$ distinto de $\{0\}$. Por tanto, $Im(u) = \bb{R}$, y en consecuencia $u$ es sobreyectiva. Por el Ejercicio~\ref{ej:operador_lineal_abierto}, $u=\Re f$ es una aplicación abierta.
    \end{description}


    Por tanto, $\alpha < \Re f(b)$ para todo $b \in B$, y concluimos que:
    \[
        \Re f(a) < \alpha < \Re f(b) \quad \forall a \in A, \; \forall b \in B.
    \]


    Veamos ahora que no siempre es posible encontrar $f \in X^*$ tal que $\sup \Re f(A) < \inf \Re f(B)$. Consideremos el espacio normado $X = \bb{R}^2$ con la norma euclídea, y los conjuntos:
    \begin{align*}
        A &= \bb{R}^- \times \bb{R} = \{(x, y) \in \bb{R}^2 : x < 0\}, \\
        B &= \bb{R}^+ \times \bb{R} = \{(x, y) \in \bb{R}^2 : x > 0\}.
    \end{align*}

    Estos conjuntos son no vacíos, abiertos, convexos y disjuntos.
    \begin{description}
        \item[Opción 1.] Supongamos que existe un funcional $f \in X^*$ tal que se tenga $\sup \Re f(A) < \inf \Re f(B)$. En particular, se tiene que (usando que $\Re f = f$ por ser el cuerpo escalar $\bb{R}$):
        \begin{equation*}
            f(a) < \alpha < f(b) \quad \forall a \in A, \forall b \in B
        \end{equation*}
        para algún $\alpha \in \bb{R}$. Como $X = \bb{R}^2$ tiene por base canónica a $\{e_1 = (1,0),\, e_2 = (0,1)\}$, por la linealidad de $f \in X^*$ todo punto $(x,y) \in \bb{R}^2$ verifica:
        \[
            f(x, y) = f(x e_1 + y e_2) = x\,f(e_1) + y\,f(e_2) = c_1 x + c_2 y,
        \]
        donde $c_1 := f(1, 0) \in \bb{R}$ y $c_2 := f(0, 1) \in \bb{R}$. Como $f$ separa estrictamente punto a punto a $A$ y $B$, veamos que se necesita $c_2 = 0$ y $c_1 > 0$.
        \begin{enumerate}
            \item \textbf{Anulación de la segunda componente ($c_2 = 0$):} Para todo $y \in \bb{R}$ y todo $\varepsilon > 0$, los puntos $(-\varepsilon, y) \in A$ y $(\varepsilon, y) \in B$ satisfacen:
            \[
                -\varepsilon c_1 + y c_2 < \alpha < \varepsilon c_1 + y c_2.
            \]
            Haciendo tender $\varepsilon \to 0^+$, obtenemos:
            \[
                y c_2 \leq \alpha \leq y c_2 \implies y c_2 = \alpha \quad \forall y \in \bb{R}.
            \]
            En particular, evaluando en $y = 0$ se tiene $\alpha = 0$, y evaluando en $y = 1$ se deduce que $c_2 = 0$.

            \item \textbf{Positividad de la primera componente ($c_1 > 0$):} Con $c_2 = 0$ y $\alpha = 0$, el funcional se reduce a $f(x, y) = c_1 x$. Evaluando en $(1, 0) \in B$, la condición $f(1, 0) > \alpha = 0$ implica directamente que $c_1 > 0$.
        \end{enumerate}

        En consecuencia, todo funcional que separe estrictamente punto a punto a $A$ y $B$ debe ser de la forma $f(x, y) = c_1 x$ con $c_1 > 0$. Si además exigimos $\|f\| = 1$, entonces $c_1 = 1$, es decir, $f$ es únicamente el funcional primera coordenada $\pi_1(x, y) = x$.

        Sin embargo:
        \begin{align*}
            \sup f(A) &= \sup \{ x : x < 0\} = 0, \\
            \inf f(B) &= \inf \{ x : x > 0\} = 0.
        \end{align*}
        Por tanto, $\sup f(A) = 0 = \inf f(B)$, lo que contradice $\sup \Re f(A) < \inf \Re f(B)$.
        \item[Opción 2.] Observemos que sus adherencias son:
        \[
            \overline{A} = \left]-\infty, 0\right] \times \bb{R} \quad \text{y} \quad \overline{B} = \left[0, +\infty\right[ \times \bb{R},
        \]
        por lo que $\overline{A} \cap \overline{B} = \{0\} \times \bb{R} \neq \emptyset$. Supongamos, por reducción al absurdo, que existe $f \in X^*$ tal que:
        \[
            \sup \Re f(A) < \inf \Re f(B).
        \]
        Como $\Re f : X \to \bb{R}$ es continua, se cumple que $\Re f(\overline{A}) \subseteq \overline{\Re f(A)}$ y, además, $\Re f(\overline{B}) \subseteq \overline{\Re f(B)}$. En particular, tomando cualquier punto $z \in \overline{A} \cap \overline{B}$ (por ejemplo, $z = (0, 0)$), se tiene que:
        \[
            \Re f(z) \leq \sup \Re f(A) < \inf \Re f(B) \leq \Re f(z),
        \]
        lo cual es una contradicción. 
    \end{description}
    
    
    En consecuencia, no existe ningún $f \in X^*$ que verifique $\sup \Re f(A) < \inf \Re f(B)$.
\end{ejercicio}
