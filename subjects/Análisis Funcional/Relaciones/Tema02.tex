\section{Espacios de Hilbert}

\begin{ejercicio}
Sean $\{x_n\}_{n \in \bb{N}}$ e $\{y_n\}_{n \in \bb{N}}$ sucesiones en la bola unidad de un espacio prehilbertiano. Probar que si $\|x_n + y_n\| \longrightarrow 2$ entonces $\|x_n - y_n\| \longrightarrow 0$.
\end{ejercicio}

\begin{ejercicio}
Sea $\{x_n\}_{n \in \bb{N}}$ una sucesión en un espacio prehilbertiano $X$ y $x \in X$ tal que $\{(x_n \mid x)\} \longrightarrow \|x\|^2$ y $\{\|x_n\|\} \longrightarrow \|x\|$. Probar que $\{x_n\} \longrightarrow x$.
\end{ejercicio}

\begin{ejercicio}
Sea $X$ un espacio prehilbertiano real. Demostrar que para $x, y \in X$ equivalen:
\begin{enumerate}[label=\textbf{\alph*)}]
    \item $x \perp y$.
    \item $\|x + ty\| = \|x - ty\|$ para todo $t \in \bb{R}$.
    \item $\|x + ty\| \geq \|x\|$ para todo $t \in \bb{R}$.
\end{enumerate}
\end{ejercicio}

\begin{ejercicio}
Calcular la proyección ortogonal en $\ell_2$ sobre el subespacio $M$ generado por los vectores $v_1 = (1, 1, 0, 0, \dots)$, $v_2 = (1, 0, 1, 0, \dots)$ y $v_3 = (1, 0, 0, 1, 0, 0, \dots)$.
\end{ejercicio}

\begin{ejercicio}
Calcular el mínimo valor de la integral
\[
\int_{-1}^1 |\sen(\pi t) - a - bt|^2\,dt
\]
para $a, b \in \bb{C}$.
\end{ejercicio}

\begin{ejercicio}
Calcular el mínimo valor de
\[
\int_{-1}^1 |x^3 - a - bx - cx^2|^2\,dx
\]
siendo $a, b, c$ números reales.
\end{ejercicio}

\begin{ejercicio}
Sea $M = \{x \in \ell_2 : x(1) = x(2) = x(3)\}$. Probar que $M$ es el complemento ortogonal del subespacio de $\ell_2$ generado por los vectores $u = (1, -1, 0, 0, \dots)$ y $v = (0, 1, -1, 0, \dots)$. Obtener la descomposición de cualquier vector de $\ell_2$ como suma de un elemento de $M$ y otro de $M^\perp$.
\end{ejercicio}
