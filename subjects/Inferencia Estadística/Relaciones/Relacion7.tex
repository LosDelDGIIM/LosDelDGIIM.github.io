\begin{ejercicio}
    Se toma una observación de una variable con distribución de Poisson para contrastar que la media vale 1 frente a que vale 2.
    \begin{enumerate}[label=\alph*)]
        \item Construir un test no aleatorizado con nivel de significación $0.05$ para el contraste planteado. Calcular las probabilidades de cometer error de tipo 1 y de tipo 2, el tamaño y la potencia del test frente a la hipótesis alternativa.
        \item ¿Cómo debe aleatorizarse el test para alcanzar el tamaño $0.05$? ¿Cuál es la potencia de este test?
    \end{enumerate}
    
    \textbf{Solución}
    \begin{enumerate}[label=\alph*)]
        \item Sea $X\rightsquigarrow \cc{P}(\lm)$, queremos resolver el contraste:
            \begin{equation*}
                \left\{\begin{array}{l}
                    H_0 : \lm = 1 \\
                    H_1 : \lm = 2
                \end{array}\right.
            \end{equation*}
            Trabajaremos por tanto con:
            \begin{equation*}
                \Theta_0 = \{1\}, \qquad \Theta_1 = \{2\}, \qquad \Theta = \{1,2\}
            \end{equation*}
            y escribiremos $\lm_0 = 1$, $\lm_1 = 2$. Pensamos la forma que ha de tener el test. Puesto que $E_\lm[X]=\lm$, haciendo uso de que el espacio muestral de $X$ es $\mathbb{R}^+_0$:
            \begin{figure}[H]
                \centering
                \begin{tikzpicture}
                % Dibuja la recta real
                \draw[-Stealth] (-1,0) -- (3,0) node[right] {};

                % Marca los puntos 1 y 2
                \foreach \x in {0,1,2} {
                    \draw (\x,0.1) -- (\x,-0.1) node[below] {$\x$};
                }
                \end{tikzpicture}
            \end{figure}
            Es claro que cuanto más a la izquierda nos encontremos con dicha observación $X$, no rechazaremos la hipótesis nula, así como que cuanto más a la derecha estemos la rechazaremos. Por tanto, establecemos un cierto punto $c\in \mathbb{R}^+_0$ que delimite la región crítica del test (que será la región $\left]c,+\infty\right[$). Para determinar dicho punto $c$ imponemos un nivel de significación $\alpha = 0.05$. Por tanto, el test será de la forma:
            \begin{equation*}
                \varphi(X) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X> c \\
                    0 & \text{si\ } X \leq c
                \end{array}\right. 
            \end{equation*}
            Imponemos nivel de significación $\alpha$:
            \begin{equation*}
                \alpha \geq \sup_{\lm \in \Theta_0}\beta_\varphi(\lm) = \sup_{\lm\in \{\lm_0\}} \beta_\varphi(\lm) = \beta_\varphi(\lm_0) = E_{\lm_0}[\varphi]  = P_{\lm_0}[X>c]  = 1-P_{\lm_0}[X\leq c]
            \end{equation*}

            Haciendo uso de que $X\rightsquigarrow \cc{P}(\lm)$, $\alpha = 0.05$ y $\lm_0 = 1$, tenemos que:
            \begin{itemize}
                \item Para $c=0$:
                    \begin{equation*}
                        P_1[X\leq 0] = P_1[X = 0] = 0.3679 \quad\Longrightarrow\quad 1-P_1[X\leq 0] = 0.6321 > \alpha
                    \end{equation*}
                \item Para $c=1$:
                    \begin{align*}
                        P_1[X\leq 1] &= P_1[X=0] + P_1[X=1] = 0.3679 + 0.3679 = 0.7358  \\
                                     &\Longrightarrow 1-P_1[X\leq 1] = 0.2642 > \alpha
                    \end{align*}
                \item Para $c=2$:
                    \begin{align*}
                        P_1[X\leq 2] &= P_1[X\leq 1] + P_1[X=2] = 0.7358 + 0.1839 = 0.9197 \\
                                     &\Longrightarrow 1-P_1[X\leq 2] = 0.0803 > \alpha
                    \end{align*}
                \item Para $c=3$:
                    \begin{align*}
                        P_1[X\leq 3] &= P_1[X\leq 2] + P_1[X=3] = 0.9197 + 0.0613 = 0.981 \\
                                     &\Longrightarrow 1-P_1[X\leq 3] = 0.019 \leq \alpha
                    \end{align*}
            \end{itemize}

            Como $1-P_1[X\leq c]$ es decreciente en $c$, cualquier valor de $c\geq 3$ es válido para imponer un tamaño menor o igual que $\alpha$. Calculamos la potencia del test, pues nos quedaremos con el valor de $c$ que maximice la potencia del test en la hipótesis alternativa, que es:
            \begin{equation*}
                \beta_\varphi(\lm_1) = E_{\lm_1}[\varphi] = P_{\lm_1}[X>c] = 1-P_{\lm_1}[X\leq c]
            \end{equation*}

            Como $1-P_{\lm_1}[X\leq c]$ es decreciente en $c$, para maximizar la potencia del test debemos tomar el menor valor de $c$ posible, que es $c=3$. Por tanto, $c=3$. El test por tanto es:
            \begin{equation*}
                \varphi(X) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X> 3 \\
                    0 & \text{si\ } X \leq 3
                \end{array}\right. 
            \end{equation*}
            Calculamos ahora los valores pedidos.
            \begin{description}
                \item [Probabilidad de cometer error tipo 1.] 
                    \begin{equation*}
                        P_{\lm\in \Theta_0}[\varphi(X)] = P_{\lm_0}[X>3] = P_1[X>3] = 1-P_1[X\leq 3] = 0.019
                    \end{equation*}
                \item [Probabilidad de cometer error tipo 2.]
                    \begin{align*}
                        P_{\lm \in \Theta_1}\left[\varphi(X)=0\right] &= P_{\lm_1}[X\leq 3] = P_2[X\leq 3] = \sum_{k=0}^{3} P_2[X=k] \\
                                                                           &= (0.1353 + 0.2707 + 0.2707 + 0.1804) = 0.8571
                    \end{align*}
                \item [Tamaño del test.]
                    \begin{equation*}
                        \sup_{\lm \in \Theta_0}\beta_\varphi(\lm) = \beta_\varphi(\lm_0) = E_{\lm_0}[\varphi] = P_{\lm_0}[X>3] = 0.019
                    \end{equation*}
                \item [Potencia frente a hipótesis alternativa.] La potencia del test es:
                    \begin{align*}
                        \beta_\varphi(\lm_1) &= E_{\lm_1}[\varphi] = P_{\lm_1}[X>3] = P_2[X>3] = 1-P_2[X\leq 3] \\
                                                                   &= 1- 0.8571 = 0.1429
                    \end{align*}
            \end{description}
        \item Para alcanzar un tamaño de $0.05$ mediante un test aleatoriezado, lo que hacemos ahora es considerar un test del tipo:
            \begin{equation*}
                \varphi(X) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X>3 \\
                    \gamma & \text{si\ } X=3 \\
                    0 & \text{si\ } X<3 
                \end{array}\right. 
            \end{equation*}
            Para cierto $\gamma\in [0,1]$, de forma que a la hora de imponer un tamaño menor o igual que $\alpha=0.05$ calculemos $\gamma$ para obtener la igualdad. De esta forma, el tamaño del test es:
            \begin{equation*}
                \sup_{\lm \in \Theta_0}\beta_\varphi(\lm) = \beta_\varphi(\lm_0) = E_{\lm_0}[\varphi] = P_{\lm_0}[X>3] + \gamma P_{\lm_0}[X=3] = 0.019 + \gamma 0.0613
            \end{equation*}
            Si imponemos que sea igual a $\alpha$:
            \begin{equation*}
                0.05 = \alpha = 0.019 + \gamma \cdot 0.0613 \quad\Longrightarrow\quad \gamma = \frac{0.05 - 0.019}{0.0613} \approx 0.5057096
            \end{equation*}
            Calculamos la potencia del test:
            \begin{equation*}
                \sup_{\lm \in \Theta_1}\beta_\varphi(\lm) = E_{\lm_1}[\varphi] = P_{\lm_1}[X>3] + \gamma P_{\lm_1}[X=3] = 0.1429 + \gamma 0.1804 \approx 0.23413
            \end{equation*}
    \end{enumerate}
\end{ejercicio}

\begin{ejercicio}
    Una urna contiene 10 bolas, blancas y negras. Para contrastar que el número de bolas blancas es 5 frente a que dicho número es 6 o 7, se extraen tres bolas con reemplazamiento y se rechaza $H_0$ sólo si se obtienen 2 o 3 bolas blancas. Calcular el tamaño de este test y la potencia frente a las alternativas.\\

    \noindent 
    Sea $X$ una variable aleatoria que representa si se obtiene una bola blanca en una extracción, por lo que:
    \begin{equation*}
        X\equiv \text{``Se obtiene una bola blanca en una extracción''} \rightsquigarrow B(1,p)
    \end{equation*}
    Se quiere resolver el contraste de hipótesis:
    \begin{equation*}
        \begin{cases}
            H_0 :& p = p_0 =  \nicefrac{5}{10} \\
            H_1 :& p \in \{\nicefrac{6}{10}, \nicefrac{7}{10}\}
        \end{cases}
    \end{equation*}
    Por lo que consideraremos:
    \begin{equation*}
        \Theta_0 = \{\nicefrac{5}{10}\}, \qquad \Theta_1 = \{\nicefrac{6}{10}, \nicefrac{7}{10}\}, \qquad \Theta = \{\nicefrac{5}{10}, \nicefrac{6}{10}, \nicefrac{7}{10}\}
    \end{equation*}
    Y tenemos $(X_1, X_2, X_3)$ una m.a.s. de $X$. El test que se propone es:
    \begin{equation*}
        \varphi(X_1, X_2, X_3) = \left\{\begin{array}{ll}
            1 & \text{si\ } X_1+X_2+X_3 \in \{2,3\} \\
            0 & \text{en otro caso\ } 
        \end{array}\right. 
    \end{equation*}
    Por la reproductividad de la binomial tenemos que $X_1+X_2+X_3\rightsquigarrow B(3,p)$. Calculamos: 
    \begin{description}
        \item [Tamaño del test.] 
            \begin{align*}
                \sup_{p\in \Theta_0} \beta_\varphi(p) &= \beta_\varphi(p_0) = E_{p_0}[\varphi] = P_{p_0}[X_1+X_2+X_3 = 2] + P_{p_0}[X_1 + X_2 + X_3 = 3] \\
                                                      &= P_{\nicefrac{1}{2}}[X_1+X_2+X_3 = 2] + P_{\nicefrac{1}{2}}[X_1 + X_2 + X_3 = 3]  \\
                                                      &= 0.375 + 0.125 = 0.5
            \end{align*}
        \item [Potencia frente a hipótesis alternativas.] 
            Calculamos cada una de ellas:
            \begin{itemize}
                \item Para $p = \nicefrac{6}{10}$:
                    \begin{align*}
                        \beta_\varphi(p) &= E_p[\varphi] = P_p[X_1 + X_2 + X_3 = 2] + P_p[X_1+X_2+X_3 = 3] \\
                                         &= P_{0.6}[X_1 + X_2 + X_3 =2] + P_{0.6}[X_1+X_2+X_3 =3]
                    \end{align*}

                    Puesto que $X_1+X_2+X_3\rightsquigarrow B(3,p)$ pero la Tabla de la binomial sólo nos da valores para $p\leq 0.5$, hacemos uso de la simetría de la binomial, y consideramos $Y=n-(X_1+X_2+X_3)\rightsquigarrow B(3,1-p)$, de forma que:
                    \begin{align*}
                        \beta_\varphi(p) &= P_{0.6}[X_1 + X_2 + X_3 =2] + P_{0.6}[X_1+X_2+X_3 =3] \\
                                         &= P[Y=3-2] + P[Y=3-3] = P[Y=1] + P[Y=0] \\
                                         &= 0.432 + 0.216 = 0.648
                    \end{align*}
                \item para $p=\nicefrac{7}{10}$:
                    \begin{align*}
                        \beta_\varphi(p) &= P_p[X_1 + X_2 + X_3 = 2] + P_p[X_1+X_2+X_3 = 3] \\
                                         &= P_{0.7}[X_1 + X_2 + X_3 =2] + P_{0.7}[X_1+X_2+X_3 =3]
                    \end{align*}
                    Si consideramos $Y\rightsquigarrow B(3,1-p) = B(3,\ 0.3)$ tenemos:
                    \begin{align*}
                        \beta_\varphi(p) &= P_{0.7}[X_1 + X_2 + X_3 =2] + P_{0.7}[X_1+X_2+X_3 =3] \\
                                         &= P[Y=1] + P[Y=0] = 0.441 + 0.343 = 0.784
                    \end{align*}
            \end{itemize}
    \end{description}
\end{ejercicio}

\begin{ejercicio}
    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de una variable aleatoria con distribución de Poisson de parámetro $\lm$. Encontrar el test más potente de tamaño $\alpha$ para resolver el problema de contraste
    \begin{equation*}
        \begin{cases}
            H_0 :& \lm = \lm_0 \\
            H_1 :& \lm = \lm_1
        \end{cases}
    \end{equation*}
    \textit{Aplicación:} En una centralita telefónica el número de llamadas por minuto sigue una distribución de Poisson. Si en cinco minutos se han recibido 12 llamadas, ¿puede aceptarse que el número medio de llamadas por minuto es $1.5$, frente a que dicho número es 2, al nivel de significación $0.05$? Calcular la potencia del test obtenido.\\

    \noindent
    Sea $X\rightsquigarrow \cc{P}(\lm)$, en este caso tenemos:
    \begin{equation*}
        \Theta_0 = \{\lm_0\}, \qquad \Theta_1 = \{\lm_1\}, \qquad \Theta = \{\lm_0,\lm_1\}
    \end{equation*}

    Consideramos la función de distribución de la muestra $(X_1, \ldots, X_n)$. Para $x_1, \ldots, x_n \in \mathbb{N}_0$:
    \begin{equation*}
        P_{\lm}[X_1 = x_1, \ldots, X_n = x_n] \stackrel{\text{iid.}}{=} \prod_{i=1}^{n} P_{\lm}[X=x_i] = \prod_{i=1}^{n} e^{-\lm} \frac{\lm^{x_i}}{x_i!} = e^{-n\lm} \cdot \frac{\lm^{\sum\limits_{k=1}^{n}x_i}}{\prod\limits_{i=1}^{n}x_i!} > 0
    \end{equation*}
    Notemos que, si alguno de los $x_i$ es negativo, la probabilidad de que se dé esa observación es nula. Notaremos:
    \begin{align*}
        f_1^n(x_1, \ldots, x_n) &= P_{\lm_1}[X_1=x_1, \ldots, X_n=x_n]\\
        f_0^n(x_1, \ldots, x_n) &= P_{\lm_0}[X_1=x_1, \ldots, X_n=x_n]
    \end{align*}
    
    Como nos encontramos ante un contraste simple frente a simple y queremos hayar el test más potente de tamaño $\alpha$, recurrimos a un test de Neymann-Pearson, que será de la forma:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } f_1^n(X_1, \ldots, X_n) > k f_0^n(X_1, \ldots, X_n) \\
             \gamma & \text{si\ } f_1^n(X_1, \ldots, X_n) = k f_0^n(X_1, \ldots, X_n) \\
             0 & \text{si\ } f_1^n(X_1, \ldots, X_n) < k f_0^n(X_1, \ldots, X_n)
        \end{array}\right.
    \end{equation*}
    con $k\in \mathbb{R}^+_0$ y $\gamma\in [0,1]$ que nos disponemos a calcular. Veamos cuándo se anula cada una de las funciones de distribución de la muestra:
    \begin{itemize}
        \item $f_1^n(x_1, \ldots, x_n) = 0 \Longleftrightarrow \exists i\in \{1, \ldots, n\} : x_i < 0$
        \item $f_0^n(x_1, \ldots, x_n) = 0 \Longleftrightarrow \exists i\in \{1, \ldots, n\} : x_i < 0$
    \end{itemize}

    Por tanto, $f_0^n$ se anula si y solo si $f_1^n$ se anula, y en ese caso el valor de $\varphi(X_1, \ldots, X_n)$ no importa, pues la probabilidad de que se dé esa observación es nula. Como esto sólo ocurre si alguno de los $X_i$ es negativo, fijamos:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = 0 \quad\text{si\ } \exists i\in \{1, \ldots, n\} : X_i < 0
    \end{equation*}

    En adelante, suponemos que $X_i \geq 0$ para todo $i\in \{1, \ldots, n\}$, por lo que $f_0(X_1, \ldots, X_n) > 0$. Por tanto, podemos reescribir el test como:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } \lm(X_1, \ldots, X_n) > k \\
             \gamma & \text{si\ } \lm(X_1, \ldots, X_n) = k \\
             0 & \text{si\ } \lm(X_1, \ldots, X_n) < k
        \end{array}\right. 
    \end{equation*}
    donde:
    \begin{align*}
        \lm(x_1,\ldots, x_n) &= \frac{P_{\lm_1}[X_1=x_1,\ldots, X_n=x_n]}{P_{\lm_0}[X_1=x_1,\ldots, X_n=x_n]} = \frac{e^{-n\lm_1} \cdot \frac{\lm_1^{\sum\limits_{i=1}^{n}x_i}}{\prod\limits_{i=1}^{n}x_i!} }{e^{-n\lm_0} \cdot \frac{\lm_0^{\sum\limits_{i=1}^{n}x_i}}{\prod\limits_{i=1}^{n}x_i!} } = e^{n(\lm_0-\lm_1)} {\left(\frac{\lm_1}{\lm_0}\right)}^{\sum\limits_{i=1}^{n}x_i}
    \end{align*}

    Puesto que desconocemos la distribución de $\lm$, no es directo calcular el valor de $k$ que nos imponga un tamaño $\alpha$. Para ello, puesto que $\lambda$ solo depende de la muestra a través de la suma de las observaciones, consideramos el estadístico:
    \begin{equation*}
        T(X_1, \ldots, X_n) = \sum\limits_{i=1}^{n}X_i
    \end{equation*}

    Por la reproductividad de la Poisson, tenemos que $T(X_1, \ldots, X_n)\rightsquigarrow \cc{P}(n\lm)$. De esta forma, tenemos $\lambda=g(T)$, con:
    \begin{equation*}
        g(t) = e^{n(\lm_0-\lm_1)}{\left(\frac{\lm_1}{\lm_0}\right)}^{t}
    \end{equation*}

    Estudiamos la monotonía de $g$. Como $\lm_1,\lm_0>0$, tenemos que $g(t)>0$ para todo $t\in \mathbb{R}^+_0$. Como el logaritmo es estrictamente creciente, estudiamos la monotonía de $\ln g$:
    \begin{align*}
        \ln g(t) &= n(\lm_0-\lm_1) + t\ln\left(\frac{\lm_1}{\lm_0}\right) \\
        \frac{d}{dt}\ln g(t) &= \ln\left(\frac{\lm_1}{\lm_0}\right)>0\iff \frac{\lm_1}{\lm_0}>1 \iff \lm_1>\lm_0
    \end{align*}

    Por tanto, sabiendo que $\lm_0\neq \lm_1$, distinguimos dos casos:
    \begin{description}
        \item [Supuesto que $\lm_0<\lm_1$.] Tenemos entonces que $g$ es estrictamente creciente. Por tanto, para cada $k\in \mathbb{R}^+_0$ existe un único $c=g^{-1}(k)\in \mathbb{R}^+_0$ tal que $g(c)=k$.

            Tal y como se observa en la Figura \ref{fig:lambda_T_relation_ej7.3.1}, tenemos que:
            \begin{equation*}
                \begin{array}{l}
                    \lm > k \iff T > c \\
                    \lm < k \iff T < c \\
                    \lm = k \iff T = c
                \end{array}
            \end{equation*}
            Por tanto, podemos reescribir el test como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } \sum\limits_{i=1}^{n} X_i > c \\
                     \gamma& \text{si\ } \sum\limits_{i=1}^{n}X_i = c \\
                     0& \text{si\ } \sum\limits_{i=1}^{n}X_i < c
                \end{array}\right. 
            \end{equation*}
            Imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \sup_{\lm \in \Theta_0}\beta_\varphi(\lm) = \beta_\varphi(\lm_0) = E_{\lm_0}[\varphi] = P_{\lm_0}[T > c] + \gamma P_{\lm_0}[T=c]
            \end{equation*}
            Para cierto $\alpha$ calcularemos los valores de $c$ y $\gamma$.
        \item [Supuesto que $\lm_1 < \lm_0$.] En este caso tendremos que $g$ es estrictamente decreciente, y de forma análoga al caso anterior, para cada $k\in \mathbb{R}^+_0$ existe un único $c=g^{-1}(k)\in \mathbb{R}^+_0$ tal que $g(c)=k$.
            \begin{figure}[h]
                \centering
                \begin{tikzpicture}[>=Stealth, scale=1.15]

                % --- Definición de estilos ---
                \tikzset{

                    reg_sup/.style={ultra thick, orange!90!black},
                    reg_inf/.style={ultra thick, red!75!black},

                }

                \draw[eje] (-0.4, 0) -- (5.5, 0)
                    node[right] {$T = \sum\limits_{i=1}^n X_i$};

            \begin{equation*}
                \begin{array}{l}
                    \lm > k \iff T < c \\
                    \lm < k \iff T > c \\
                    \lm = k \iff T = c
                \end{array}
            \end{equation*}

            Por tanto, podemos reescribir el test como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } \sum\limits_{i=1}^{n}X_i<c \\
                     \gamma& \text{si\ } \sum\limits_{i=1}^{n}X_i = c \\
                     0 & \text{si\ } \sum\limits_{i=1}^{n}X_i > c
                \end{array}\right. 
            \end{equation*}
            Imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \sup_{\lm \in \Theta_0}\beta_\varphi(\lm) = \beta_\varphi(\lm_0) = E_{\lm_0}[\varphi] = P_{\lm_0}[T < c] + \gamma P_{\lm_0}[T=c]
            \end{equation*}
            
            Los valores de $c$ y $\gamma$ se calcularán para cierto $\alpha$.
    \end{description}~\\

    \noindent
    \textit{Aplicación.} Sea ahora $X$ la variable aleatoria que representa el número de llamadas por minuto, por lo que:
    \begin{equation*}
        X \equiv \text{``Número de llamadas por minuto''}\rightsquigarrow\cc{P}(\lm)
    \end{equation*}
    En 5 observaciones $(x_1, \ldots, x_5)$ se ha obtenido:
    \begin{equation*}
        \sum_{i=1}^{n} x_i = 12
    \end{equation*}
    Y planteamos el contraste de hipótesis:
    \begin{equation*}
        \left\{\begin{array}{l}
            H_0 : \lm = \lm_0 = 1.5 \\
            H_1 : \lm = \lm_1 = 2
        \end{array}\right.
    \end{equation*}
    Sea $\alpha = 0.05$, como $\lm_0<\lm_1$ nos encontramos en el primer caso, por lo que usamos el test:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } \sum\limits_{i=1}^{n}X_i>c \\
            \gamma& \text{si\ } \sum\limits_{i=1}^{n}X_i = c \\
            0 & \text{si\ } \sum\limits_{i=1}^{n}X_i < c
        \end{array}\right. 
    \end{equation*}
    Imponiendo tamaño $\alpha$:
    \begin{equation*}
        0.05 = \alpha = \sup_{\lm \in \Theta_0}\beta_\varphi(\lm) = P_{\lm_0}[T > c] + \gamma P_{\lm_0}[T=c] = P_{1.5}[T>c] + \gamma P_{1.5}[T=c]
    \end{equation*}
    Recordamos que $T\rightsquigarrow \cc{P}(n\lm)$, y como nos encontramos bajo hipótesis nula, tenemos que $T\rightsquigarrow \cc{P}(5\cdot 1.5) \equiv \cc{P}(7.5)$. Buscamos en primer lugar los valores de $c$ de forma que se tenga $P[T>c]\leq \alpha$:
    \begin{itemize}
        \item Para $c=13$, se tiene:
            \begin{equation*}
                P[T>c] = P[T\geq 14] = 0.0216 \leq 0.05 = \alpha
            \end{equation*}
        \item Para $c=12$, se tiene:
            \begin{equation*}
                P[T>c] = P[T\geq 13] = 0.0427 \leq 0.05 = \alpha
            \end{equation*}
        \item Para $c=11$ se tiene:
            \begin{equation*}
                P[T>c] = P[T\geq 12] = 0.0793 > 0.05 = \alpha
            \end{equation*}
    \end{itemize}

    Por tanto, puesto que se ha de cumplir con el tamaño del test, descartamos $c<12$. Ahora, para los valores de $c$ que cumplen con el tamaño, calculamos $\gamma$:
    \begin{equation*}
        \alpha = 0.05 = P[T>c] + \gamma P[T=c] \quad\Longrightarrow\quad \gamma = \frac{0.05-P[T>c]}{P[T=c]}
    \end{equation*}
    \begin{itemize}
        \item Para $c=13$, se tiene:
            \begin{equation*}
                \gamma = \frac{0.05-0.0216}{0.0211} \approx 1.34>1
            \end{equation*}
            Como $\gamma$ ha de estar en el intervalo $[0,1]$, descartamos este valor de $c$, y consecuentemente, descartamos también todos los valores de $c>13$.
        \item Para $c=12$, se tiene:
            \begin{equation*}
                \gamma = \frac{0.05-0.0427}{0.0366} \approx 0.199454 \in [0,1]
            \end{equation*}
    \end{itemize}

    De esta forma, el único valor de $c$ que nos permite cumplir con el tamaño del test es $c=12$. Por tanto, el test de Neymann-Pearson de tamaño $\alpha=0.05$ es:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } \sum\limits_{i=1}^{n}X_i>12 \\
             0.199454 & \text{si\ } \sum\limits_{i=1}^{n}X_i = 12 \\
             0 & \text{si\ } \sum\limits_{i=1}^{n}X_i < 12
        \end{array}\right. 
    \end{equation*}
    Como $\sum\limits_{i=1}^{5} x_i = 12$, tenemos que:
    \begin{equation*}
        \varphi(x_1, \ldots, x_5) = 0.199454
    \end{equation*}

    Por tanto, la probabilidad de rechazar la hipótesis nula, y por tanto de rechazar que el número medio de llamadas por minuto es $1.5$, es de aproximadamente $0.199454$. Por tanto, la evidencia no es suficiente para rechazar la hipótesis nula al nivel de significación $0.05$. Lo mejor sería repetir el experimento para obtener una decisión clara. Calculamos ahora la potencia del test:
    \begin{equation*}
        \sup_{\lm \in \Theta_1}\beta_\varphi(\lm) = \beta_\varphi(\lm_1) = E_{\lm_1}[\varphi] = P_{\lm_1}[T>12] + 0.199454\cdot  P_{\lm_1}[T=12]
    \end{equation*}
    Recordemos que $T\rightsquigarrow \cc{P}(5\lm_1)\equiv \cc{P}(10)$:
    \begin{equation*}
        \sup_{\lm\in \Theta_1}\beta_\varphi(\lm) = P_2[T> 12] + 0.199454 \cdot P_2[T=12] = 0.2085 + 0.199454 \cdot 0.0948 = 0.227408
    \end{equation*}
\end{ejercicio}

\begin{ejercicio}
    Dada una muestra de tamaño $n$ de una variable con distribución $\cc{N}(\mu, \sigma_0^2)$, deducir el test más potente de tamaño arbitrario para contrastar hipótesis simples sobre $\mu$.\\

    Consideramos el contraste de hipótesis:
    \begin{align*}
        \begin{cases}
            H_0: & \mu = \mu_0 \\
            H_1: & \mu = \mu_1
        \end{cases}
    \end{align*}
    
    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de la variable $X\rightsquigarrow \cc{N}(\mu, \sigma_0^2)$. Calculamos la función de densidad conjunta de la muestra:
    \begin{align*}
        f_\mu^n(x_1, \ldots, x_n) &= \prod_{i=1}^{n} f_\mu(x_i) = \prod_{i=1}^{n} \frac{1}{\sqrt{2\pi}\sigma_0} e^{-\frac{(x_i-\mu)^2}{2\sigma_0^2}} = \frac{1}{(2\pi)^{\nicefrac{n}{2}}\sigma_0^n} \exp\left(-\frac{1}{2\sigma_0^2}\sum_{i=1}^{n}(x_i-\mu)^2\right)
        =\\&= \frac{1}{(2\pi)^{\nicefrac{n}{2}}\sigma_0^n} \exp\left(-\frac{1}{2\sigma_0^2}\left[\sum_{i=1}^{n}x_i^2 - 2\mu\sum_{i=1}^{n}x_i + n\mu^2\right]\right)
    \end{align*}
    
    Notamos:
    \begin{align*}
        f_0^n(x_1, \ldots, x_n) &= f_{\mu_0}^n(x_1, \ldots, x_n) \\
        f_1^n(x_1, \ldots, x_n) &= f_{\mu_1}^n(x_1, \ldots, x_n)
    \end{align*}

    El test de Neyman-Pearson de tamaño $\alpha$ para el contraste de hipótesis anunciado será de la forma:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } f_1^n(X_1, \ldots, X_n) > k f_0^n(X_1, \ldots, X_n) \\
            \gamma & \text{si\ } f_1^n(X_1, \ldots, X_n) = k f_0^n(X_1, \ldots, X_n) \\
            0 & \text{si\ } f_1^n(X_1, \ldots, X_n) < k f_0^n(X_1, \ldots, X_n)
        \end{array}\right.
    \end{equation*}
    Para ciertas constantes $\gamma$ y $k$. Como la función de densidad conjunta de la muestra es estrictamente positiva, podemos reescribir el test como:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } \lm(X_1, \ldots, X_n) > k \\
            \gamma & \text{si\ } \lm(X_1, \ldots, X_n) = k \\
            0 & \text{si\ } \lm(X_1, \ldots, X_n) < k
        \end{array}\right.
    \end{equation*}
    donde:
    \begin{align*}
        \lm&(x_1, \ldots, x_n) = \frac{f_1^n(x_1, \ldots, x_n)}{f_0^n(x_1, \ldots, x_n)} =\\&= \exp\left(-\frac{1}{2\sigma_0^2}\left[\cancel{\sum_{i=1}^{n}x_i^2} - 2\mu_1\sum_{i=1}^{n}x_i + n\mu_1^2 - \cancel{\sum_{i=1}^{n}x_i^2} + 2\mu_0\sum_{i=1}^{n}x_i - n\mu_0^2\right]\right) =\\&= \exp\left(-\frac{1}{2\sigma_0^2}\left[- 2(\mu_1-\mu_0)\sum_{i=1}^{n}x_i + n(\mu_1^2-\mu_0^2)\right]\right)
    \end{align*}

    Puesto que no conocemos la distribución de $\lm$, no es directo calcular el valor de $k$ que nos imponga un tamaño $\alpha$. Puesto que $\lambda$ solo depende de la muestra a través de la suma de las observaciones, consideramos el estadístico:
    \begin{equation*}
        T(X_1, \ldots, X_n) = \sum\limits_{i=1}^{n} X_i
    \end{equation*}

    Por la reproductividad de la normal, tenemos que $T(X_1, \ldots, X_n)\rightsquigarrow \cc{N}(n\mu, n\sigma_0^2)$. De esta forma, tenemos $\lambda=g(T)$, con:
    \begin{equation*}
        g(t) = \exp\left(-\frac{1}{2\sigma_0^2}\left[- 2(\mu_1-\mu_0)t + n(\mu_1^2-\mu_0^2)\right]\right)
    \end{equation*}

        \ln g(t) &= -\frac{1}{2\sigma_0^2}\left[- 2(\mu_1-\mu_0)t + n(\mu_1^2-\mu_0^2)\right] \\

            \caption{Representación gráfica de la relación entre $\lambda$ y $T$.}
            \label{fig:lambda_T_relation_ej7.4.2}
        \end{figure}

        Tal y como se observa en la Figura \ref{fig:lambda_T_relation_ej7.4.2}, tenemos que:
        \begin{equation*}
            \begin{array}{l}
                \lm > k \iff T < c \\
                \lm < k \iff T > c \\
                \lm = k \iff T = c
            \end{array}
        \end{equation*}

        Por tanto, podemos reescribir el test como:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \sum\limits_{i=1}^{n}X_i<c \\
                 \gamma& \text{si\ } \sum\limits_{i=1}^{n}X_i = c \\
                    0 & \text{si\ } \sum\limits_{i=1}^{n}X_i > c
            \end{array}\right. 
        \end{equation*}

        De nuevo, como $T\rightsquigarrow \cc{N}(n\mu, n\sigma_0^2)$ es una variable continua, tenemos que $P_{\mu_0}[T=c]=0$, por lo que el valor de $\gamma$ no afecta al tamaño del test ni a su potencia. Por tanto, podemos fijar $\gamma=0$ y el test queda como:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \sum\limits_{i=1}^{n}X_i<c \\
                 0 & \text{si\ } \sum\limits_{i=1}^{n}X_i \geq c
            \end{array}\right.
        \end{equation*}

        Imponiendo tamaño $\alpha$:
        \begin{equation*}
            \alpha = \sup_{\mu \in \Theta_0}\beta_\varphi(\mu) = \beta_\varphi(\mu_0) = E_{\mu_0}[\varphi] = P_{\mu_0}[T < c] = 1 - P_{\mu_0}[T \geq c]
            \Longrightarrow
            P_{\mu_0}[T \geq c] = 1 - \alpha
        \end{equation*}

        Tipificando y considerando $Z\rightsquigarrow \cc{N}(0,1)$:
        \begin{equation*}
            1-\alpha = P_{\mu_0}[T \geq c] = P\left[\frac{T-n\mu_0}{\sqrt{n}\sigma_0} \geq \frac{c-n\mu_0}{\sqrt{n}\sigma_0}\right] = P\left[Z \geq \frac{c-n\mu_0}{\sqrt{n}\sigma_0}\right]
        \end{equation*}

        Por tanto:
        \begin{equation*}
            \frac{c-n\mu_0}{\sqrt{n}\sigma_0} = z_{1-\alpha} \quad\Longrightarrow\quad c = n\mu_0 + z_{1-\alpha}\sqrt{n}\sigma_0
        \end{equation*}

        Por tanto, el test de Neyman-Pearson de tamaño $\alpha$ es:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \sum\limits_{i=1}^{n}X_i<n\mu_0 + z_{1-\alpha}\sqrt{n}\sigma_0 \\
                 0 & \text{si\ } \sum\limits_{i=1}^{n}X_i \geq n\mu_0 + z_{1-\alpha}\sqrt{n}\sigma_0
            \end{array}\right.
        \end{equation*}

        La potencia del test es, puesto que bajo $H_1$ tenemos que $T\rightsquigarrow \cc{N}(n\mu_1, n\sigma_0^2)$:
        \begin{align*}
            \sup_{\mu \in \Theta_1}\beta_\varphi(\mu) &= \beta_\varphi(\mu_1) = E_{\mu_1}[\varphi] = P_{\mu_1}[T < n\mu_0 + z_{1-\alpha}\sqrt{n}\sigma_0] =\\&= P\left[Z < \frac{n\mu_0 + z_{1-\alpha}\sqrt{n}\sigma_0 - n\mu_1}{\sqrt{n}\sigma_0}\right] = P\left[Z < z_{1-\alpha} - \frac{\sqrt{n}(\mu_1-\mu_0)}{\sigma_0}\right]
        \end{align*}
    \end{description}
\end{ejercicio}

\begin{ejercicio}
    Dada una muestra de tamaño $n$ de una variable aleatoria con distribución $\cc{U}(-\theta,\theta)$, deducir el test más potente de tamaño $\alpha$ para contrastar $H_0 : \theta = \theta_0$ frente a $H_1 : \theta = \theta_1$ y calcular su potencia. ¿Cuál es el test óptimo fijado un nivel de significación arbitrario?\\

    Consideramos el contraste de hipótesis:
    \begin{align*}
        \begin{cases}
            H_0: & \theta = \theta_0 \\
            H_1: & \theta = \theta_1
        \end{cases}
    \end{align*}

    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de la variable $X\rightsquigarrow \cc{U}(-\theta,\theta)$. Calculamos la función de densidad conjunta de la muestra:
    \begin{align*}
        f_\theta^n(x_1, \ldots, x_n) &= \prod_{i=1}^{n} f_\theta(x_i) = \prod_{i=1}^{n} \frac{1}{2\theta} = \frac{1}{(2\theta)^n}\qquad \text{si\ } -\theta < x_i < \theta, \forall i=1,\ldots,n
    \end{align*}

    Notemos que $-\theta< x_i < \theta$ para todo $i=1,\ldots,n$ es equivalente a que se cumpla que $\max\{|x_1|, \ldots, |x_n|\} = |x|_{(n)} < \theta$. Por tanto, podemos reescribir la función de densidad conjunta como:
    \begin{equation*}
        f_\theta^n(x_1, \ldots, x_n) = \begin{cases}
            \frac{1}{(2\theta)^n} & \text{si\ } |x|_{(n)} < \theta \\
            0 & \text{en otro caso}
        \end{cases}
    \end{equation*}

    Notamos:
    \begin{align*}
        f_0^n(x_1, \ldots, x_n) &= f_{\theta_0}^n(x_1, \ldots, x_n) \\
        f_1^n(x_1, \ldots, x_n) &= f_{\theta_1}^n(x_1, \ldots, x_n)
    \end{align*}

    El test de Neyman-Pearson de tamaño $\alpha$ para el contraste de hipótesis anunciado será de la forma:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } f_1^n(X_1, \ldots, X_n) > k f_0^n(X_1, \ldots, X_n) \\
            \gamma & \text{si\ } f_1^n(X_1, \ldots, X_n) = k f_0^n(X_1, \ldots, X_n) \\
            0 & \text{si\ } f_1^n(X_1, \ldots, X_n) < k f_0^n(X_1, \ldots, X_n)
        \end{array}\right.
    \end{equation*}
    con $k\in \mathbb{R}^+_0$ y $\gamma\in [0,1]$ que nos disponemos a a determinar. Veamos cuándo se anula cada una de las funciones de densidad conjunta:
    \begin{align*}
        f_0^n(x_1, \ldots, x_n) = 0 &\iff |x|_{(n)} \geq \theta_0 \\
        f_1^n(x_1, \ldots, x_n) = 0 &\iff |x|_{(n)} \geq \theta_1
    \end{align*}

    Dado que el soporte de $f_0^n$ y $f_1^n$ depende de $\theta_0$ y $\theta_1$, distinguimos dos casos, según el orden de $\theta_0$ y $\theta_1$.
    \begin{description}
        \item[Caso $\theta_0 < \theta_1$.] Analizamos dónde se anula cada una de las funciones de densidad conjunta según el valor de $|x|_{(n)}$ (véase la Figura \ref{fig:densidades_conjuntas_ej7.5.1}).

Si $\theta_0 < |x|_{(n)} < \theta_1$, entonces $f_1^n>0$ y $f_0^n=0$, por lo que el test nos dice que (independientemente del valor de $k$ elegido) debemos rechazar $H_0$. Por último, si $|x|_{(n)} < \theta_0$, entonces $f_1^n>0$ y $f_0^n>0$, por lo que podemos dividir ambas funciones de densidad conjunta. Por tanto, el test de Neyman-Pearson de tamaño $\alpha$ queda como:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_0 \leq |X|_{(n)} \leq \theta_1 \\
                0 & \text{si\ } \theta_1 < |X|_{(n)} \\
                1 & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \\
                \gamma & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \\
                0 & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k
            \end{array}\right.
        \end{equation*}

        En el caso de que $|X|_{(n)} < \theta_0$, calculamos la razón de verosimilitudes:
        \begin{equation*}
            \lambda(X_1, \ldots, X_n) = \frac{f_1^n(X_1, \ldots, X_n)}{f_0^n(X_1, \ldots, X_n)} = \frac{(2\theta_0)^n}{(2\theta_1)^n} = \left(\frac{\theta_0}{\theta_1}\right)^n
        \end{equation*}

        Como el cociente $\lambda(X_1, \ldots, X_n)$ no depende de la muestra, elegimos la constante $k=\lambda(X_1, \ldots, X_n)$ y el test queda como:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_0 \leq |X|_{(n)} \leq \theta_1 \\
                0 & \text{si\ } \theta_1 < |X|_{(n)} \\
                \gamma & \text{si\ } |X|_{(n)} < \theta_0
            \end{array}\right.

            = \gamma
        \end{equation*}
        
        Por tanto, el test más potente de tamaño $\alpha$ cuando $\theta_0 < \theta_1$ es:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_0 \leq |X|_{(n)} \leq \theta_1 \\
                0 & \text{si\ } \theta_1 < |X|_{(n)} \\
                \alpha & \text{si\ } |X|_{(n)} < \theta_0
            \end{array}\right.
        \end{equation*}

        La potencia del test es:
        \begin{align*}
            \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = P_{\theta_1}[\theta_0 \leq |X|_{(n)} \leq \theta_1] + \alpha P_{\theta_1}[|X|_{(n)} < \theta_0]
        \end{align*}

        Calculamos cada una de las probabilidades:
        \begin{align*}
            P_{\theta_1}[|X|_{(n)} < \theta_0] &= P_{\theta_1}[|X_1| < \theta_0, \ldots, |X_n| < \theta_0] = \prod_{i=1}^{n} P_{\theta_1}[|X| < \theta_0] = \left(P_{\theta_1}[|X| < \theta_0]\right)^n\\
            P_{\theta_1}[|X| < \theta_0] &= \int_{-\theta_0}^{\theta_0} f_{\theta_1}(x) dx = \int_{-\theta_0}^{\theta_0} \frac{1}{2\theta_1} dx = \frac{\theta_0 - (-\theta_0)}{2\theta_1} = \frac{\theta_0}{\theta_1}\\
            P_{\theta_1}[|X|_{(n)} < \theta_0] &= \left(\frac{\theta_0}{\theta_1}\right)^n\\
            P_{\theta_1}[\theta_0 \leq |X|_{(n)} \leq \theta_1] &= P_{\theta_1}[|X|_{(n)} \leq \theta_1] - P_{\theta_1}[|X|_{(n)} < \theta_0] = 1 - \left(\frac{\theta_0}{\theta_1}\right)^n
        \end{align*}

        Por tanto, la potencia del test es:
        \begin{align*}
            \beta_\varphi(\theta_1) &= \left(1-\left(\frac{\theta_0}{\theta_1}\right)^n\right) + \alpha \left(\frac{\theta_0}{\theta_1}\right)^n = 1 - (1-\alpha)\left(\frac{\theta_0}{\theta_1}\right)^n
        \end{align*}

        Observamos que, si $|x|_{(n)} > \theta_0$, entonces $f_1^n=f_0^n=0$ (suceso de probabilidad nula bajo ambas hipótesis, donde fijamos $\varphi=0$ aunque su valor no afecte al tamaño ni a la potencia del test). Si $|x|_{(n)} < \theta_0$, entonces $f_0^n>0$, por lo que podemos dividir ambas funciones de densidad conjunta. Por tanto, el test de Neyman-Pearson de tamaño $\alpha$ queda como:

            \item[Subcaso $k=0$.] En este caso, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \iff |X|_{(n)} < \theta_1 \\
                    \gamma & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \iff \theta_1 \leq |X|_{(n)} < \theta_0\\
                    0 & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \iff \text{(nunca)}\\
                    0 & \text{si\ } |X|_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            Por tanto, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } |X|_{(n)} < \theta_1 \\
                    \gamma & \text{si\ } \theta_1 \leq |X|_{(n)} < \theta_0 \\
                    0 & \text{si\ } |X|_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            Imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = P_{\theta_0}[|X|_{(n)} < \theta_1] + \gamma P_{\theta_0}[\theta_1 \leq |X|_{(n)} < \theta_0]
            \end{equation*}

            Calculamos cada una de las probabilidades:
            \begin{align*}
                P_{\theta_0}[|X|_{(n)} < \theta_1] &= P_{\theta_0}[|X_1| < \theta_1, \ldots, |X_n| < \theta_1] = \left(P_{\theta_0}[|X| < \theta_1]\right)^n\\
                P_{\theta_0}[|X| < \theta_1] &= \int_{-\theta_1}^{\theta_1} f_{\theta_0}(x) dx = \int_{-\theta_1}^{\theta_1} \frac{1}{2\theta_0} dx = \frac{\theta_1 - (-\theta_1)}{2\theta_0} = \frac{\theta_1}{\theta_0}\\
                P_{\theta_0}[|X|_{(n)} < \theta_1] &= \left(\frac{\theta_1}{\theta_0}\right)^n\\
                P_{\theta_0}[\theta_1 \leq |X|_{(n)} < \theta_0] &= P_{\theta_0}[|X|_{(n)} < \theta_0] - P_{\theta_0}[|X|_{(n)} < \theta_1] = 1 - \left(\frac{\theta_1}{\theta_0}\right)^n
            \end{align*}

            Por tanto, imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \left(\frac{\theta_1}{\theta_0}\right)^n + \gamma \left(1 - \left(\frac{\theta_1}{\theta_0}\right)^n\right) \Longrightarrow \gamma = \frac{\alpha - \left(\nicefrac{\theta_1}{\theta_0}\right)^n}{1 - \left(\nicefrac{\theta_1}{\theta_0}\right)^n}
            \end{equation*}

            Para que $\gamma \in [0,1]$ (observando que $\gamma\leq 1$ se cumple, pues $\alpha \leq 1$) debemos imponer $\gamma\geq 0$, lo que nos lleva a la condición:
            \begin{equation*}
                \alpha \geq \left(\frac{\theta_1}{\theta_0}\right)^n
            \end{equation*}

            Bajo esta condición, el test más potente de tamaño $\alpha$ cuando $\theta_0 > \theta_1$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } |X|_{(n)} < \theta_1 \\
                    \dfrac{\alpha - \left(\nicefrac{\theta_1}{\theta_0}\right)^n}{1 - \left(\nicefrac{\theta_1}{\theta_0}\right)^n} & \text{si\ } \theta_1 \leq |X|_{(n)} < \theta_0 \\
                    0 & \text{si\ } |X|_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            La potencia del test es:
            \begin{align*}
                \beta_\varphi(\theta_1) &= E_{\theta_1}[\varphi] = P_{\theta_1}[|X|_{(n)} < \theta_1] + \gamma P_{\theta_1}[\theta_1 \leq |X|_{(n)} < \theta_0] =\\&= 1 + \gamma \cdot 0 = 1
            \end{align*}

            \item[Subcaso $k=\left(\nicefrac{\theta_0}{\theta_1}\right)^n$.] En este caso, observando de nuevo la Figura~\ref{fig:razon_verosimilitudes_ej7.5.2}, vemos que el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \iff \text{(nunca)} \\
                    \gamma & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \iff |X|_{(n)} < \theta_1 \\
                    0 & \text{si\ } |X|_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \iff \theta_1 \leq |X|_{(n)} < \theta_0\\
                    0 & \text{si\ } |X|_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            Por tanto, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \gamma & \text{si\ } |X|_{(n)} < \theta_1 \\
                    0 & \text{si\ } \theta_1 \leq |X|_{(n)}
                \end{array}\right.
            \end{equation*}

            Imponiendo tamaño $\alpha$:
            \begin{multline*}
                \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = \gamma P_{\theta_0}[|X|_{(n)} < \theta_1] = \gamma \left(\frac{\theta_1}{\theta_0}\right)^n \Longrightarrow \\ \Longrightarrow\gamma = \frac{\alpha}{\left(\nicefrac{\theta_1}{\theta_0}\right)^n}
                = \alpha\left(\frac{\theta_0}{\theta_1}\right)^n
            \end{multline*}

            Para que $\gamma \in [0,1]$ debemos imponer $\gamma\leq 1$, lo que nos lleva a la condición:
            \begin{equation*}
                \alpha \leq \left(\frac{\theta_1}{\theta_0}\right)^n
            \end{equation*}

            Bajo esta condición, el test más potente de tamaño $\alpha$ cuando $\theta_0 > \theta_1$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \gamma = \alpha\left(\frac{\theta_0}{\theta_1}\right)^n & \text{si\ } |X|_{(n)} < \theta_1 \\
                    0 & \text{si\ } \theta_1 \leq |X|_{(n)}
                \end{array}\right.
            \end{equation*}

            La potencia del test es:
            \begin{align*}
                \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = \gamma P_{\theta_1}[|X|_{(n)} < \theta_1] = \gamma \cdot 1 = \gamma = \alpha\left(\frac{\theta_0}{\theta_1}\right)^n
            \end{align*}
        \end{description}
    \end{description}


    Veamos ahora cuál es el test óptimo fijado un nivel de significación $\alpha'$ arbitrario, debemos elegir $\alpha\in \left[0,\alpha'\right]$ de manera que la potencia del test sea máxima.
    \begin{description}
        \item[Caso $\theta_0 < \theta_1$.] La potencia del test es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = 1 - (1-\alpha)\left(\frac{\theta_0}{\theta_1}\right)^n
        \end{equation*}

        Buscamos estudiar la monotonía de la potencia del test respecto a $\alpha$:
        \begin{equation*}
            \frac{d}{d\alpha}\beta_\varphi(\theta_1) = \frac{d}{d\alpha}\left(1 - (1-\alpha)\left(\frac{\theta_0}{\theta_1}\right)^n\right) = \left(\frac{\theta_0}{\theta_1}\right)^n > 0
        \end{equation*}
        Esto nos dice que la potencia del test es creciente en $\alpha$, por lo que para maximizarla, debemos elegir $\alpha = \alpha'$.

        Por tanto, el test óptimo de tamaño $\alpha'$ cuando $\theta_0 < \theta_1$ es:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_0 \leq |X|_{(n)} \leq \theta_1 \\
                0 & \text{si\ } \theta_1 < |X|_{(n)} \\
                \alpha' & \text{si\ } |X|_{(n)} < \theta_0
            \end{array}\right.
        \end{equation*}

        La potencia del test óptimo es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = 1 - (1-\alpha')\left(\frac{\theta_0}{\theta_1}\right)^n
        \end{equation*}

        \item[Caso $\theta_0 > \theta_1$.] En este caso, debemos estudiar los dos subcasos que hemos visto anteriormente:
        \begin{description}
            \item[Subcaso $\alpha'\leq \left(\nicefrac{\theta_1}{\theta_0}\right)^n$.] La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = \alpha\left(\frac{\theta_0}{\theta_1}\right)^n
            \end{equation*}

            Buscamos estudiar la monotonía de la potencia del test respecto a $\alpha$:
            \begin{equation*}
                \frac{d}{d\alpha}\beta_\varphi(\theta_1) = \frac{d}{d\alpha}\left(\alpha\left(\frac{\theta_0}{\theta_1}\right)^n\right) = \left(\frac{\theta_0}{\theta_1}\right)^n > 0
            \end{equation*}
            Esto nos dice que la potencia del test es creciente en $\alpha$, por lo que para maximizarla, debemos elegir $\alpha = \alpha'$. Por tanto, el test óptimo de tamaño $\alpha'$ cuando $\theta_0 > \theta_1$ y $\alpha'\leq \left(\nicefrac{\theta_1}{\theta_0}\right)^n$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \alpha'\left(\frac{\theta_0}{\theta_1}\right)^n & \text{si\ } |X|_{(n)} < \theta_1 \\
                    0 & \text{si\ } \theta_1 \leq |X|_{(n)}
                \end{array}\right.
            \end{equation*}

            La potencia del test óptimo es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = \alpha'\left(\frac{\theta_0}{\theta_1}\right)^n
            \end{equation*}

            \item[Subcaso $\alpha' > \left(\nicefrac{\theta_1}{\theta_0}\right)^n$.] En este caso, podemos seleccionar $\alpha\in \left[0,\left(\nicefrac{\theta_1}{\theta_0}\right)^n\right]$ o $\alpha\in \left[\left(\nicefrac{\theta_1}{\theta_0}\right)^n,\alpha'\right]$.
            \begin{itemize}
                \item Si $\alpha\in \left[0,\left(\nicefrac{\theta_1}{\theta_0}\right)^n\right]$, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = \alpha\left(\frac{\theta_0}{\theta_1}\right)^n
                \end{equation*}

                Esta potencia es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \left(\nicefrac{\theta_1}{\theta_0}\right)^n$. En este caso, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = 1
                \end{equation*}
                \item Si $\alpha\in \left[\left(\nicefrac{\theta_1}{\theta_0}\right)^n,\alpha'\right]$, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = 1
                \end{equation*}

                Esta potencia es constante en $\alpha$, por lo que para maximizarla podemos elegir cualquier $\alpha\in \left[\left(\nicefrac{\theta_1}{\theta_0}\right)^n,\alpha'\right]$. Para minimizar el tamaño del test, elegimos $\alpha = \left(\nicefrac{\theta_1}{\theta_0}\right)^n$.
            \end{itemize}

            En ambos casos, la potencia del test es máxima y vale $1$. Por tanto, el test óptimo de tamaño $\alpha'$ cuando $\theta_0 > \theta_1$ y $\alpha' > \left(\nicefrac{\theta_1}{\theta_0}\right)^n$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } |X|_{(n)} < \theta_1 \\
                    0 & \text{si\ } \theta_1 \leq |X|_{(n)}
                \end{array}\right.
            \end{equation*}
            La potencia del test óptimo es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = 1
            \end{equation*}
        \end{description}
    \end{description}
\end{ejercicio}

\begin{ejercicio}
    Deducir el test más potente de tamaño arbitrario para contrastar $H_0 : \theta = \theta_0$ frente a $H_1 : \theta = \theta_1$, basándose en una muestra de tamaño $n$ de una variable aleatoria con función de densidad
    \begin{equation*}
        f_\theta(x) = \frac{\theta}{x^2}, \qquad x>\theta
    \end{equation*}
    Deducir el test óptimo para un nivel de significación arbitrario.\\

    Consideramos el contraste de hipótesis:
    \begin{align*}
        \begin{cases}
            H_0: & \theta = \theta_0 \\
            H_1: & \theta = \theta_1
        \end{cases}
    \end{align*}
    con $\theta_0, \theta_1 > 0$ y $\theta_0 \neq \theta_1$. Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de la variable $X$ con densidad la dada. Calculamos la función de densidad conjunta de la muestra:
    \begin{align*}
        f_\theta^n(x_1, \ldots, x_n) &= \prod_{i=1}^{n} f_\theta(x_i) = \prod_{i=1}^{n} \frac{\theta}{x_i^2} = \frac{\theta^n}{\prod\limits_{i=1}^{n} x_i^2} \qquad \text{si\ } x_i > \theta, \ \forall i=1,\ldots,n
    \end{align*}

    Notemos que $x_i > \theta$ para todo $i=1,\ldots,n$ es equivalente a que se cumpla que $\min\{x_1, \ldots, x_n\} = x_{(1)} > \theta$. Por tanto, podemos reescribir la función de densidad conjunta como:
    \begin{equation*}
        f_\theta^n(x_1, \ldots, x_n) = \begin{cases}
            \dfrac{\theta^n}{\prod\limits_{i=1}^{n} x_i^2} & \text{si\ } x_{(1)} > \theta \\[12pt]
            0 & \text{en otro caso}
        \end{cases}
    \end{equation*}

    Notamos:
    \begin{align*}
        f_0^n(x_1, \ldots, x_n) &= f_{\theta_0}^n(x_1, \ldots, x_n) \\
        f_1^n(x_1, \ldots, x_n) &= f_{\theta_1}^n(x_1, \ldots, x_n)
    \end{align*}

    El test de Neyman-Pearson de tamaño $\alpha$ para el contraste de hipótesis anunciado será de la forma:
    \begin{equation*}
        \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
            1 & \text{si\ } f_1^n(X_1, \ldots, X_n) > k f_0^n(X_1, \ldots, X_n) \\
            \gamma & \text{si\ } f_1^n(X_1, \ldots, X_n) = k f_0^n(X_1, \ldots, X_n) \\
            0 & \text{si\ } f_1^n(X_1, \ldots, X_n) < k f_0^n(X_1, \ldots, X_n)
        \end{array}\right.
    \end{equation*}
    con $k \in \mathbb{R}^+_0$ y $\gamma \in [0,1]$ que nos disponemos a determinar. Veamos cuándo se anula cada una de las funciones de densidad conjunta:
    \begin{align*}
        f_0^n(x_1, \ldots, x_n) = 0 &\iff x_{(1)} \leq \theta_0 \\
        f_1^n(x_1, \ldots, x_n) = 0 &\iff x_{(1)} \leq \theta_1
    \end{align*}

    Dado que el soporte de $f_0^n$ y $f_1^n$ depende de $\theta_0$ y $\theta_1$, distinguimos dos casos, según el orden de $\theta_0$ y $\theta_1$.
    \begin{description}

            \end{array}\right.
        \end{equation*}

        En el caso de que $X_{(1)} > \theta_0$, calculamos la razón de verosimilitudes:

                0 & \text{si\ } \theta_0 < X_{(1)} \leq \theta_1 \\

        Como $\lambda$ únicamente puede tomar los valores $0$ y $\left(\nicefrac{\theta_1}{\theta_0}\right)^n$ (véase la Figura~\ref{fig:razon_verosimilitudes_ej7.6.1}), solo debemos estudiar dos posibles valores de $k$:

        \begin{description}
            \item[Subcaso $k = \left(\nicefrac{\theta_1}{\theta_0}\right)^n$.] En este caso, observando la Figura~\ref{fig:razon_verosimilitudes_ej7.6.1}, vemos que el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \iff \text{(nunca)} \\
                    \gamma & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \iff X_{(1)} > \theta_1 \\
                    0 & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \iff \theta_0 < X_{(1)} \leq \theta_1 \\
                    0 & \text{si\ } X_{(1)} \leq \theta_0
                \end{array}\right.
            \end{equation*}

            Por tanto, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \gamma & \text{si\ } X_{(1)} > \theta_1 \\
                    0 & \text{si\ } X_{(1)} \leq \theta_1
                \end{array}\right.
            \end{equation*}

            Imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = \gamma P_{\theta_0}[X_{(1)} > \theta_1]
            \end{equation*}

            Calculamos cada una de las probabilidades:
            \begin{align*}
                P_{\theta_0}[X_{(1)} > \theta_1] &= P_{\theta_0}[X_1 > \theta_1, \ldots, X_n > \theta_1] = \prod_{i=1}^{n} P_{\theta_0}[X > \theta_1] = \left(P_{\theta_0}[X > \theta_1]\right)^n \\
                P_{\theta_0}[X > \theta_1] &= \int_{\theta_1}^{+\infty} f_{\theta_0}(x)\,dx = \int_{\theta_1}^{+\infty} \frac{\theta_0}{x^2}\,dx = \theta_0 \left[-\frac{1}{x}\right]_{\theta_1}^{+\infty} = \frac{\theta_0}{\theta_1} \\
                P_{\theta_0}[X_{(1)} > \theta_1] &= \left(\frac{\theta_0}{\theta_1}\right)^n
            \end{align*}

            Por tanto, imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \gamma \left(\frac{\theta_0}{\theta_1}\right)^n \Longrightarrow \gamma = \frac{\alpha}{\left(\nicefrac{\theta_0}{\theta_1}\right)^n} = \alpha\left(\frac{\theta_1}{\theta_0}\right)^n
            \end{equation*}

            Para que $\gamma \in [0,1]$ debemos imponer $\gamma \leq 1$, lo que nos lleva a la condición:
            \begin{equation*}
                \alpha \leq \left(\frac{\theta_0}{\theta_1}\right)^n
            \end{equation*}

            Bajo esta condición, el test más potente de tamaño $\alpha$ cuando $\theta_0 < \theta_1$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \gamma = \alpha\left(\dfrac{\theta_1}{\theta_0}\right)^n & \text{si\ } X_{(1)} > \theta_1 \\[8pt]
                    0 & \text{si\ } X_{(1)} \leq \theta_1
                \end{array}\right.
            \end{equation*}

            La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = \gamma P_{\theta_1}[X_{(1)} > \theta_1] = \gamma \cdot 1 = \alpha\left(\frac{\theta_1}{\theta_0}\right)^n
            \end{equation*}

            \item[Subcaso $k = 0$.] En este caso, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \iff X_{(1)} > \theta_1 \\
                    \gamma & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \iff \theta_0 < X_{(1)} \leq \theta_1 \\
                    0 & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \iff \text{(nunca)} \\
                    0 & \text{si\ } X_{(1)} \leq \theta_0
                \end{array}\right.
            \end{equation*}

            Por tanto, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X_{(1)} > \theta_1 \\
                    \gamma & \text{si\ } \theta_0 < X_{(1)} \leq \theta_1 \\
                    0 & \text{si\ } X_{(1)} \leq \theta_0
                \end{array}\right.
            \end{equation*}

            Imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = P_{\theta_0}[X_{(1)} > \theta_1] + \gamma P_{\theta_0}[\theta_0 < X_{(1)} \leq \theta_1]
            \end{equation*}

            Calculamos las probabilidades involucradas:
            \begin{align*}
                P_{\theta_0}[X_{(1)} > \theta_1] &= \left(\frac{\theta_0}{\theta_1}\right)^n \\
                P_{\theta_0}[\theta_0 < X_{(1)} \leq \theta_1] &= P_{\theta_0}[X_{(1)} > \theta_0] - P_{\theta_0}[X_{(1)} > \theta_1] = 1 - \left(\frac{\theta_0}{\theta_1}\right)^n
            \end{align*}

            Por tanto, imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \left(\frac{\theta_0}{\theta_1}\right)^n + \gamma \left(1 - \left(\frac{\theta_0}{\theta_1}\right)^n\right) \Longrightarrow \gamma = \frac{\alpha - \left(\nicefrac{\theta_0}{\theta_1}\right)^n}{1 - \left(\nicefrac{\theta_0}{\theta_1}\right)^n}
            \end{equation*}

            Para que $\gamma \in [0,1]$ (observando que $\gamma \leq 1$ se cumple, pues $\alpha \leq 1$) debemos imponer $\gamma \geq 0$, lo que nos lleva a la condición:
            \begin{equation*}
                \alpha \geq \left(\frac{\theta_0}{\theta_1}\right)^n
            \end{equation*}

            Bajo esta condición, el test más potente de tamaño $\alpha$ cuando $\theta_0 < \theta_1$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X_{(1)} > \theta_1 \\[6pt]
                    \dfrac{\alpha - \left(\nicefrac{\theta_0}{\theta_1}\right)^n}{1 - \left(\nicefrac{\theta_0}{\theta_1}\right)^n} & \text{si\ } \theta_0 < X_{(1)} \leq \theta_1 \\[12pt]
                    0 & \text{si\ } X_{(1)} \leq \theta_0
                \end{array}\right.
            \end{equation*}

            La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = P_{\theta_1}[X_{(1)} > \theta_1] + \gamma P_{\theta_1}[\theta_0 < X_{(1)} \leq \theta_1] = 1 + \gamma \cdot 0 = 1
            \end{equation*}
        \end{description}

        \item[Caso $\theta_0 > \theta_1$.] Analizamos dónde se anula cada una de las funciones de densidad conjunta según el valor de $x_{(1)}$ (véase la Figura \ref{fig:densidades_conjuntas_ej7.6.2}).
        \begin{figure}[h]

                \filldraw[black] (2, 0) circle (2pt) node[below=4pt] {$\theta_1$};
                \filldraw[black] (5, 0) circle (2pt) node[below=4pt] {$\theta_0$};

                % Etiquetas de f_0 y f_1 en cada tramo
                \node[above=4pt, font=\small] at (0.75, 0.55) {$f_1^n = 0$};
                \node[above=4pt, font=\small] at (0.75, 0.05) {$f_0^n = 0$};

                \node[above=4pt, font=\small] at (3.5, 0.55) {$f_1^n > 0$};
                \node[above=4pt, font=\small] at (3.5, 0.05) {$f_0^n = 0$};

                \node[above=4pt, font=\small] at (6.2, 0.55) {$f_1^n > 0$};
                \node[above=4pt, font=\small] at (6.2, 0.05) {$f_0^n > 0$};

                % Divisiones verticales sutiles
                \draw[dashed, gray] (2, -0.2) -- (2, 1.1);
                \draw[dashed, gray] (5, -0.2) -- (5, 1.1);
            \end{tikzpicture}
            \caption{Análisis de los soportes de $f_0^n$ y $f_1^n$ según el valor de $x_{(1)}$.}
            \label{fig:densidades_conjuntas_ej7.6.2}
        \end{figure}

        Observamos que, si $x_{(1)} \leq \theta_1$, entonces $f_1^n = f_0^n = 0$ (suceso de probabilidad nula bajo ambas hipótesis, donde fijamos $\varphi = 0$ aunque su valor no afecte al tamaño ni a la potencia del test). Si $\theta_1 < x_{(1)} \leq \theta_0$, entonces $f_1^n > 0$ y $f_0^n = 0$, por lo que el test nos dice que (independientemente del valor de $k$ elegido) debemos rechazar $H_0$. Por último, si $x_{(1)} > \theta_0$, entonces $f_1^n > 0$ y $f_0^n > 0$, por lo que podemos dividir ambas funciones de densidad conjunta. Por tanto, el test de Neyman-Pearson de tamaño $\alpha$ queda como:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_1 < X_{(1)} \leq \theta_0 \\
                0 & \text{si\ } X_{(1)} \leq \theta_1 \\
                1 & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \\
                \gamma & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \\
                0 & \text{si\ } X_{(1)} > \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k
            \end{array}\right.
        \end{equation*}

        En el caso de que $X_{(1)} > \theta_0$, calculamos la razón de verosimilitudes:
        \begin{equation*}
            \lambda(X_1, \ldots, X_n) = \frac{f_1^n(X_1, \ldots, X_n)}{f_0^n(X_1, \ldots, X_n)} = \left(\frac{\theta_1}{\theta_0}\right)^n
        \end{equation*}

        Como el cociente $\lambda(X_1, \ldots, X_n)$ no depende de la muestra, elegimos la constante $k = \lambda(X_1, \ldots, X_n)$ y el test queda como:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_1 < X_{(1)} \leq \theta_0 \\
                0 & \text{si\ } X_{(1)} \leq \theta_1 \\
                \gamma & \text{si\ } X_{(1)} > \theta_0
            \end{array}\right.
        \end{equation*}

        Imponiendo tamaño $\alpha$:
        \begin{equation*}
            \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = \gamma P_{\theta_0}[X_{(1)} > \theta_0] = \gamma
        \end{equation*}

        Por tanto, el test más potente de tamaño $\alpha$ cuando $\theta_0 > \theta_1$ es:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_1 < X_{(1)} \leq \theta_0 \\
                0 & \text{si\ } X_{(1)} \leq \theta_1 \\
                \alpha & \text{si\ } X_{(1)} > \theta_0
            \end{array}\right.
        \end{equation*}

        La potencia del test es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = P_{\theta_1}[\theta_1 < X_{(1)} \leq \theta_0] + \alpha P_{\theta_1}[X_{(1)} > \theta_0]
        \end{equation*}

        Calculamos cada una de las probabilidades:
        \begin{align*}
            P_{\theta_1}[X_{(1)} > \theta_0] &= P_{\theta_1}[X_1 > \theta_0, \ldots, X_n > \theta_0] = \left(P_{\theta_1}[X > \theta_0]\right)^n \\
            P_{\theta_1}[X > \theta_0] &= \int_{\theta_0}^{+\infty} f_{\theta_1}(x)\,dx = \int_{\theta_0}^{+\infty} \frac{\theta_1}{x^2}\,dx = \theta_1 \left[-\frac{1}{x}\right]_{\theta_0}^{+\infty} = \frac{\theta_1}{\theta_0} \\
            P_{\theta_1}[X_{(1)} > \theta_0] &= \left(\frac{\theta_1}{\theta_0}\right)^n \\
            P_{\theta_1}[\theta_1 < X_{(1)} \leq \theta_0] &= P_{\theta_1}[X_{(1)} > \theta_1] - P_{\theta_1}[X_{(1)} > \theta_0] = 1 - \left(\frac{\theta_1}{\theta_0}\right)^n
        \end{align*}

        Por tanto, la potencia del test es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = \left(1 - \left(\frac{\theta_1}{\theta_0}\right)^n\right) + \alpha \left(\frac{\theta_1}{\theta_0}\right)^n = 1 - (1-\alpha)\left(\frac{\theta_1}{\theta_0}\right)^n
        \end{equation*}
    \end{description}

    Veamos ahora cuál es el test óptimo fijado un nivel de significación $\alpha'$ arbitrario; debemos elegir $\alpha \in [0,\alpha']$ de manera que la potencia del test sea máxima.
    \begin{description}
        \item[Caso $\theta_0 < \theta_1$.] En este caso, debemos estudiar los dos subcasos que hemos visto anteriormente:
        \begin{description}
            \item[Subcaso $\alpha' \leq \left(\nicefrac{\theta_0}{\theta_1}\right)^n$.] La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = \alpha\left(\frac{\theta_1}{\theta_0}\right)^n
            \end{equation*}

            Buscamos estudiar la monotonía de la potencia del test respecto a $\alpha$:
            \begin{equation*}
                \frac{d}{d\alpha}\beta_\varphi(\theta_1) = \frac{d}{d\alpha}\left(\alpha\left(\frac{\theta_1}{\theta_0}\right)^n\right) = \left(\frac{\theta_1}{\theta_0}\right)^n > 0
            \end{equation*}
            Esto nos dice que la potencia del test es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \alpha'$. Por tanto, el test óptimo de nivel $\alpha'$ cuando $\theta_0 < \theta_1$ y $\alpha' \leq \left(\nicefrac{\theta_0}{\theta_1}\right)^n$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \alpha'\left(\dfrac{\theta_1}{\theta_0}\right)^n & \text{si\ } X_{(1)} > \theta_1 \\[8pt]
                    0 & \text{si\ } X_{(1)} \leq \theta_1
                \end{array}\right.
            \end{equation*}

            La potencia del test óptimo es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = \alpha'\left(\frac{\theta_1}{\theta_0}\right)^n
            \end{equation*}

            \item[Subcaso $\alpha' > \left(\nicefrac{\theta_0}{\theta_1}\right)^n$.] En este caso, podemos seleccionar $\alpha \in \left[0, \left(\nicefrac{\theta_0}{\theta_1}\right)^n\right]$ o $\alpha \in \left[\left(\nicefrac{\theta_0}{\theta_1}\right)^n, \alpha'\right]$.
            \begin{itemize}
                \item Si $\alpha \in \left[0, \left(\nicefrac{\theta_0}{\theta_1}\right)^n\right]$, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = \alpha\left(\frac{\theta_1}{\theta_0}\right)^n
                \end{equation*}
                Esta potencia es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \left(\nicefrac{\theta_0}{\theta_1}\right)^n$. En este caso, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = 1
                \end{equation*}
                \item Si $\alpha \in \left[\left(\nicefrac{\theta_0}{\theta_1}\right)^n, \alpha'\right]$, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = 1
                \end{equation*}
                Esta potencia es constante en $\alpha$, por lo que para maximizarla podemos elegir cualquier $\alpha \in \left[\left(\nicefrac{\theta_0}{\theta_1}\right)^n, \alpha'\right]$. Para minimizar el tamaño del test, elegimos $\alpha = \left(\nicefrac{\theta_0}{\theta_1}\right)^n$.
            \end{itemize}

            En ambos casos, la potencia del test es máxima y vale $1$. Por tanto, el test óptimo de nivel $\alpha'$ cuando $\theta_0 < \theta_1$ y $\alpha' > \left(\nicefrac{\theta_0}{\theta_1}\right)^n$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } X_{(1)} > \theta_1 \\
                    0 & \text{si\ } X_{(1)} \leq \theta_1
                \end{array}\right.
            \end{equation*}
            La potencia del test óptimo es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = 1
            \end{equation*}
        \end{description}

        \item[Caso $\theta_0 > \theta_1$.] La potencia del test es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = 1 - (1-\alpha)\left(\frac{\theta_1}{\theta_0}\right)^n
        \end{equation*}

        Buscamos estudiar la monotonía de la potencia del test respecto a $\alpha$:
        \begin{equation*}
            \frac{d}{d\alpha}\beta_\varphi(\theta_1) = \frac{d}{d\alpha}\left(1 - (1-\alpha)\left(\frac{\theta_1}{\theta_0}\right)^n\right) = \left(\frac{\theta_1}{\theta_0}\right)^n > 0
        \end{equation*}
        Esto nos dice que la potencia del test es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \alpha'$.

        Por tanto, el test óptimo de nivel $\alpha'$ cuando $\theta_0 > \theta_1$ es:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } \theta_1 < X_{(1)} \leq \theta_0 \\
                0 & \text{si\ } X_{(1)} \leq \theta_1 \\
                \alpha' & \text{si\ } X_{(1)} > \theta_0
            \end{array}\right.
        \end{equation*}

        La potencia del test óptimo es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = 1 - (1-\alpha')\left(\frac{\theta_1}{\theta_0}\right)^n
        \end{equation*}
    \end{description}
\end{ejercicio}

\begin{ejercicio}
    Construir el test de Neyman-Pearson de tamaño $\alpha$ para contrastar $H_0:\theta = \theta_0$ frente a $H_1 : \theta = \theta_1$, basándose en una muestra de tamaño $n$ de una variable aleatoria con función de densidad
    \begin{equation*}
        f_\theta(x) = \frac{1}{x\ln \theta}, \qquad 1<x<\theta
    \end{equation*}
    Deducir el test óptimo para un nivel de significación arbitrario.\\

    Consideramos el contraste de hipótesis:
    \begin{align*}
        \begin{cases}
            H_0: & \theta = \theta_0 \\
            H_1: & \theta = \theta_1
        \end{cases}
    \end{align*}
    con $\theta_0, \theta_1 > 1$ y $\theta_0 \neq \theta_1$. Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de la variable $X$ con densidad la dada. Calculamos la función de densidad conjunta de la muestra:
    \begin{align*}
        f_\theta^n(x_1, \ldots, x_n) &= \prod_{i=1}^{n} f_\theta(x_i) = \prod_{i=1}^{n} \frac{1}{x_i\ln \theta} = \frac{1}{(\ln \theta)^n \prod\limits_{i=1}^{n} x_i} \qquad \text{si\ } 1 < x_i < \theta, \ \forall i=1,\ldots,n
    \end{align*}

    Notemos que $1 < x_i < \theta$ para todo $i=1,\ldots,n$ es equivalente a que se cumpla que $\min\{x_1, \ldots, x_n\} = x_{(1)} > 1$ y $\max\{x_1, \ldots, x_n\} = x_{(n)} < \theta$. Por tanto, podemos reescribir la función de densidad conjunta como:
    \begin{equation*}
        f_\theta^n(x_1, \ldots, x_n) = \begin{cases}
            \dfrac{1}{(\ln \theta)^n \prod\limits_{i=1}^{n} x_i} & \text{si\ } 1 < x_{(1)} \text{\ y\ } x_{(n)} < \theta \\[14pt]
            0 & \text{en otro caso}
        \end{cases}
    \end{equation*}

    Notamos:
    \begin{align*}
        f_0^n(x_1, \ldots, x_n) &= f_{\theta_0}^n(x_1, \ldots, x_n) \\
        f_1^n(x_1, \ldots, x_n) &= f_{\theta_1}^n(x_1, \ldots, x_n)
    \end{align*}

            \gamma & \text{si\ } f_1^n(X_1, \ldots, X_n) = k f_0^n(X_1, \ldots, X_n) \\

        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \\
                \gamma & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \\
                0 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \\
                0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } X_{(n)} \geq \theta_0
            \end{array}\right.
        \end{equation*}

        En el caso de que $1 < X_{(1)}$ y $X_{(n)} < \theta_0$, calculamos la razón de verosimilitudes:
        \begin{equation*}
            \lambda(X_1, \ldots, X_n) = \frac{f_1^n(X_1, \ldots, X_n)}{f_0^n(X_1, \ldots, X_n)} = \begin{cases}
                \left(\dfrac{\ln \theta_0}{\ln \theta_1}\right)^n & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\[12pt]
                0 & \text{si\ } 1 < X_{(1)} \text{\ y\ } \theta_1 \leq X_{(n)} < \theta_0
            \end{cases}
        \end{equation*}

        Como $\lambda$ únicamente puede tomar los valores $0$ y $\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n$ (véase la Figura~\ref{fig:razon_verosimilitudes_ej7.7.2}), solo debemos estudiar dos posibles valores de $k$:

        \begin{description}
            \item[Subcaso $k = 0$.] En este caso, el test queda como:
            \begin{equation*}
                \hspace{-2.5cm}\varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \iff 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\
                    \gamma & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \iff 1 < X_{(1)} \text{\ y\ } \theta_1 \leq X_{(n)} < \theta_0 \\
                    0 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \iff \text{(nunca)} \\
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } X_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            Por tanto, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\
                    \gamma & \text{si\ } 1 < X_{(1)} \text{\ y\ } \theta_1 \leq X_{(n)} < \theta_0 \\
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } X_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            Imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = P_{\theta_0}[X_{(n)} < \theta_1] + \gamma P_{\theta_0}[\theta_1 \leq X_{(n)} < \theta_0]
            \end{equation*}

            Calculamos cada una de las probabilidades:
            \begin{align*}
                P_{\theta_0}[X_{(n)} < \theta_1] &= P_{\theta_0}[X_1 < \theta_1, \ldots, X_n < \theta_1] = \left(P_{\theta_0}[X < \theta_1]\right)^n \\
                P_{\theta_0}[X < \theta_1] &= \int_{1}^{\theta_1} f_{\theta_0}(x)\,dx = \int_{1}^{\theta_1} \frac{1}{x \ln \theta_0}\,dx = \frac{1}{\ln \theta_0}\Big[\ln x\Big]_{1}^{\theta_1} = \frac{\ln \theta_1}{\ln \theta_0} \\
                P_{\theta_0}[X_{(n)} < \theta_1] &= \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n \\
                P_{\theta_0}[\theta_1 \leq X_{(n)} < \theta_0] &= P_{\theta_0}[X_{(n)} < \theta_0] - P_{\theta_0}[X_{(n)} < \theta_1] = 1 - \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n
            \end{align*}

            Por tanto, imponiendo tamaño $\alpha$:
            \begin{equation*}
                \alpha = \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n + \gamma \left(1 - \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n\right) \Longrightarrow \gamma = \frac{\alpha - \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n}{1 - \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n}
            \end{equation*}

            Para que $\gamma \in [0,1]$ (observando que $\gamma \leq 1$ se cumple, pues $\alpha \leq 1$) debemos imponer $\gamma \geq 0$, lo que nos lleva a la condición:
            \begin{equation*}
                \alpha \geq \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n
            \end{equation*}

            Bajo esta condición, el test más potente de tamaño $\alpha$ cuando $\theta_0 > \theta_1$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\[6pt]
                    \dfrac{\alpha - \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n}{1 - \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n} & \text{si\ } 1 < X_{(1)} \text{\ y\ } \theta_1 \leq X_{(n)} < \theta_0 \\[12pt]
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } X_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = P_{\theta_1}[X_{(n)} < \theta_1] + \gamma P_{\theta_1}[\theta_1 \leq X_{(n)} < \theta_0] = 1 + \gamma \cdot 0 = 1
            \end{equation*}

            \item[Subcaso $k = \left(\nicefrac{\ln \theta_0}{\ln \theta_1}\right)^n$.] En este caso, observando de nuevo la Figura~\ref{fig:razon_verosimilitudes_ej7.7.2}, vemos que el test queda como:
            \begin{equation*}
                \hspace{-2.5cm}\varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) > k \iff \text{(nunca)} \\
                    \gamma & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) = k \iff 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\
                    0 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0 \text{\ y\ } \lambda(X_1, \ldots, X_n) < k \iff 1 < X_{(1)} \text{\ y\ } \theta_1 \leq X_{(n)} < \theta_0 \\
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } X_{(n)} \geq \theta_0
                \end{array}\right.
            \end{equation*}

            Por tanto, el test queda como:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \gamma & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } \theta_1 \leq X_{(n)}
                \end{array}\right.
            \end{equation*}

            Imponiendo tamaño $\alpha$:
            \begin{multline*}
                \alpha = \sup_{\theta \in \Theta_0}\beta_\varphi(\theta) = \beta_\varphi(\theta_0) = E_{\theta_0}[\varphi] = \gamma P_{\theta_0}[X_{(n)} < \theta_1] = \gamma \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n \Longrightarrow \\ \Longrightarrow \gamma = \frac{\alpha}{\left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n} = \alpha\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
            \end{multline*}

            Para que $\gamma \in [0,1]$ debemos imponer $\gamma \leq 1$, lo que nos lleva a la condición:
            \begin{equation*}
                \alpha \leq \left(\frac{\ln \theta_1}{\ln \theta_0}\right)^n
            \end{equation*}

            Bajo esta condición, el test más potente de tamaño $\alpha$ cuando $\theta_0 > \theta_1$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \gamma = \alpha\left(\dfrac{\ln \theta_0}{\ln \theta_1}\right)^n & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\[8pt]
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } \theta_1 \leq X_{(n)}
                \end{array}\right.
            \end{equation*}

            La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = E_{\theta_1}[\varphi] = \gamma P_{\theta_1}[X_{(n)} < \theta_1] = \gamma \cdot 1 = \gamma = \alpha\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
            \end{equation*}
        \end{description}
    \end{description}

    Veamos ahora cuál es el test óptimo fijado un nivel de significación $\alpha'$ arbitrario; debemos elegir $\alpha \in [0,\alpha']$ de manera que la potencia del test sea máxima:
    \begin{description}
        \item[Caso $\theta_0 < \theta_1$.] La potencia del test es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = 1 - (1-\alpha)\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
        \end{equation*}

        Buscamos estudiar la monotonía de la potencia del test respecto a $\alpha$:
        \begin{equation*}
            \frac{d}{d\alpha}\beta_\varphi(\theta_1) = \frac{d}{d\alpha}\left(1 - (1-\alpha)\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n\right) = \left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n > 0
        \end{equation*}
        Esto nos dice que la potencia del test es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \alpha'$.

        Por tanto, el test óptimo de nivel $\alpha'$ cuando $\theta_0 < \theta_1$ es:
        \begin{equation*}
            \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } \theta_0 \leq X_{(n)} < \theta_1 \\
                0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } \theta_1 \leq X_{(n)} \\
                \alpha' & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_0
            \end{array}\right.
        \end{equation*}

        La potencia del test óptimo es:
        \begin{equation*}
            \beta_\varphi(\theta_1) = 1 - (1-\alpha')\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
        \end{equation*}

        \item[Caso $\theta_0 > \theta_1$.] En este caso, debemos estudiar los dos subcasos que hemos visto anteriormente:
        \begin{description}
            \item[Subcaso $\alpha' \leq \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n$.] La potencia del test es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = \alpha\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
            \end{equation*}

            Buscamos estudiar la monotonía de la potencia del test respecto a $\alpha$:
            \begin{equation*}
                \frac{d}{d\alpha}\beta_\varphi(\theta_1) = \frac{d}{d\alpha}\left(\alpha\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n\right) = \left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n > 0
            \end{equation*}
            Esto nos dice que la potencia del test es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \alpha'$. Por tanto, el test óptimo de nivel $\alpha'$ cuando $\theta_0 > \theta_1$ y $\alpha' \leq \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    \alpha'\left(\dfrac{\ln \theta_0}{\ln \theta_1}\right)^n & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\[8pt]
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } \theta_1 \leq X_{(n)}
                \end{array}\right.
            \end{equation*}

            La potencia del test óptimo es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = \alpha'\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
            \end{equation*}

            \item[Subcaso $\alpha' > \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n$.] En este caso, podemos seleccionar $\alpha \in \left[0, \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n\right]$ o $\alpha \in \left[\left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n, \alpha'\right]$.
            \begin{itemize}
                \item Si $\alpha \in \left[0, \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n\right]$, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = \alpha\left(\frac{\ln \theta_0}{\ln \theta_1}\right)^n
                \end{equation*}
                Esta potencia es creciente en $\alpha$, por lo que para maximizarla debemos elegir $\alpha = \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n$. En este caso, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = 1
                \end{equation*}
                \item Si $\alpha \in \left[\left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n, \alpha'\right]$, la potencia del test es:
                \begin{equation*}
                    \beta_\varphi(\theta_1) = 1
                \end{equation*}
                Esta potencia es constante en $\alpha$, por lo que para maximizarla podemos elegir cualquier $\alpha \in \left[\left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n, \alpha'\right]$. Para minimizar el tamaño del test, elegimos $\alpha = \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n$.
            \end{itemize}

            En ambos casos, la potencia del test es máxima y vale $1$. Por tanto, el test óptimo de nivel $\alpha'$ cuando $\theta_0 > \theta_1$ y $\alpha' > \left(\nicefrac{\ln \theta_1}{\ln \theta_0}\right)^n$ es:
            \begin{equation*}
                \varphi(X_1, \ldots, X_n) = \left\{\begin{array}{ll}
                    1 & \text{si\ } 1 < X_{(1)} \text{\ y\ } X_{(n)} < \theta_1 \\
                    0 & \text{si\ } X_{(1)} \leq 1 \text{\ o\ } \theta_1 \leq X_{(n)}
                \end{array}\right.
            \end{equation*}
            La potencia del test óptimo es:
            \begin{equation*}
                \beta_\varphi(\theta_1) = 1
            \end{equation*}
        \end{description}
    \end{description}
\end{ejercicio}


\begin{ejercicio}
    Sea $X$ una observación de una variable aleatoria con función de densidad
    \begin{equation*}
        f_\theta(x) = \frac{1}{\theta} e^{\nicefrac{-x}{\theta}}, \qquad x>0
    \end{equation*}
    Construir el test de razón de verosimilitudes de tamaño $\alpha$ arbitrario para contrastar
    \begin{align*}
        \begin{cases}
            H_0: & \theta = \theta_0 \\
            H_1: & \theta < \theta_0
        \end{cases}
    \end{align*}

    Planteamos el contraste de hipótesis anterior sobre el parámetro $\theta \in \Theta$, donde:
    \begin{equation*}
        \Theta_0 = \{\theta_0\}, \qquad \Theta_1 = \left]0, \theta_0\right[, \qquad \Theta = \Theta_0 \cup \Theta_1 = \left]0, \theta_0\right].
    \end{equation*}

    Dado que disponemos de una única observación $X = x > 0$ (es decir, una muestra aleatoria simple de tamaño $n = 1$), la función de verosimilitud coincide directamente con la función de densidad:
    \begin{equation*}
        L_x(\theta) = f_\theta(x) = \frac{1}{\theta} e^{\nicefrac{-x}{\theta}}, \qquad \forall \theta \in \Theta.
    \end{equation*}

    Empleamos el Test de Razón de Verosimilitudes (TRV) de tamaño $\alpha \in \left]0, 1\right[$ para contrastar las hipótesis, cuya expresión es:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } \lambda(x) < c \\
            0, & \text{si } \lambda(x) \geq c
        \end{cases}
    \end{equation*}
    donde $c \in \left]0, 1\right]$ es una constante que se determina imponiendo el tamaño $\alpha$ y $\lambda(x)$ es la razón de verosimilitudes:
    \begin{equation*}
        \lambda(x) = \frac{\sup\limits_{\theta \in \Theta_0} L_x(\theta)}{\sup\limits_{\theta \in \Theta} L_x(\theta)}.
    \end{equation*}

    Para hallar el denominador, estudiamos el comportamiento de $L_x(\theta)$ sobre el espacio $\Theta = \left]0, \theta_0\right]$. Como $L_x(\theta) > 0$ para todo $\theta \in \Theta$ y el logaritmo neperiano es estrictamente creciente, derivamos la log-verosimilitud:
    \begin{align*}
        \ln\left(L_x(\theta)\right) &= -\ln(\theta) - \frac{x}{\theta}, \\
        \frac{d}{d\theta} \ln\left(L_x(\theta)\right) &= -\frac{1}{\theta} + \frac{x}{\theta^2} = \frac{x - \theta}{\theta^2} = 0 \Longleftrightarrow \theta = x.
    \end{align*}
    Observamos que $\frac{d}{d\theta} \ln(L_x(\theta)) > 0$ si $\theta < x$ y $\frac{d}{d\theta} \ln(L_x(\theta)) < 0$ si $\theta > x$. Por tanto:
    \begin{equation*}
        \sup\limits_{\theta \in \Theta} L_x(\theta) =
        \begin{cases}
            L_x(x), & \text{si } 0 < x < \theta_0, \\
            L_x(\theta_0), & \text{si } x \geq \theta_0.
        \end{cases}
    \end{equation*}

    Calculamos ahora el numerador y el denominador de $\lambda(x)$:
    \begin{itemize}
        \item \textbf{Supremo sobre $\Theta_0 = \{\theta_0\}$:}
        Al tratarse de una hipótesis nula simple, el conjunto paramétrico consta de un único punto, por lo que:
        \begin{equation*}
            \sup\limits_{\theta \in \Theta_0} L_x(\theta) = L_x(\theta_0) = \frac{1}{\theta_0} e^{\nicefrac{-x}{\theta_0}}.
        \end{equation*}

        \item \textbf{Supremo sobre $\Theta = \left]0, \theta_0\right]$:}
        Distinguimos dos casos según la posición de $x$ respecto a $\theta_0$:
        \begin{itemize}
            \item Si $x \geq \theta_0$, el supremo sobre $\Theta$ se alcanza en $\theta = \theta_0$:
            \begin{equation*}
                \sup\limits_{\theta \in \Theta} L_x(\theta) = L_x(\theta_0).
            \end{equation*}
            En consecuencia, la razón de verosimilitudes vale:
            \begin{equation*}
                \lambda(x) = \frac{L_x(\theta_0)}{L_x(\theta_0)} = 1.
            \end{equation*}

            \item Si $0 < x < \theta_0$, el punto $\theta = x$ pertenece a $\Theta = \left]0, \theta_0\right]$ y maximiza $L_x(\theta)$ en él, por lo que:
            \begin{equation*}
                \sup\limits_{\theta \in \Theta} L_x(\theta) = L_x(x) = \frac{1}{x} e^{-1}.
            \end{equation*}
            En este caso, la razón de verosimilitudes es:
            \begin{equation*}
                \lambda(x) = \frac{L_x(\theta_0)}{L_x(x)} = \frac{\frac{1}{\theta_0} e^{\nicefrac{-x}{\theta_0}}}{\frac{1}{x} e^{-1}} = \frac{x}{\theta_0} e^{1 - \nicefrac{x}{\theta_0}}.
            \end{equation*}
        \end{itemize}
    \end{itemize}

    En resumen, la función $\lambda(x)$ queda definida a trozos por:
    \begin{equation*}
        \lambda(x) =
        \begin{cases}
            \dfrac{x}{\theta_0} e^{1 - \nicefrac{x}{\theta_0}}, & \text{si } 0 < x < \theta_0 \\[8pt]
            1, & \text{si } x \geq \theta_0
        \end{cases}
    \end{equation*}

    Recordamos que el test de razón de verosimilitudes es:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } \lambda(x) < c \\[2pt]
            0, & \text{si } \lambda(x) \geq c
        \end{cases}
    \end{equation*}

    Puesto que no conocemos la distribución de $\lambda(X)$ bajo $H_0$, no podemos determinar el valor crítico $c$ que imponga tamaño $\alpha$. Por tanto, debemos estudiar la relación entre $\lambda(x)$ y $x$ para poder expresar el test en función de $x$ y así poder determinar el valor crítico $k'$ que imponga tamaño $\alpha$. Para estudiar la monotonía de $\lambda(x)$, puesto que esta es estrictamente positiva, podemos estudiar la monotonía de $\ln(\lambda(x))$:
    \begin{align*}
        \ln(\lambda(x)) &= \ln\left(\frac{x}{\theta_0} e^{1 - \nicefrac{x}{\theta_0}}\right) = \ln\left(\frac{x}{\theta_0}\right) + 1 - \frac{x}{\theta_0} = \ln(x) - \ln(\theta_0) + 1 - \frac{x}{\theta_0}\qquad \forall x \in \left]0, \theta_0\right[.\\
        \frac{d}{dx} \ln(\lambda(x)) &= \frac{1}{x} - \frac{1}{\theta_0} = \frac{\theta_0 - x}{x\theta_0} > 0 \qquad \forall x \in \left]0, \theta_0\right[.
    \end{align*}

    Por tanto, $\ln(\lambda(x))$ (y por tanto $\lambda(x)$) es estrictamente creciente en $\left]0, \theta_0\right[$. La relación entre $\lambda(x)$ y $x$ se ilustra en la Figura~\ref{fig:razon_verosimilitudes_exponencial}.

\begin{ejercicio}
    En base a una observación de $X\rightsquigarrow \{\cc{B}(n,p) : p\in \left]0,1\right[\}$, deducir el test de razón de verosimilitudes para contrastar la hipótesis de que el parámetro $p$ no supera un determinado valor, $p_0$.\\

    Planteamos el contraste de hipótesis anterior sobre el parámetro $p \in \Theta$, donde:
    \begin{equation*}
        \Theta_0 = \left]0, p_0\right], \qquad \Theta_1 = \left]p_0, 1\right[, \qquad \Theta = \Theta_0 \cup \Theta_1 = \left]0, 1\right[,
    \end{equation*}
    con $p_0 \in \left]0, 1\right[$ fijo. El contraste es:
    \begin{align*}
        \begin{cases}
            H_0: & p \leq p_0 \\
            H_1: & p > p_0
        \end{cases}
    \end{align*}

    Dado que disponemos de una única observación $X = x \in \{0, 1, \ldots, n\}$ (es decir, una muestra aleatoria simple de tamaño $1$), la función de verosimilitud coincide directamente con la función de masa de probabilidad de $\cc{B}(n, p)$:
    \begin{equation*}
        L_x(p) = P_p[X = x] = \binom{n}{x} p^x (1 - p)^{n - x}, \qquad \forall p \in \Theta.
    \end{equation*}

    Empleamos el Test de Razón de Verosimilitudes (TRV) de nivel de significación $\alpha \in \left]0, 1\right[$ para contrastar las hipótesis, cuya expresión es:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } \lambda(x) < c, \\
            0, & \text{si } \lambda(x) \geq c,
        \end{cases}
    \end{equation*}
    donde $c \in \left]0, 1\right]$ es una constante que se determina a partir del nivel de significación $\alpha$ y $\lambda(x)$ es la razón de verosimilitudes:
    \begin{equation*}
        \lambda(x) = \frac{\sup\limits_{p \in \Theta_0} L_x(p)}{\sup\limits_{p \in \Theta} L_x(p)}.
    \end{equation*}

    Para hallar tanto el numerador como el denominador de $\lambda(x)$, distinguimos tres casos según el valor observado $x \in \{0, 1, \ldots, n\}$:
    \begin{itemize}
        \item \textbf{Caso $x = 0$:}
        La función de verosimilitud es $L_0(p) = (1 - p)^n$, que es estrictamente decreciente en $\left]0, 1\right[$. Por tanto:
        \begin{equation*}
            \sup\limits_{p \in \Theta_0} L_0(p) = \lim_{p \to 0^+} (1 - p)^n = 1, \qquad \sup\limits_{p \in \Theta} L_0(p) = \lim_{p \to 0^+} (1 - p)^n = 1,
        \end{equation*}
        de donde se obtiene directamente que:
        \begin{equation*}
            \lambda(0) = \frac{1}{1} = 1.
        \end{equation*}

        \item \textbf{Caso $x = n$:}
        La función de verosimilitud es $L_n(p) = p^n$, que es estrictamente creciente en $\left]0, 1\right[$. Por tanto:
        \begin{equation*}
            \sup\limits_{p \in \Theta_0} L_n(p) = L_n(p_0) = p_0^n, \qquad \sup\limits_{p \in \Theta} L_n(p) = \lim_{p \to 1^-} p^n = 1,
        \end{equation*}
        de donde se obtiene directamente que:
        \begin{equation*}
            \lambda(n) = \frac{p_0^n}{1} = p_0^n.
        \end{equation*}

        \item \textbf{Caso $x \in \{1, \ldots, n - 1\}$:}
        En este caso, $L_x(p) > 0$ para todo $p \in \left]0, 1\right[$ y podemos maximizar la log-verosimilitud derivando respecto a $p$:
        \begin{align*}
            \ln\left(L_x(p)\right) &= \ln\binom{n}{x} + x\ln(p) + (n - x)\ln(1 - p), \\
            \frac{d}{dp} \ln\left(L_x(p)\right) &= \frac{x}{p} - \frac{n - x}{1 - p} = \frac{x - np}{p(1 - p)} = 0 \Longleftrightarrow p = \frac{x}{n}.
        \end{align*}
        Observamos que $\frac{d}{dp} \ln(L_x(p)) > 0$ si $p < \frac{x}{n}$ y $\frac{d}{dp} \ln(L_x(p)) < 0$ si $p > \frac{x}{n}$. Por tanto, $p = \frac{x}{n} \in \Theta$ es el único máximo absoluto de $L_x(p)$ en $\Theta = \left]0, 1\right[$, lo que nos da el denominador:
        \begin{equation*}
            \sup\limits_{p \in \Theta} L_x(p) = L_x\left(\frac{x}{n}\right) = \binom{n}{x} \left(\frac{x}{n}\right)^x \left(1 - \frac{x}{n}\right)^{n - x}.
        \end{equation*}

        Para el numerador $\sup\limits_{p \in \Theta_0} L_x(p)$ (con $\Theta_0 = \left]0, p_0\right]$), distinguimos dos subcasos según la posición de $\frac{x}{n}$ respecto a $p_0$:
        \begin{itemize}
            \item Si $\frac{x}{n} \leq p_0$ (es decir, $x \leq np_0$), el máximo global $p = \frac{x}{n}$ pertenece a $\Theta_0$, por lo que:
            \begin{equation*}
                \sup\limits_{p \in \Theta_0} L_x(p) = L_x\left(\frac{x}{n}\right) \Longrightarrow \lambda(x) = \frac{L_x\left(\frac{x}{n}\right)}{L_x\left(\frac{x}{n}\right)} = 1.
            \end{equation*}

            \item Si $\frac{x}{n} > p_0$ (es decir, $x > np_0$), como $L_x(p)$ es estrictamente creciente en $\left]0, \frac{x}{n}\right]$ y $\Theta_0 \subseteq \left]0, \frac{x}{n}\right]$, el supremo sobre $\Theta_0$ se alcanza en el extremo superior $p = p_0$:
            \begin{equation*}
                \sup\limits_{p \in \Theta_0} L_x(p) = L_x(p_0) = \binom{n}{x} p_0^x (1 - p_0)^{n - x}.
            \end{equation*}
            En este caso, la razón de verosimilitudes es:
            \begin{equation*}
                \lambda(x) = \frac{L_x(p_0)}{L_x\left(\frac{x}{n}\right)} = \frac{\cancel{\binom{n}{x}} p_0^x (1 - p_0)^{n - x}}{\cancel{\binom{n}{x}} \left(\frac{x}{n}\right)^x \left(1 - \frac{x}{n}\right)^{n - x}} = \left(\frac{np_0}{x}\right)^x \left(\frac{n(1 - p_0)}{n - x}\right)^{n - x}.
            \end{equation*}
        \end{itemize}
    \end{itemize}

    Agrupando todos los casos, la razón de verosimilitudes $\lambda: \{0, 1, \ldots, n\} \to \mathbb{R}$ queda completamente definida por:
    \begin{equation*}
        \lambda(x) =
        \begin{cases}
            1, & \text{si } x \in \{0, 1, \ldots, \lfloor np_0 \rfloor\}, \\[6pt]
            \left(\dfrac{np_0}{x}\right)^x \left(\dfrac{n(1 - p_0)}{n - x}\right)^{n - x}, & \text{si } x \in \{\lfloor np_0 \rfloor + 1, \ldots, n - 1\}, \\[10pt]
            p_0^n, & \text{si } x = n.
        \end{cases}
    \end{equation*}

    Estudiamos ahora la monotonía de la sucesión $\{\lambda(x)\}_{x=0}^n$:
    \begin{itemize}
        \item Para $x \in \{0, 1, \ldots, \lfloor np_0 \rfloor\}$, $\lambda(x) = 1$ es constante.
        \item Para el salto de $x_0 = \lfloor np_0 \rfloor$ a $x_0 + 1 = \lfloor np_0 \rfloor + 1$ (si $x_0 + 1 < n$), observamos que la función auxiliar $g(t) = \left(\frac{np_0}{t}\right)^t \left(\frac{n(1 - p_0)}{n - t}\right)^{n - t}$ definida en $\left]0, n\right[$ cumple $g(np_0) = 1$ y tiene derivada logarítmica:
        \begin{align*}
            \ln g(t) &= t\ln(np_0) - t\ln t + (n - t)\ln(n(1 - p_0)) - (n - t)\ln(n - t), \\
            \frac{d}{dt}\ln g(t) &= \ln(np_0) - \ln t - 1 - \ln(n(1 - p_0)) + \ln(n - t) + 1 = \ln\left(\frac{p_0(n - t)}{t(1 - p_0)}\right) < 0 \qquad \forall t \in \left]np_0, n\right[,
        \end{align*}
        ya que $t > np_0 \iff p_0(n - t) < t(1 - p_0)$. Por tanto, $g(t)$ es estrictamente decreciente en $\left[np_0, n\right[$, lo que garantiza directamente que:
        \begin{equation*}
            \lambda(\lfloor np_0 \rfloor) = 1 = g(np_0) > g(\lfloor np_0 \rfloor + 1) = \lambda(\lfloor np_0 \rfloor + 1) > \cdots > \lambda(n - 1).
        \end{equation*}
        \item Finalmente, en el último punto $x = n$, basta comprobar que $\lambda(n - 1) > \lambda(n) = p_0^n$:
        \begin{equation*}
            \frac{\lambda(n - 1)}{\lambda(n)} = \frac{\left(\frac{np_0}{n - 1}\right)^{n - 1} \left(n(1 - p_0)\right)^1}{p_0^n} = \left(\frac{n}{n - 1}\right)^{n - 1} \frac{n(1 - p_0)}{p_0} > 1
        \end{equation*}
        (en el caso en que $n - 1 > np_0$, es decir, $\frac{1 - p_0}{p_0} > \frac{1}{n}$, pues $\left(\frac{n}{n - 1}\right)^{n - 1} > 1$ y $\frac{n(1 - p_0)}{p_0} > 1$).
    \end{itemize}

    En conclusión, los valores $\lambda(x)$ son iguales a $1$ para $x \leq np_0$ y estrictamente decrecientes para los enteros $x \geq \lfloor np_0 \rfloor$ hasta alcanzar el mínimo $\lambda(n) = p_0^n$ (véase la Figura~\ref{fig:razon_verosimilitudes_binomial}).

Así, el test de razón de verosimilitudes se reescribe en términos de $X$ como:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } x > k, \\
            0, & \text{si } x \leq k.
        \end{cases}
    \end{equation*}

    Determinamos ahora el valor crítico $k \in \{\lfloor np_0 \rfloor, \ldots, n\}$ imponiendo el nivel de significación $\alpha \in \left]0, 1\right[$:
    \begin{align*}
        \alpha &\geq \sup_{p \in \Theta_0} \beta_\varphi(p) = \sup_{p \in \Theta_0} E_p[\varphi(X)] = \sup_{p \in \Theta_0} P_p[X > k] = \sup_{p \in \Theta_0} \left(\sum_{t = k + 1}^{n} P_p[X = t]\right).
    \end{align*}

    Veamos ahora que, fijado $t \in \{k + 1, \ldots, n\}$, la función $p \mapsto P_p[X = t]$ es estrictamente creciente en $\Theta_0 = \left]0, p_0\right]$. Para ello, si tomamos:
    \begin{equation*}
        h(p) = P_p[X = t] = \binom{n}{t} p^t (1 - p)^{n - t} \Longrightarrow \ln h(p) = \ln\binom{n}{t} + t\ln p + (n - t)\ln(1 - p),
    \end{equation*}
    derivando respecto a $p$ obtenemos:
    \begin{equation*}
        \frac{d}{dp} \ln h(p) = \frac{t}{p} - \frac{n - t}{1 - p} = \frac{t - np}{p(1 - p)}.
    \end{equation*}
    Observamos ahora que, para todo $p \in \Theta_0 = \left]0, p_0\right]$ y todo $t \geq k + 1$, se cumple:
    \begin{equation*}
        t \geq k + 1 \geq \lfloor np_0 \rfloor + 1 > np_0 \geq np,
    \end{equation*}
    por lo que $t - np > 0$ y, en consecuencia, $\frac{d}{dp}\ln h(p) > 0$ en $\Theta_0$. De donde deducimos que $h(p) = P_p[X = t]$ es estrictamente creciente en $\Theta_0$ para cada $t \in \{k + 1, \ldots, n\}$, y por tanto también lo es su suma. Así, el supremo anterior se alcanza en el extremo superior $p = p_0$:
    \begin{equation*}
        \alpha \geq \sup_{p \in \Theta_0} \beta_\varphi(p) = \sup_{p \in \Theta_0} \left(\sum_{t = k + 1}^{n} P_p[X = t]\right) = \sum_{t = k + 1}^{n} P_{p_0}[X = t] = \sum_{t = k + 1}^{n} \binom{n}{t} p_0^t (1 - p_0)^{n - t}.
    \end{equation*}

    Dado que la función de potencia para $p \in \Theta_1 = \left]p_0, 1\right[$ viene dada por:
    \begin{equation*}
        \beta_\varphi(p) = E_p[\varphi(X)] = P_p[X > k] = \sum_{t = k + 1}^{n} \binom{n}{t} p^t (1 - p)^{n - t}, \qquad \forall p \in \left]p_0, 1\right[,
    \end{equation*}
    y esta expresión es decreciente en $k$, para maximizar la potencia del test en $H_1$ elegimos como punto crítico el menor entero $k \in \{\lfloor np_0 \rfloor, \ldots, n\}$ que cumpla:
    \begin{equation*}
        \sum_{t = k + 1}^{n} \binom{n}{t} p_0^t (1 - p_0)^{n - t} \leq \alpha
    \end{equation*}

    Con este valor de $k$, el test de razón de verosimilitudes de nivel $\alpha$ queda:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } x > k, \\
            0, & \text{si } x \leq k,
        \end{cases}
    \end{equation*}
    y su potencia frente a la hipótesis alternativa es:
    \begin{equation*}
        \beta_\varphi(p) = \sum_{t = k + 1}^{n} \binom{n}{t} p^t (1 - p)^{n - t}, \qquad \forall p \in \left]p_0, 1\right[.
    \end{equation*}

        f_\theta(x) = \theta x^{\theta-1}, \qquad 0<x<1
    \end{equation*}
    Basándose en una observación de $X$, deducir el test de razón de verosimilitudes de tamaño arbitrario para contrastar
    \begin{align*}
        \begin{cases}
            H_0: & \theta \leq \theta_0 \\
            H_1: & \theta > \theta_0
        \end{cases}
    \end{align*}

    Planteamos el contraste de hipótesis anterior sobre el parámetro $\theta \in \Theta$, donde:
    \begin{equation*}
        \Theta_0 = \left]0, \theta_0\right], \qquad \Theta_1 = \left]\theta_0, +\infty\right[, \qquad \Theta = \Theta_0 \cup \Theta_1 = \left]0, +\infty\right[,
    \end{equation*}
    con $\theta_0 > 0$ fijo.

    Dado que disponemos de una única observación $X = x \in \left]0, 1\right[$ (es decir, una muestra aleatoria simple de tamaño $1$), la función de verosimilitud coincide directamente con la función de densidad:
    \begin{equation*}
        L_x(\theta) = f_\theta(x) = \theta x^{\theta - 1}, \qquad \forall \theta \in \Theta.
    \end{equation*}

    Empleamos el Test de Razón de Verosimilitudes (TRV) de tamaño $\alpha \in \left]0, 1\right[$ para contrastar las hipótesis, cuya expresión es:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } \lambda(x) < c, \\
            0, & \text{si } \lambda(x) \geq c,
        \end{cases}
    \end{equation*}
    donde $c \in \left]0, 1\right]$ es una constante que se determina imponiendo el tamaño $\alpha$ y $\lambda(x)$ es la razón de verosimilitudes:
    \begin{equation*}
        \lambda(x) = \frac{\sup\limits_{\theta \in \Theta_0} L_x(\theta)}{\sup\limits_{\theta \in \Theta} L_x(\theta)}.
    \end{equation*}

    Para hallar el denominador, estudiamos el comportamiento de $L_x(\theta)$ sobre el espacio $\Theta = \left]0, +\infty\right[$. Como $L_x(\theta) > 0$ para todo $\theta \in \Theta$ (al ser $x \in \left]0, 1\right[$) y el logaritmo neperiano es estrictamente creciente, derivamos la log-verosimilitud respecto a $\theta$:
    \begin{align*}
        \ln\left(L_x(\theta)\right) &= \ln(\theta) + (\theta - 1)\ln(x), \\
        \frac{d}{d\theta} \ln\left(L_x(\theta)\right) &= \frac{1}{\theta} + \ln(x) = 0 \Longleftrightarrow \theta = -\frac{1}{\ln(x)} = \frac{1}{-\ln(x)} > 0.
    \end{align*}
    Observamos que $\frac{d}{d\theta} \ln(L_x(\theta)) > 0$ si $\theta < -\frac{1}{\ln(x)}$ y $\frac{d}{d\theta} \ln(L_x(\theta)) < 0$ si $\theta > -\frac{1}{\ln(x)}$. Por tanto, $\widehat{\theta} = -\frac{1}{\ln(x)} \in \Theta$ es el único máximo absoluto de $L_x(\theta)$ en $\Theta = \left]0, +\infty\right[$, lo que nos proporciona el denominador:
    \begin{equation*}
        \sup\limits_{\theta \in \Theta} L_x(\theta) = L_x\left(-\frac{1}{\ln(x)}\right) = -\frac{1}{\ln(x)} x^{-\frac{1}{\ln(x)} - 1} = -\frac{1}{x\ln(x)} e^{\ln(x) \cdot \left(-\frac{1}{\ln(x)}\right)} = -\frac{1}{e x \ln(x)}.
    \end{equation*}

    Calculamos ahora el numerador $\sup\limits_{\theta \in \Theta_0} L_x(\theta)$, distinguiendo dos casos según la posición de $-\frac{1}{\ln(x)}$ respecto a $\theta_0$ (notando que, como $\ln(x) < 0$ para $x \in \left]0, 1\right[$):
    \begin{equation*}
        -\frac{1}{\ln(x)} \leq \theta_0 \Longleftrightarrow -\frac{1}{\theta_0} \geq \ln(x) \Longleftrightarrow x \leq e^{-\nicefrac{1}{\theta_0}}.
    \end{equation*}
    \begin{itemize}
        \item \textbf{Si $0 < x \leq e^{-\nicefrac{1}{\theta_0}}$:} entonces $-\frac{1}{\ln(x)} \leq \theta_0$, por lo que el máximo global pertenece a $\Theta_0 = \left]0, \theta_0\right]$ y se cumple:
        \begin{equation*}
            \sup\limits_{\theta \in \Theta_0} L_x(\theta) = L_x\left(-\frac{1}{\ln(x)}\right) \Longrightarrow \lambda(x) = \frac{L_x\left(-\frac{1}{\ln(x)}\right)}{L_x\left(-\frac{1}{\ln(x)}\right)} = 1.
        \end{equation*}

        \item \textbf{Si $e^{-\nicefrac{1}{\theta_0}} < x < 1$:} entonces $-\frac{1}{\ln(x)} > \theta_0$. Como $L_x(\theta)$ es estrictamente creciente en $\left]0, -\frac{1}{\ln(x)}\right]$ y $\Theta_0 \subset \left]0, -\frac{1}{\ln(x)}\right]$, el supremo sobre $\Theta_0$ se alcanza en el extremo superior $\theta = \theta_0$:
        \begin{equation*}
            \sup\limits_{\theta \in \Theta_0} L_x(\theta) = L_x(\theta_0) = \theta_0 x^{\theta_0 - 1}.
        \end{equation*}
        En este caso, la razón de verosimilitudes es:
        \begin{equation*}
            \lambda(x) = \frac{L_x(\theta_0)}{L_x\left(-\frac{1}{\ln(x)}\right)} = \frac{\theta_0 x^{\theta_0 - 1}}{-\dfrac{1}{e x \ln(x)}} = -e \theta_0 x^{\theta_0} \ln(x).
        \end{equation*}
    \end{itemize}

    En resumen, la función $\lambda(x)$ queda definida a trozos en $\left]0, 1\right[$ por:
    \begin{equation*}
        \lambda(x) =
        \begin{cases}
            1, & \text{si } 0 < x \leq e^{-\nicefrac{1}{\theta_0}}, \\[6pt]
            -e \theta_0 x^{\theta_0} \ln(x), & \text{si } e^{-\nicefrac{1}{\theta_0}} < x < 1.
        \end{cases}
    \end{equation*}

    Recordamos que el test de razón de verosimilitudes es:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } \lambda(x) < c, \\[2pt]
            0, & \text{si } \lambda(x) \geq c.
        \end{cases}
    \end{equation*}

    Puesto que no conocemos directamente la distribución de $\lambda(X)$ bajo $H_0$, estudiamos la monotonía de $\lambda(x)$ en $\left]e^{-\nicefrac{1}{\theta_0}}, 1\right[$ para poder expresar el test en función de $x$ y así determinar el valor crítico $k'$ que imponga tamaño $\alpha$. Como $\lambda(x)>0$ y el logaritmo neperiano es estrictamente creciente, estudiamos la monotonía de $\ln(\lambda(x))$:
    \begin{align*}
        \ln(\lambda(x)) &= \ln(-e \theta_0 x^{\theta_0} \ln(x)) = \ln(-e) + \ln(\theta_0) + \theta_0 \ln(x) + \ln(\ln(x))\\
        \frac{d}{dx} \ln(\lambda(x)) &= \frac{\theta_0}{x} + \frac{1}{x \ln(x)} = \frac{\theta_0 \ln(x) + 1}{x \ln(x)}.
    \end{align*}
    
    Para todo $x \in \left]e^{-\nicefrac{1}{\theta_0}}, 1\right[$ se tiene $\ln(x) > -\frac{1}{\theta_0}$, luego $\theta_0 \ln(x) + 1 > 0$ y, por tanto, $\lambda'(x) < 0$. Así, $\lambda(x)$ es estrictamente decreciente en $\left]e^{-\nicefrac{1}{\theta_0}}, 1\right[$, con:
    \begin{equation*}
        \lim\limits_{x \to \left(e^{-\nicefrac{1}{\theta_0}}\right)^+} \lambda(x) = 1\qquad \lim\limits_{x \to 1^-} \lambda(x) = 0
    \end{equation*}

Así, el test de razón de verosimilitudes se reescribe equivalentemente en términos de $X$ como:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } x > k', \\
            0, & \text{si } x \leq k'.
        \end{cases}
    \end{equation*}

    Determinamos ahora el valor crítico $k'$ imponiendo que el test tenga tamaño $\alpha \in \left]0, 1\right[$:
    \begin{align*}
        \alpha &= \sup_{\theta \in \Theta_0} \beta_\varphi(\theta) = \sup_{\theta \leq \theta_0} E_\theta[\varphi(X)] = \sup_{\theta \leq \theta_0} P_\theta[X > k'] = \sup_{\theta \leq \theta_0} \int_{k'}^{1} \theta x^{\theta - 1} \, dx =\\&= \sup_{\theta \leq \theta_0} \Big[x^\theta\Big]_{k'}^{1} = \sup_{\theta \leq \theta_0} \left(1 - (k')^\theta\right).
    \end{align*}
    Dado que $k' \in \left]0, 1\right[$, la función $\theta \mapsto (k')^\theta$ es estrictamente decreciente en $\theta$, por lo que $\theta \mapsto 1 - (k')^\theta$ es estrictamente creciente y alcanza su supremo sobre $\Theta_0 = \left]0, \theta_0\right]$ en el extremo superior $\theta = \theta_0$:
    \begin{equation*}
        \alpha = 1 - (k')^{\theta_0} \Longleftrightarrow (k')^{\theta_0} = 1 - \alpha \Longleftrightarrow k' = (1 - \alpha)^{\nicefrac{1}{\theta_0}}.
    \end{equation*}
    
    Por tanto, el test de razón de verosimilitudes de tamaño $\alpha \in \left]0, 1\right[$ queda completamente determinado por:
    \begin{equation*}
        \varphi(x) =
        \begin{cases}
            1, & \text{si } x > (1 - \alpha)^{\nicefrac{1}{\theta_0}}, \\
            0, & \text{si } x \leq (1 - \alpha)^{\nicefrac{1}{\theta_0}},
        \end{cases}
    \end{equation*}
    
    Su función de potencia frente a la hipótesis alternativa es:
    \begin{equation*}
        \beta_\varphi(\theta) = E_\theta[\varphi(X)] = P_\theta[X > k'] = 1 - (k')^\theta = 1 - (1 - \alpha)^{\nicefrac{\theta}{\theta_0}}, \qquad \forall \theta \in \left]\theta_0, +\infty\right[.
    \end{equation*}

\begin{ejercicio}
    A partir de las siguientes observaciones de muestras independientes de dos poblaciones normales, contrastar, al nivel de significación $0.01$, si la media de la primera población supera en al menos una unidad la media de la segunda.\newline

    \noindent
    Planteamos el constrate de hipótesis:
    \begin{equation*}
        \left\{\begin{array}{l}
            H_0 : \mu_1-\mu_2 \geq \mu_0 = 1 \\
            H_1 : \mu_1-\mu_2 < \mu_0
        \end{array}\right.
    \end{equation*}
    Sean $X\rightsquigarrow \cc{N}(\mu_1, \sigma^2)$, $Y\rightsquigarrow\cc{N}(\mu_2,\sigma^2)$, tenemos dos observaciones: $(x_1, \ldots, x_8)$ de $X$ y $(y_1, \ldots, y_9)$ de $Y$ (tomamos $n_1=8$ y $n_2 = 9$). Usaremos el estadístico:
    \begin{equation*}
        T = \frac{\overline{X}-\overline{Y}-(\mu_1-\mu_2)}{S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}}\rightsquigarrow t_{n_1+n_2-2} , \qquad S_p = \sqrt{\frac{(n_1-1)S_1^2 + (n_2-1)S_2^2}{n_1+n_2-2}}
    \end{equation*}
    Por lo que calculamos $t_{exp}$: 
    \begin{equation*}
        t_{exp} = \frac{\overline{x}-\overline{y}-\mu_0}{s_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}}
    \end{equation*}

    para lo que necesitaremos:
    \begin{equation*}
        \overline{x} = 129, \qquad \overline{y} = 126, \qquad s_1^2 = 79.142857, \qquad s_2^2 = 121
    \end{equation*}
    Por lo que:
    \begin{gather*}
        s_p = \sqrt{\frac{(n_1-1)s_1^2 + (n_2-1)s_2^2}{n_1+n_2-2}} = \sqrt{\frac{7\cdot 79.142857 + 8\cdot 121}{8+9-2}}  \approx 10.073066 \\

    \end{gather*}
    Sea $\alpha = 0.01$, vimos en teoría que la región crítica era:

        t_{exp} < -t_{n_1+n_2-2;\alpha} = -t_{15;0.01} = 2.6025
    \end{equation*}
    Por tanto, como $0.408611 < 2.6025$ estamos en la región crítica, por lo que podemos rechazar a nivel de significación $0.01$ que la media de la primera población supera en al menos una unidad la media de la segunda.
