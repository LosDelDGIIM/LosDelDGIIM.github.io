\section{Estimación puntual. Insesgadez y mínima varianza}

\begin{ejemplo}
    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de una variable aleatoria $X$ con distribución $\cc{P}(\lm)$, $\lm > 0$. La media muestral $\ol{X}$ es un estimador de $\lm$. Además, cualquier función medible de la muestra que sea independiente del parámetro $\lm$ y tome valores positivos es un estimador del parámetro.
    \begin{equation*}
        \ol{X} = \frac{1}{n} \sum_{i=1}^{n} X_i \geq 0
    \end{equation*}
    donde la positividad de $\ol{X}$ se debe a que $X_i \geq 0$ para todo $i = 1, \ldots, n$.

    Por lo tanto, $\ol{X}$ es un estimador de $\lm$.
\end{ejemplo}

\begin{ejemplo}
    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de una variable aleatoria $X$ con distribución $\cc{B}(1, p)$, $p \in \left]0, 1\right[$. La media muestral $\ol{X}$ es un estimador de $p$. Además, cualquier función medible de la muestra que sea independiente del parámetro $p$ y tome valores en el intervalo $(0, 1)$ es un estimador del parámetro.
    \begin{equation*}
        \ol{X} = \frac{1}{n} \sum_{i=1}^{n} X_i \in \left[0, 1\right]
    \end{equation*}
    donde la pertenencia al intervalo $\left[0, 1\right]$ se debe a que $X_i \in \{0, 1\}$ para todo valor $i = 1, \ldots, n$. Estrictamente hablando, como $p$ no puede tomar los valores $0$ y $1$, entonces $\ol{X}$ no podría tomar esos valores. No obstante, estos valores son muy particulares, puesto que implicarían que todos los $X_i$ toman el mismo valor, lo que podría hacernos pensar que no se trata de un experimento aleatorio. Por tanto, en estos casos, que también contenga a los extremos del intervalo no es un problema.\\
    
    Por lo tanto, $\ol{X}$ es un estimador de $p$.
\end{ejemplo}

\begin{prop}
    Sea $X$ una v.a. con función de distribución en una familia de distribuciones paramétricas, $F \in \{F_\theta : \theta \in \Theta\}$ y $(X_1, \ldots, X_n)$ una m.a.s. de $X$. Sea $T = T(X_1, \ldots, X_n)$ un estimador de $\theta$. El error cuadrático medio (ECM) de $T$ se define como:
    \begin{equation*}
        ECM_T(\theta) := E_\theta\left[(T - \theta)^2\right]
    \end{equation*}

    El ECM puede descomponerse en términos de la varianza y una función denominada sesgo:
    \begin{equation*}
        ECM_T(\theta) = Var_\theta[T] + B^2_T(\theta)\qquad \text{donde } B_T(\theta) := E_\theta[T] - \theta
    \end{equation*}
    \begin{proof}
        Tenemos que:
        \begin{align*}
            ECM_T(\theta)
            &= E_\theta\left[(T - \theta)^2\right] 
            = E_\theta\left[(T - E_\theta[T] + E_\theta[T] - \theta)^2\right] \\
            &= E_\theta\left[(T - E_\theta[T])^2 + 2(T - E_\theta[T])(E_\theta[T] - \theta) + (E_\theta[T] - \theta)^2\right] \\
            &= Var_\theta[T] + 2(E_\theta[T] - \theta)E_\theta[T - E_\theta[T]] + (E_\theta[T] - \theta)^2 \\
            &= Var_\theta[T] + 2(E_\theta[T] - \theta)\cancel{(E_\theta[T] - E_\theta[T])} + (E_\theta[T] - \theta)^2 \\
            &= Var_\theta[T] + (E_\theta[T] - \theta)^2 \\
            &= Var_\theta[T] + B^2_T(\theta)
        \end{align*}
    \end{proof}
\end{prop}

\begin{ejemplo}
    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de alguna población. Probar que, en general, la cuasidesviación típica no es un estimador insesgado de la desviación típica poblacional.
    \begin{proof}
        Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de $X\rightsquigarrow \cc{N}(\mu, \sigma^2)$, $\mu \in \bb{R}$ y $\sigma^2 > 0$. Sabemos que $E[S^2] = \sigma^2$. Queremos ver si $E[S] = \sigma$. Sabemos que:
        \begin{align*}
            Y = \frac{(n-1)S^2}{\sigma^2} \sim \chi^2(n-1)
        \end{align*}
        Por tanto:
        \begin{align*}
            E\left[\sqrt{Y}\right] &= \int_0^\infty \sqrt{y} f_Y(y) dy = \int_0^\infty \sqrt{y}\dfrac{1}{2^{(n-1)/2}\Gamma((n-1)/2)} y^{(n-3)/2} e^{-y/2} dy
            =\\&= \dfrac{1}{2^{(n-1)/2}\Gamma((n-1)/2)} \int_0^\infty y^{n/2-1} e^{-y/2} dy
            \AstIg\\&\AstIg \dfrac{1}{2^{(n-1)/2}\Gamma((n-1)/2)} \cdot \dfrac{\Gamma(n/2)}{(1/2)^{n/2}}
        \end{align*}
        donde en $(\ast)$ hemos aplicado que la función de densidad de la distribución $\chi^2(n)$ integra $1$. Por tanto:
        \begin{align*}
            E[\sqrt{Y}] &= E\left[\sqrt{\frac{(n-1)S^2}{\sigma^2}}\right] = \dfrac{\sqrt{n-1}E[S]}{\sigma}
        \end{align*}

        Por tanto:
        \begin{align*}
            E[S] = \dfrac{\sigma}{\sqrt{n-1}} E[\sqrt{Y}] = \dfrac{\sigma}{\sqrt{n-1}} \cdot \dfrac{1}{2^{(n-1)/2}\Gamma((n-1)/2)} \cdot \dfrac{\Gamma(n/2)}{(1/2)^{n/2}}
            \neq \sigma
        \end{align*}
        Por tanto, $S$ no es un estimador insesgado de $\sigma$.
    \end{proof}
\end{ejemplo}

\begin{ejemplo}
    Sea $(X_1, \ldots, X_n)$ una m.a.s. de una distribución binomial $\cc{B}(1, p)$. Probar que
    \begin{equation*}
        \dfrac{T^2 - T}{n^2 - n}
    \end{equation*}
    con $T = \sum\limits_{i=1}^{n} X_i$ es un estimador insesgado para la función paramétrica $g(p) = p^2$. Probar, además, que para $n = 1$ no existe un estimador insesgado de $p^2$.
    \begin{proof}
        Es directo ver que se trata de un estimador. Veamos que es insesgado, para lo que calculamos su esperanza:
        \begin{align*}
            E\left[\dfrac{T^2 - T}{n^2 - n}\right]
            &= \dfrac{1}{n^2 - n} E[T^2 - T]
            = \dfrac{1}{n^2 - n} \left(E[T^2] - E[T]\right)
            \AstIg\\&\AstIg \dfrac{1}{n^2 - n} \left(Var[T] + E[T]^2 - E[T]\right) = \dfrac{1}{n^2 - n} \left(np(1-p) + (np)^2 - np\right) =\\&= \dfrac{1}{n^2 - n} \left(n^2p^2 - np^2\right) = p^2
        \end{align*}
        donde en $(\ast)$ hemos aplicado la identidad $Var[T] = E[T^2] - E[T]^2$. Por tanto, $\dfrac{T^2 - T}{n^2 - n}$ es un estimador insesgado de $p^2$. Veamos ahora que para el valo de $n = 1$ no existe un estimador insesgado de $p^2$. Supongamos que existe un estimador insesgado $T(X)$ de $p^2$ basado en una muestra de tamaño $1$. Entonces, por definición de insesgadez, se cumple que:
        \begin{equation*}
            p^2 = E_p[T(X)] = T(0)(1-p) + T(1)p \Longrightarrow p^2 = T(0) - T(0)p + T(1)p
        \end{equation*}
        La expresión de la derecha es una ecuación polinómica de primer grado en $p$, mientras que $p^2$ es un polinomio de segundo grado en $p$. Llegamos por tanto a una contradicción, ya que no puede haber un polinomio de segundo grado igual a un polinomio de primer grado. Por tanto, no existe un estimador insesgado de $p^2$ basado en una muestra de tamaño $1$.
    \end{proof}
\end{ejemplo}

\begin{ejemplo}
    Sea $X$ una v.a. con distribución $\cc{P}(\lm)$, ¿existe algún estimador insesgado de $\nicefrac{1}{\lm}$ basado en una muestra de tamaño $1$?\\

    Supongamos que existe un estimador insesgado $T(X)$ de $\nicefrac{1}{\lm}$ basado en una muestra de tamaño $1$. Entonces, por definición de insesgadez, se cumple que:
    \begin{equation*}
        \dfrac{1}{\lm} = E_\lm[T(X)] = \sum_{x=0}^{\infty} T(x) \dfrac{e^{-\lm}\lm^x}{x!}
        \Longrightarrow
        e^{\lm} = \sum_{x=0}^{\infty} T(x) \dfrac{\lm^{x+1}}{x!}
    \end{equation*}

    Empleando la serie de Taylor de $e^{\lm}$, obtenemos:
    \begin{equation*}
        \sum_{x=0}^{\infty} \dfrac{\lm^x}{x!} = \sum_{x=0}^{\infty} T(x) \dfrac{\lm^{x+1}}{x!}
    \end{equation*}

    La primera sumatoria tiene un término independiente de $\lm$, mientras que la segunda sumatoria no. Como ha de ser cierta la igualdad para todo $\lm > 0$, se llega a una contradicción. Por tanto, no existe un estimador insesgado de $\nicefrac{1}{\lm}$ basado en una muestra de tamaño $1$.
\end{ejemplo}

\begin{teo}[Rao-Blackwell]
    Sea $X$ una v.a. con función de distribución en una familia de distribuciones paramétricas, $F \in \{F_\theta : \theta \in \Theta\}$, y $(X_1, \ldots, X_n)$ una m.a.s. de $X$. Si $T(X_1, \ldots, X_n)$ es un estadístico suficiente para la familia $\{F_\theta : \theta \in \Theta\}$ y $S(X_1, \ldots, X_n)$ es un estimador insesgado de $g(\theta)$ con momento de segundo orden finito, entonces:
    \begin{enumerate}
        \item $E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]$ es un estimador insesgado de $g(\theta)$ y tiene momento de segundo orden finito.
        \item $Var_\theta\left[E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]\right] \leq Var_\theta[S(X_1, \ldots, X_n)]$, $\forall \theta \in \Theta$.
    \end{enumerate}
    \begin{proof}
        Demostramos cada una de las afirmaciones por separado:
        \begin{enumerate}
            \item En primer lugar, veamos que $E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]$ es un estadístico. Es claro que $E_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]$ es un función de la m.a.s., pero hemos de ver que no depende del parámetro $\theta$:
            \begin{multline*}
                E_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]
                =\\= \int_{\chi^n} S(x_1, \ldots, x_n) f_{X_1, \ldots, X_n \mid T}(x_1, \ldots, x_n)\ dx_1 \ldots dx_n
            \end{multline*}
            Como $T$ es un estadístico suficiente, la función de densidad condicionada $f_{X_1, \ldots, X_n \mid T}(x_1, \ldots, x_n)$ no depende de $\theta$. Por tanto, $E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]$ no depende del parámetro $\theta$, luego es un estadístico. Además, como $S(X_1, \ldots, X_n)$ es un estimador de $g(\theta)$, toma valores en el rango de $g(\theta)$, luego al tomar la esperanza condicionada también tomará valores en el rango de $g(\theta)$, luego es un estimador de $g(\theta)$. Además, como $S$ es un estimador insesgado de $g(\theta)$, entonces $E_\theta[S(X_1, \ldots, X_n)] = g(\theta)$, y por la propiedad de la esperanza condicionada, tenemos que:
            \begin{align*}
                E_\theta[E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]] = E_\theta[S(X_1, \ldots, X_n)] = g(\theta)
            \end{align*}
            Por tanto, $E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]$ es un estimador insesgado de $g(\theta)$.

            Veamos ahora que tiene momento de segundo orden finito. Como $S$ tiene momento de segundo orden finito, entonces su momento de segundo orden condicionado también es finito, luego:
            \begin{multline*}
                Var_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)] =\\= E_\theta[S^2(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)] - E_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]^2 \geq 0
            \end{multline*}

            Por tanto:
            \begin{multline*}
                E_\theta[S^2(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)] \geq E_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]^2
            \end{multline*}

            Tomando esperanza:
            \begin{multline*}
                E_\theta[E_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]^2] \leq\\\leq E_\theta [E_\theta[S^2(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]]
                = E_\theta[S^2(X_1, \ldots, X_n)] < \infty
            \end{multline*}

            \item Empleamos la descomposición de la varianza:
            \begin{align*}
                &Var_\theta[S(X_1, \ldots, X_n)]
                =\\&= E_\theta[Var_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]] + Var_\theta[E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]]
            \end{align*}

            Como $E_\theta[Var_\theta[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]] \geq 0$, se sigue que:
            \begin{align*}
                Var_\theta[S(X_1, \ldots, X_n)] \geq Var_\theta[E[S(X_1, \ldots, X_n) \mid T(X_1, \ldots, X_n)]]
            \end{align*}


        \end{enumerate}
    \end{proof}
\end{teo}

\begin{ejemplo}
    Sea $(X_1, \ldots, X_n)$ una m.a.s. con distribución $\cc{B}(1, p)$, $p \in \left[0, 1\right]$. Encontrar el UMVUE para $p$.\\

    En el ejemplo de la página~\pageref{ejemplo:suficiencia-completitud-Bernoulli}, vimos que:
    \begin{equation*}
        T = \sum_{i=1}^{n} X_i
    \end{equation*}
    es un estadístico suficiente y completo para $p$. Además, $T\rightsquigarrow \cc{B}(n, p)$, por lo que:
    \begin{equation*}
        E[T] = np
        \Longrightarrow
        E\left[\dfrac{T}{n}\right] = p
    \end{equation*}

    Por tanto, de ser $S = \nicefrac{T}{n}$ estimador, es insesgado de $p$. Como $X$ toma valores en $\{0, 1\}$, entonces $T$ toma valores en $\{0, 1, \ldots, n\}$ y $S$ toma valores en $\left\{0, \nicefrac{1}{n}, \ldots, 1\right\}$. Por tanto, $S$ es un estimador insesgado de $p$. Nos falta ver que tiene momento de segundo orden finito. Como $S$ es una función de $T$, entonces:
    \begin{equation*}
        Var[S] = Var\left[\dfrac{T}{n}\right] = \dfrac{1}{n^2} Var[T] = \dfrac{1}{n^2} np(1-p)<\infty
    \end{equation*}

    Por tanto, $S$ es un estimador insesgado de $p$ con momento de segundo orden finito. Por tanto, por el teorema de Lehmann-Scheffé, el UMVUE de $p$ es:
    \begin{equation*}
        E[S \mid T] = E\left[\dfrac{T}{n} \mid T\right] = \dfrac{T}{n} 
        = \dfrac{1}{n} \sum_{i=1}^{n} X_i = \ol{X}
    \end{equation*}

    Por tanto, el UMVUE de $p$ es la media muestral $\ol{X}$.
\end{ejemplo}

\begin{ejemplo}
    Sea $(X_1, \ldots, X_n)$ una m.a.s. con distribución $\cc{U}(0, \theta)$, $\theta > 0$. Encontrar el UMVUE para $\theta$ y $\nicefrac{1}{\theta}$.\\

    En el ejemplo de la página~\pageref{ejemplo:suficiencia-completitud-Uniforme}, vimos que:
    \begin{equation*}
        T = X_{(n)} = \max\{X_1, \ldots, X_n\}
    \end{equation*}
    es un estadístico suficiente y completo para $\theta$. Además, su función de densidad es:
    \begin{equation*}
        f_{X_{(n)}}(x) = \dfrac{n}{\theta^n} x^{n-1}\qquad \text{para } 0 < x < \theta
    \end{equation*}

    Por tanto:
    \begin{align*}
        E_\theta[X_{(n)}] &= \int_0^\theta x \dfrac{n}{\theta^n} x^{n-1} dx = \dfrac{n}{\theta^n} \int_0^\theta x^n dx = \dfrac{n}{\theta^n} \left[\dfrac{x^{n+1}}{n+1}\right]_0^\theta = \dfrac{n}{\theta^n} \cdot \dfrac{\theta^{n+1}}{n+1} = \dfrac{n}{n+1} \theta
    \end{align*}

    Por tanto, de ser $S = \dfrac{n+1}{n} X_{(n)}$ estimador, es insesgado de $\theta$. Como $X_{(n)}$ toma valores en $\left]0, \theta\right[$, entonces $S$ toma valores en $\left]0, \dfrac{n+1}{n}\theta\right[\subset \bb{R}^+$. Por tanto, $S$ es un estimador insesgado de $\theta$. Nos falta ver que tiene momento de segundo orden finito:
    \begin{align*}
        Var[S] = Var\left[\dfrac{n+1}{n} X_{(n)}\right] = \left(\dfrac{n+1}{n}\right)^2 Var[X_{(n)}] < \infty
    \end{align*}
    Por tanto, por el teorema de Lehmann-Scheffé, el UMVUE de $\theta$ es:
    \begin{equation*}
        E[S \mid T] = E\left[\dfrac{n+1}{n} X_{(n)} \mid X_{(n)}\right] = \dfrac{n+1}{n} X_{(n)}
    \end{equation*}~\\

    Buscamos ahora el UMVUE de $\nicefrac{1}{\theta}$. Para ello, buscamos una función $h(T)$ tal que:
    \begin{equation*}
        E_\theta[h(T)] = \frac{1}{\theta}
    \end{equation*}

    Tenemos que:
    \begin{align*}
        E_\theta[h(T)] &= \int_0^\theta h(x) \dfrac{n}{\theta^n} x^{n-1} dx = \dfrac{n}{\theta^n} \int_0^\theta h(x) x^{n-1} dx = \frac{1}{\theta}
    \end{align*}
    Multiplicando por $\theta^n$ y dividiendo por $n$, obtenemos:
    \begin{equation*}
        \int_0^\theta h(x) x^{n-1} dx = \frac{\theta^{n-1}}{n}
    \end{equation*}
    
    Por el teorema fundamental del cálculo, tenemos que:
    \begin{equation*}
        h(\theta) \theta^{n-1} = \frac{(n-1)\theta^{n-2}}{n} \Longrightarrow h(\theta) = \frac{n-1}{n} \cdot \frac{1}{\theta}
    \end{equation*}

    Por tanto, consideramos el estadístico:
    \begin{equation*}
        S = \frac{n-1}{n} \cdot \frac{1}{X_{(n)}}
    \end{equation*}

    Como $X_{(n)}$ toma valores en $\left]0, \theta\right[$, entonces $S$ toma valores en un subconjunto de $\bb{R}^+$, luego es un estimador de $\nicefrac{1}{\theta}$. Además, veamos que es insesgado:
    \begin{align*}
        E_\theta[S] &= E_\theta\left[\frac{n-1}{n} \cdot \frac{1}{X_{(n)}}\right] = \frac{n-1}{n} \int_0^\theta \frac{1}{x} \dfrac{n}{\theta^n} x^{n-1} dx =\\&= \frac{n-1}{\theta^n} \int_0^\theta x^{n-2} dx = \frac{n-1}{\theta^n} \left[\frac{x^{n-1}}{n-1}\right]_0^\theta = \frac{1}{\theta}
    \end{align*}

    Por tanto, $S$ es un estimador insesgado de $\nicefrac{1}{\theta}$. Nos falta ver que tiene momento de segundo orden finito:
    \begin{align*}
        Var[S] = Var\left[\frac{n-1}{n} \cdot \frac{1}{X_{(n)}}\right] = \left(\frac{n-1}{n}\right)^2 Var\left[\frac{1}{X_{(n)}}\right]
    \end{align*}

    Comprobamos que $Var\left[\frac{1}{X_{(n)}}\right] < \infty$:
    \begin{align*}
        E\left[\frac{1}{X_{(n)}^2}\right] &= \int_0^\theta \frac{1}{x^2} \dfrac{n}{\theta^n} x^{n-1} dx = \frac{n}{\theta^n} \int_0^\theta x^{n-3} dx = \frac{n}{\theta^n} \left[\frac{x^{n-2}}{n-2}\right]_0^\theta = \frac{n}{n-2} \cdot \frac{1}{\theta^2} < \infty
    \end{align*}

    Por tanto, $S$ tiene momento de segundo orden finito. Por tanto, por el teorema de Lehmann-Scheffé, el UMVUE de $\nicefrac{1}{\theta}$ es:
    \begin{equation*}
        E[S \mid T] = E\left[\frac{n-1}{n} \cdot \frac{1}{X_{(n)}} \mid X_{(n)}\right] = \frac{n-1}{n} \cdot \frac{1}{X_{(n)}}
    \end{equation*}
\end{ejemplo}

\begin{ejercicio}
    Sea $(X_1, \ldots, X_n)$ una m.a.s. con distribución $\cc{P}(\lm)$, $\lm > 0$. Encontrar el UMVUE de $\lm^s$, $s \in \bb{N}$.\\
    
    Buscamos un estadístico suficiente y completo. Por el Ejercicio~\ref{ej:2.3.7}.\ref{ej:2.3.7:poisson}, sabemos que:
    \begin{equation*}
        T = \sum_{i=1}^{n} X_i
    \end{equation*}
    es un estadístico suficiente y completo para $\lm$. Para encontrar el UMVUE de $\lm^s$, buscamos una función $h(T)$ tal que:
    \begin{equation*}
        E_\lm[h(T)] = \lm^s
    \end{equation*}

    Como $T\rightsquigarrow \cc{P}(n\lm)$, tenemos que:
    \begin{align*}
        E_\lm[h(T)] &= \sum_{t=0}^{\infty} h(t) \dfrac{e^{-n\lm}(n\lm)^t}{t!} = \lm^s
        \Longrightarrow
        e^{n\lm} \lm^s = \sum_{t=0}^{\infty} h(t) \dfrac{(n\lm)^t}{t!}
    \end{align*}

    Empleando la serie de Taylor de $e^{n\lm}$, obtenemos:
    \begin{align*}
        \sum_{t=0}^{\infty} h(t) \dfrac{n^t}{t!}\lm^t
        = \sum_{t=0}^{\infty} h(t) \dfrac{(n\lm)^t}{t!}
        = \lm^s \sum_{t=0}^{\infty} \dfrac{(n\lm)^t}{t!}
        = \sum_{t=0}^{\infty} \dfrac{n^t}{t!} \lm^{s+t}
    \end{align*}

    En la sumatoria de la derecha, hacemos el cambio de variable $k = s + t$, de modo que $t = k - s$. Por tanto:
    \begin{align*}
        \sum_{t=0}^{\infty} h(t) \dfrac{n^t}{t!}\lm^t
        = \sum_{k=s}^{\infty} \dfrac{n^{k-s}}{(k-s)!} \lm^k
    \end{align*}

    Como la igualdad ha de cumplirse para todo $\lm > 0$, se sigue que:
    \begin{itemize}
        \item Para $t < s$:
        \begin{equation*}
            h(t) \frac{n^t}{t!} = 0 \Longrightarrow h(t) = 0
        \end{equation*}
        \item Para $t \geq s$:
        \begin{equation*}
            h(t) \frac{n^t}{t!} = \frac{n^{t-s}}{(t-s)!} \Longrightarrow h(t) = \frac{n^{t-s}}{(t-s)!} \cdot \frac{t!}{n^t} = \frac{t!}{(t-s)! n^s}
        \end{equation*}
    \end{itemize}

    Por tanto, consideramos el estadístico:
    \begin{equation*}
        S = h(T) = \begin{cases}
            0 & \text{si } T < s \\
            \dfrac{T!}{(T-s)! n^s} & \text{si } T \geq s
        \end{cases}
    \end{equation*}

    Como $T$ toma valores en $\bb{N}$, entonces $S$ toma valores en $\bb{R}^+$, luego es un estimador de $\lm^s$. Además, se ha construido de manera que es insesgado:
    \begin{align*}
        E_\lm[S] = E_\lm[h(T)] = \lm^s
    \end{align*}

    Nos falta ver que tiene momento de segundo orden finito:
    \begin{align*}
        Var[S] = Var[h(T)] < \infty
    \end{align*}
    Por ser $S$ un polinomio de $T$, y como $T$ tiene momentos de todos los órdenes finitos, se sigue que $S$ tiene momento de segundo orden finito. Por tanto, por el teorema de Lehmann-Scheffé, el UMVUE de $\lm^s$ es:
    \begin{equation*}
        E[S \mid T] = E[h(T) \mid T] = h(T) = \begin{cases}
            0 & \text{si } T < s \\
            \dfrac{T!}{(T-s)! n^s} & \text{si } T \geq s
        \end{cases}
    \end{equation*}
\end{ejercicio}
