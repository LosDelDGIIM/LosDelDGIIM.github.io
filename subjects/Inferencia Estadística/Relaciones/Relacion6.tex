\section{Estimación por intervalos de confianza}

\begin{ejercicio}
    Sea $\overline{X}$ la media de una muestra aleatoria de tamaño $n$ de una población $\cc{N}(\mu, 16)$ Encontrar el menor valor de $n$ para que $\left]\overline{X}-1,\overline{X}+1\right[$ sea un intervalo de confianza para $\mu$ al nivel de confianza $0.9$.\\

    \noindent
    Si $\overline{X}$ es la media de una m.a.s. de tamaño $n$ de una población $\cc{N}(\mu,16)$, tenemos entonces que $\overline{X}\rightsquigarrow\cc{N}(\mu, \frac{16}{n})$. Calculamos $n$ para que el intervalo $\left]\overline{X}-1,\overline{X}+1\right[$ sea un intervalo de confianza para $\mu$ a nivel $0.9$, buscando:
    \begin{align*}
        P_\mu&\left[\overline{X}-1 \leq \mu \leq \overline{X}+1\right] \geq 0.9 \Longleftrightarrow P_\mu[\mu-1\leq \overline{X}\leq \mu+1] \geq 0.9 \\
        &\stackrel{\text{tipif.}}{\Longleftrightarrow} P_\mu\left[\frac{-\sqrt{n}}{4} \leq Z \leq \frac{\sqrt{n}}{4}\right] \geq 0.9 
        \Longleftrightarrow 2P_\mu\left[Z\leq \frac{\sqrt{n}}{4}\right] - 1 \geq 0.9 \\
        &\Longleftrightarrow 2P_\mu\left[Z\leq \frac{\sqrt{n}}{4}\right] \geq 1.9 
        \Longleftrightarrow P_\mu\left[Z\leq \frac{\sqrt{n}}{4}\right] \geq 0.95 
    \end{align*}
    Consultando la tabla de la normal $\cc{N}(0,1)$, tenemos que la primera abscisa en la que se alcanza una probabilidad superior a $0.95$ es $1.65$, por lo que:
    \begin{equation*}
        \frac{\sqrt{n}}{4} \geq 1.65 \Longleftrightarrow \sqrt{n} \geq 6.6 \Longleftrightarrow n \geq 43.56
    \end{equation*}
    Por tanto, el menor valor de $n$ para el cual el intervalo $\left]\overline{X}-1,\overline{X}+1\right[$ es un intervalo de confianza para $\mu$ a nivel de confianza $0.9$ es $44$.
\end{ejercicio}

\begin{ejercicio}
    La altura en cm. de los individuos varones de una población sigue una distribución $\cc{N}(\mu,\ 56.25)$. Si en una muestra aleatoria simple de tamaño $12$ de dicha población se obtiene una altura media de $175$ cm., determinar un intervalo de confianza para $\mu$ al nivel de confianza $0.95$. ¿Qué tamaño de muestra es necesario para que el intervalo de confianza a dicho nivel tenga longitud menor que 1 cm?\\

    \noindent
    Buscamos un intervalo de confianza para $\mu$ en una población normal con varianza $\sigma_0$ conocida a nivel de confianza $1-\alpha = 0.95$ (por lo que $\alpha= 0.05$). Tenemos pues una muestra aleatoria simple $(X_1, \ldots, X_n)$ de $X\rightsquigarrow\cc{N}(\mu,56.25)$. Usaremos el método de la cantidad pivotal, usando para ello la función pivote (donde $n=12$):
    \begin{equation*}
        T(X_1, \ldots, X_n; \mu) = \frac{\overline{X}-\mu}{\nicefrac{\sigma_0}{\sqrt{n}}} \rightsquigarrow\cc{N}(0,1)
    \end{equation*}
    
    Comprobamos que cumple las condiciones necesarias para poder aplicar el método de la cantidad pivotal:
    \begin{itemize}
        \item Es estrictamente monótona respecto $\mu$.
        \begin{equation*}
            \frac{\partial T}{\partial \mu} = -\frac{\sqrt{n}}{\sigma_0} < 0
        \end{equation*}

        En efecto, es estrictamente decreciente respecto $\mu$.
        \item Si tomamos $\lm$ de forma que:
            \begin{equation*}
                \lm = \frac{\overline{X}-\mu}{\nicefrac{\sigma_0}{\sqrt{n}}} \Longrightarrow \mu = \overline{X} - \lm \frac{\sigma_0}{\sqrt{n}}
            \end{equation*}
    \end{itemize}
    Por tanto, el intervalo de confianza a considerar será:
    \begin{equation*}
        \left]\overline{X}-\lm_2 \frac{\sigma_0}{\sqrt{n}}, \overline{X}-\lm_1 \frac{\sigma_0}{\sqrt{n}}\right[
    \end{equation*}
    donde $\lm_1$ y $\lm_2$ están sujetos a la restricción $P_\mu[\lm_1<T<\lm_2]=1-\alpha$. Aunque no se pide explícitamente proporcionar el intervalo de confianza de mínima longitud esperada uniforme, se calcula este por ser el más interesante. Para ello, calculamos la longitud esperada del intervalo:
    \begin{equation*}
        E[L] = E\left[(\lm_2-\lm_1)\frac{\sigma_0}{\sqrt{n}}\right] = (\lm_2-\lm_1)\frac{\sigma_0}{\sqrt{n}}
    \end{equation*}
    Como $\nicefrac{\sigma_0}{\sqrt{n}}$ es una constante positiva, bastará minimizar la cantidad $\lm_2-\lm_1$ (sujeta a la restricción $P_\mu[\lm_1<T<\lm_2]=1-\alpha$) para minimizar la longitud del intervalo. La restricción mencionada puede reescribirse como: 
    \begin{equation*}
        1-\alpha = P_\mu[\lm_1<T<\lm_2] = F_Z(\lm_2) - F_Z(\lm_1)
    \end{equation*}
    Consideramos por el método de los multiplicadores de Lagrange:
    \begin{equation*}
        F(\lm_1,\lm_2) = (\lm_2-\lm_1) - \lm(F_Z(\lm_2) - F_Z(\lm_1) - (1-\alpha))
    \end{equation*}
    y buscamos los valores de $\lm_1$ y $\lm_2$ que minimicen la expresión, por lo que calculamos sus derivadas parciales:
    \begin{equation*}
        \dfrac{\partial F}{\partial \lm_1} = -1+\lm f_Z(\lm_1) \qquad \dfrac{\partial F}{\partial \lm_2} = 1-\lm f_Z(\lm_2)
    \end{equation*}
    Si igualamos ambas a cero y tratamos de despejar $\lm$:
    \begin{equation*}
        \left.\begin{array}{l}
            0 = \dfrac{\partial F}{\partial \lm_1} = -1+\lm f_Z(\lm_1) \\ \\
            0 = \dfrac{\partial F}{\partial \lm_2} = 1-\lm f_Z(\lm_2)
        \end{array}\right\} \Longrightarrow \left.\begin{array}{l}
            \lm = \dfrac{1}{f_Z(\lm_1)} \\ \\
            \lm = \dfrac{1}{f_Z(\lm_2)}
        \end{array}\right\} \Longrightarrow f_Z(\lm_1) = f_Z(\lm_2)
    \end{equation*}
    Como $f_Z$ es una función simétrica respecto al origen, las únicas posibilidades son bien $\lm_1 = \lm_2$ bien $\lm_1 = -\lm_2$. Como estos valores han de cumplir la restricción $P_\mu[\lm_1<T<\lm_2]=1-\alpha > 0$, la primera opción es imposible, por lo que tiene que ser $\lm_1 = -\lm_2$. Como han de dejar entre ellos un valor de $1-\alpha$, han de ser:
    \begin{equation*}
        \lm_2 = z_{\nicefrac{\alpha}{2}}, \qquad \lm_1 = -\lm_2 = -z_{\nicefrac{\alpha}{2}}
    \end{equation*}
    Por tanto, el intervalo de confianza a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]\overline{X}-z_{\nicefrac{\alpha}{2}}\frac{\sigma_0}{\sqrt{n}}, \overline{X}+z_{\nicefrac{\alpha}{2}}\frac{\sigma_0}{\sqrt{n}}\right[
    \end{equation*}
    A partir de los datos proporcionados, tenemos que $\alpha=0.05$, $n=12$, la desviación típica es $\sigma_0 = \sqrt{56.25} = 7.5$. Consultando la tabla de la normal $\cc{N}(0,1)$, tenemos que $z_{\nicefrac{\alpha}{2}} = z_{0.025} = 1.96$. Por tanto, el intervalo de confianza a nivel de confianza $0.95$ será:
    \begin{equation*}
        \left]\overline{X}-1.96\cdot \frac{7.5}{3.4641},\overline{X}+1.96\cdot \frac{7.5}{3.4641}\right[ = \left]\ol{X} - 4.2435, \ol{X} + 4.2435\right[
    \end{equation*}

    Como en el enunciado se afirma que $\overline{X}(x_1, \ldots, x_{12})=175$, el intervalo de confianza a nivel de confianza $0.95$ para esta muestra es:
    \begin{equation*}
        \left]175-4.2435, 175+4.2435\right[ = \left]170.7565, 179.2435\right[
    \end{equation*}


    Buscamos ahora el tamaño de la muestra para que la longitud del intervalo sea menor que $1$cm. Para ello, vemos que nuestro intervalo tiene longitud:
    \begin{equation*}
        2z_{\nicefrac{\alpha}{2}}\frac{\sigma_0}{\sqrt{n}} = 2\cdot 1.96 \cdot \frac{7.5}{\sqrt{n}} = \frac{29.4}{\sqrt{n}}  \leq 1 \Longleftrightarrow 29.4 \leq \sqrt{n} \Longleftrightarrow 864.36 \leq n
    \end{equation*}
    Por tanto, necesitaremos al menos una muestra de tamaño $865$ para que el intervalo de confianza a dicho nivel tenga longitud menor que 1 cm.
\end{ejercicio}

\begin{ejercicio}\label{ej:3_rel6}
    Una fábrica produce tornillos cuyo diámetro medio es $3$ mm. Se seleccionan aleatoria e independientemente 12 de estos tornillos y se miden sus diámetros, que resultan ser $3.01,\ 3.05,\ 2.99,\ 2.99,\ 3.00,\ 3.02,\ 2.98,\ 2.99,\ 2.97,\ 2.97,\ 3.02$ y $3.01$. Suponiendo que el diámetro es una variable aleatoria con distribución normal, determinar un intervalo de confianza para la varianza al nivel de confianza $0.99$, y una cota superior de confianza al mismo nivel. Interpretar los resultados en términos de la desviación típica del diámetro de los tornillos.\\


    Sea $(X_1, \ldots, X_n)$ una m.a.s. de $X\rightsquigarrow\cc{N}(3,\sigma^2)$, vamos a calcular un intervalo de confianza para $\sigma^2$ en una población normal con media $\mu_0$ conocida a nivel de confianza $1-\alpha$ mediante el método de la cantidad pivotal, usando la función pivote:
    \begin{equation*}
        T(X_1, \ldots, X_n;\sigma^2) = \frac{\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}}{\sigma^2} \rightsquigarrow\chi^2(n)
    \end{equation*}
    
    Veamos que cumple las condiciones necesarias para poder aplicar el método de la cantidad pivotal:
    \begin{itemize}
        \item Es estrictamente monótona en $\sigma^2$.
        
        \begin{equation*}
            \frac{\partial T}{\partial \sigma^2} = -\frac{\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}}{{\sigma^2}^2} < 0
        \end{equation*}

        En efecto, es estrictamente decreciente respecto $\sigma^2$.
        \item Si tenemos $\lm$ con:
            \begin{equation*}
                \lm = \frac{\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}}{\sigma^2} \rightsquigarrow\chi^2(n) \Longrightarrow \sigma^2 = \frac{\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}}{\lm}
            \end{equation*}
    \end{itemize}
    
    Por tanto, el intervalo de confianza a considerar será:
    \begin{equation*}
        \left]\frac{1}{\lm_2}\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2},\frac{1}{\lm_1}\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}\right[
    \end{equation*}
    donde $\lm_1$ y $\lm_2$ están sujetos a la restricción:
    \begin{equation*}
        1-\alpha \leq P_{\sigma^2}[\lm_1<T<\lm_2] = F_T(\lm_2) - F_T(\lm_1) \qquad (T\rightsquigarrow\chi^2(n))
    \end{equation*}

    Aunque no se pide explícitamente proporcionar el intervalo de confianza de mínima longitud esperada uniforme, se calcula este por ser el más interesante.
    Tenemos que la longitud esperada del intervalo es:
    \begin{equation*}
        E[L] = E\left[\left(\frac{1}{\lm_1}-\frac{1}{\lm_2}\right)\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}\right] = \left(\frac{1}{\lm_1}-\frac{1}{\lm_2}\right)E\left[\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2}\right]
    \end{equation*}
    Donde la última esperanza es una cantidad que no depende de $\lm_1,\lm_2$ y positiva, por lo que trataremos de minimizar simplemente $\nicefrac{1}{\lm_1}-\nicefrac{1}{\lm_2}$. Mediante el método de los multiplicadores de Lagrange:
    \begin{equation*}
        F(\lm_1,\lm_2) = \left(\frac{1}{\lm_1}-\frac{1}{\lm_2}\right) - \lm(F_T(\lm_2)-F_T(\lm_1)-(1-\alpha))
    \end{equation*}

    Calculamos las derivadas parciales:
    \begin{equation*}
        \dfrac{\partial F}{\partial \lm_1} = -\frac{1}{{\lm_1}^2} + \lm f_T(\lm_1) \qquad \dfrac{\partial F}{\partial \lm_2} = \frac{1}{{\lm_2}^2} - \lm f_T(\lm_2)
    \end{equation*}

    Igualando ambas a cero y tratando de despejar $\lm$:
    \begin{equation*}
        \left.\begin{array}{l}
            0 = \dfrac{\partial F}{\partial \lm_1} = -\dfrac{1}{{\lm_1}^2} + \lm f_T(\lm_1) \\ \\
            0 = \dfrac{\partial F}{\partial \lm_2} = \dfrac{1}{{\lm_2}^2} - \lm f_T(\lm_2)
        \end{array}\right\} \Longrightarrow \left.\begin{array}{l}
            \lm = \dfrac{1}{f_T(\lm_1){\lm_1}^2} \\ \\
            \lm = \dfrac{1}{f_T(\lm_2){\lm_2}^2}
        \end{array}\right\}
    \end{equation*}

    Igualando ambas expresiones obtenemos:
    \begin{equation*}
        \frac{f_T(\lm_1)}{f_T(\lm_2)} = \frac{\lm_2^2}{\lm_1^2}
    \end{equation*}

    Debido a la asimetría de la distribución $\chi^2(n)$, no es posible despejar de forma sencilla $\lm_1$ y $\lm_2$ de esta ecuación, por lo que en la práctica se toman los valores que dejan a sus lados colas de probabilidad iguales, es decir:
    \begin{equation*}
        \lm_1 = \chi^2_{n,1-\nicefrac{\alpha}{2}}, \qquad \lm_2 = \chi^2_{n,\nicefrac{\alpha}{2}}
    \end{equation*}
    
    En definitiva, consideraremos el intervalo:
    \begin{equation*}
        \left]\frac{1}{\chi^2_{n,\nicefrac{\alpha}{2}}}\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2},\frac{1}{\chi^2_{n,1-\nicefrac{\alpha}{2}}}\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2} \right[
    \end{equation*}

    Para determinar una cota superior, lo que hacemos es tomar $\lm_2 = \infty$, por lo que:
    \begin{equation*}
        1-\alpha = F_T(\lm_2)-F_T(\lm_1) = 1-F_T(\lm_1) \Longrightarrow F_T(\lm_1) = \alpha
    \end{equation*}
    Luego podemos tomar $\lm_1 = \chi^2_{n,1-\alpha}$, obteniendo la cota superior:
    \begin{equation*}
        \frac{1}{\chi^2_{n,1-\alpha}}\sum\limits_{i=1}^n{(X_i-\mu_0)}^{2} 
    \end{equation*}

    Empleamos ahora los datos proporcionados en el enunciado para calcular el intervalo de confianza y la cota superior. Tenemos que $\mu_0=3$, $n=12$ y $\alpha = 0.01$. Calculamos la suma de cuadrados:
    \begin{align*}
        \sum_{i=1}^{n}{(x_i - \mu_0)}^{2} &= (3.01-3)^2 + (3.05-3)^2 + (2.99-3)^2 + (2.99-3)^2 + (3-3)^2 + (3.02-3)^2 \\
        &+ (2.98-3)^2 + (2.99-3)^2 + (2.97-3)^2 + (2.97-3)^2 + (3.02-3)^2 + (3.01-3)^2 \\
        &= 0.006
    \end{align*}

    Además, consultando la tabla de la distribución $\chi^2(n)$, tenemos que:
    \begin{align*}
        \chi^2_{n,\nicefrac{\alpha}{2}} &= \chi^2_{12,0.005} = 28.2997 \\
        \chi^2_{n,1-\nicefrac{\alpha}{2}} &= \chi^2_{12,0.995} = 3.0738 \\
        \chi^2_{n,1-\alpha} &= \chi^2_{12,0.99} = 3.5706
    \end{align*}

    Por tanto, el intervalo de confianza a nivel de confianza $0.99$ evaluado en la muestra obtenida será:
    \begin{equation*}
        \left]\frac{1}{28.2997} \cdot 0.0059, \frac{1}{3.0738} \cdot 0.0059\right[ = \left]0.0002084,0.00191945\right[
    \end{equation*}
    Y la cota superior será, evaluada en la muestra obtenida:
    \begin{equation*}
        \frac{1}{3.5706} \cdot 0.0059 = 0.00165238
    \end{equation*}

\begin{ejercicio}\label{ej:2.6.5}
    Dos muestras independientes, cada una de tamaño 7, de poblaciones normales con igual varianza, producen medias $4.8$ y $5.4$ y cuasivarianzas muestrales $8.38$ y $7.62$, respectivamente. Encontrar un intervalo de confianza para la diferencia de medias al nivel de confianza $0.95$.\\

    \noindent
    Tenemos por un lado $(X_1, \ldots, X_{n_1})$ m.a.s.  de $X\rightsquigarrow\cc{N}(\mu_1,\sigma^2_1)$ y, por otro lado $(Y_1, \ldots, Y_{n_2})$ m.a.s. de $Y\rightsquigarrow\cc{N}(\mu_2,\sigma_2^2)$. Vamos a calcular un intervalo de confianza para $\mu_1-\mu_2$ en una población normal con varianzas desconocidas pero iguales a nivel de confianza $1-\alpha$ mediante el método de la cantidad pivotal, usando la función pivote:
    \begin{equation*}
        T(X_1, \ldots, X_{n_1}, Y_1, \ldots, Y_{n_2}; \mu_1-\mu_2) = \frac{(\overline{X}-\overline{Y})-(\mu_1-\mu_2)}{S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}} \rightsquigarrow t(n_1+n_2-2)
    \end{equation*}
    donde $S_p^2 = \frac{(n_1-1)S_1^2 + (n_2-1)S_2^2}{n_1+n_2-2}$ es la cuasivarianza muestral combinada. Veamos que cumple las condiciones necesarias para poder aplicar el método de la cantidad pivotal:
    \begin{itemize}
        \item Es estrictamente monótona en $\mu_1-\mu_2$.
        \begin{equation*}
            \frac{\partial T}{\partial (\mu_1-\mu_2)} = -\frac{1}{S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}} < 0
        \end{equation*}
        En efecto, es estrictamente decreciente respecto $\mu_1-\mu_2$.
        \item Si tenemos $\lm$ con:
            \begin{equation*}
                \lm = \frac{(\overline{X}-\overline{Y})-(\mu_1-\mu_2)}{S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}}\Longrightarrow \mu_1-\mu_2 = \overline{X}-\overline{Y} - \lm S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}
            \end{equation*}
    \end{itemize}
    Por tanto, el intervalo de confianza a considerar será:
    \begin{equation*}
        \left]\overline{X}-\overline{Y}-\lm_2 S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}},\overline{X}-\overline{Y}-\lm_1 S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}\right[
    \end{equation*}
    donde $\lm_1$ y $\lm_2$ están sujetos a la restricción $P_{\mu_1-\mu_2}[\lm_1<T<\lm_2]=1-\alpha$. Aunque no se pide explícitamente proporcionar el intervalo de confianza de mínima longitud esperada uniforme, se calcula este por ser el más interesante. Para ello, calculamos la longitud esperada del intervalo:
    \begin{equation*}
        E[L] = E\left[(\lm_2-\lm_1)S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}\right] = (\lm_2-\lm_1)E\left[S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}\right]
    \end{equation*}
    Como $E\left[S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}\right]$ es una cantidad que no depende de $\lm_1,\lm_2$ y positiva, bastará minimizar la cantidad $\lm_2-\lm_1$ (sujeta a $P_{\mu_1-\mu_2}[\lm_1<T<\lm_2]=1-\alpha$) para minimizar la longitud del intervalo. La restricción mencionada puede reescribirse como:
    \begin{equation*}
        1-\alpha = P_{\mu_1-\mu_2}[\lm_1<T<\lm_2] = F_T(\lm_2) - F_T(\lm_1)
    \end{equation*}
    Consideramos por el método de los multiplicadores de Lagrange:
    \begin{equation*}
        F(\lm_1,\lm_2) = (\lm_2-\lm_1) - \lm(F_T(\lm_2) - F_T(\lm_1) - (1-\alpha))
    \end{equation*}
    y buscamos los valores de $\lm_1$ y $\lm_2$ que minimicen la expresión, por lo que calculamos sus derivadas parciales:
    \begin{equation*}
        \dfrac{\partial F}{\partial \lm_1} = -1+\lm f_T(\lm_1) \qquad \dfrac{\partial F}{\partial \lm_2} = 1-\lm f_T(\lm_2)
    \end{equation*}
    Si igualamos ambas a cero y tratamos de despejar $\lm$:
    \begin{equation*}
        \left.\begin{array}{l}
            0 = \dfrac{\partial F}{\partial \lm_1} = -1+\lm f_T(\lm_1) \\ \\
            0 = \dfrac{\partial F}{\partial \lm_2} = 1-\lm f_T(\lm_2)
        \end{array}\right\} \Longrightarrow \left.\begin{array}{l}
            \lm = \dfrac{1}{f_T(\lm_1)} \\ \\
            \lm = \dfrac{1}{f_T(\lm_2)}
        \end{array}\right\} \Longrightarrow f_T(\lm_1) = f_T(\lm_2)
    \end{equation*}
    Como $f_T$ es una función simétrica respecto al origen, las únicas posibilidades son bien $\lm_1 = \lm_2$ bien $\lm_1 = -\lm_2$. Como estos valores han de cumplir la restricción $P_{\mu_1-\mu_2}[\lm_1<T<\lm_2]=1-\alpha > 0$, la primera opción es imposible, por lo que tiene que ser $\lm_1 = -\lm_2$. Como han de dejar entre ellos un valor de $1-\alpha$, han de ser:
    \begin{equation*}
        \lm_2 = t_{n_1+n_2-2,\nicefrac{\alpha}{2}}, \qquad \lm_1 = -\lm_2 = -t_{n_1+n_2-2,\nicefrac{\alpha}{2}}
    \end{equation*}
    Por tanto, el intervalo de confianza a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]\overline{X}-\overline{Y}-t_{n_1+n_2-2,\nicefrac{\alpha}{2}}S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}},\overline{X}-\overline{Y}+t_{n_1+n_2-2,\nicefrac{\alpha}{2}}S_p\sqrt{\frac{1}{n_1}+\frac{1}{n_2}}\right[
    \end{equation*}

    A partir de los datos proporcionados, tenemos que $\alpha=0.05$, $n_1=n_2=7$, $\overline{x} = 4.8$, $\overline{y} = 5.4$, $s_1^2 = 8.38$ y $s_2^2 = 7.62$. Calculamos la cuasivarianza muestral combinada evaluada en la muestra obtenida:

    \end{equation*}

\begin{ejercicio}
    Sea $(X_1, \ldots, X_n)$ una muestra aleatoria simple de una variable aleatoria con distribución $\cc{U}(0,\theta)$. Dado un nivel de confianza arbitrario, calcular el intervalo de confianza para $\theta$ de menor longitud media uniformemente basado en un estadistico suficiente.\\

    \noindent
    Si la muestra es de $X\rightsquigarrow \cc{U}(0,\theta)$ y consideramos un nivel de confianza $1-\alpha$ para $\alpha \in \left]0,1\right[$, calculamos un estadístico suficiente para $\theta$:
    \begin{equation*}
        f_\theta^n(x_1, \ldots, x_n) \stackrel{\text{iid.}}{=} \prod_{i=1}^{n}f_\theta(x_i) = \prod_{i=1}^{n}\frac{1}{\theta} = \frac{1}{\theta^n} \qquad x_i \in \left]0,\theta\right[
    \end{equation*}
    Por tanto, se necesita que $0<x_i<\theta$ para todo $i=1,\ldots,n$, es decir, es necesario que $0<X_{(1)}<X_{(n)}<\theta$. Por tanto, la función de densidad conjunta de la muestra es:
    \begin{equation*}
        f_\theta^n(x_1, \ldots, x_n) = \ind_{\mathbb{R}^+}(X_{(1)}) \ind_{\mathbb{R}^-}(X_{(n)}-\theta)\frac{1}{\theta^n}
    \end{equation*}
    Tomando:
    \begin{gather*}
        h(X_1, \ldots, X_n) = \ind_{\mathbb{R}^+}(X_{(1)}), \qquad T(X_1, \ldots, X_n) = X_{(n)} \\
        g_\theta(t) = \ind_{\mathbb{R}^-}(t-\theta)\frac{1}{\theta^n}
    \end{gather*}
    Tenemos por el Teorema de factorización de Neymann-Fisher que el estadístico $T(X_1, \ldots, X_n) = X_{(n)}$ es suficiente para $\theta$. Calculamos su función de distribución:
    \begin{align*}
        F_T(t) &= P_\theta[T\leq t] = P_\theta[X_{(n)}\leq t] = P_\theta[X_1\leq t, \ldots, X_n\leq t] = \prod_{i=1}^{n}P_\theta[X_i\leq t] =\\&= \prod_{i=1}^{n}F_X(t) = {(F_X(t))}^{n} \\
        F_X(t) &= P_\theta[X\leq t] = \int_{0}^{t}f_\theta(x)dx = \int_{0}^{t}\frac{1}{\theta}dx = \frac{t}{\theta} \qquad t\in [0,\theta]
    \end{align*}

    Por tanto, tenemos que:
    \begin{equation*}
        F_T(t) = {(F_X(t))}^{n} = {\left(\frac{t}{\theta}\right)}^{n} \qquad t\in [0,\theta]
    \end{equation*}

    Por tanto, una función pivote para aplicar el método de la cantidad pivotal es:
    \begin{equation*}
        T(X_1, \ldots, X_n;\theta) = F_T(T) = {\left(\frac{X_{(n)}}{\theta}\right)}^{n} \rightsquigarrow \cc{U}(0,1)
    \end{equation*}
    
    Comprobamos que cumple las condiciones necesarias para poder aplicar el método de la cantidad pivotal:
    \begin{itemize}
        \item La función es estrictamente monótona en $\theta$.
        \begin{equation*}
            \frac{\partial T}{\partial \theta} = -n{\left(\frac{X_{(n)}}{\theta}\right)}^{n-1}\frac{X_{(n)}}{\theta^2} < 0
        \end{equation*}
        En efecto, es estrictamente decreciente respecto $\theta$.

        \item Si tenemos $\lm$ de forma que:
            \begin{equation*}
                \lm = {\left(\frac{X_{(n)}}{\theta}\right)}^{n} \Longrightarrow \theta = \frac{X_{(n)}}{\sqrt[n]{\lm}}
            \end{equation*}
    \end{itemize}
    
    Por tanto, el intervalo de confianza para $\theta$ a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]\frac{X_{(n)}}{\sqrt[n]{\lm_2}},\frac{X_{(n)}}{\sqrt[n]{\lm_1}} \right[
    \end{equation*}
    donde $\lm_1$ y $\lm_2\in \bb{R}$ verifican:
    \begin{equation*}
        1-\alpha\leq P_\theta[\lm_1<T(X_1, \ldots, X_n;\theta)<\lm_2] = F_{T(X_1, \ldots, X_n;\theta)}(\lm_2) - F_{T(X_1, \ldots, X_n;\theta)}(\lm_1) = \lm_2-\lm_1
    \end{equation*}
    La longitud esperada del intervalo es:
    \begin{equation*}
        E[L] = E\left[\left(\frac{1}{\sqrt[n]{\lm_1}} - \frac{1}{\sqrt[n]{\lm_2}}\right) X_{(n)}\right] = \left(\frac{1}{\sqrt[n]{\lm_1}} - \frac{1}{\sqrt[n]{\lm_2}}\right)E[X_{(n)}]
    \end{equation*}

    Como $E[X_{(n)}]$ es una cantidad que no depende de $\lm_1,\lm_2$ y positiva, bastará minimizar la cantidad $\frac{1}{\sqrt[n]{\lm_1}} - \frac{1}{\sqrt[n]{\lm_2}}$ (sujeta a $P_\theta[\lm_1<T(X_1, \ldots, X_n;\theta)<\lm_2]\geq 1-\alpha$) para minimizar la longitud del intervalo. Buscamos minimizar la siguiente expresión:
    \begin{equation*}
        F(\lm_1,\lm_2) = \left(\frac{1}{\sqrt[n]{\lm_1}} - \frac{1}{\sqrt[n]{\lm_2}}\right)
    \end{equation*}

    Empleando que $\lm_2=\lm_1 + 1-\alpha$ (por la restricción), tenemos que:
    \begin{equation*}
        F(\lm_1) = \frac{1}{\sqrt[n]{\lm_1}} - \frac{1}{\sqrt[n]{\lm_1 + 1-\alpha}}
        = \lm_1^{-\nicefrac{1}{n}} - (\lm_1 + 1-\alpha)^{-\nicefrac{1}{n}}
    \end{equation*}

    Calculamos la derivada de $F$ respecto $\lm_1$:
    \begin{equation*}
        \frac{\partial F}{\partial \lm_1} = -\frac{1}{n\sqrt[n]{\lm_1^{n+1}}} + \frac{1}{n\sqrt[n]{(\lm_1 + 1-\alpha)^{n+1}}}
    \end{equation*}

    Como $0<\lm_1<\lm_1 + 1-\alpha$ y la potencia $-\nicefrac{1}{n}$ negativa es estrictamente decreciente, tenemos que:
    \begin{equation*}
        \frac{\partial F}{\partial \lm_1} = -\frac{1}{n\sqrt[n]{\lm_1^{n+1}}} + \frac{1}{n\sqrt[n]{(\lm_1 + 1-\alpha)^{n+1}}} < 0
    \end{equation*}
    Por tanto, $F$ es estrictamente decreciente respecto $\lm_1$, y por tanto, para minimizar $F$ (y por tanto la longitud del intervalo), debemos tomar $\lm_1$ lo más grande posible, es decir, $\lm_1 = 1-\alpha \Longrightarrow \lm_2 = 1$. Por tanto, el intervalo de confianza de menor longitud media uniformemente a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]X_{(n)},\frac{X_{(n)}}{\sqrt[n]{1-\alpha}}\right[
    \end{equation*}
\end{ejercicio}

\begin{ejercicio}
    Utilizando la desigualdad de Chebychev, dar un intervalo de confianza para $p$ a nivel de confianza arbitrario, basado en una muestra de tamaño arbitrario de una variable aleatoria con distribución $\cc{B}(1,p)$.\\

    Sea $(X_1, \ldots, X_n)$ una m.a.s. de $X\rightsquigarrow \cc{B}(1,p)$. Buscamos por el método de Chebyshev un intervalo de confianza para $p$ a nivel de confianza $1-\alpha$. Para ello, es necesario tener un estimador insesgado de $p$. Si consideramos:
    \begin{equation*}
        T(X_1, \ldots, X_n) = \overline{X}
    \end{equation*}
    tenemos que $\overline{X}\in [0,1]$, así como que $\overline{X}$ es insesgado, puesto que:
    \begin{equation*}
        E[\overline{X}] = E\left[\frac{1}{n}\sum_{i=1}^{n}X_i\right] = \frac{1}{n}\sum_{i=1}^{n}E[X_i] = \frac{1}{n}\sum_{i=1}^{n}p = \frac{np}{n} = p
    \end{equation*}

    Buscamos acotar $Var[\ol{X}]$, y para ello puesto que las variables $X_i$ son independientes, tenemos que:
    \begin{align*}
        Var[\ol{X}] &= Var\left[\frac{1}{n}\sum_{i=1}^{n}X_i\right] = \frac{1}{n^2}Var\left[\sum_{i=1}^{n}X_i\right] = \frac{1}{n^2}\sum_{i=1}^{n}Var[X_i] =\\&= \frac{1}{n^2}\sum_{i=1}^{n}p(1-p) = \frac{np(1-p)}{n^2} = \frac{p(1-p)}{n}
    \end{align*}

    Puesto que se busca una cota superior de $Var[\ol{X}]$ suficientemente ajustada (para obtener así un intervalo de confianza más pequeño), consideramos la función:
    \Func{f}{[0,1]}{\bb{R}^+}{p}{p(1-p)}

    La función $f$ es continua en el intervalo cerrado $[0,1]$, por lo que alcanza su máximo en dicho intervalo. Calculamos la derivada de $f$:
    \begin{equation*}
        f'(p) = 1-2p=0\iff p=\frac{1}{2}
    \end{equation*}
    Además, como $f''(p)=-2<0$, tenemos que $p=\nicefrac{1}{2}$ es un máximo relativo, y por tanto, la abcisa del máximo absoluto de $f$ en $[0,1]$. Su ordenada es:
    \begin{equation*}
        f\left(\frac{1}{2}\right) = \frac{1}{4}
    \end{equation*}

    Por tanto, tenemos que:
    \begin{equation*}
        Var[\ol{X}] = \frac{p(1-p)}{n} \leq \frac{1}{4n} \qquad \forall p\in [0,1]
    \end{equation*}

    Por la desigualdad de Chebyshev, tenemos que:
    \begin{equation*}
        P\left[\left|\ol{X}-p\right|< k\right] \geq 1-\frac{Var[\ol{X}]}{k^2} \geq 1-\frac{1}{4nk^2}\qquad \forall k>0
    \end{equation*}

    Por tanto, para un nivel de confianza $1-\alpha$, tenemos que:
    \begin{equation*}
        1-\alpha = 1-\frac{1}{4nk^2} \iff k = \frac{1}{2\sqrt{n\alpha}}
    \end{equation*}

    Por tanto, el intervalo de confianza para $p$ a nivel de confianza $1-\alpha$ es:
    \begin{equation*}
        \left]\ol{X}-\frac{1}{2\sqrt{n\alpha}},\ol{X}+\frac{1}{2\sqrt{n\alpha}}\right[
    \end{equation*}
\end{ejercicio}

\begin{ejercicio}
    Para una muestra de tamaño $n$ de una variable aleatoria con función de densidad
    \begin{equation*}
        f_\theta(x) = \frac{2x}{\theta^2}, \qquad 0<x<\theta
    \end{equation*}
    encontrar el intervalo de confianza para $\theta$ de menor longitud media uniformemente a nivel de confianza $1-\alpha$, basado en un estadístico suficiente.\\

    Buscamos en primer lugar un estadístico suficiente de $\theta$:
    \begin{equation*}
        f_\theta^n(X_1, \ldots, X_n) \stackrel{\text{iid.}}{=} \prod_{i=1}^{n}f_\theta(X_i) = \prod_{i=1}^{n}\frac{2X_i}{\theta^2} = \frac{2^n}{\theta^{2n}}\prod_{i=1}^{n}X_i \qquad 0<X_i<\theta
    \end{equation*}

    Por tanto, es necesario que $0<X_i<\theta$ para todo $i=1,\ldots,n$, es decir, es necesario que $0<X_{(1)}<X_{(n)}<\theta$. Por tanto, la función de densidad conjunta de la muestra es:
    \begin{equation*}
        f_\theta^n(X_1, \ldots, X_n) = \frac{2^n}{\theta^{2n}}\ind_{\mathbb{R}^+}(X_{(1)})\ind_{\mathbb{R} ^-}(X_{(n)}-\theta)\prod_{i=1}^n X_i
    \end{equation*}
    Si tomamos:
    \begin{gather*}
        h(X_1, \ldots, X_n) = \ind_{\mathbb{R}^+}(X_{(1)})\prod_{i=1}^n X_i, \qquad T(X_1, \ldots, X_n) = X_{(n)} \\
        g_\theta(t) = \frac{2^n}{\theta^{2n}}\ind_{\mathbb{R}^-}(t-\theta)
    \end{gather*}
    Tenemos por el Teorema de factorización de Neymann-Fisher que $X_{(n)}$ es un estadístico suficiente para $\theta$. Calculamos su función de distribución:
    \begin{align*}
        F_T(t) &= P_\theta[T\leq t] = P_\theta[X_{(n)}\leq t] = P_\theta[X_1\leq t, \ldots, X_n\leq t] = \prod_{i=1}^{n}P_\theta[X_i\leq t] =\\&= \prod_{i=1}^{n}F_X(t) = {(F_X(t))}^{n}\\
        F_X(t) &= P_\theta[X\leq t] = \int_{0}^{t}f_\theta(x)dx = \int_{0}^{t}\frac{2x}{\theta^2}dx = \left[\frac{x^2}{\theta^2}\right]_0^t = \frac{t^2}{\theta^2}\qquad t\in [0,\theta]
    \end{align*}
    Por tanto, tenemos que:
    \begin{equation*}
        F_T(t) = {(F_X(t))}^{n} = {\left(\frac{t}{\theta}\right)}^{2n}\qquad t\in [0,\theta]
    \end{equation*}

    Por tanto, una función pivote para aplicar el método de la cantidad pivotal es:
    \begin{equation*}
        T(X_1, \ldots, X_n;\theta) = F_T(X_{(n)}) = {\left(\frac{X_{(n)}}{\theta}\right)}^{2n} \rightsquigarrow \cc{U}(0,1)
    \end{equation*}
    
    Comprobemos que cumple las condiciones necesarias para poder aplicar el método de la cantidad pivotal:
    \begin{itemize}
        \item La función es estrictamente monótona en $\theta$.
        \begin{equation*}
            \frac{\partial T}{\partial \theta} = -2n{\left(\frac{X_{(n)}}{\theta}\right)}^{2n-1}\frac{X_{(n)}}{\theta^2} < 0
        \end{equation*}
        En efecto, es estrictamente decreciente respecto $\theta$.

        \item Si tenemos $\lm$ de forma que:
            \begin{equation*}
                \lm = {\left(\frac{X_{(n)}}{\theta}\right)}^{2n} \Longrightarrow \theta = \frac{X_{(n)}}{\sqrt[2n]{\lm}}
            \end{equation*}
    \end{itemize}

    Por tanto, el intervalo de confianza para $\theta$ a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]\frac{X_{(n)}}{\sqrt[2n]{\lm_2}},\frac{X_{(n)}}{\sqrt[2n]{\lm_1}} \right[
    \end{equation*}
    donde $\lm_1$ y $\lm_2\in \bb{R}$ verifican:
    \begin{equation*}
        1-\alpha\leq P_\theta[\lm_1<T(X_1, \ldots, X_n;\theta)<\lm_2] = F_{T(X_1, \ldots, X_n;\theta)}(\lm_2) - F_{T(X_1, \ldots, X_n;\theta)}(\lm_1) = \lm_2-\lm_1
    \end{equation*}

    La longitud esperada del intervalo es:
    \begin{equation*}
        E[L] = E\left[\left(\frac{1}{\sqrt[2n]{\lm_1}} - \frac{1}{\sqrt[2n]{\lm_2}}\right) X_{(n)}\right] = \left(\frac{1}{\sqrt[2n]{\lm_1}} - \frac{1}{\sqrt[2n]{\lm_2}}\right)E[X_{(n)}]
    \end{equation*}

    Para minimizar la longitud esperada del intervalo, necesitamos encontrar los valores de $\lm_1$ y $\lm_2$ que minimicen la expresión anterior. Definimos la función a minimizar, empleando que $\lm_2 = \lm_1 + 1 - \alpha$:
    \begin{equation*}
        F(\lm_1) = \frac{1}{\sqrt[2n]{\lm_1}} - \frac{1}{\sqrt[2n]{\lm_1 + 1 - \alpha}} = \lm_1^{-\frac{1}{2n}} - (\lm_1 + 1 - \alpha)^{-\frac{1}{2n}}
    \end{equation*}

    Calculamos su derivada:
    \begin{equation*}
        F'(\lm_1) = -\frac{1}{2n}\lm_1^{-\frac{1}{2n}-1} + \frac{1}{2n}(\lm_1 + 1 - \alpha)^{-\frac{1}{2n}-1} = \left(\frac{1}{2n}\right)\left[(\lm_1 + 1 - \alpha)^{-\frac{1}{2n}-1} - \lm_1^{-\frac{1}{2n}-1}\right]
    \end{equation*}
    Como $0\leq \lm_1 < \lm_1 + 1 - \alpha$, y la potencia $-\frac{1}{2n}-1$ negativa es estrictamente decreciente, tenemos que:
    \begin{equation*}
        \lm_1^{-\frac{1}{2n}-1} > (\lm_1 + 1 - \alpha)^{-\frac{1}{2n}-1} \implies F'(\lm_1) < 0
    \end{equation*}

    Por tanto, $F$ es estrictamente decreciente respecto $\lm_1$, y para minimizar $F$ (y por tanto la longitud del intervalo), debemos tomar $\lm_1$ lo más grande posible, es decir, $\lm_1 = 1 - \alpha \Longrightarrow \lm_2 = 1$. Por tanto, el intervalo de confianza de menor longitud media uniformemente a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]X_{(n)},\frac{X_{(n)}}{\sqrt[2n]{1 - \alpha}}\right[
    \end{equation*}
\end{ejercicio}

\begin{ejercicio} 
    Para una muestra de tamaño $n$ de una variable aleatoria con función de densidad
    \begin{equation*}
        f_\theta(x) = \frac{\theta}{x^2}, \qquad x>\theta
    \end{equation*}
    encontrar el intervalo de confianza para $\theta$ de menor longitud media uniformemente a nivel de confianza $1 - \alpha$, basado en el estimador máximo verosímil de $\theta$.\\

    \noindent
    La función de verosimilitud es:
    \begin{equation*}
        L_{x_1,\ldots,x_n}(\theta) = f_\theta^n(x_1, \ldots, x_n) = \prod_{i=1}^{n}f_\theta(x_i) = \prod_{i=1}^{n} \frac{\theta}{x_i^2} \qquad \forall x_i > \theta > 0
    \end{equation*}
    Por tanto, tenemos que:
    \begin{equation*}
        L_{x_1,\ldots,x_n}(\theta) = \left\{\begin{array}{ll}
            \prod\limits_{i=1}^{n}\frac{\theta}{x_i^2} & \text{si\ } \theta < x_{(1)}  \\
             0 & \text{en otro caso} 
        \end{array}\right. 
    \end{equation*}
    Y vemos que $L_{x_1,\ldots,x_n}(\theta)$ es creciente, con lo que alcanza su máximo en $\theta = x_{(1)}$. Por tanto, de ser un estimador, el estimador máximo verosímil de $\theta$ es:
    \begin{equation*}
        \hat{\theta} = X_{(1)}
    \end{equation*}

    Como la única condición que se impone a $\theta$ es $\theta\in\bb{R}^+$ y $x_i>\theta$, tenemos que $\hat{\theta}$ es un estimador de $\theta$, luego el EMV es $\hat{\theta} = X_{(1)}$. Por tanto, para construir un intervalo de confianza basado en el EMV, necesitamos encontrar la distribución de $X_{(1)}$.
    \begin{align*}
        P[X_{(1)}\leq t] &= 1-P[X_{(1)}>t] = 1-P[X_1>t, \ldots, X_n>t] = 1-\prod_{i=1}^{n}P[X_i>t] = 1-\prod_{i=1}^{n}(1-P[X_i\leq t]) =\\&= 1-\prod_{i=1}^{n}(1-F_X(t)) = 1-{(1-F_X(t))}^{n}\\
        F_X(t) &= P[X\leq t] = \int_{\theta}^{t}f_\theta(x)dx = \int_{\theta}^{t}\frac{\theta}{x^2}dx = \left[\frac{-\theta}{x}\right]_\theta^t = \frac{-\theta}{t} + 1 \qquad t>\theta
    \end{align*}

    Por tanto, tenemos que:
    \begin{equation*}
        F_{X_{(1)}}(t) = 1-{(1-F_X(t))}^{n} = 1 - {\left(1-\left(1-\frac{\theta}{t}\right)\right)}^{n} = 1-{\left(\frac{\theta}{t}\right)}^{n} \qquad t> \theta
    \end{equation*}
    Por tanto, una función pivote para aplicar el método de la cantidad pivotal es:
    \begin{equation*}
        T(X_1, \ldots, X_n;\theta) = 1-{\left(\frac{\theta}{X_{(1)}}\right)}^{n} \rightsquigarrow \cc{U}(0,1)
    \end{equation*}

    Comprobamos que cumple las condiciones necesarias para poder aplicar el método de la cantidad pivotal:
    \begin{itemize}
        \item La función es estrictamente monótona en $\theta$.
        \begin{equation*}
            \frac{\partial T}{\partial \theta} = -n{\left(\frac{\theta}{X_{(1)}}\right)}^{n-1}\frac{1}{X_{(1)}} = \frac{-n\theta^{n-1}}{X_{(1)}^n} < 0 \qquad \forall \theta
        \end{equation*}
        En efecto, es estrictamente decreciente respecto $\theta$.
        \item Si tenemos $\lm$ de forma que:
            \begin{equation*}
                \lm = 1-{\left(\frac{\theta}{X_{(1)}}\right)}^{n} \Longrightarrow \theta = X_{(1)}\sqrt[n]{1-\lm}
            \end{equation*}
    \end{itemize}

    Por tanto, el intervalo de confianza para $\theta$ a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]X_{(1)}\sqrt[n]{1-\lm_2},X_{(1)}\sqrt[n]{1-\lm_1}\right[
    \end{equation*}
    donde $\lm_1$ y $\lm_2\in \bb{R}$ verifican:
    \begin{equation*}
        1-\alpha\leq P_\theta[\lm_1<T(X_1, \ldots, X_n;\theta)<\lm_2] = F_{T(X_1, \ldots, X_n;\theta)}(\lm_2) - F_{T(X_1, \ldots, X_n;\theta)}(\lm_1) = \lm_2-\lm_1
    \end{equation*}

    La longitud esperada del intervalo es:
    \begin{equation*}
        E[L] = E\left[\left(\sqrt[n]{1-\lm_1} - \sqrt[n]{1-\lm_2}\right) X_{(1)}\right] = \left(\sqrt[n]{1-\lm_1} - \sqrt[n]{1-\lm_2}\right)E[X_{(1)}]
    \end{equation*}

    Como $E[X_{(1)}]$ es una cantidad que no depende de $\lm_1,\lm_2$ y positiva, bastará minimizar la cantidad $\sqrt[n]{1-\lm_1} - \sqrt[n]{1-\lm_2}$ (sujeta a la restricción dada por la expresión $P_\theta[\lm_1<T(X_1, \ldots, X_n;\theta)<\lm_2]\geq 1-\alpha$) para minimizar la longitud del intervalo. Buscamos minimizar la siguiente expresión:
    \begin{equation*}
        F(\lm_1,\lm_2) = \sqrt[n]{1-\lm_1} - \sqrt[n]{1-\lm_2}
    \end{equation*}

    Como $\lm_2=\lm_1 + 1-\alpha$ (por la restricción), tenemos que:
    \begin{equation*}
        F(\lm_1) = \sqrt[n]{1-\lm_1} - \sqrt[n]{1-\lm_1 - 1+\alpha} = \sqrt[n]{1-\lm_1} - \sqrt[n]{\alpha - \lm_1}
        = (1-\lm_1)^{\nicefrac{1}{n}} - (\alpha - \lm_1)^{\nicefrac{1}{n}}
    \end{equation*}

    Calculamos la derivada de $F$ respecto $\lm_1$:
    \begin{equation*}
        \frac{\partial F}{\partial \lm_1} = -\frac{1}{n}(1-\lm_1)^{\nicefrac{1}{n}-1} + \frac{1}{n}(\alpha - \lm_1)^{\nicefrac{1}{n}-1} = \frac{1}{n}\left[(\alpha - \lm_1)^{\nicefrac{1}{n}-1} - (1-\lm_1)^{\nicefrac{1}{n}-1}\right]
    \end{equation*}

    Como $\alpha< 1$, entonces $\alpha - \lm_1 < 1-\lm_1$. Además:
    \begin{equation*}
        \alpha-\lm_1>0\iff \alpha+1-\lm_2-\alpha>0\iff \lm_2<1
    \end{equation*}
    Por tanto, tenemos que $0<\alpha-\lm_1<1-\lm_1$, y como la potencia $\nicefrac{1}{n}-1$ negativa es estrictamente decreciente, tenemos que:
    \begin{equation*}
        (\alpha - \lm_1)^{\nicefrac{1}{n}-1} > (1-\lm_1)^{\nicefrac{1}{n}-1}
    \end{equation*}

    Por tanto, $\frac{\partial F}{\partial \lm_1} > 0$, y por tanto, $F$ es estrictamente creciente respecto $\lm_1$, y para minimizar $F$ (y por tanto la longitud del intervalo), debemos tomar $\lm_1$ lo más pequeño posible, es decir, $\lm_1 = 0 \Longrightarrow \lm_2 = 1-\alpha$. Por tanto, el intervalo de confianza de menor longitud media uniformemente a nivel de confianza $1-\alpha$ será:
    \begin{equation*}
        \left]X_{(1)}\sqrt[n]{\alpha},X_{(1)}\right[
    \end{equation*}
\end{ejercicio}
