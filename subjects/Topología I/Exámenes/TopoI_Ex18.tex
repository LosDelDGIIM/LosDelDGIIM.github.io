    \begin{ejercicio} 
    (2,5 puntos) Define espacio topológico T1 y demuestra que $(X, T)$ es T1 si y sólo si $\{x\}$ es cerrado para todo $x \in X$.
    
    Un espacio topológico $(X, T)$ es T1 si para cada par de puntos $x, y\in X$, $x\neq y$ existen un $U, V\in T$ con $x\in U$ y $y\in V$ tales que $y\notin U$ y $x\notin V$.\\

    Veamos ahora por doble implicación que $(X, T)$ es T1 si y sólo si $\{x\}$ es cerrado para todo $x \in X$:
    \begin{itemize}
        \item[$\Rightarrow$] Supongamos que $(X, T)$ es T1, vamos a ver que sea $x\in X$ arbitrario entonces $\{x\} = \overline{\{x\}}$. Está claro que $\{x\} \subset \overline{\{x\}}$. Para la otra inclusión sea $y \in \overline{\{x\}}$, entonces para todo $V\in T$ con $y\in V$ se tiene que $V\cap \{x\} \neq \varnothing$, lo que implica que $x\in V$. Si fuese $x \neq y$, como $(X, T)$ es T1 existiría $V' \in T$ con $y \in V'$ y $x \notin V'$, contradicción. Por tanto $x = y$ (y en particular $y \in \{x\}$).
        \item[$\Leftarrow$] Supongamos que $\{x\}$ es cerrado para todo $x \in X$, vamos a ver que $(X, T)$ es T1. Sea $x, y \in X$ con $x \neq y$. Entonces $x \in X \setminus \{y\}$ y $y \in X \setminus \{x\}$. Como $U = X \setminus \{y\}$ y $V =X \setminus \{x\}$ son abiertos (complementario de cerrado), existen $U, V \in T$ tales que $x \in U$ y $y \in V$ y además, $y \notin U$ y $x \notin V$.
    \end{itemize} 
    \end{ejercicio}

    \begin{ejercicio}
    Sea $X$ un conjunto y $A \subset X$ con $A \neq \varnothing$ y $A \neq X$. Se considera la siguiente familia de subconjuntos de $X$:
    \[
        B = \{\{x\} \cup A : x \in X\}.
    \]
    \begin{enumerate}
        \item (1 punto) Prueba que existe una única topología $T$ en $X$ para la cual $B$ es base.\\
        Vamos a probar que $\cc{B} = \{B\subseteq X: B = \{x\}\cup A \text{ con } x \in X\}\subseteq \cc{P}(X)$ es una base de una topología en $X$. Para ello probemos que cumple las siguientes propiedades:
        \begin{enumerate}
            \item[(B1)] $X = \bigcup_{B \in B} B$
            \begin{itemize}
                \item[$\subseteq$] Trivial porque sea $x \in X$, entonces existe un $B\in \cc{B}$ tal que $x \in B = \{x\} \cup A$.
                \item[$\supseteq$] Sea $x \in \bigcup_{B \in B} B$, entonces existe un $B_x\in \cc{B}$ tal que $x \in B_x = \{x\} \cup A$ por definición con $x\in X$.
            \end{itemize}
            \item[(B2)] Sea $B_1, B_2 \in \cc{B}$ y $x \in B_1 \cap B_2$, existe $B_3 \in \cc{B}$ tal que $x \in B_3 \subset B_1 \cap B_2$. Sean $B_1 = \{x_1\}\cup A$ y $B_2 = \{x_2\}\cup A$, distinguimos los siguientes casos:
            \begin{enumerate}
                \item Si $x\in A$ bastaría tomar $B_3 = \{x\} \cup A = A$ porque claramente $x\in B_3 = A \subset B_1 \cap B_2$.
                \item Si $x\notin A$ entonces $x = x_1$ y $x = x_2$ para estar en la intersección. Supongamos que $x = x_1$, entonces tomando $B_3 = B_1 = \{x\} \cup A$ cumple que $x \in B_3 \subset B_1 \cap B_2$.
            \end{enumerate}
        \end{enumerate}
        Así quedaría demostrado que existe una única topología $T$ en $X$ para la cual $B$ es base.
        \item (0,5 puntos) Demuestra que si $U \subset X$ con $U \neq \varnothing$, entonces $U \in T$ si y sólo si $A \subset U$.\\
        Demostremos la doble implicación:
        \begin{itemize}
            \item[$\Rightarrow$] Supongamos que $U \in T$, entonces $U = \bigcup_{B_i\in \cc{B}} B_i = \bigcup_{i\in I} (\{x_i\} \cup A) = (\bigcup_{i\in I} \{x_i\}) \cup A$, por lo que, $A \subset U$.
            \item[$\Leftarrow$] Supongamos que $A \subset U$, entonces para todo $a \in A$ existe un $B_a = \{a\} \cup A = A$ con $a\in B_a\subset A\subset U$ luego $U\in \cc{N}_a, \forall a \in A$. Por otro lado si $b\in U\setminus A$ existe un $B_b = \{b\} \cup A$ con $b\in B_b\subset U$ y por lo tanto $U \in \cc{N}_b$. Luego $U\in \cc{N}_x, \forall x \in U$ lo que implica que $U \in T$.
        \end{itemize}
        \item (1 punto) Identifica las topologías inducidas sobre $A$ y $X \setminus A$.
        
        Comencemos con la topología inducida $(A, T_{|A})$ que sabemos que tiene como base de la topología $\cc{B}_{|A} = \{B\cap A : B \in \cc{B}\} = \{(\{x\} \cup A)\cap A : x\in X\} = \{\varnothing\} \cup\{A\}$. De esta manera si escribimos todo abierto como unicón de sus bases tenemos sea $V\in T_{|A}$ entonces: 
        \begin{itemize}
            \item $V = \bigcup_{B\in\cc{B}_{|A}} B = A$
            \item $V = \varnothing$
        \end{itemize}
        Luego $T_{|A} = \{\varnothing, A\}$ es la topología trivial.\\

        Por otro lado, la topología inducida $(X\setminus A, T_{|X\setminus A})$ tiene como base de la topología $\cc{B}_{|X\setminus A} = \{B\cap (X\setminus A) : B \in \cc{B}\} = \{(\{x\} \cup A)\cap (X\setminus A) : x\in X\} = \{\{x\} : x\in X\setminus A\}$. De esta manera si escribimos todo abierto como unión de sus bases tenemos sea $V\in T_{|X\setminus A}$ entonces $V = \bigcup_{B\in\cc{B}_{|X\setminus A}} B = \bigcup_{x\in V} \{x\} = V$. Con lo que claramente se ve que $T_{|X\setminus A} = \cc{P}(X\setminus A)$ es la topología discreta.
        
        \item (1 punto) Estudia si $(X, T)$ es 1AN y 2AN.\\
        Vamos a probar que $(X, T)$ es 1AN. Para ello observemos que sea $x\in X$ entonces $\cc{B}_x=\{\{x\}\cup A\}$ es una base de entornos de $x$ pues sea $U\in T$ arbitrario, con $x\in U$, como $U \neq \varnothing$ y $U \in T$, por el apartado 2 se tiene que $A \subset U$. Como además $x \in U$, de alguna de las maneras, $\exists V = \{x\} \cup A$ entorno básico de $x$ tal que $x\in V\subset U$.\\
        Habiendo demostrado que en efecto $\cc{B}_x=\{\{x\}\cup A\}$ es base de entornos de $x$, el resultado es inmediato pues $\cc{B}_x$ es finita, de hecho tiene un único elemento, luego es numerable y $(X, T)$ es 1AN.\\
        
        Ahora vamos a demostrar que $(X, T)$ puede ser o no 2AN dependiendo de $A$:
        \begin{itemize}
            \item Si $X\setminus A$ es numerable (o finito), entonces existe una cantidad numerable de puntos $\{x_n\}_{n\in I} = X\setminus A$ (con $I \subseteq \bb{N}$) y por lo tanto tomando $\cc{B}_n = \{A\} \cup \{\{x_n\}\cup A : n\in I\}$ tenemos una familia numerable. Comprobemos que es efectivamente una base de $T$:
            
            Sabemos por el apartado 1 que $\cc{B} = \{\{x\}\cup A : x\in X\}$ es una base de $T$. Basta observar que $\cc{B}_n = \cc{B}$, ya que para todo $x \in X$:
            \begin{itemize}
                \item Si $x \in A$, entonces $\{x\} \cup A = A \in \cc{B}_n$.
                \item Si $x \in X \setminus A$, entonces $x = x_n$ para algún $n \in I$, luego $\{x\} \cup A = \{x_n\} \cup A \in \cc{B}_n$.
            \end{itemize}
            Por tanto, $\cc{B}_n$ es una base numerable de $T$ y $(X, T)$ es 2AN.

            \item Si $X\setminus A$ es no numerable, supongamos por reducción al absurdo que $(X, T)$ tiene una base numerable $\cc{B}_n$. Tomemos $I = X \setminus A$ (no numerable) y la familia de abiertos $\{U_i\}_{i\in I}$ definida por $U_i = \{x_i\} \cup A$ para cada $x_i \in X\setminus A$. Por definición de base, para cada $i\in I$ existe $B_i \in \cc{B}_n$ tal que $x_i \in B_i \subset U_i = \{x_i\}\cup A$. Como $B_i \in T$ y $B_i \neq \varnothing$, por el apartado 2 se tiene que $A \subset B_i$, lo que fuerza a que $B_i = \{x_i\}\cup A$. Si tomamos dos puntos $x_i, x_j \in X\setminus A$ con $x_i \neq x_j$, entonces $B_i \neq B_j$ (pues $x_i \notin B_j$) y por lo tanto $\{B_i\}_{i\in I}$ es una familia de abiertos no numerable, lo que contradice que $\cc{B}_n$ sea numerable pues estamos inyectando un conjunto no numerable en uno numerable. Luego $(X, T)$ no es 2AN.
        \end{itemize}
        \end{enumerate}
    \end{ejercicio}

    \begin{ejercicio}
    En $\mathbb{R}^2$ denotamos por $d$ a la distancia usual y definimos la familia de subconjuntos $T$ de la siguiente forma:
    \[
        T = \{U \subset \mathbb{R}^2 : (0,0) \notin U\} \cup \{U \subset \mathbb{R}^2 : \exists \varepsilon > 0 \text{ con } B_d((0,0), \varepsilon) \subset U\}.
    \]
    \begin{enumerate}
        \item (1,5 punto) Demuestra que $T$ es una topología en $\mathbb{R}^2$. Compárala con la topología usual de $\mathbb{R}^2$.\\
        Para ello vamos a probar qye $T\subseteq \cc{P}(\bb{R}^2)$ cumple las siguientes propiedades:
        \begin{enumerate}
            \item[(A1)] $\varnothing, \bb{R}^2 \in T$\\
            Está claro que $\varnothing \in T$ porque $(0,0) \notin \varnothing$. Por otro lado $\bb{R}^2 \in T$ porque $B_d((0,0), 1) \subset \bb{R}^2$.
            \item[(A2)] Si $\{U_i\}_{i\in I} \subset T$, entonces $\bigcup_{i\in I} U_i \in T$\\
            Sea $\{U_i\}_{i\in I} \subset T$, entonces existen dos casos:
            \begin{itemize}
                \item Si $(0,0) \notin U_i$ para todo $i\in I$, entonces $(0,0) \notin \bigcup_{i\in I} U_i$ y por lo tanto $\bigcup_{i\in I} U_i \in T$.
                \item Si existe un $i_0\in I$ tal que $(0,0) \in U_{i_0}$, entonces $\exists \varepsilon > 0$ tal que $B_d((0,0), \varepsilon) \subset U_{i_0}$. Por lo tanto $B_d((0,0), \varepsilon) \subset U_{i_0} \subset \bigcup_{i\in I} U_i$ y por lo tanto $\bigcup_{i\in I} U_i \in T$.
            \end{itemize}
            \item[(A3)] Si $U_1, U_2 \in T$, entonces $U_1 \cap U_2 \in T$\\
            Sea $U_1, U_2 \in T$. Si $U_1\cap U_2 = \varnothing \in T$ y ya lo tendríamos. Si $U_1\cap U_2 \neq \varnothing$, entonces distinguimos casos:
            \begin{itemize}
                \item Si $(0,0)\notin U_1$ (y/o análogamente $(0,0)\notin U_2$), entonces $(0,0) \notin U_1\cap U_2$ y por lo tanto $U_1 \cap U_2 \in T$.
                \item Si $(0,0) \in U_1, U_2$ entonces $\exists \delta, \epsilon > 0$ tales que $B_d((0,0), \delta) \subset U_1$ y $B_d((0,0), \epsilon) \subset U_2$. Sea $\varepsilon = \min\{\delta, \epsilon\}$, entonces $B_d((0,0), \varepsilon) \subset B_d((0,0), \delta) \cap B_d((0,0), \epsilon) \subset U_1 \cap U_2$ y por lo tanto $U_1 \cap U_2 \in T$.
            \end{itemize}
        \end{enumerate}
        Además es claro que $T_u\leq T$. Comprobémoslo de la siguiente manera: sea $B_1 = B_d((x,y), r)\in B_u$ con $(x,y) \in B_1$, $r>0$ entonces:
        \begin{itemize}
            \item Si $(x,y) = (0,0)$ entonces está claro que existe un $B_d((0,0), r') \in T$ con $r>r'>0$ tal que $(x,y)\in B_d((0,0), r') \subset B_d((0,0), r)$
            \item Si $(x,y) \neq (0,0)$ entonces para $d((x,y),(0,0)) > \epsilon > 0$ y $r > \epsilon$ se tiene $(0,0)\notin B_d((x,y), \epsilon) \Rightarrow B_d((x,y), \epsilon) \in T$ y por lo tanto $(x,y)\in B_d((x,y), \epsilon) \subset B_d((x,y), r)$
        \end{itemize}
        Por inclusión de las bases, $T_u\leq T$. Pero la otra implicación no es cierta, pues por ejemplo$\overline{B_d}((0,0), 1) \in T$ pero $\overline{B_d}((0,0), 1)\notin T_u$.
        
        \item (1 punto) Calcula una base de entornos de $(x,y) \in \mathbb{R}^2$ en $(\mathbb{R}^2, T)$:\\
        Sea $(x,y)\in\bb{R}^2$ entonces:
        \[\begin{cases}
            \cc{B}_{(x,y)} = \{\{(x,y)\}\} & \text{ si }(x,y)\neq (0,0)\\
            \cc{B}_{(0,0)} = \{B_d((0,0), r) : r > 0\}
        \end{cases}\]
        Vamos a comprobar que, efectivamente, es una base de entornos distinguiendo dos casos:
        \begin{itemize}
            \item $(x,y) = (0,0)$: Todo $B_d((0,0), r)$ es abierto en $T$ y contiene a $(0,0)$. Además, si $U$ es un entorno de $(0,0)$, existe un abierto $W\in T$ con $(0,0)\in W\subset U$, y por definición de $T$, $\exists \varepsilon > 0$ tal que $(0,0)\in V = B_d((0,0), \varepsilon) \subset W \subset U$ con $V\in\cc{B}_{(0,0)}$.
            \item $(x,y) \neq (0,0)$: Como $(0,0)\notin \{(x,y)\}$, se tiene que $V = \{(x,y)\}\in T$ es un entorno abierto de $(x,y)$. Así, para cualquier entorno $U$ de $(x,y)$ se cumple $(x,y)\in V\subset U$ con $V\in\cc{B}_{(x,y)}$.
        \end{itemize}
        Efectivamente $\cc{B}_{(x,y)}$ es una base de entornos de $(x,y)$ en $(\mathbb{R}^2, T)$.

        \item (1 punto) Calcula la adherencia y el interior de $\mathbb{Q} \times \{0\}$ y $\mathbb{Q} \times (0,1)$ en $(\mathbb{R}^2, T)$.
        \begin{itemize}
        \item $A = \mathbb{Q} \times \{0\}$\\
        Demostramos que $A^{\circ} = (\mathbb{Q}\setminus \{0\}) \times \{0\}$. Lo probaremos por doble implicación:
        \begin{itemize}
            \item[$\subseteq$] Basta ver que $(0,0) \notin A^{\circ}$. Si $(0,0) \in A^{\circ}$, existiría un abierto $U \in T$ tal que $(0,0) \in U \subset A$. Por definición de $T$, existiría $\varepsilon > 0$ tal que $B_d((0,0), \varepsilon) \subset U \subset A = \mathbb{Q} \times \{0\}$, lo cual es absurdo porque el punto $(0, \varepsilon/2) \in B_d((0,0), \varepsilon)$ no pertenece a $A$.
            \item[$\supseteq$] Como $(0,0) \notin (\mathbb{Q}\setminus \{0\}) \times \{0\}$, por definición de $T$ este conjunto es abierto en $(\mathbb{R}^2, T)$. Al estar contenido en $A$, se cumple $(\mathbb{Q}\setminus \{0\}) \times \{0\} \subset A^{\circ}$.
        \end{itemize}

        Por otro lado veamos que $\overline{A} = A = \mathbb{Q} \times \{0\}$ probando que $A$ es cerrado. Tenemos que $(0,0) \in A$, por lo que $(0,0) \notin \mathbb{R}^2 \setminus A$. Por definición de $T$, todo subconjunto que no contiene al origen es abierto, luego $\mathbb{R}^2 \setminus A \in T$ y por tanto $A$ es cerrado y $\overline{A} = A$.

        \item $B = \mathbb{Q} \times (0,1)$\\
        Es fácil ver que $B^{\circ} = B = \mathbb{Q} \times (0,1)$ pues $B\in T$. Para probarlo veamos que $(0,0)\notin B$, lo cual es inmediato ya que la segunda coordenada de $(0,0)$ es $0 \notin (0,1)$.

        Ahora veamos que $\overline{B} = (\mathbb{Q} \times (0,1)) \cup \{(0,0)\}$ por doble implicación:
        \begin{itemize}
            \item[$\supseteq$] Claramente $B \subset \overline{B}$. Para ver que $(0,0) \in \overline{B}$, sea $U \in T$ un entorno abierto cualquiera con $(0,0) \in U$. Por definición de $T$, existe $\varepsilon > 0$ tal que $B_d((0,0), \varepsilon) \subset U$. Tomando $r = \min\{\varepsilon/2, 1/2\} > 0$, se cumple $(0, r) \in B_d((0,0), \varepsilon) \subset U$ y además $(0, r) \in \mathbb{Q} \times (0,1) = B$. Luego $U \cap B \neq \varnothing$ y por tanto $(0,0) \in \overline{B}$.
            \item[$\subseteq$] Sea $F = B \cup \{(0,0)\}$. Su complementario $\mathbb{R}^2 \setminus F$ no contiene al $(0,0)$, por lo que $\mathbb{R}^2 \setminus F \in T$ y $F$ es cerrado. Como $B \subset F$ y $\overline{B}$ es el menor cerrado que contiene a $B$, se concluye que $\overline{B} \subset F$.
        \end{itemize}
        \end{itemize}
