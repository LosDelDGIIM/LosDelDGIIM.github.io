\begin{ejercicio}
Defina el concepto de \emph{base} de una topología $\cc{T}$ sobre un conjunto $X$.
Sea $(X, \cc{T})$ un espacio topológico, $\cc{B}$ una base de $\cc{T}$ y $A \subseteq X$ un subconjunto no vacío. Demuestre que la familia
\[
    \cc{B}_A = \{ B \cap A : B \in \cc{B} \}
\]
constituye una base para la topología inducida $\cc{T}_A$ sobre $A$.

\medskip
Llamamos \emph{base} de una topología $\cc{T}$ sobre un conjunto $X$ a una familia de abiertos $\cc{B} \subseteq \cc{T}$ (llamados \emph{abiertos básicos}) tal que todo abierto $U \in \cc{T}$ puede expresarse como unión de elementos de $\cc{B}$; es decir, existe una subfamilia $\{B_i\}_{i \in I} \subseteq \cc{B}$ tal que
\[
    U = \bigcup_{i \in I} B_i,
\]
o, equivalentemente, para todo $U \in \cc{T}$ y todo $x \in U$, existe $B \in \cc{B}$ tal que $x \in B \subseteq U$.

Sea ahora $(X, \cc{T})$ un espacio topológico, $\cc{B}$ una base de $\cc{T}$ y $A \subseteq X$ un subconjunto no vacío. Recordemos que la topología inducida sobre $A$ viene dada por $\cc{T}_A = \{ U \cap A : U \in \cc{T} \}$. Para probar que $\cc{B}_A$ es base de $\cc{T}_A$, comprobamos las dos condiciones de la definición:
\begin{itemize}
    \item $\cc{B}_A \subseteq \cc{T}_A$: Dado $B \cap A \in \cc{B}_A$, como $B \in \cc{B} \subseteq \cc{T}$, por definición de topología inducida se tiene directamente que $B \cap A \in \cc{T}_A$.
    \item Todo abierto de $\cc{T}_A$ es unión de elementos de $\cc{B}_A$: Sea $V \in \cc{T}_A$ arbitrario. Existe $U \in \cc{T}$ tal que $V = U \cap A$. Como $\cc{B}$ es base de $\cc{T}$, podemos escribir $\displaystyle U = \bigcup_{i \in I} B_i$ con $B_i \in \cc{B}$ para cada $i \in I$. Aplicando la distributividad de la intersección respecto de la unión:
    \[
        V = U \cap A = \left( \bigcup_{i \in I} B_i \right) \cap A = \bigcup_{i \in I} (B_i \cap A).
    \]
    Como cada término $B_i \cap A$ pertenece a $\cc{B}_A$, queda demostrado que $\cc{B}_A$ es base de $\cc{T}_A$.
\end{itemize}
\end{ejercicio}

\begin{ejercicio}
En el plano real $\bb{R}^2$, considérese la familia de subconjuntos
\[
    \cc{B} = \big\{ \{x\} \times [y, +\infty) : x, y \in \bb{R} \big\}.
\]
Se pide:
\begin{enumerate}[label=\alph*)]
    \item Demuestre que $\cc{B}$ es base de una topología $\cc{T}$ en $\bb{R}^2$.
    
    Veamos que la familia $\cc{B}$ cumple las dos condiciones caracterizadoras de base:
    \begin{itemize}
        \item Por doble inclusión veremos que $\displaystyle \bigcup_{B \in \cc{B}} B = \bb{R}^2$:
        \begin{itemize}
            \item[($\subseteq$)] Dado $(x, y) \in \displaystyle \bigcup_{B \in \cc{B}} B$, existe $B = \{x\} \times [y_0, +\infty) \in \cc{B}$ con $y_0 \leq y$ tal que $(x, y) \in B \subseteq \bb{R}^2$.
            \item[($\supseteq$)] Dado $(x, y) \in \bb{R}^2$, basta tomar $B = \{x\} \times [y, +\infty) \in \cc{B}$. Se cumple $(x, y) \in B$, luego $(x, y) \in \displaystyle \bigcup_{B \in \cc{B}} B$.
        \end{itemize}
        \item Sean $B_1 = \{x_1\} \times [y_1, +\infty)$ y $B_2 = \{x_2\} \times [y_2, +\infty)$ dos elementos de $\cc{B}$ y supongamos que $(x, y) \in B_1 \cap B_2$. Entonces $x = x_1 = x_2$ e $y \geq \max(y_1, y_2)$. Tomando $y_3 = \max(y_1, y_2)$, el conjunto $B_3 = \{x\} \times [y_3, +\infty)$ pertenece a $\cc{B}$ y cumple
        \[
            (x, y) \in B_3 \subseteq B_1 \cap B_2.
        \]
    \end{itemize}
    Por el teorema de caracterización de bases, existe una única topología $\cc{T}(\cc{B})$ en $\bb{R}^2$ que tiene a $\cc{B}$ como base.

    \item Determine qué rectas del plano $\bb{R}^2$ heredan la topología discreta como subespacios topológicos de $(\bb{R}^2, \cc{T}(\cc{B}))$.

    Una recta $L \subseteq \bb{R}^2$ hereda la topología discreta $\cc{T}_{|L}$ si y solo si todo conjunto unitario $\{(x_0, y_0)\} \subseteq L$ es abierto en $\cc{T}_{|L}$; es decir, si para cada $(x_0, y_0) \in L$ existe un abierto básico $B \in \cc{B}$ tal que $B \cap L = \{(x_0, y_0)\}$. Distinguimos dos casos:
    \begin{itemize}
        \item Rectas no verticales: ($L : y = mx + n$, con $m, n \in \bb{R}$): Dado $(x_0, y_0) \in L$, elegimos el básico $B = \{x_0\} \times [y_0, +\infty) \in \cc{B}$. Como la recta $L$ corta a cada vertical $\{x_0\} \times \bb{R}$ en un único punto, se tiene
        \[
            B \cap L = \big(\{x_0\} \times [y_0, +\infty)\big) \cap L = \{(x_0, y_0)\}.
        \]
        Por tanto, todos los puntos son abiertos en $\cc{T}_{|L}$ y las rectas no verticales sí son discretas.
        \item Rectas verticales ($L = \{x_0\} \times \bb{R}$, con $x_0 \in \bb{R}$): Para cualquier $(x_0, y_0) \in L$, todo abierto $U \in \cc{T}$ que contenga a $(x_0, y_0)$ contiene algún básico $B = \{x_0\} \times [y', +\infty)$ con $y' \leq y_0$. Entonces
        \[
            B \cap L = \{x_0\} \times [y', +\infty) = B \neq \{(x_0, y_0)\},
        \]
        ya que contiene infinitos puntos de $L$ por encima de $y_0$. Así, ningún punto es abierto en $L$ y las rectas verticales no son discretas.
    \end{itemize}

    \item Calcule el interior, la clausura y la frontera en $(\bb{R}^2, \cc{T})$ de los siguientes subconjuntos:
    \begin{enumerate}
        \item $D = \{ (x, y) \in \bb{R}^2 : x^2 + y^2 < 1 \}$.\\
        $D^{\circ} = \varnothing$.\\
        Dado cualquier $(x, y) \in D$, todo abierto básico que lo contenga es de la forma $B = \{x\} \times [y', +\infty)$ con $y' \leq y$. Como $B$ no es acotado superiormente en la segunda coordenada y $D$ sí es acotado, se cumple $B \not\subseteq D$. Luego $(x, y) \notin D^{\circ}$ y $D^{\circ} = \varnothing$.\\

        $\overline{D} = D \cup \big((-1, 1) \times (-\infty, 0]\big)$.\\
        Probamos la igualdad por doble inclusión:
        \begin{itemize}
            \item[($\supseteq$)] Como $D \subseteq \overline{D}$, basta ver que todo $(x, y) \in C \setminus D$ pertenece a $\overline{D}$. Si $(x, y) \in C \setminus D$, entonces $|x| < 1$ e $y \leq -\sqrt{1 - x^2} < 0$. Dado cualquier básico $B = \{x\} \times [y', +\infty)$ que contenga a $(x, y)$, se cumple $y' \leq y < 0 < \sqrt{1 - x^2}$, por lo que $(x, 0) \in B \cap D \neq \varnothing$. Así, $(x, y) \in \overline{D}$.
            \item[($\subseteq$)] Sea $(x, y) \in \bb{R}^2 \setminus C$. Distinguimos dos casos:
            \begin{enumerate}
                \item Si $|x| \geq 1$, el básico $B = \{x\} \times [y, +\infty)$ contiene a $(x, y)$ y es disjunto con $D$ (pues todo punto de $D$ tiene $|x| < 1$).
                \item Si $|x| < 1$ e $y \geq \sqrt{1 - x^2}$, el básico $B = \{x\} \times [y, +\infty)$ contiene a $(x, y)$ y cumple $B \cap D = \varnothing$ (pues todos sus puntos satisfacen $x^2 + \tilde{y}^2 \geq x^2 + y^2 \geq 1$).
            \end{enumerate}
            En ambos casos $(x, y) \notin \overline{D}$, de donde $\overline{D} \subseteq C$.
        \end{itemize}
        $\partial D = \overline{D} \setminus D^{\circ} = \overline{D} = D \cup \big((-1, 1) \times (-\infty, 0]\big)$.

        \item $Q = \{0\} \times \bb{Q}$.\\
        $Q^{\circ} = \varnothing$.\\
        Sea $(0, q) \in Q$. Cualquier abierto que lo contenga incluye un básico $B = \{0\} \times [y', +\infty)$ con $y' \leq q$. Por la densidad de los irracionales en $\bb{R}$, existe $p \in (\bb{R} \setminus \bb{Q}) \cap [y', +\infty)$, de modo que $(0, p) \in B$ pero $(0, p) \notin Q$. Así, $B \not\subseteq Q$ y $(0, q) \notin Q^{\circ}$.\\

        $\overline{Q} = \{0\} \times \bb{R}$.\\
        Si $x \neq 0$, el básico $B = \{x\} \times [y, +\infty)$ contiene a $(x, y)$ y verifica $B \cap Q = \varnothing$, luego $(x, y) \notin \overline{Q}$. Por el contrario, si $x = 0$, para cualquier $y \in \bb{R}$ y cualquier básico $B = \{0\} \times [y', +\infty)$ con $y' \leq y$, por densidad de $\bb{Q}$ en $\bb{R}$ existe $q \in \bb{Q} \cap [y', +\infty)$, por lo que $(0, q) \in B \cap Q \neq \varnothing$. Por tanto, $\overline{Q} = \{0\} \times \bb{R}$.\\

        $\partial Q = \overline{Q} \setminus Q^{\circ} = \{0\} \times \bb{R}$.

        \item $H = [0, +\infty) \times \bb{R}$.\\
        $H^{\circ} = H$.\\
        Para cada $(x, y) \in H$ se tiene $x \geq 0$, luego el básico $B = \{x\} \times [y, +\infty)$ satisface $(x, y) \in B \subseteq H$. Así, $H$ es abierto y $H^{\circ} = H$.\\

        $\overline{H} = H$.\\
        Su complementario es $\bb{R}^2 \setminus H = (-\infty, 0) \times \bb{R}$. Para todo $(x, y) \in \bb{R}^2 \setminus H$ se cumple $x < 0$, luego $B = \{x\} \times [y, +\infty) \subseteq (-\infty, 0) \times \bb{R}$. Por tanto, $\bb{R}^2 \setminus H$ es abierto, $H$ es cerrado y $\overline{H} = H$.\\

        $\partial H = \overline{H} \setminus H^{\circ} = \varnothing$.

    \item Estudie si el espacio topológico $(\bb{R}^2, \cc{T})$ verifica las propiedades de separación $T_1$ y $T_2$ (Hausdorff), así como el primer axioma de numerabilidad ($1\mathrm{AN}$) y el segundo axioma de numerabilidad ($2\mathrm{AN}$).
    \begin{itemize}
        \item \textbf{Axiomas de separación ($T_1$ y $T_2$):} No es $T_1$ ni, por consiguiente, $T_2$.\\
        Tomemos dos puntos distintos en la misma vertical $(x, y), (x, y') \in \bb{R}^2$ con $y < y'$. Todo abierto $U \in \cc{T}$ tal que $(x, y) \in U$ contiene al básico $\{x\} \times [y, +\infty)$. Como $y < y'$, se tiene $(x, y') \in \{x\} \times [y, +\infty) \subseteq U$. Por tanto, no existe ningún abierto que contenga a $(x, y)$ y deje fuera a $(x, y')$, lo que impide que $(\bb{R}^2, \cc{T})$ sea $T_1$ (y tampoco $T_2$).

            \draw[->, gray] (-1.5,0) -- (2.2,0) node[right] {$x$};
            \draw[->, gray] (0,-0.8) -- (0,2.5) node[above] {$y$};

            \fill[black] (0.9, 0.4) circle (2.2pt) node[left] {\small $(x,y)$};
            \fill[blue] (0.9, 1.5) circle (2.2pt) node[left] {\small $(x,y')$};
            \node[align=center, font=\footnotesize] at (0.4, -1.3) {};
        \end{tikzpicture}
        \end{center}

        \item \textbf{Primer axioma de numerabilidad ($1\mathrm{AN}$):} Sí lo verifica.\\
        Para cada punto $(x, y) \in \bb{R}^2$, la familia unitaria $\cc{B}_{(x, y)} = \big\{ \{x\} \times [y, +\infty) \big\}$ es una base de entornos finita (luego numerable) de $(x, y)$, ya que dado cualquier abierto $U \in \cc{T}$ con $(x, y) \in U$, existe un básico $\{x\} \times [y', +\infty) \subseteq U$ con $y' \leq y$, y por tanto $\{x\} \times [y, +\infty) \subseteq \{x\} \times [y', +\infty) \subseteq U$.

        \item \textbf{Segundo axioma de numerabilidad ($2\mathrm{AN}$):} No lo verifica.\\
        Por el apartado b), la recta horizontal $L = \bb{R} \times \{0\}$ es un subespacio discreto no numerable (tiene el cardinal de $\bb{R}$). Si $(\bb{R}^2, \cc{T})$ fuese $2\mathrm{AN}$, esta propiedad sería hereditaria y el subespacio $(L, \cc{T}_L)$ debería admitir una base numerable. Sin embargo, un espacio discreto solo es $2\mathrm{AN}$ si es numerable, lo cual es una contradicción.
    \end{itemize}
\end{enumerate}
