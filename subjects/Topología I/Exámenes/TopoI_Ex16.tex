    \begin{ejercicio}
       En $\bb{R}$, se considera la topología $T$ de la que conocemos una base dada por:
        \[
        \mathcal{B}=\{[-r,r]\quad : \quad r>0\}.
        \]
        \begin{enumerate}
            \item Estudiar si $(\mathbb{R}, T)$ es un espacio $T_1$. ¿Es cierto que $T\cap C_T = \{\emptyset, \mathbb{R}\}$?\\
            Veamos que $(\mathbb{R}, T)$ no es un espacio $T_1$. Se puede ver claramente tomando $x = 0$ e $y \in \mathbb{R} \setminus \{0\}$ (por ejemplo, $y \in \mathbb{R}^+$).
            Para cualquier abierto $U_y \in T$ tal que $y \in U_y$, como $\mathcal{B}$ es base de $T$, existe algún básico $[-r, r] \in \mathcal{B}$ (con $r > 0$) tal que $y \in [-r, r] \subseteq U_y$. 
            Sin embargo, como $r > 0$, se cumple que $0 \in [-r, r] \subseteq U_y$. Es decir, $0$ está en todo abierto no vacío de $T$. Por tanto, todo punto distinto de $0$ comparte todos sus entornos con el $0$, luego $(\mathbb{R}, T)$ no es $T_1$ (de hecho, ni siquiera es $T_0$).\\

            Para probar que $T\cap C_T = \{\emptyset, \mathbb{R}\}$, lo haremos por doble inclusión:
            \begin{itemize}
                \item[$\supseteq$] Es evidente que $\emptyset, \mathbb{R} \in T \cap C_T$ por definición de topología.
                \item[$\subseteq$] Ahora veamos que sea $U \in T \cap C_T$ con $U \neq \emptyset$, probemos que $U = \mathbb{R}$.\\
                Como $U \in T$ y $U \neq \emptyset$, es unión de elementos de $\mathcal{B}$ y, como hemos visto antes, todo básico contiene al $0$, luego $0 \in U$.\\
                Por otro lado, como $U \in C_T$, su complementario $V = \mathbb{R} \setminus U$ es abierto. Si $V \neq \emptyset$, al ser también unión de elementos de $\mathcal{B}$, tendríamos que $0 \in V = \mathbb{R} \setminus U$ lo cual es una contradicción porque $0\in U \Rightarrow V = \emptyset$, de donde concluimos que $U = \mathbb{R}$. 
            \end{itemize}
        \item Para cada $x\neq 0$, encontrar una base de entornos de $x$ en $(\mathbb{R}, T)$ con un único elemento.
        La base de entornos más económica (de cardinal 1) viene dada por:
        \[
            \mathcal{B}_x = \{[-x, x]\} \text{ si } x > 0, \qquad \mathcal{B}_x = \{[x, -x]\} \text{ si } x < 0
        \]
        (es decir, $\mathcal{B}_x = \{[-|x|, |x|]\}$ para todo $x \in \bb{R} \setminus \{0\}$).
        \\Probemos que $\mathcal{B}_x$ es efectivamente una base de entornos para cada $x \in \bb{R} \setminus \{0\}$:
        \begin{itemize}
            \item \textbf{Reflexividad puntual:} $\mathcal{B}_x \neq \emptyset$ y para el único elemento $V = [-|x|, |x|] \in \mathcal{B}_x$ se cumple claramente que $x \in [-|x|, |x|] = V$.
            
            \item \textbf{Intersección finita:} Sean $V_1, V_2 \in \mathcal{B}_x$. Como $\mathcal{B}_x$ es unitaria, necesariamente $V_1 = V_2 = [-|x|, |x|]$. Tomando $V_3 = [-|x|, |x|] \in \mathcal{B}_x$, se verifica trivialmente que $V_3 \subseteq V_1 \cap V_2$.
            
            \item \textbf{Axioma de Bourbaki:} Sea $V = [-|x|, |x|] \in \mathcal{B}_x$. Como $|x| > 0$, tenemos que $V \in \mathcal{B} \subseteq T$ es decir, $V$ ya es abierto, luego entorno de todos sus puntos por un entorno básico ($\forall y\in V \quad \exists V_y\in \mathcal{B}_y$ tal que  $y\in V_y\in V$)
        \end{itemize}
        Probado así que $\mathcal{B}_x$ conforma una base de entornos para cada $x \neq 0$.
                
        \item Demostrar que $(\mathbb{R}, T)$ cumple el primer axioma de numerabilidad pero no el segundo.
        \\Teniendo definida la base de entornos $\forall x\in \mathbb{R}\setminus\{0\}$, agregemos también para $x=0$, de manera que:
        \[\begin{cases}
            \mathcal{B}_x = \{[-|x|, |x|]\} \quad\text{para todo } x \in \bb{R} \setminus \{0\}\\
            \mathcal{B}_0 = \{0\} \quad\text{que es en realidad un caso particular donde }\{0\}=[-0,0]
        \end{cases}\]
        Luego $\mathcal{B}_x$ es finita $\forall x\in \bb{R}$ y, por tanto, numerable.\\

        Ahora veremos que no cumple el segundo axioma de numerabilidad. Supongamos por reducción al absurdo que $(\bb{R}, T)$ es (2AN). Como $\mathcal{B}$ es una base de la topología $T$, por ser (2AN) existirá una subfamilia numerable $\mathcal{B}_n \subseteq \mathcal{B}$ que también es base de $T$:
        \[
            \mathcal{B}_n = \{[-r_n, r_n] : n \in \bb{N}\} \quad \text{con } r_n \in \bb{R}^+.
        \]
        Sea $R = \{r_n : n \in \bb{N}\} \subseteq \bb{R}^+$. Como $R$ es numerable y $\bb{R}^+$ es no numerable, se tiene que $\bb{R}^+ \setminus R \neq \emptyset$.
        
        Tomemos $x \in \bb{R}^+ \setminus R \Rightarrow [-x, x] \in \mathcal{B} \subseteq T$, por lo que es abierto y debe expresarse como unión de elementos de $\mathcal{B}_N$:
        \[
            [-x, x] = \bigcup_{n \in J} [-r_n, r_n] \quad \text{para algún } J \subseteq \bb{N}.
        \]
        Como $x \in [-x, x]$, existe algún $n_0 \in J$ tal que $x \in [-r_{n_0}, r_{n_0}]$, lo que implica que $x \le r_{n_0}$. Por otro lado, como $[-r_{n_0}, r_{n_0}] \subseteq [-x, x]$, también se cumple que $r_{n_0} \le x$. De ambas desigualdades deducimos que $r_{n_0} = x$, lo cual es una contradicción pues $x \notin R$. Por tanto, no existe base numerable y $(\bb{R}, T)$ NO es (2AN).
            
        \item Calcular el cierre en $(\mathbb{R}, T)$ del conjunto $A=\{1\}$ y el interior del conjunto $C=(-3,5)$
        \begin{itemize}
    \item \textbf{Probemos que $\overline{A} = \bb{R} \setminus ]-1, 1[$:}
    \begin{itemize}
        \item[$\subseteq$)] Probemos por contrarrecíproco que si $x \in ]-1, 1[ = \mathbb{R}\setminus(\bb{R}\setminus ]-1,1[)$, entonces $x \notin \overline{A}$ (es decir, que ningún punto fuera de $\bb{R} \setminus ]-1, 1[$ pertenece a la adherencia)\\
        Si $x \in ]-1, 1[ \setminus \{0\}$, sabemos que $|x| < 1$. Considerando su entorno básico $[-|x|, |x|] \in \mathcal{B}_x$, es evidente que $[-|x|, |x|] \cap \{1\} = \emptyset$ pues $|x| < 1$. Por tanto, $x \notin \overline{A}$. \\
        Por contrarrecíproco, todo $x \in \overline{A}$ cumple que $x \in \bb{R} \setminus ]-1, 1[$.

        \item[$\supseteq$)] Sea $x \in \bb{R} \setminus ]-1, 1[$ arbitrario, es decir, $|x| \ge 1$. Como $x \neq 0$, su base de entornos es $\mathcal{B}_x = \{[-|x|, |x|]\}$. Dado que $|x| \ge 1$, se tiene que $[-1, 1] \subseteq [-|x|, |x|]$, por lo que:
        \[
            [-|x|, |x|] \cap \{1\} = \{1\} \neq \emptyset.
        \]
        Como el único entorno básico de $x$ corta a $A$, todo entorno de $x$ corta a $A$, de donde $x \in \overline{A}$.
    \end{itemize}

    \item \textbf{Ahora veamos que $C^\circ = ]-3, 3[$:}
    \begin{itemize}
        \item[$\subseteq$)] Probemos por contrarrecíproco que si $x \in \bb{R} \setminus ]-3, 3[$, entonces $x \notin C^\circ$. \\
        Sea $x \in \bb{R} \setminus ]-3, 3[$, es decir, $|x| \ge 3$. Como $x \neq 0$, su único entorno básico es $[-|x|, |x|]$. Dado que $|x| \ge 3 \Rightarrow [-|x|, |x|]\nsubseteq C$, lo que equivale a probar que $[-|x|, |x|]\cap(\bb{R}\setminus C)\neq \emptyset$.\\
        Efectivamente $-3\in [-|x|, |x|]\cap(\bb{R}\setminus C) \Rightarrow [-|x|, |x|] \nsubseteq C$, lo que implica que ningún entorno de $x$ está contenido en $C$, luego $x \notin C^\circ$. Por contrarrecíproco, $C^\circ \subseteq ]-3, 3[$.

        \item[$\supseteq$)] Sea $x \in ]-3, 3[$. Si $x = 0$, el básico $[-1, 1]$ es un entorno de $0$ contenido en $(-3, 5)$, luego $0 \in C^\circ$. \\
        Si $x \in ]-3, 3[ \setminus \{0\}$, entonces $|x| < 3$, por lo que su entorno básico cumple:
        \[
            x \in [-|x|, |x|] \subseteq ]-3, 3[ \subseteq (-3, 5) = C,
        \]
        de donde concluimos directamente que $x \in C^\circ$.
    \end{itemize}
\end{itemize}
        \item Probar que si un subconjunto $D\subseteq \mathbb{R}$ es discreto, entonces consta de un único punto.\\
        Veamos que si $(D, T_{|D}) = (D, T_d)$ , por definición de topología inducida tendremos que:
        \[
            T_{|D} = \{U \cap D : U \in T\} = \left\{ U \cap D : U = \bigcup_{i \in I} B_i \text{ con } B_i \in \mathcal{B} \right\}
        \]
        Y por tanto su base es $\mathcal{B}_D = \{B \cap D : B \in \mathcal{B}\} = \{[-r, r] \cap D : r > 0\}$.
        
        Como $(D, T_{|D})$ es discreto, todo subconjunto unitario $\{x\}\in D$ es abierto en $T_{|D}$, por lo que debe expresarse como unión de elementos de $\mathcal{B}_D$ (en particular, existe algún $r > 0$ tal que $\{x\} = [-r, r] \cap D$).
        
        Supongamos por reducción al absurdo que $D$ contiene al menos dos puntos distintos $x, y \in D$ ($x \neq y$). Como $\{x\} \in T_{|D}$, existe un básico $[-r_1, r_1] \in \mathcal{B}$ (con $r_1 > 0$) tal que:
        \[
            \{x\} = [-r_1, r_1] \cap D.
        \]
        Como $x \in [-r_1, r_1]$, se tiene que $|x| \le r_1$. Además, como $y \in D$ e $y \neq x$, necesariamente $y \notin [-r_1, r_1]$, por lo que $|y| > r_1$. Combinando ambas desigualdades obtenemos:
        \[
            |x| \le r_1 < |y| \implies |x| < |y|.
        \]
        Ahora, como $\{y\} \in T_{|D}$, existe $r_2 > 0$ tal que $\{y\} = [-r_2, r_2] \cap D$), por un razonamiento totalmente análogo se deduce que:
        \[
            |y| \le r_2 < |x| \implies |y| < |x|,
        \]
        llegando a la contradicción $|x| < |y| < |x|$. Por tanto, $D$ no puede tener más de un punto.
        \end{enumerate}
    \end{ejercicio}
