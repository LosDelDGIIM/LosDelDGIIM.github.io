\chapter{Dominios de factorización única}

\begin{definicion}[Divisor propio]
    Sea $A$ un DI y $a \in A$. Un divisor no trivial de $a$ se denomina \textbf{divisor propio de $a$}, es decir, un elemento $b \in A$ tal que:
    \[
        b \notin \mathcal{U}(A), \quad b \mid a \quad \text{y} \quad a \nmid b.
    \]
\end{definicion}

\begin{definicion}[Dominio de factorización única]
    Sea $A$ un DI. Diremos que es un \textbf{dominio de factorización única} (abreviado DFU) si todo elemento no nulo y no invertible $a \in A \setminus (\{0\} \cup \mathcal{U}(A))$ admite una factorización:
    \[
        a = p_1 p_2 \cdots p_s,
    \]
    donde cada $p_i$ es irreducible para todo $i \in \{1, \dots, s\}$. Además, dicha factorización es esencialmente única: si
    \[
        a = q_1 q_2 \cdots q_r
    \]
    es otra factorización en irreducibles, entonces $r = s$ y existe una reordenación tal que:
    \[
        p_i \sim q_i \quad \forall i \in \{1, \dots, s\}.
    \]
\end{definicion}

\begin{ejemplo}
    Dos factorizaciones son esencialmente iguales en $\mathbb{Z}$ cuando los factores coinciden salvo asociados:
    \begin{align*}
        -6 &= (-2) \cdot 3, \\
        -6 &= 2 \cdot (-3),
    \end{align*}
    puesto que $2 \sim -2$ y $3 \sim -3$.
\end{ejemplo}

En cualquier dominio de integridad $A$, podemos elegir un conjunto $P \neq \emptyset$ de representantes de sus elementos irreducibles en el siguiente sentido:
\begin{enumerate}[label=\arabic*)]
    \item Para todo $p \in P$, $p$ es irreducible.
    \item Si $q \in A$ es irreducible, existe un único $p \in P$ tal que $p \sim q$.
    \item Si $p, p' \in P$ con $p \neq p'$, entonces $p \not\sim p'$.
\end{enumerate}

\begin{ejemplo}
    Ejemplos habituales del conjunto $P$ son:
    \begin{itemize}
        \item En $\mathbb{Z}$: $P = \{p \in \mathbb{Z} \mid p \text{ es irreducible y } p > 0\}$.
        \item En $K[x]$ ($K$ cuerpo): $P$ es el conjunto de todos los polinomios irreducibles mónicos.
    \end{itemize}
\end{ejemplo}

Si $A$ es un DFU, todo elemento $a \in A \setminus \{0\}$ se expresa de forma esencialmente única como:
\[
    a = u p_1^{e_1} p_2^{e_2} \cdots p_r^{e_r},
\]
con $u \in \mathcal{U}(A)$, $r \geq 0$, elementos $p_i \in P$ distintos dos a dos y exponentes enteros $e_i \geq 1$.

\begin{notacion}
    Para cada $a \in A \setminus \{0\}$ y cada $p \in P$, denotaremos por $e(p,a)$ al exponente del irreducible $p$ en la factorización de $a$, conveniendo que $e(p,a) = 0$ si $p$ no divide a $a$. Así, todo $a \in A \setminus \{0\}$ se escribe como:
    \[
        a = u \prod\limits_{p \in P} p^{e(p,a)}, \quad \text{con } u \in \mathcal{U}(A).
    \]
\end{notacion}

\begin{lema}
    Sea $A$ un DFU con $a, b \in A \setminus \{0\}$ y sea $p \in P$. Entonces:
    \[
        e(p,ab) = e(p,a) + e(p,b).
    \]
\end{lema}

\begin{proof}
    Como $A$ es un DFU, existen $u, v \in \mathcal{U}(A)$ tales que:
    \[
        a = u \prod\limits_{p \in P} p^{e(p,a)}, \qquad b = v \prod\limits_{p \in P} p^{e(p,b)}.
    \]
    Multiplicando ambas expresiones:
    \[
        ab = \left(u \prod\limits_{p \in P} p^{e(p,a)}\right) \left(v \prod\limits_{p \in P} p^{e(p,b)}\right) = uv \prod\limits_{p \in P} p^{e(p,a) + e(p,b)}.
    \]
    Por otro lado, la factorización canónica de $ab$ viene dada por $ab = w \prod\limits_{p \in P} p^{e(p,ab)}$ para algún $w \in \mathcal{U}(A)$. Por la unicidad de la descomposición en irreducibles, deducimos que $w = uv$ y que:
    \[
        e(p,ab) = e(p,a) + e(p,b) \quad \forall p \in P. \qedhere
    \]
\end{proof}

\begin{lema}
    Sea $A$ un DFU y sean $a, c \in A \setminus \{0\}$. Entonces:
    \[
        a \mid c \iff e(p,a) \leq e(p,c) \quad \forall p \in P.
    \]
\end{lema}

\begin{proof}~
\begin{description}
    \item[$\implies$)] Si $a \mid c$, existe $c' \in A$ tal que $c = ac'$. Aplicando el lema anterior:
    \[
        e(p,c) = e(p,a) + e(p,c') \quad \forall p \in P.
    \]
    Dado que $e(p,c') \geq 0$, se concluye que $e(p,c) \geq e(p,a)$.

    \item[$\impliedby$)] Sean las descomposiciones $a = u \prod\limits_{p \in P} p^{e(p,a)}$ y $c = v \prod\limits_{p \in P} p^{e(p,c)}$ con $u, v \in \mathcal{U}(A)$. Reescribimos $c$ agrupando factores:
    \begin{align*}
        c &= v \prod\limits_{p \in P} p^{e(p,c) - e(p,a) + e(p,a)} \\
          &= v \prod\limits_{p \in P} p^{e(p,c) - e(p,a)} \prod\limits_{p \in P} p^{e(p,a)} \\
          &= \left(v u^{-1} \prod\limits_{p \in P} p^{e(p,c) - e(p,a)}\right) a.
    \end{align*}
    Como $e(p,c) - e(p,a) \geq 0$ para todo $p$, el término entre corchetes pertenece a $A$, luego $a \mid c$.
\end{description}
\end{proof}

\begin{prop}\label{propExistenMCDDFU}
    Sea $A$ un DFU y consideremos dos elementos no nulos descompuestos como:
    \[
        a = u \prod\limits_{p \in P} p^{e(p,a)}, \qquad b = v \prod\limits_{p \in P} p^{e(p,b)}, \quad u, v \in \mathcal{U}(A).
    \]
    Entonces existen $\mcd(a,b)$ y $\mcm(a,b)$, dados por:
    \begin{align*}
        \mcd(a,b) &= \prod\limits_{p \in P} p^{\min\{e(p,a),\, e(p,b)\}}, \\
        \mcm(a,b) &= \prod\limits_{p \in P} p^{\max\{e(p,a),\, e(p,b)\}}.
    \end{align*}
\end{prop}

\begin{proof}
    Definimos $\alpha := \prod\limits_{p \in P} p^{\min\{e(p,a),\, e(p,b)\}}$ y $\beta := \prod\limits_{p \in P} p^{\max\{e(p,a),\, e(p,b)\}}$.
    \begin{itemize}
        \item Para el máximo común divisor:
        \begin{itemize}
            \item Como $\min\{e(p,a), e(p,b)\} \leq e(p,a)$ y $\leq e(p,b)$, por el lema anterior se tiene $\alpha \mid a$ y $\alpha \mid b$.
            \item Si $t \in A$ satisface $t \mid a$ y $t \mid b$, entonces $e(p,t) \leq e(p,a)$ y $e(p,t) \leq e(p,b)$. Por tanto, $e(p,t) \leq \min\{e(p,a), e(p,b)\} = e(p,\alpha)$, lo cual prueba que $t \mid \alpha$.
        \end{itemize}
        Concluimos que $\alpha = \mcd(a,b)$.

        \item Para el mínimo común múltiplo, el razonamiento es completamente dual sustituyendo el mínimo por el máximo.
    \end{itemize}
\end{proof}

\begin{prop}\label{propPrimoIrreducible}
    Sea $A$ un DI:
    \begin{enumerate}[label=\arabic*)]
        \item Todo elemento primo de $A$ es irreducible.
        \item Si $A$ es además un DFU, todo elemento irreducible de $A$ es primo.
    \end{enumerate}
\end{prop}
\begin{proof}
    \leavevmode
    \begin{enumerate}[label=\arabic*)]
        \item Sea $p \in A$ primo. Si $a \mid p$, existe $b \in A$ tal que $p = ab$. Como $p$ es primo y $p \mid ab$, se tiene $p \mid a$ o $p \mid b$:
        \begin{itemize}
            \item Si $p \mid a$, al ser también $a \mid p$, resulta $p \sim a$.
            \item Si $p \mid b$, existe $c \in A$ tal que $b = pc$. Entonces $p = apc$, de donde $1 = ac$, lo que implica que $a \in \mathcal{U}(A)$.
        \end{itemize}
        Así, los únicos divisores de $p$ son sus asociados y las unidades, de modo que $p$ es irreducible.

        \item Supongamos que $A$ es un DFU y sea $p \in A$ irreducible (en particular $p \neq 0$ y $p \notin \mathcal{U}(A)$). Sean $a, b \in A$ tales que $p \mid ab$. Tomando exponentes:
        \[
            1 = e(p,p) \leq e(p,ab) = e(p,a) + e(p,b).
        \]
        Al ser cantidades enteras no negativas, se sigue que $e(p,a) \geq 1$ (luego $p \mid a$) o bien $e(p,b) \geq 1$ (luego $p \mid b$). Por tanto, $p$ es primo. \qedhere
    \end{enumerate}
\end{proof}

Aunque el siguiente corolario es trivial a partir de la Proposición~\ref{propPrimoIrreducible}, conviene resaltarlo.
\begin{coro}
    En un DFU $A$, un elemento $p \in A$ es primo si y solo si es irreducible.
\end{coro}

\begin{teo}[Caracterización de los DFU]\label{teoCaracDFU}
    Sea $A$ un dominio de integridad. Los siguientes enunciados son equivalentes:
    \begin{enumerate}
        \item $A$ es un DFU.
        \item Todo elemento de $A \setminus (\{0\} \cup \mathcal{U}(A))$ se expresa como producto de irreducibles y, para cualesquiera $a, b \in A$, existen $\mcd(a,b)$ y $\mcm(a,b)$.
        \item Todo elemento de $A \setminus (\{0\} \cup \mathcal{U}(A))$ se expresa como producto de irreducibles y todo elemento irreducible de $A$ es primo.
    \end{enumerate}
\end{teo}

\begin{proof}\
\begin{description}
    \item[$1. \Longrightarrow 2.$)] Es consecuencia directa de la definición de DFU y de la Proposición~\ref{propExistenMCDDFU}.
    
    \item[$2. \Longrightarrow 3.$)] Basta probar que todo elemento irreducible es primo. Sea $p$ irreducible y supongamos que no es primo: existen $a, b \in A$ tales que $p \mid ab$, $p \nmid a$ y $p \nmid b$. Al ser $p$ irreducible y no dividir a $a$ ni a $b$, se tiene:
    \[
        \mcd(a,p) = 1 = \mcd(b,p).
    \]
    Por las propiedades del máximo común divisor, $\mcd(ab, p) = \mcd(b, p) = 1$. Pero como $p \mid ab$, se cumple que $\mcd(ab, p) \sim p$, lo que implica $p \sim 1$, en contradicción con que $p$ es irreducible.

    
    \item[$3. \Longrightarrow 1.$] Debemos probar la unicidad esencial de las factorizaciones. Supongamos que:
    \[
        p_1 p_2 \cdots p_n = q_1 q_2 \cdots q_m,
    \]
    siendo todos ellos irreducibles. Procedemos por inducción sobre $n$:
    \begin{itemize}
        \item Si $n = 1$, $p_1 = q_1 \cdots q_m$. Como $p_1$ es irreducible y por hipótesis es primo, $p_1$ divide a algún factor $q_j$ de forma $q_j=p_1\widehat{q_j}$. Diviendo a ambos lados por $p$, se obtiene:
        \[
            1 = q_1 \cdots \widehat{q_j} \cdots q_m,
        \]
        Como ningún producto de irreducibles es una unidad, se concluye que $m = 1$ y $q_1 \sim p_1$.
        \item Supongamos cierto el resultado para $n - 1$. En la igualdad $p_1 \cdots p_n = q_1 \cdots q_m$, el elemento primo $p_n$ divide al producto del segundo miembro, luego existe $j \in \{1, \dots, m\}$ tal que $p_n \mid q_j$. Al ser ambos irreducibles, $p_n \sim q_j$, de modo que $p_n = u q_j$ para cierta unidad $u \in \mathcal{U}(A)$. Simplificando $p_n$:
        \[
            p_1 \cdots p_{n-1} = u^{-1} q_1 \cdots \widehat{q_j} \cdots q_m.
        \]
        Por hipótesis de inducción, $n - 1 = m - 1$ (es decir, $n = m$) y los factores restantes son asociados dos a dos. \qedhere
    \end{itemize}
\end{description}
\end{proof}

\begin{lema}
    Sea $A$ un DE con función euclídea $\phi$, y sean $a, b \in A$ tales que $a$ es un divisor propio de $b$. Entonces:
    \[
        \phi(a) < \phi(b).
    \]
\end{lema}

\begin{proof}
    Sean $a, b \in A \setminus \mathcal{U}(A)$ no nulos. Como $a$ es un divisor propio de $b$, existe $c \in A$ no nulo y no invertible tal que $b = ac$. Por la propiedad de la función euclídea, $\phi(b) = \phi(ac) \geq \phi(a)$. Supongamos que $\phi(a) = \phi(b)$ y lleguemos a una contradicción. Realizando la división euclídea de $a$ entre $b=ac$, existen $q, r \in A$ tales que:
    \[
        a = acq + r, \quad \text{con } r = 0 \quad \text{o} \quad \phi(r) < \phi(ac).
    \]
    \begin{itemize}
        \item Si $r = 0$, entonces $a = acq$. Al ser $A$ un dominio y $a \neq 0$, se cancela $a$ y resulta $1 = cq$, lo que implicaría $c \in \mathcal{U}(A)$, contradicción.
        \item Si $r \neq 0$, entonces $\phi(r) < \phi(ac)$. Despejando $r$:
        \[
            r = a(1 - cq).
        \]
        Como $c \notin \mathcal{U}(A)$, $1 - cq \neq 0$. Por las propiedades de la función euclídea:
        \[
            \phi(r) = \phi(a(1 - cq)) \geq \phi(a) = \phi(ac),
        \]
        lo que contradice que $\phi(r) < \phi(ac)$.
    \end{itemize}
    Dado que siempre se verifica $\phi(ac) \geq \phi(a)$, concluimos estrictamente que se cumple $\phi(a) < \phi(ac)= \phi(b)$.
\end{proof}

\begin{teo}\label{teoDEEntoncesDFU}
    Todo dominio euclídeo es un dominio de factorización única.
\end{teo}

\begin{proof}
    Sea $A$ un DE con función euclídea $\phi$. Como en todo DE existen el máximo común divisor y el mínimo común múltiplo, por el Teorema~\ref{teoCaracDFU} basta demostrar que todo elemento $a \in A \setminus (\{0\} \cup \mathcal{U}(A))$ se puede descomponer en producto de irreducibles.

    Supongamos que existe algún elemento no nulo y no invertible que no admite tal factorización. En particular, $a$ no es irreducible, luego admite un divisor propio $a_1 \in A$ con $a = a_1 b_1$, donde $a_1$ o $b_1$ no se descompone en irreducibles.

    Suponiendo sin pérdida de generalidad que es $a_1$, este no puede ser irreducible y posee a su vez otro divisor propio $a_2$. Reiterando este proceso, construiríamos una sucesión infinita de divisores propios:
    \[
        a = a_0, \, a_1, \, a_2, \, \dots, \, a_n, \, \dots
    \]
    donde cada $a_i$ es divisor propio de $a_{i-1}$. Por el lema anterior:
    \[
        \phi(a_0) > \phi(a_1) > \phi(a_2) > \cdots > \phi(a_n) > \cdots
    \]
    lo cual conforma una sucesión estrictamente decreciente de números naturales, contradicción con el principio del buen orden de $\mathbb{N}$. Concluimos que todo elemento se descompone en producto de irreducibles.
\end{proof}

Los siguientes corolarios son consecuencia inmediata del Teorema~\ref{teoDEEntoncesDFU} y de los ejemplos de dominios euclídeos vistos en el tema anterior, pero conviene destacarlos por su importancia histórica y matemática.

\begin{coro}[Teorema fundamental de la aritmética]
    El anillo $\mathbb{Z}$ es un DFU.
\end{coro}

\begin{coro}
    Si $K$ es un cuerpo, el anillo de polinomios $K[x]$ es un DFU.
\end{coro}

\begin{coro}
    Para cada $n \in \{-2, -1, 2, 3\}$, el anillo $\mathbb{Z}[\sqrt{n}]$ es un DFU.
\end{coro}


Veamos no obstante que el recíproco del Teorema~\ref{teoDEEntoncesDFU} no es cierto: existen DFU que no son DE. Un ejemplo de ello es el anillo de polinomios $\mathbb{Z}[x]$, que más adelante veremos que es un DFU.
\begin{prop}
    El anillo de polinomios $\mathbb{Z}[x]$ no es un dominio euclídeo.
\end{prop}

\begin{proof}
    Basta encontrar un ideal de $\mathbb{Z}[x]$ que no sea principal. Consideremos el conjunto:
    \[
        I = \left\{p(x) = \sum\limits_{i=0}^n a_i x^i \in \mathbb{Z}[x] \;\middle|\; a_0 \in 2\mathbb{Z}\right\}.
    \]

    Comprobemos que se cumplen las condiciones de ideal. Sean $f,g\in I$ y $h \in \mathbb{Z}[x]$, dados por:
    \[
        f(x) = \sum\limits_{i=0}^n a_i x^i, \quad g(x) = \sum\limits_{i=0}^m b_i x^i, \quad h(x) = \sum\limits_{i=0}^r c_i x^i.
    \]
    \begin{itemize}
        \item $f + g = \sum\limits_{i=0}^{\max\{n,m\}} (a_i + b_i) x^i \in I$, ya que su término independiente es $a_0 + b_0 \in 2\mathbb{Z}$.
        \item $hf = \sum\limits_{i=0}^{n+r} d_i x^i \in I$, donde $d_i = \sum\limits_{j=0}^{\min\{i,n\}} c_j a_{i-j}$. En particular, $d_0 = c_0 a_0 \in 2\mathbb{Z}$, luego $hf \in I$.
    \end{itemize}

    Por tanto, $I$ es un ideal de $\mathbb{Z}[x]$. Supongamos que fuese principal, es decir, $I =~f\mathbb{Z}[x]$ para algún $f \in \mathbb{Z}[x]$. Dado que el polinomio constante $2$ pertenece a $I$, debe existir $g \in \mathbb{Z}[x]$ tal que $2 = fg$. Tomando grados:
    \[
        0 = \grd(2) = \grd(f) + \grd(g) \implies \grd(f) = 0 = \grd(g),
    \]
    luego $f$ es una constante entera $a \in \mathbb{Z}$. Puesto que $2 = ab$, los únicos divisores en $\mathbb{Z}$ son $a = \pm 1$ o $a = \pm 2$:
    \begin{itemize}
        \item Si $a = \pm 1$, entonces $I = \mathbb{Z}[x]$, lo cual contradice que $1 \notin I$ (su término independiente es impar).
        \item Si $a = \pm 2$, entonces todos los polinomios de $I$ tendrían coeficientes pares, lo cual contradice que $x \in I$ (cuyo término independiente es $0 \in 2\mathbb{Z}$ pero su coeficiente principal es $1$).
    \end{itemize}
    Deducimos que $I$ no es principal, luego $\mathbb{Z}[x]$ no es un DE.
\end{proof}

\section{Irreducibles y primos en el anillo de enteros cuadráticos}

\begin{lema}
    Sean $\alpha, \beta \in \mathbb{Z}[\sqrt{n}]$ tales que $\alpha$ es divisor propio de $\beta$. Entonces $N(\alpha)$ es un divisor propio de $N(\beta)$ en $\mathbb{Z}$.
\end{lema}

\begin{proof}
    Al ser $\alpha$ un divisor propio de $\beta$, existe $\gamma \in \mathbb{Z}[\sqrt{n}] \setminus \mathcal{U}(\mathbb{Z}[\sqrt{n}])$ tal que $\beta = \alpha \gamma$. Tomando normas:
    \[
        N(\beta) = N(\alpha \gamma) = N(\alpha) N(\gamma).
    \]
    Como $\alpha, \gamma \notin \mathcal{U}(\mathbb{Z}[\sqrt{n}])$, sus normas enteras satisfacen $N(\alpha), N(\gamma) \notin \{\pm 1\} = \mathcal{U}(\mathbb{Z})$. Por consiguiente, $N(\alpha)$ divide estrictamente a $N(\beta)$ en $\mathbb{Z}$ sin ser unidad ni asociado.
\end{proof}

\begin{teo}
    En $\mathbb{Z}$ existen infinitos números primos.
\end{teo}

\begin{proof}
    Supongamos por reducción al absurdo que únicamente existen finitos primos positivos $\{p_1, p_2, \dots, p_n\}$. Consideremos el entero:
    \[
        a = p_1 p_2 \cdots p_n + 1.
    \]
    Como $a > 1$ y $\mathbb{Z}$ es un DFU, $a$ posee algún divisor primo $p_i$. Entonces $p_i \mid a$ y $p_i \mid p_1 \cdots p_n$, de donde $p_i \mid (a - p_1 \cdots p_n) = 1$, contradicción con que $p_i$ es primo.
\end{proof}

\begin{prop}
    Sea $n \in \mathbb{Z}$ no cuadrado perfecto y sea $\alpha \in \mathbb{Z}[\sqrt{n}]$.
    \begin{enumerate}[label=\arabic*)]
        \item Si $N(\alpha) = \pm p$ con $p \in \mathbb{Z}$ primo, entonces $\alpha$ es irreducible en $\mathbb{Z}[\sqrt{n}]$.
        \item Si $\alpha$ es primo en $\mathbb{Z}[\sqrt{n}]$, entonces $N(\alpha) \in \{\pm p, \pm p^2\}$ con $p \in \mathbb{Z}$ primo. Además, si $N(\alpha) = \pm p^2$, entonces $\alpha \sim p$ en $\mathbb{Z}[\sqrt{n}]$.
    \end{enumerate}
\end{prop}

\begin{proof}
    \leavevmode
    \begin{enumerate}[label=\arabic*)]
        \item Si $\alpha = \beta \gamma$, entonces $N(\alpha) = N(\beta) N(\gamma) = \pm p$. Dado que $p$ es primo en $\mathbb{Z}$, necesariamente $N(\beta) = \pm 1$ o $N(\gamma) = \pm 1$, lo que significa que $\beta \in \mathcal{U}(\mathbb{Z}[\sqrt{n}])$ o $\gamma \in \mathcal{U}(\mathbb{Z}[\sqrt{n}])$. Por tanto, $\alpha$ es irreducible.

        \item Como $\alpha$ es primo, $\alpha \neq 0$ y $N(\alpha) \notin \{0, \pm 1\}$. Consideremos la igualdad en $\mathbb{Z}$:
        \[
            N(\alpha) = \alpha \overline{\alpha} = \pm p_1 p_2 \cdots p_k, \quad \text{con } p_i \text{ primos enteros.}
        \]
        En $\mathbb{Z}[\sqrt{n}]$, $\alpha \mid N(\alpha)$, por lo que $\alpha \mid p_1 \cdots p_k$. Siendo $\alpha$ un elemento primo, debe existir algún índice tal que $\alpha \mid p_i$. Llamemos $p = p_i$.

        Existe por tanto $\beta \in \mathbb{Z}[\sqrt{n}]$ tal que $p = \alpha \beta$. Tomando normas:
        \[
            p^2 = N(p) = N(\alpha) N(\beta).
        \]
        Dado que $N(\alpha) \neq \pm 1$, las únicas posibilidades enteras son $N(\alpha) = \pm p$ o $N(\alpha) = \pm p^2$. En este último caso, $N(\beta) = \pm 1$, lo que implica $\beta \in \mathcal{U}(\mathbb{Z}[\sqrt{n}])$ y por ende $\alpha \sim p$. \qedhere
    \end{enumerate}
\end{proof}

\begin{lema}
    Sea $\alpha = a + bi \in \mathbb{Z}[i]$ con $a, b \neq 0$. Entonces:
    \[
        \alpha \text{ es primo o irreducible} \iff N(\alpha) \text{ es un número primo en } \mathbb{Z}.
    \]
\end{lema}
\begin{proof}
    \leavevmode
    \begin{description}
        \item[$\implies$)] Si $\alpha$ es primo o irreducible, en particular es primo, luego $N(\alpha) = a^2 + b^2\in\{\pm p, \pm p^2\}$ con $p$ primo en $\mathbb{Z}$. Como $a, b \neq 0$, se tiene $N(\alpha) = a^2 + b^2 > 1$, luego necesariamente $N(\alpha)\in \{p, p^2\}$.
        
        Si $N(\alpha) = p^2$, entonces $\alpha \sim p$, luego $\exists \beta \in \mathcal{U}(\mathbb{Z}[i])=\{\pm 1, \pm i\}$ tal que $\alpha = \beta p$, lo que contradice que $a, b \neq 0$. Por tanto, $N(\alpha) = p$ es primo en $\mathbb{Z}$.

        \item[$\impliedby$)] Si $N(\alpha) = p$ es primo en $\mathbb{Z}$, entonces $\alpha$ es irreducible en $\mathbb{Z}[i]$ por la Proposición anterior. Como $\mathbb{Z}[i]$ es un DFU, todo irreducible es primo.
    \end{description}
\end{proof}

\begin{ejemplo}
    \leavevmode
    \begin{enumerate}[label=\arabic*)]
        \item Factorizar $\alpha = 11 + 7i$ en $\mathbb{Z}[i]$.

        Al ser $\mathbb{Z}[i]$ un DFU, irreducible equivale a primo. Calculamos su norma:
        \[
            N(\alpha) = 11^2 + 7^2 = 121 + 49 = 170 = 2 \cdot 5 \cdot 17.
        \]
        Como ningún primo divide a $N(\alpha)$ con multiplicidad superior a $1$, buscamos divisores irreducibles de normas $2$, $5$ y $17$.

        Para norma $2$: $a^2 + b^2 = 2 \implies a = \pm 1, \, b = \pm 1$. Tomamos $1 + i$:
        \[
            \frac{11 + 7i}{1 + i} = \frac{(11 + 7i)(1 - i)}{2} = \frac{18 - 4i}{2} = 9 - 2i \in \mathbb{Z}[i].
        \]
        Luego $\alpha = (1 + i)(9 - 2i)$, siendo $1 + i$ irreducible por tener norma prima $2$.

        Para norma $5$: $a^2 + b^2 = 5 \implies a + bi \in \{\pm 2 \pm i, \, \pm 1 \pm 2i\}$. Probando con $1 + 2i$:
        \[
            \frac{9 - 2i}{1 + 2i} = \frac{(9 - 2i)(1 - 2i)}{5} = \frac{5 - 20i}{5} = 1 - 4i \in \mathbb{Z}[i].
        \]
        Como $N(1 - 4i) = 1^2 + (-4)^2 = 17$, que es primo, concluimos la factorización completa en irreducibles:
        \[
            11 + 7i = (1 + i)(1 + 2i)(1 - 4i).
        \]

        \item Sea $\beta = 2i \in \mathbb{Z}[i]$. Descomponiendo en irreducibles:
        \[
            \beta = 2i = -i (1+i)^2.
        \]
        Aplicando la Proposición~\ref{propExistenMCDDFU}:
        \begin{align*}
            \mcd(\alpha, \beta) &= 1 + i, \\
            \mcm(\alpha, \beta) &= (1 + i)^2 (1 + 2i)(1 - 4i) = 2i(9 - 2i) = 4 + 18i.
        \end{align*}
    \end{enumerate}
\end{ejemplo}

\begin{ejemplo}
    En dominios que no son DFU, existen elementos irreducibles que no son primos. Consideremos el anillo $\mathbb{Z}[\sqrt{-5}]$ y el elemento $\alpha = 1 + \sqrt{-5}$. Su norma es $N(\alpha) = 1^2 + 5(1)^2 = 6$. 

    Si $\alpha = \beta \gamma$, entonces $N(\beta) N(\gamma) = 6$. Como la ecuación $a^2 + 5b^2 = 2$ o $3$ no tiene soluciones enteras, necesariamente $N(\beta) = 1$ o $N(\gamma) = 1$, con lo que $\alpha$ es irreducible.

    Sin embargo, $\alpha$ no es primo: $\alpha \mid 6 = 2 \cdot 3$, pero $\alpha \nmid 2$ y $\alpha \nmid 3$ en $\mathbb{Z}[\sqrt{-5}]$ porque $N(\alpha) = 6$ no divide a $N(2) = 4$ ni a $N(3) = 9$.
\end{ejemplo}

\section{Factorización en el anillo de polinomios}

Sabemos que, si $K$ es un cuerpo, el anillo de polinomios $K[x]$ es un DE y por tanto un DFU. Nuestro primer objetivo es demostrar que si $A$ es un DFU, entonces $A[x]$ también lo es (lo que nos permitirá demostrar, por ejemplo, que $\bb{Z}[x]$ es un DFU). Para ello, necesitamos caracterizar los elementos irreducibles de $A[x]$.
\begin{prop} \label{propIrreduciblesDFUxPolinomios}
    Sea $A$ un DFU y sea $a \in A$. Entonces:
    \[
        a \text{ es irreducible en } A \iff a \text{ es irreducible en } A[x].
    \]

    Notemos que, en particular, si $A$ es un cuerpo, entonces no hay elementos irreducibles en $A[x]$ de grado $0$, puesto que todos los elementos de $A \setminus \{0\}$ son unidades.
\end{prop}
\begin{proof}\leavevmode
\begin{description}
    \item[$\implies$)] Sea $a \in A$ irreducible en $A$. En particular $a \neq 0$ y $a \notin \mathcal{U}(A) = \mathcal{U}(A[x])$. Si $a = fg$ en $A[x]$, tomando grados:
    \[
        0 = \grd(a) = \grd(f) + \grd(g) \implies \grd(f) = 0 = \grd(g).
    \]
    Luego $f, g \in A$. Al ser $a$ irreducible en $A$, uno de los factores es unidad en $A$ (y por tanto en $A[x]$). Así, $a$ es irreducible en $A[x]$.

    \item[$\impliedby$)] Si $a$ es reducible en $A$, admite divisores no triviales en $A$, los cuales son inmediatamente divisores no triviales en $A[x]$, contradicción.
\end{description}
\end{proof}

\begin{definicion}[Contenido de un polinomio]
    Sea $A$ un DFU y consideremos $f(x) = \sum\limits_{i=0}^n a_i x^i \in A[x]$ con $n \geq 1$. Se define el \textbf{contenido de $f$}, denotado $C(f)$, como:
    \[
        C(f) := \mcd(a_0, a_1, \dots, a_n).
    \]
    Diremos que $f$ es un \textbf{polinomio primitivo} si $C(f) = 1$ (salvo asociados).
\end{definicion}

\begin{lema}\label{lemaPropsContenido}
    Sea $A$ un DFU y $K = \mathbb{Q}(A)$ su cuerpo de fracciones. Se verifica:
    \begin{enumerate}[label=\roman*)]
        \item Para todo $a \in A$ y todo $f \in A[x]$ con $\grd(f) \geq 1$, $C(af) = a C(f)$.
        \item Todo $f \in A[x]$ con $\grd(f) \geq 1$ se expresa de forma única como $f = af'$, donde $a = C(f)$ y $f' \in A[x]$ es primitivo.
        \item Todo $\phi \in K[x]$ con $\grd(\phi) \geq 1$ se expresa de forma esencialmente única como $\phi = \nicefrac{a}{b} f$, con $\nicefrac{a}{b} \in K$ y $f \in A[x]$ primitivo.
    \end{enumerate}
\end{lema}

\begin{proof}
    \leavevmode
    \begin{enumerate}[label=\roman*)]
        \item Por las propiedades del máximo común divisor:
        \[
            C(af) = \mcd(a a_0, \dots, a a_n) = a \mcd(a_0, \dots, a_n) = a C(f).
        \]

        \item Tomando $a = C(f)$, cada coeficiente $a_i$ se escribe como $a_i = a a_i'$ con $a_i' \in A$. El polinomio $f' = \sum\limits_{i=0}^n a_i' x^i$ satisface $C(f') = \mcd(a_0', \dots, a_n') = 1$, luego es primitivo. Si $f = b g'$ con $g'$ primitivo, $C(f) = C(b g') = b C(g') = b$, de donde $a = b$ y por simplificación $f' = g'$.

        \item Multiplicando $\phi$ por el producto de los denominadores de sus coeficientes, obtenemos un polinomio en $A[x]$. Extrayendo el contenido mediante el apartado ii), se deduce inmediatamente el resultado. \qedhere
    \end{enumerate}
\end{proof}

\begin{lema}[de Gauss]\label{lemaGauss}
    Sea $A$ un DFU.
    El producto de dos polinomios primitivos en $A[x]$ es primitivo.
\end{lema}

\begin{proof}
    Sean $f(x) = \sum\limits_{i=0}^n a_i x^i$ y $g(x) = \sum\limits_{j=0}^m b_j x^j$ dos polinomios primitivos en $A[x]$. Supongamos por reducción al absurdo que su producto $fg$ no es primitivo: existe un elemento irreducible $p \in A$ tal que $p \mid C(fg)$, es decir, $p$ divide a todos los coeficientes de $fg$.

    Como $A$ es un DFU, el elemento $p$ es primo. Al ser $f$ primitivo, $p$ no divide a todos sus coeficientes; consideremos el menor índice $r$ tal que $p \nmid a_r$. Análogamente, sea $s$ el menor índice tal que $p \nmid b_s$.

    El coeficiente de grado $r + s$ del producto $fg$ es:
    \[
        c_{r+s} = \sum\limits_{i+j = r+s} a_i b_j = a_r b_s + \sum\limits_{\substack{i+j = r+s \\ i < r}} a_i b_j + \sum\limits_{\substack{i+j = r+s \\ i > r}} a_i b_j.
    \]
    Por la elección minimal de $r$ y $s$, si $i < r$ se tiene $p \mid a_i$, y si $i > r$ entonces necesariamente $j < s$, con lo cual $p \mid b_j$. Así, $p$ divide a todos los términos de las sumas secundarias.

    Dado que por hipótesis $p \mid c_{r+s}$, se concluye que $p \mid a_r b_s$. Siendo $p$ primo, $p \mid a_r$ o $p \mid b_s$, contradicción. Por tanto, $C(fg) = 1$.
\end{proof}

\begin{coro}
    Sean $f, g \in A[x]$ con $\grd(f), \grd(g) \geq 1$. Entonces:
    \[
        C(fg) = C(f) C(g).
    \]
\end{coro}

\begin{proof}
    Factorizando $f = C(f)f'$ y $g = C(g)g'$ con $f', g'$ primitivos:
    \[
        C(fg) = C\bigl(C(f)C(g) f'g'\bigr) = C(f)C(g) C(f'g') = C(f)C(g),
    \]
    donde en el último paso hemos aplicado el Lema de Gauss.
\end{proof}

\begin{teo}\label{teoEqIrreducibles}
    Sea $A$ un DFU y $K = \mathbb{Q}(A)$. Sea $\phi \in K[x]$ con $\grd(\phi) \geq 1$ y escribamos $\phi = \nicefrac{a}{b} f$ con $f \in A[x]$ primitivo. Los siguientes enunciados son equivalentes:
    \begin{enumerate}[label=\arabic*)]
        \item $\phi$ es irreducible en $K[x]$.
        \item $f$ es irreducible en $K[x]$.
        \item $f$ es irreducible en $A[x]$.
    \end{enumerate}
\end{teo}

\begin{proof}\leavevmode
\begin{description}
    \item[$1) \iff 2)$] Puesto que $\nicefrac{a}{b} \in \mathcal{U}(K[x])$, los polinomios $\phi$ y $f$ son asociados en $K[x]$, luego comparten sus propiedades de divisibilidad.

    \item[$2) \implies 3)$] Supongamos que $f$ no es irreducible es $A[x]$; es decir, $f = f_1 f_2$ en $A[x]$. Si ambos factores tuviesen grado estricto $\geq 1$, esta factorización sería válida en $K[x]$, contradiciendo que $f$ es irreducible en $K[x]$. Por tanto, alguno de ellos, digamos $f_1$, debe tener grado $0$, es decir, $f_1 \in A$. Entonces $C(f) = C(f_1 f_2) = f_1 C(f_2)$. Al ser $f$ primitivo ($C(f) = 1$), se concluye que $f_1 \in \mathcal{U}(A) = \mathcal{U}(A[x])$, luego $f$ es irreducible en $A[x]$.

    \item[$3) \implies 2)$] Supongamos que $f$ no es irreducible en $K[x]$; es decir, $f = \phi_1 \phi_2$ en $K[x]$ con $\grd(\phi_i) \geq 1$. Escribiendo $\phi_i = \frac{a_i}{b_i} f_i$ con $f_i \in A[x]$ primitivos:
    \[
        f = \frac{a_1 a_2}{b_1 b_2} f_1 f_2 \implies b_1 b_2 f = a_1 a_2 f_1 f_2.
    \]
    Tomando contenidos a ambos lados:
    \[
        b_1 b_2 C(f) = a_1 a_2 C(f_1) C(f_2) \implies b_1 b_2 = a_1 a_2.
    \]
    Sustituyendo resulta $f = f_1 f_2$ en $A[x]$ con $\grd(f_1), \grd(f_2) \geq 1$, contradicción con que $f$ es irreducible en $A[x]$.
\end{description}
\end{proof}

\begin{coro}
    Sea $A$ un DFU y $f \in A[x]$ con $\grd(f) \geq 1$. Entonces:
    \[
        f \text{ es irreducible en } A[x] \iff f \text{ es primitivo e irreducible en } K[x].
    \]
\end{coro}
\begin{proof}
    Para que la demostración sea directa a partir del Teorema~\ref{teoEqIrreducibles}, basta notar que si $f$ es irreducible en $A[x]$, entonces es primitivo. Considerando la factorización $f = C(f) f'$ con $f'$ primitivo, si $C(f) \notin \mathcal{U}(A)$, entonces $f$ sería reducible en $A[x]$, luego necesariamente $C(f) \in \mathcal{U}(A)$, es decir, $f$ es primitivo.
\end{proof}

En este punto, podemos demostrar el resultado que nos permitirá concluir que $\mathbb{Z}[x]$ es un DFU.
\begin{teo}[de Gauss]\label{teoGauss}
    Si $A$ es un DFU, entonces $A[x]$ es un DFU.
\end{teo}

\begin{proof}
    Sea $f \in A[x] \setminus (\{0\} \cup \mathcal{U}(A[x]))$. Veamos que $f$ se descompone en irreducibles de $A[x]$.
    \begin{itemize} 
        \item Si $\grd(f) = 0$, $f \in A$ se factoriza en irreducibles de $A$, que siguen siendo irreducibles en $A[x]$.
        \item Si $\grd(f) \geq 1$, escribimos $f = a f'$ con $a = C(f) \in A$ y $f'$ primitivo. Factorizamos $a$ en irreducibles de $A$, obteniendo $a = p_1 \cdots p_r$. Por su parte, notando $K=\bb{Q}(A)$, $K[x]$ es un DE y por tanto DFU, por lo que $f'$ se descompone en $K[x]$ como $f' = \phi_1 \cdots \phi_s$. Escribiendo cada $\phi_i = \frac{a_i}{b_i} f_i$ con $f_i \in A[x]$ primitivo e irreducible en $K[x]$ (y por tanto irreducible en $A[x]$):
        \[
            f' = \left(\prod\limits_{i=1}^s \frac{a_i}{b_i}\right) f_1 \cdots f_s.
        \]
        Tomando contenidos deducimos que $\prod \frac{a_i}{b_i} = 1$, con lo que:
        \begin{align*}
            f &= a f' = p_1 \cdots p_r \cdot f_1 \cdots f_s
        \end{align*}
        es una factorización completa en irreducibles de $A[x]$.
    \end{itemize}

    Para verificar que $A[x]$ es DFU, por el Teorema~\ref{teoCaracDFU} basta ver que todo irreducible $f \in A[x]$ es primo:
    \begin{itemize}
        \item Si $\grd(f) = 0$, $f$ es un irreducible de $A$, luego es primo en $A$ y se comprueba directamente que también lo es en $A[x]$.
        \item Si $\grd(f) \geq 1$, $f$ es irreducible y primitivo, luego es primo en $K[x]$. Si $f \mid gh$ en $A[x]$, entonces $f \mid g$ o $f \mid h$ en $K[x]$. Supongamos $f \mid g$: existe $\phi \in K[x]$ tal que $g = \phi f$. Escribiendo $\phi = \nicefrac{a}{b} g'$ con $g'$ primitivo, $bg = a g' f$. Tomando contenidos, $b C(g) = a$, luego $\nicefrac{a}{b} = C(g) \in A$, lo que prueba que $\phi \in A[x]$ y $f \mid g$ en $A[x]$. \qedhere
    \end{itemize}
\end{proof}

\section{Criterios básicos de irreducibilidad de polinomios}

En todo lo que sigue, $K$ denotará un cuerpo.

\begin{lema}
    Dos polinomios mónicos en $K[x]$ son asociados si y solo si son iguales.
\end{lema}
\begin{proof}
    Sean $\phi, \psi \in K[x]$ mónicos. Son asociados si y solo si existe $c \in K \setminus \{0\}$ tal que $\phi = c \psi$. Igualando coeficientes principales, $1 = c \cdot 1 \implies c = 1$, luego $\phi = \psi$.
\end{proof}



Por tanto, puesto que todo polinomio en $K[x]$ es asociado a un polinomio mónico (dividiendo por su coeficiente principal) y este es único, podemos restringirnos a estudiar la irreducibilidad de polinomios mónicos.

\begin{prop}
    Todo polinomio de grado $1$ en $K[x]$ es irreducible.
\end{prop}

\begin{proof}
    Sea $\phi \in K[x]$ con $\grd(\phi) = 1$. Si $\phi = \phi' \phi''$, tomando grados: $$1 = \grd(\phi') + \grd(\phi'')$$Por tanto, uno de los polinomios debe tener grado $0$, luego es una unidad en $K[x]$. Sus únicos divisores son los triviales, de modo que es irreducible.
\end{proof}

Por tanto, los polinomios irreducibles mónicos de grado $1$ en $K[x]$ son exactamente los de la forma $x + a$ con $a \in K$.\\

Aunque su demostración excede el alcance de este curso (se verá en múltiples ocasiones a lo largo de la carrera, como en Variable Compleja I), es importante conocer el siguiente resultado:
\begin{teo}[Fundamental del álgebra, Gauss]
    Los únicos polinomios irreducibles en $\mathbb{C}[x]$ son los de grado $1$.
\end{teo}

\begin{teo}[de Ruffini]\label{teoRuffini}
    Sean $\phi \in K[x]$ y $a \in K$. El resto de la división de $\phi$ entre $x - a$ es $\phi(a)$.
\end{teo}
\begin{proof}
    Por el algoritmo de división euclídea:
    \[
        \phi(x) = (x - a)q(x) + r, \quad \text{con } r = 0 \quad \text{o} \quad \grd(r) < 1.
    \]
    Así, $r \in K$ es una constante. Evaluando en $x = a$:
    \[
        \phi(a) = (a - a)q(a) + r = r.
    \]
\end{proof}

\begin{prop}\label{propFactorGrd1Raiz}
    Sea $\phi \in K[x]$ con $\grd(\phi) \geq 2$. Entonces $\phi$ tiene un factor de grado $1$ si y solo si tiene alguna raíz en $K$.
\end{prop}
\begin{proof}Demostramos la equivalencia por doble implicación.
\begin{description}
    \item[$\implies$)] Si $a_1 x + a_0 \mid \phi$ con $a_1 \neq 0$, entonces $x - (-a_0 a_1^{-1}) \mid \phi$, luego $a = -a_0 a_1^{-1} \in K$ es raíz de $\phi$.

    \item[$\impliedby$)] Si $\phi(a) = 0$, por el Teorema de Ruffini el resto es cero, luego $x - a \mid \phi$.
\end{description}
\end{proof}

\begin{coro}[Criterio de la raíz]\label{corCritRaiz}
    Sea $\phi \in K[x]$ con $\grd(\phi) \in \{2, 3\}$. Entonces $\phi$ es irreducible en $K[x]$ si y solo si no tiene raíces en $K$.
\end{coro}

\begin{proof}
    Si fuese reducible, al menos uno de sus factores debe tener grado $1$, lo cual equivale por la proposición anterior a poseer una raíz en $K$.
\end{proof}

\begin{teo}
    Los polinomios irreducibles mónicos en $\mathbb{R}[x]$ son los de grado $1$ y los de grado $2$ con discriminante negativo:
    \[
        \{x + a \mid a \in \mathbb{R}\} \cup \{x^2 + bx + c \mid b, c \in \mathbb{R}, \, b^2 - 4c < 0\}.
    \]
\end{teo}

\begin{ejemplo}
    Notemos que podemos listar todos los polinomios mónicos irreducibles de grado $2$ o $3$ en $\mathbb{Z}_p[x]$, considerando todos los mónicos y quedándonos con los que no tengan raíces en $\mathbb{Z}_p$. Por ejemplo, los de grado $2$ en $\mathbb{Z}_2[x]$ y $\mathbb{Z}_3[x]$ son:
    \begin{enumerate}[label=\arabic*)]
        \item En $\mathbb{Z}_2[x]$, los polinomios mónicos de grado $2$ son de la forma $x^2 +bx + c$ con $b, c \in \mathbb{Z}_2$.
        \begin{align*}
            b=0 \Rightarrow & \left\{ \begin{array}{llll}
                c=0 & x^2   & \text{reducible} & (x^2)(0)=0   \\
                c=1 & x^2+1 & \text{reducible} & (x^2+1)(1)=0
            \end{array} \right.\\
            b=1 \Rightarrow & \left\{ \begin{array}{llll}
                c=0 & x^2+x   & \text{reducible}   & (x^2+x)(1)=0           \\
                c=1 & x^2+x+1 & \text{irreducible} & \text{no tiene raíces}
            \end{array} \right.
        \end{align*}
        \\item En $\mathbb{Z}_3[x]$, los polinomios mónicos de grado $2$ son de la forma $x^2 +bx + c$ con $b, c \in \mathbb{Z}_3$.
        \begin{align*}
            b=0 \Rightarrow & \left\{ \begin{array}{llll}
                c=0 & x^2   & \text{reducible}   & (x^2)(0) = 0           \\
                c=1 & x^2+1 & \text{irreducible} & \text{no tiene raíces} \\
                c=2 & x^2+2 & \text{reducible}   & (x^2+2)(2)=0
            \end{array} \right.\\
            b=1 \Rightarrow & \left\{ \begin{array}{llll}
                c=0 & x^2+x   & \text{reducible}   & (x^2+x)(2) = 0         \\
                c=1 & x^2+x+1 & \text{reducible}   & (x^2+x+1)(1)=0         \\
                c=2 & x^2+x+2   & \text{irreducible} & \text{no tiene raíces}
            \end{array} \right.\\
            b=2 \Rightarrow & \left\{ \begin{array}{llll}
                c=0 & x^2+2x   & \text{reducible}   & (x^2+2x)(0) = 0        \\
                c=1 & x^2+2x+1 & \text{reducible}   & (x^2+2x+1)(2)=0        \\
                c=2 & x^2+2x+2 & \text{irreducible} & \text{no tiene raíces}
            \end{array} \right.
        \end{align*}
    \end{enumerate}
\end{ejemplo}

Para obtener los polinomios de grado $4$, los listamos y descartamos los que tengan raíces (puesto que tendrán un factor de grado $1$). Para ver si tienen un factor de grado $2$, podemos dividir entre los irreducibles de grado $2$ que ya hemos listado. Si el resto es nulo, entonces el polinomio tiene un factor de grado $2$ y es reducible; si no, es irreducible.

\begin{ejemplo}
    \leavevmode
    \begin{enumerate}[label=\arabic*)]
        \item Factorizar $f = x^4 + x^3 + x^2 + x + 1 \in \mathbb{Z}_2[x]$.
        
        Evaluando: $f(0) = 1$ y $f(1) = 1$, luego no tiene raíces ni factores de grado $1$ ni $3$. El único irreducible cuadrático candidato es $x^2 + x + 1$. Al dividir:
        \[
            f = (x^2 + x + 1)^2 + x +1
        \]
        obtenemos resto no nulo, luego $f$ no tiene factores de grado $2$. Por tanto, $f$ es irreducible en $\mathbb{Z}_2[x]$.


        \item Factorizar $f = x^5 + x^4 + x^2 + 1 \in \mathbb{Z}_3[x]$.
        
        Evaluando: $f(0) = 1$, $f(1) =1$ y $f(2) = 2$, luego no tiene raíces ni factores de grado $1$ ni $4$. Puede tener factores de grado $2$ o $3$. Buscamos factores irreducibles de grado $2$, que pueden ser: $x^2 + 1$, $x^2 + x + 2$ o $x^2 + 2x + 2$.
        \begin{align*}
            f &= (x^2 + 1)(x^3 + x^2 + 2x) + x + 1 \\
            f &= (x^2 + x + 2)(x^3 + x) + x + 1 \\
            f &= (x^2 + 2x + 2)(x^2 + 2x) + 2
        \end{align*}
        Luego $f$ no tiene factores de grado $2$, luego tampoco tiene factores de grado $3$. Por tanto, $f$ es irreducible en $\mathbb{Z}_3[x]$.

        \item Factorizar $f = x^7 + 2x^5 + x^4 + 2x^3 + x + 2 \in \mathbb{Z}_3[x]$.
        
        Evaluando: $f(0) = 2$, $f(1) = 0$, luego $x - 1 = x + 2 \mid f$. Dividiendo:
        \[
            f = (x + 2)(x^6 + x^5 + x^3 + 1) = (x + 2)g.
        \]
        Factorizamos $g$:
        Evaluando: $g(2) = 0$, luego $x - 2= x + 1 \mid g$. Dividiendo:
        \[
            g = (x + 1)(x^5 + x^2 + 2x + 1) = (x + 1)h.
        \]
        Factorizamos $h$:
        Evaluando: $h(0), h(1), h(2) \neq 0$, luego $h$ no tiene raíces ni factores de grado $1$ ni $4$. Buscamos factores de grado $2$ en $\mathbb{Z}_3[x]$, que pueden ser: $x^2 + 1$, $x^2 + x + 2$ o $x^2 + 2x + 2$.
        \[
            h = (x^2 + 1)(x^3 + 2x + 1)
        \]
        Factorizamos $x^3 + 2x + 1$. No tiene raíces en $\mathbb{Z}_3[x]$, luego es irreducible por el Criterio de la Raíz. En resumen:
        \[
            f = (x + 2)(x + 1)(x^2 + 1)(x^3 + 2x + 1)
        \]

\section{Polinomios sobre $\mathbb{Z}$ y $\mathbb{Q}$}

Antes de comenzar a estudiar criterios de irreducibilidad más avanzados, conviene recordar lo que ya sabemos sobre polinomios con coeficientes enteros y racionales.
\begin{itemize}
    \item Por la Proposición~\ref{propIrreduciblesDFUxPolinomios}, tenemos que los irreducibles de grado $0$ de $\mathbb{Z}[x]$ son exactamente los irreducibles de $\mathbb{Z}$ y, por el Teorema~\ref{teoCaracDFU}, estos son los primos de $\mathbb{Z}$.
    \item Sea $f\in \bb{Z}\in \bb{Z}[x]$ primitivo con $\grd(f)>1$. Por el Teorema~\ref{teoEqIrreducibles}, $f$ es irreducible en $\bb{Z}[x]$ si y solo si es irreducible en $\bb{Q}[x]$.
    \item Sea $f\in \bb{Q}[x]$ con $\grd(f)>0$, y empleamos su factorización $f = \nicefrac{a}{b} f'$ con $f'\in \bb{Z}[x]$ primitivo. Por el Teorema~\ref{teoEqIrreducibles}, $f$ es irreducible en $\bb{Q}[x]$ si y solo si $f'$ es irreducible en $\bb{Z}[x]$.
    \item Como todo polinomio de grado $1$ en $\bb{Q}[x]$ es irreducible, todo polinomio de grado $1$ en $\bb{Z}[x]$ es irreducible si y solo si es primitivo.
    \item Sea $f\in \bb{Q}[x]$ con $\grd(f)=2$ o $\grd(f)=3$. Por el Corolario~\ref{corCritRaiz}, $f$ es irreducible en $\bb{Q}[x]$ si y solo si no tiene raíces en $\bb{Q}$.
    \item Sea $f\in \bb{Z}[x]$ con $\grd(f)=2$ o $\grd(f)=3$. Por el Teorema~\ref{teoEqIrreducibles}, $f$ es irreducible en $\bb{Z}[x]$ si y solo si es primitivo y no tiene raíces en $\bb{Q}$.
\end{itemize}

Una vez recordados estos resultados, podemos enunciar criterios de irreducibilidad más avanzados.
\begin{prop}\label{propbxmenosA}
    Sea $f(x) = \sum\limits_{i=0}^n a_i x^i \in \mathbb{Z}[x]$ con $n \geq 1$. Si $\nicefrac{a}{b} \in \mathbb{Q}$ es una raíz y $\mcd(a,b) = 1$, entonces $bx - a$ divide a $f$ en $\mathbb{Z}[x]$.
\end{prop}

\begin{proof}
    Por el Teorema de Ruffini, $x - \nicefrac{a}{b}$ divide a $f$ en $\mathbb{Q}[x]$, luego $bx - a$ también divide a $f$ en $\mathbb{Q}[x]$. Existe por tanto $g \in \mathbb{Q}[x]$ tal que $f = (bx - a)g$. Escribiendo $g = \frac{c}{d} g'$ con $c, d \in \mathbb{Z}$ coprimos y $g' \in \mathbb{Z}[x]$ primitivo:
    \[
        df = c(bx - a)g'.
    \]
    Tomando contenidos:
    \[
        d C(f) = c C(bx - a) C(g') = c \cdot 1 \cdot 1 = c.
    \]
    Deducimos que $\nicefrac{c}{d} = C(f) \in \mathbb{Z}$, con lo cual $g \in \mathbb{Z}[x]$.
\end{proof}

\begin{coro}\label{corpropbxmenosA}
    Bajo las hipótesis de la proposición anterior, necesariamente:
    \[
        a \mid a_0 \quad \text{y} \quad b \mid a_n \quad \text{en } \mathbb{Z}.
    \]
\end{coro}

\begin{proof}
    Si $f(x) = (bx - a)(c_{n-1} x^{n-1} + \dots + c_0)$, igualando coeficientes extremos resulta $a_0 = -a c_0$ y $a_n = b c_{n-1}$, de donde se siguen las divisibilidades.
\end{proof}

\begin{coro}
    \leavevmode
    \begin{enumerate}[label=\arabic*)]
        \item Si $f(x) \in \mathbb{Z}[x]$ es un polinomio mónico, todas sus raíces racionales son enteras.
        \item Si $n \in \mathbb{Z}$ no es un cuadrado perfecto en $\mathbb{Z}$, entonces $\sqrt{n} \notin \mathbb{Q}$.
    \end{enumerate}
\end{coro}
\begin{proof}
    \leavevmode
    \begin{enumerate}[label=\arabic*)]
        \item Sea $f(x) = x^n + a_{n-1} x^{n-1} + \dots + a_0$ mónico y supongamos que $\nicefrac{a}{b} \in \mathbb{Q}$ es una raíz con $\mcd(a,b) = 1$. Por el Corolario~\ref{corpropbxmenosA}, $b \mid 1$, luego $b = \pm 1$ y $\nicefrac{a}{b} = \pm a \in \mathbb{Z}$.
        \item Si $\sqrt{n}$ no es un cuadrado perfecto, entonces $x^2 - n$ no tiene raíces enteras, luego no tiene raíces racionales por el punto anterior, por lo que $\sqrt{n} \notin \mathbb{Q}$.
    \end{enumerate}
\end{proof}

\begin{ejemplo}
    Factorizar $f = 20x^4 - 10x^3 - 80x^2 + 80x - 20$ en $\mathbb{Z}[x]$ y en $\mathbb{Q}[x]$.\\

    Extrayendo el contenido: $f = 10 g$, con $g = 2x^4 - x^3 - 8x^2 + 8x - 2$ primitivo. Las posibles raíces racionales $\nicefrac{a}{b}$ de $g$ cumplen $a \mid 2$ y $b \mid 2$, luego $\nicefrac{a}{b} \in \{\pm 1, \pm 2, \pm \nicefrac{1}{2}\}$. Evaluando, encontramos $g(\nicefrac{1}{2}) = 0$, luego $2x - 1 \mid g$:
    \begin{equation*}
        \polylongdiv[style=D]{2x^4 - x^3 - 8x^2 + 8x - 2}{2x - 1}
    \end{equation*}

    Por tanto, $g = (2x - 1)(x^3 - 4x + 2)$ y finalmente el polinomio cúbico es mónico; sus únicas raíces racionales podrían ser $\pm 1$ o $\pm 2$, pero ninguna lo anula. Al ser de grado $3$ sin raíces en $\mathbb{Q}$, es irreducible. Por tanto:
    \begin{itemize}
        \item En $\mathbb{Z}[x]$: $f = 2 \cdot 5 (2x - 1)(x^3 - 4x + 2)$.
        \item En $\mathbb{Q}[x]$: $f = 20 \left(x - \frac{1}{2}\right)(x^3 - 4x + 2)$.
    \end{itemize}
\end{ejemplo}

\begin{ejemplo}
    Factorizar en $\Q[x]$: $\phi=x^3+\nicefrac{1}{2}x^2-x-3$.\\

    Tenemos que $\phi = \dfrac{1}{2}(2x^3+x^2-2x-6)=\dfrac{1}{2}f$, con $f \in \Z[x]$ primitivo. Las posibles raíces racionales $\nicefrac{a}{b}$ de $f$ cumplen $a \mid -6$ y $b \mid 2$, luego $\nicefrac{a}{b} \in \{\pm 1, \pm 2, \pm 3, \pm 6, \pm \nicefrac{1}{2}, \pm \nicefrac{3}{2}\}$. Evaluando, encontramos que $f(\nicefrac{3}{2}) = 0$, luego $2x - 3 \mid f$:
    \begin{equation*}
        \polylongdiv[style=D]{2x^3 + x^2 - 2x - 6}{2x - 3}
    \end{equation*}
    Por tanto, $f = (2x - 3)(x^2 + 2x + 2)$. El polinomio cuadrático es mónico; sus únicas raíces racionales podrían ser $\pm 1$ o $\pm 2$, pero ninguna lo anula. Al ser de grado $2$ sin raíces en $\mathbb{Q}$, es irreducible. Por tanto:
    \[
        \phi = \dfrac{1}{2}(2x - 3)(x^2 + 2x + 2) = \left(x - \dfrac{3}{2}\right)(x^2 + 2x + 2)
    \]
\end{ejemplo}



\subsection{Criterio de reducción}

Para cada primo $p\in \bb{Z}$, consideramos la proyección canónica en $\bb{Z}_p$, que vimos que se trata de un homeomorfismo.
\Func{R_p}{\bb{Z}}{\bb{Z}_p}{a}{[a]}

Este homomorfismo induce un homomorfismo en el anillo de polinomios:
\Func{R_p}{\bb{Z}[x]}{\bb{Z}_p[x] }{ \sum_{i=0}^n a_i x^i }{ \sum_{i=0}^n [a_i] x^i }

\begin{prop}[Criterio de reducción]
    Sea $f \in \mathbb{Z}[x]$ con $\grd(f) \geq 1$ tal que $\grd(R_p(f)) = \grd(f)$. Si $R_p(f)$ no tiene factores de grado $r$ en $\mathbb{Z}_p[x]$, entonces $f$ no tiene factores de grado $r$ en $\mathbb{Z}[x]$.

    En particular, si $R_p(f)$ es irreducible en $\mathbb{Z}_p[x]$, entonces $f$ es irreducible en $\mathbb{Z}[x]$.
\end{prop}

\begin{proof}
    Supongamos que $f = gh$ en $\mathbb{Z}[x]$ con $\grd(g) = r$. Aplicando el homomorfismo $R_p$:
    \[
        R_p(f) = R_p(g) R_p(h).
    \]
    Como $\grd(R_p(f)) = \grd(f) = \grd(g) + \grd(h)$ y los grados tras reducción solo pueden mantenerse o disminuir, necesariamente $\grd(R_p(g)) = \grd(g) = r$, contradiciendo que $R_p(f)$ no tiene factores de grado $r$.
\end{proof}

\begin{prop}[Criterio de Eisenstein]
    Sea $f = a_0 + a_1x + \dots + a_nx^n \in \mathbb{Z}[x]$ con $\operatorname{grd}(f) = n \geq 1$ un polinomio primitivo.
    Entonces, $f$ es irreducible en $\mathbb{Z}[x]$ (y por tanto también en $\mathbb{Q}[x]$) si existe un número primo $p \in \mathbb{Z}$ tal que se verifica alguna de las siguientes condiciones:
    \begin{enumerate}[label=\roman*)]
        \item $p \mid a_i \quad \forall i \in \{0, \dots, n-1\}, \quad p^2 \nmid a_0$.
        \item $p \mid a_i \quad \forall i \in \{1, \dots, n\}, \quad p^2 \nmid a_n$.
    \end{enumerate}
\end{prop}

\begin{proof}
    \leavevmode
    \begin{enumerate}[label=\roman*)]
        \item Por reducción al aburso. Supongamos que $f$ es reducible en $\mathbb{Z}[x]$. Entonces existen $g, h \in \mathbb{Z}[x]$ con $\operatorname{grd}(g) = r \geq 1$ y $\operatorname{grd}(h) = s \geq 1$ (donde $r + s = n$) tales que:
        \[
            f = gh
        \]
        Escribamos $g = b_0 + b_1x + \dots + b_rx^r$ y $h = c_0 + c_1x + \dots + c_sx^s$.

        Al igualar términos independientes tenemos $a_0 = b_0c_0$. Como $p \mid a_0$, por ser $p$ primo se deduce que $p \mid b_0$ o $p \mid c_0$. Además, dado que $p^2 \nmid a_0$, $p$ no puede dividir simultáneamente a ambos. Supongamos, sin pérdida de generalidad, que:
        \[
            p \mid b_0 \quad \text{y} \quad p \nmid c_0
        \]
        Puesto que $f$ es primitivo y $p \mid a_i$ para todo $i \in \{0, \dots, n-1\}$, necesariamente $p \nmid a_n$ (de lo contrario, $p$ dividiría a todos los coeficientes, contradiciendo $\operatorname{mcd}(a_0, \dots, a_n) = 1$). Por tanto:
        \[
            a_n = b_rc_s \implies p \nmid b_rc_s \implies p \nmid b_r \quad \text{y} \quad p \nmid c_s
        \]
        En particular, como $p \mid b_0$ y $p \nmid b_r$, existe un primer índice $k$ tal que $p \nmid b_k$, con $0 < k \le r \le n - 1 < n$.

        Consideremos el coeficiente $k$-ésimo de $f$:
        \[
            a_k = b_0c_k + b_1c_{k-1} + \dots + b_{k-1}c_1 + b_kc_0
        \]
        Como $k < n$, por hipótesis tenemos que $p \mid a_k$. Por la elección minimal de $k$, sabemos que $p \mid b_j$ para todo $j < k$, lo que implica que:
        \[
            p \mid (b_0c_k + b_1c_{k-1} + \dots + b_{k-1}c_1)
        \]
        Despejando, se obtiene:
        \[
            b_kc_0 = a_k - \sum_{j=0}^{k-1} b_jc_{k-j} \implies p \mid b_kc_0
        \]
        Al ser $p$ primo, debe ocurrir que $p \mid b_k$ o $p \mid c_0$, lo cual entra en contradicción directa con el hecho de que $p \nmid b_k$ y $p \nmid c_0$.

        Por consiguiente, $f$ no admite tal factorización y es irreducible en $\mathbb{Z}[x]$.

        \item Consideremos el polinomio recíproco de $f$:
        \[
            f^*(x) = x^n f\left(\frac{1}{x}\right) = a_n + a_{n-1}x + \dots + a_1x^{n-1} + a_0x^n
        \]
        Llamando $a'_i$ al coeficiente de $x^i$ en $f^*(x)$, tenemos $a'_i = a_{n-i}$ para cada $i \in \{0, \dots, n\}$. De este modo:
        \begin{itemize}
            \item $a'_0 = a_n$, por lo que $p \mid a'_0$ y $p^2 \nmid a'_0$.
            \item Para cada $i \in \{0, \dots, n-1\}$, el coeficiente es $a'_i = a_{n-i}$ con $n-i \in \{1, \dots, n\}$, luego $p \mid a'_i$.
        \end{itemize}
        Por tanto, el polinomio $f^*(x)$ verifica las hipótesis del primer apartado, luego $f^*(x)$ es irreducible en $\mathbb{Z}[x]$.

        Demostramos ahora lo pedido por reducción al absurdo.
        Si $f(x)$ fuera reducible en $\mathbb{Z}[x]$, existirían $g, h \in \mathbb{Z}[x]$ no invertibles tales que $f(x) = g(x)h(x)$. Tomando recíprocos se obtendría $f^*(x) = g^*(x)h^*(x)$, donde ni $g^*$ ni $h^*$ son unidades, lo que contradice la irreducibilidad de $f^*(x)$. Concluimos que $f$ es irreducible en $\mathbb{Z}[x]$.
    \end{enumerate}
\end{proof}
