\section{Curvas simples}

\subsection{Definición y primeras observaciones}

\begin{definicion}
    Una \textbf{curva simple} es un subconjunto $C\subset \bb{R}^2$ no vacío tal que $\forall p\in C$ se tiene que 
    \begin{itemize}
        \item $\exists\ V$ entorno abierto de $p$ en $C$.
        \item $\exists\ I\subset \bb{R}$ intervalo abierto.
        \item $\exists\ \alpha:I \to \bb{R}^2$ curva parametrizada regular
    \end{itemize} 
    de forma que $\alpha(I)=V$, y la aplicación $\alpha:I\to V$ es un homeomorfismo.
\end{definicion}

\begin{observacion}\
    \begin{enumerate}
        \item La aplicación $\alpha$ la llamaremos \textbf{parametrización local} de $C$.
        \item $C$ no puede tener ``esquinas'' (o ``picos'') ya que $\alpha$ es regular.
        \item $C$ no puede tener autointersecciones. Ni siquiera el Folium de Descartes puede ser una curva simple, a pesar de que en este caso $\alpha$ es inyectiva.
        \item Los entornos $V$ son todos de la forma
        \begin{gather*}
            V = U\cap C
        \end{gather*}
        donde $U$ es un entorno abierto de $p$ en $\bb{R}^2$.
        \item Los entornos $V$ de la definición son todos conexos.
    \end{enumerate}
\end{observacion}

\subsection{Cambios de parametrización}

\begin{ejemplo}\label{ej:cambio}
    Supongamos que tenemos una curva simple $C\subset \bb{R}^2$ y consideramos las parametrizaciones $\alpha: I \to C$ y $\beta:J \to C$ de forma que $W = \alpha(I)\cap \beta(J)\neq \emptyset$.\\

    Tenemos que $\alpha^{-1}(W)\subset I$ y que $\beta^{-1}(W)\subset J$. Además, como $W$ es abierto, tendremos que $\alpha^{-1}(W)$ es abierto en $I$ y $\beta^{-1}(W)$ es abierto en $J$ y por ser abiertos de un abierto, ambos serán abiertos en $\bb{R}$.\\

    Además, podemos considerar los \textbf{cambios de parametrización}
    \begin{align*}
        \Phi &= \beta^{-1}\circ \alpha : \alpha^{-1}(W) \to \beta^{-1}(W)\\
        \Phi^{-1} = \Psi &= \alpha^{-1}\circ \beta : \beta^{-1}(W) \to \alpha^{-1}(W)
    \end{align*}
    Veamos que $\Phi$ y $\Psi$ son \textbf{difeomorfismos}. Para ello, tan solo tendremos que verlo para $\Phi$. Además, bastará con ver que $\Phi$ es derivable (diferenciable), ya que tenemos que es una biyección y su inversa será $\Psi$, donde $\alpha$ y $\beta$ juegan papeles inversos por lo que será equivalente a ver que $\Phi$ lo sea.\\

    Recordamos la situación en la que estamos
    \begin{gather*}
        \xymatrix@R=3mm{
            \alpha^{-1}(W) \ar[r]^{\Phi}  & \beta^{-1}(W) \ar[r]^{\beta_1} & \bb{R} \\
            t_0 \ar@{|->}[r] & \Phi(t_0) \ar@{|->}[r] & \beta(\Phi(t_0))
        }
    \end{gather*}
    Además, como $\beta$ es regular tenemos que $\beta'(\Phi(t_0)) \neq (0,0)$. Supongamos sin pérdida de generalidad que $\beta_1'(\Phi(t_0)) \neq 0$ (donde $\beta_1$ es la primera componente de $\beta$) y estamos en condiciones de aplicarle el Teorema de la Función Inversa. Este nos dice que existe un $W_2$ entorno abierto de $\Phi(t_0)$ en $\beta^{-1}(W)$ (que también será abierto en $\bb{R}$ por ser abierto de abierto) y existe también un $W_1$ entorno abierto de $\beta_1(\Phi(t_0))$ en $\bb{R}$ de forma que $\beta_1(W_2) = W_1$ y tal que $\beta_{1|_{W_2}} : W_2 \to W_1$ es un difeomorfismo. Además,
    \begin{gather*}
        \Phi = \beta^{-1}\circ \alpha \quad \Rightarrow \quad \alpha = \beta \circ \Phi \quad \Rightarrow \quad \alpha_1 = \beta_1 \circ \Phi
    \end{gather*}
    Como $\Phi$ es continua, en particular será continua en $t_0$ por lo que tenemos un entorno abierto de $\Phi(t_0)$ que es $W_2$. Por la definición topológica de continuidad tendremos que existe un $W_3$ entorno abierto de $t_0$ en $\alpha^{-1}(W)$ y por tanto en $\bb{R}$ de forma que $\Phi(W_3) \subset W_2$.\\

    Vamos ahora a considerar la restricción
    \begin{gather*}
        \alpha_{1|_{W_3}} = (\beta_1\circ \Phi)_{|_{W_3}} = \beta_{1|_{W_2}} \circ \Phi_{|_{W_3}}
    \end{gather*}
    Además, por ser una parametrización, $\alpha$ es derivable y por tanto su primera componente $\alpha_1$ también lo será (y en particular su restricción también). Del Teorema de la Función Inversa anterior sabemos que $\beta_{1|_{W_2}}$ es un difeomorfismo. Por tanto podemos pasar $\beta_{1|_{W_2}}$ al otro lado (usando su inversa) y obtener
    \begin{gather*}
        \Phi_{|_{W_3}} = \beta_{1|_{W_2}}^{-1} \circ \alpha_{1|_{W_3}}
    \end{gather*}
    y llegamos a la conclusión de que $\Phi_{|_{W_3}}$ es derivable, pero $W_3$ es un entorno abierto de $t_0$. Hemos llegado a que para un $t_0$ cualquiera del dominio del cambio de parámetros existe un entorno suyo en el que $\Phi$ es derivable. En particular lo será en $t_0$.
\end{ejemplo}

Del ejemplo anterior (Ejemplo~\ref{ej:cambio}) podemos concluir la siguiente proposición:\\
\begin{prop}\label{prop:cambio}
    Sean $C\subset \bb{R}^2$ una curva simple y $\alpha:I \to C$, $\beta: J \to C$ dos parametrizaciones suyas cuyas imágenes se solapan, es decir, tal que 
    \begin{gather*}
        W = \alpha(I)\cap \beta(J) \neq \emptyset
    \end{gather*}
    Entonces las aplicaciones \textbf{cambio de parámetros} siguientes
    \begin{align*}
        \Phi &= \beta^{-1}\circ \alpha : \alpha^{-1}(W) \to \beta^{-1}(W)\\
        \Psi &= \alpha^{-1}\circ \beta : \beta^{-1}(W) \to \alpha^{-1}(W)
    \end{align*}
    son \underline{difeomorfismos} entre abiertos de $\bb{R}$ (de $I$ y $J$ respectivamente).
\end{prop}

\subsection{Recta tangente}

\begin{definicion}
    Sea $C\subset \bb{R}^2$ una curva simple y un punto $p\in C$. Definimos la \textbf{recta tangente} a la curva $C$ en el punto $p$ como la recta afín $T_pC$ que pasa por $p$ y tiene dirección $\alpha'(t_0)$, es decir,
    \begin{gather*}
        T_pC = p + \langle \alpha'(t_0) \rangle
    \end{gather*}
    donde $\alpha: I \to C$ es una parametrización con $\alpha(t_0)=p$.
\end{definicion}

\begin{observacion}
    En primer lugar es fácil ver que $T_pC$ es una recta, ya que $\alpha'(t_0)\neq 0$ por ser una curva regular. \\
    
    Nos planteamos además si está bien definida, es decir, qué ocurre si tomamos una parametrización distinta. Para ello consideramos $\beta: J \to C$ otra parametrización con $\beta(u_0)=p$, donde $u_0\in J$ y queremos ver la relación entre $\alpha'(t_0)$ y $\beta'(u_0)$.\\

    Sabemos que $\Phi = \beta^{-1} \circ \alpha$ es un difeomorfismo (Proposición~\ref{prop:cambio}). Además, esto podemos aplicarlo porque sabemos que se solapan, ya que 
    \begin{gather*}
        \left.
        \begin{array}{l}
            p = \alpha(t_0) \in \alpha (I)\\[0.8em]
            p = \beta(u_0) \in \beta(J)
        \end{array}
        \right\} \Rightarrow \alpha(I) \cap \beta(J) \neq \emptyset,\quad \Phi(t_0) = u_0
    \end{gather*} 
    Además $\alpha = \beta \circ \Phi$ y estamos en condiciones de derivar en $t_0$ aplicando la regla de la cadena.
    \begin{gather*}
        \alpha'(t_0) = (\beta \circ \Phi)'(t_0) = \Phi'(t_0)\beta'(\Phi(t_0)) = \Phi'(t_0)\beta'(u_0) \neq 0
    \end{gather*}
    Por colinealidad tenemos que 
    \begin{gather*}
        \langle \alpha'(t_0) \rangle = \langle \beta'(u_0) \rangle
    \end{gather*}
    lo que nos confirma que la definición anterior está bien hecha y no depende de la parametrización escogida.
\end{observacion}

\subsection{Construcción de curvas simples}

\begin{ejemplo}\label{ej:construccion} Vamos a ver a continuación una serie de ejemplos que nos darán las 5 formas de construir curvas simples:
    \begin{enumerate}
        \item\label{ej:abierto} Sea $C\subset \bb{R}^2$ una curva simple y consideramos $C'\subset C$ un abierto suyo. Entonces $C'$ es también una curva simple.
        \begin{proof}
            Para probarlo tenemos que ver que para todo punto existe una parametrización regular. Así consideramos un $p'\in C'$ y en particular $p'\in C$ por lo que existirá un $V\subset C$ entorno de $p'$, un intervalo abierto $I\subset \bb{R}$ y una curva regular $\alpha : I \to \bb{R}^2$ tal que $\alpha(I)=V$ y tal que $\alpha : I \to V$ es un homeomorfismo.\\

            Podríamos ahora pensar en elegir un $V'=V\cap C'$ pero a poco que lo pensemos no vale porque $V'$ puede ser no conexo. Por ello consideramos $V'$ como la componente conexa de $V\cap C'$ que contiene a $p'$, que será un entorno abierto y conexo de $p'$ en $C'$ (por ser $V\cong I$ localmente conexo, las componentes conexas de sus abiertos son abiertas).\\

            Con esta elección es natural considerar $I'=\alpha^{-1}(V') \subset \alpha^{-1}(V) \subset I \subset \bb{R}$. Por ser $\alpha$ continua tendremos que $I'$ es un abierto de $\bb{R}$ y por ser homeomorfismo $\alpha^{-1}$ será también continua lo que nos garantiza que $I'$ es conexo. Concluimos que $I'$ es un intervalo abierto de $\bb{R}$.\\

            Finalmente consideramos $\hat{\alpha} = \alpha_{|I'}$ que será una curva regular de $I'$ en $\bb{R}^2$ (por ser restricción en el dominio de una curva regular). \\
            
            Queremos ver ahora que se verifican las condiciones para que $C'$ sea una curva simple. Tenemos que 
            \begin{gather*}
                \hat{\alpha}(I') = \hat{\alpha} (\alpha^{-1} (V')) = V'
            \end{gather*}
            y además $\hat{\alpha}:I'\to V'$ es un homeomorfismo por ser restricción de $\alpha$ que ya lo era.
        \end{proof}

        \item\label{ej:union} Consideramos $C\subset \bb{R}^2$ un subconjunto y $\{C_i\}_{i\in I}$ una familia de subconjuntos de $C$, es decir, $C_i\subset C$ para todo $i\in I$ tal que cada $C_i$ es una curva simple abierta en $C$ y tal que 
        \begin{gather*}
            C = \bigcup_{i\in I} C_i
        \end{gather*}
        Entonces $C$ es una curva simple.
        \begin{proof}
            La demostración se deja como ejercicio
        \end{proof}

        \item\label{ej:grafo} Grafos de funciones derivables. Consideramos un intervalo abierto $I\subset \bb{R}$ y una función $f: I\to \bb{R}$ derivable. De esta forma podemos tomar
        \begin{gather*}
            Grafo(f) = \{(x,y) \in \bb{R}^2 \ : \ x\in I,\ y=f(x)\} \subset \bb{R}^2
        \end{gather*}
        y dicho grafo será una curva simple.
        \begin{proof}
            Comencemos nombrando $C=Grafo(f)\neq \emptyset$ y podemos reescribir
            \begin{gather*}
                C = \{(x,f(x)) : x\in I\}
            \end{gather*}
            Tomamos un $p=(x,f(x))\in C$ donde $x\in I$. Elegimos $V=C$ por lo que trivialmente $V$ será un entorno abierto de $p$ en $C$. También tomamos el $I$ dominio de la función que ya es un intervalo abierto de $\bb{R}$. Nos queda por escoger un $\alpha:I \to \bb{R}^2$ que a poco que pensemos es claro que nos vale
            \begin{gather*}
                \alpha(t) = (t,f(t))
            \end{gather*}  
            que es derivable por serlo la identidad y $f$. Además, tenemos que $\alpha'(t) = (1, f'(t)) \neq (0,0)$ por lo que claramente $\alpha$ es una curva regular. Tenemos que
            \begin{gather*}
                \alpha(I) = \{(t, f(t)) : t\in I\} = Grafo(f) = C = V
            \end{gather*}
            Veamos que $\alpha : I \to V$ es un homeomorfismo. Para ello consideramos $\pi:\bb{R}^2 \to \bb{R}$ la primera proyección, es decir, tal que
            \begin{gather*}
                \pi(x,y) = x
            \end{gather*}
            y se verificará que 
            \begin{gather*}
                (\pi \circ \alpha)(t) = \pi ( \alpha(t)) = \pi(t, f(t)) = t\\
                (\alpha \circ \pi)(x, f(x)) = \alpha(\pi(x, f(x))) = \alpha(x) = (x, f(x))
            \end{gather*}
            Concluimos que $\pi$ es la inversa de $\alpha$ y además es continua (ya que la proyección siempre lo es\footnote{Contenido estudiado en Topología I}) por lo que $\alpha : I \to V$ es un homeomorfismo.
        \end{proof}
        Una consecuencia clara es que todas las rectas de $\bb{R}^2$ son curvas simples (las rectas verticales son grafos de funciones de $y$, intercambiando los papeles de $x$ e $y$). Por la primera observación (apartado~\ref{ej:abierto}) tendremos también que todos los segmentos abiertos de $\bb{R}^2$ son curvas simples, y por la segunda (apartado~\ref{ej:union}) también lo será cualquier unión de ellos en la que cada segmento sea abierto en la unión (no vale, por ejemplo, para dos segmentos que se cruzan).

        no trivial\footnote{Un valor regular no trivial es aquel $a\in \bb{R}$ que además verifica $F^{-1}(\{a\}) \neq \emptyset$.}

        Tomamos un punto $p\in C = F^{-1}(\{a\})$ y por ser $a$ regular tenemos que 
        \begin{gather*}
            \nabla F_p = \left( \frac{\partial F}{\partial x} (p), \frac{\partial F}{\partial y}(p)\right) \neq (0,0)
        \end{gather*}
        Vamos a suponer sin pérdida de generalidad que $\frac{\partial F}{\partial y}(p) \neq 0$. Estamos en la situación en la que $p=(x,y)$ y tenemos que $F(x,y) = a$ que es una ecuación y esto nos debería recordar al Teorema de la Función Implícita que nos permite despejar una variable en función de la otra de forma local, ya que tenemos todas las hipótesis necesarias. De esta forma sabemos que $\exists U_p$ entorno abierto de $p$ en $O$ (y por tanto en $\bb{R}^2$), $\exists I_p\subset \bb{R}$ un intervalo abierto y $\exists f_p : I_p \to \bb{R}$ tales que 
        \begin{gather*}
            C\cap U_p = F^{-1}(\{a\})\cap U_p = Grafo(f_p) = \{(x,y)\in \bb{R}^2 \ : \ x\in I_p,\ y=f_p(x)\}
        \end{gather*}
        y por ser grafo de una función diferenciable tenemos que $C\cap U_p$ es una curva simple. Además, podemos escribir
        \begin{gather*}
            C = \bigcup_{p\in C} (C\cap U_p) 
        \end{gather*}
        y además $(C\cap U_p)$ es abierto en $C$ porque $U_p$ es abierto en $\bb{R}^2$ por lo que su intersección con $C$ será un abierto en la topología inducida. De esta forma tenemos que $C$ es una curva simple.\\

        Hemos demostrado que las imágenes inversas de un valor regular no trivial de una función de dos variables son curvas simples (no necesariamente conexas). Además a estas curvas las llamaremos \textbf{curvas de nivel} de la función.\\

        Todos los grafos en particular son de este tipo, ya que tenemos $F(x,y)=a$ y tomando $F(x,y) = y - f(x)$ y $a=0$ tenemos $y=f(x)$ que era el caso que estudiábamos en el punto anterior (apartado~\ref{ej:grafo}). En este caso sí serán siempre conexos.

        \item\label{ej:periodica} Consideramos $\alpha: \bb{R} \to \bb{R}^2$ curva regular $L-$periódica (con $L>0$) tal que $\alpha_{|_{[0,L)}}$ es inyectiva. Veamos que el siguiente conjunto
        \begin{gather*}
            C = traza(\alpha) = \alpha(\bb{R}) = \alpha([0,L]) = \alpha([0,L))
        \end{gather*}
        es una curva simple homeomorfa a $S^1$.\\

        Lo primero que observamos es que $C$ es la imagen por $\alpha$, que es una función continua, de un compacto por lo que $C$ es un compacto. Además es no trivial, ya que $C$ es también $\alpha(\bb{R})$ y $\bb{R}$ no es vacío teniendo necesariamente alguna imagen. Tenemos
        \begin{gather*}
            \bb{R} \overset{\alpha}{\longrightarrow} C \quad \text{ continua y sobreyectiva}
        \end{gather*} 
        y así habrá una relación de equivalencia $R_\alpha$ inducida por $\alpha$ en $\bb{R}$ donde si $t,t'\in \bb{R}$ se tiene que 
        \begin{gather*}
            t R_\alpha t' \sii \alpha(t) = \alpha(t')
        \end{gather*}
        Podemos ver esta relación como
        \begin{gather*}
            \left.
            \begin{array}{l}
                t = kL + m,\ \  k\in \bb{Z},\ \  m\in [0,L)\\
                t' = k'L + m',\ \ k'\in \bb{Z},\ \  m'\in [0,L)
            \end{array}
            \right\} \Rightarrow 
            \begin{array}{c}
                t R_\alpha t' \sii \alpha(kL + m) = \alpha(k'L + m') \\ 
                \sii \alpha(m)=\alpha(m') \sii m=m' \sii\\
                t-t' = (k-k')L\in L\bb{Z}
            \end{array}
        \end{gather*}
        de esta forma podemos considerar el conjunto cociente
        \begin{gather*}
            \xymatrix{
                \bb{R} \ar[r]^\alpha \ar[d]_\pi & C \\
                S^1 \cong \bb{R} / R_\alpha \quad \ar@{-->}[ur]_{\tilde{\alpha}}
            }
        \end{gather*}
        y tenemos que existe $\tilde{\alpha} : \bb{R}/R_\alpha \to C$ continua y biyectiva. Para ver que es un homeomorfismo basta con ver que $\tilde{\alpha}$ es cerrada, y lo es porque sale de un compacto ($S^1$) y llega a un espacio T2 ($C$ es T2 porque lo es $\bb{R}^2$ y es una propiedad hereditaria). Por tanto
        \begin{gather*}
            C \cong \bb{R}/R_\alpha \cong S^1
        \end{gather*}
        De esta forma podemos tomar un $p\in C=tr(\alpha)$ por lo que habrá un $t\in \bb{R}$ tal que $p=\alpha(t)$. Vamos a considerar el intervalo $(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2})\subset \bb{R}$ que contiene a $t$ y es abierto en $\bb{R}$. Sabemos que la aplicación
        \begin{gather*}
            \alpha_{|_{\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)}} : \left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right) \longrightarrow \alpha\left(\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)\right)
        \end{gather*}
        es un homeomorfismo. Tomamos por tanto $V = \alpha\left(\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)\right)$ y sabemos que $\alpha(t)=p$ está en dicho $V$. Además, por ser $\tilde{\alpha}$ un homeomorfismo y ser la imagen por $\pi$ de dicho intervalo un abierto de $S^1$, $V$ es abierto en $C$. De esta forma tenemos que $V$ es un entorno de $p$ en $C$.\\

        Definimos $I = \left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)$ que es claramente un intervalo abierto de $\bb{R}$ y definimos $\beta = \alpha_{|_{\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)}}$. De esta forma tendremos
        \begin{gather*}
            \beta: I \longrightarrow V
        \end{gather*}
        y podemos escribir
        \begin{align*}
            \beta = (\tilde{\alpha}\circ \pi)_{|_{\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)}}
            = \tilde{\alpha} \circ \pi_{|_{\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)}}
        \end{align*}
        y por ser $\tilde{\alpha}$ un homeomorfismo y $\pi_{|_{\left(t- \nicefrac{L}{2}, t+ \nicefrac{L}{2}\right)}}$ un homeomorfismo sobre su imagen (que es un abierto de $S^1$), su composición será un homeomorfismo sobre su imagen.\\
    \end{enumerate}
\end{ejemplo}

\section{Clasificación de las curvas simples}

\subsection{Curvas de Jordan}

\begin{definicion}
    A las curvas simples de $\bb{R}^2$ que son homeomorfas a $S^1$ se les llama \textbf{curvas de Jordan}.
\end{definicion}

\subsection{Parametrizaciones por el arco y solapamiento de entornos.}

Vamos a ver ahora que todas las curvas simples conexas de $\bb{R}^2$ son trazas de curvas regulares en $\bb{R}^2$ (aunque la otra inclusión no se da\footnote{Basta pensar en una curva regular con autointersecciones o el Folium de Descartes}).

\begin{observacion}
    En la definición de curvas simples podemos exigir que las parametrizaciones sean curvas parametrizadas por el arco (ya que toda curva regular se puede reparametrizar por el arco).\\

    Consideramos $C\subset \bb{R}^2$ una curva simple y $\alpha:I \to C$, $\beta: J \to C$ parametrizaciones por el arco. Supongamos que
    \begin{gather*}
        \alpha(I)\cap \beta(J)=W\neq \emptyset, \qquad \alpha(I)\not\subset\beta(J), \qquad \beta(J)\not\subset \alpha(I)
    \end{gather*}
    Ya sabemos (Proposición~\ref{prop:cambio}) que los cambios de parametrización eran
    \begin{gather*}
        \Phi = \beta^{-1} \circ \alpha : \alpha^{-1}(W) \to \beta^{-1}(W)\\
        \Psi = \alpha^{-1} \circ \beta : \beta^{-1}(W) \to \alpha^{-1}(W)
    \end{gather*}
    y sabemos que $\Phi$ y $\Psi$ son difeomorfismos (Proposición~\ref{prop:cambio}). Vamos a definir el siguiente conjunto:
    \begin{gather*}
        X=\{(s_1, s_2) \in I\times J \ : \ \alpha(s_1)=\beta(s_2)\} \subset I \times J \subset \bb{R}^2
    \end{gather*}
    y de esta forma podemos reescribir
    \begin{gather*}
        X=\{(s_1, s_2) \in I\times J \ : \ s_1 \in \alpha^{-1}(W), \ s_2 = \Phi(s_1)\} = Grafo(\Phi) = Grafo(\Psi) 
    \end{gather*}
    El sitio donde coinciden dos funciones continuas es cerrado en el producto por lo que $X$ será cerrado en $I\times J$.\\

    Veamos a continuación una serie de observaciones, que serán cada una consecuencias de la anterior:

    \begin{enumerate}
        \item\label{obs:cc_grafos} Todas las c.c de $X$ son grafos simultáneamente sobre intervalos abiertos de $\alpha^{-1}(W)$ y sobre intervalos abiertos de $\beta^{-1}(W)$, que son cerrados en $I\times J$.\\
        
        Si $\Phi$ es un cambio de parámetros, entonces podemos escribir $\alpha = \beta \circ \Phi$ en $\alpha^{-1}(W)$. Si derivamos en cualquier punto del dominio $s\in \alpha^{-1}(W)$ obtenemos
        \begin{gather*}
            \alpha'(s) = (\beta \circ \Phi)'(s) \overset{R.C.}{=} \Phi'(s)\beta'(\Phi(s))
        \end{gather*}
        Donde hemos aplicado la regla de la cadena. Tomando módulos obtenemos 
        \begin{gather*}
            1 = |\alpha'(s)| = |\Phi'(s)| \cdot |\beta'(\Phi(s))| = |\Phi'(s)| \cdot 1 = |\Phi'(s)|
        \end{gather*}
        \item Por tanto $\Phi'(s) = 1$ o bien $\Phi'(s) = -1$ en cada componente conexa de $\alpha^{-1}(W)$. Eso implica que las componentes conexas de $X$ son segmentos (de recta) de pendiente $1$ o $-1$ que son cerradas en $I\times J$.

\subsection{Teorema de clasificación}

\begin{teo}[Clasificación de las curvas simples en $\bb{R}^2$]\label{teo:clasif}
    Sea $C\subset \bb{R}^2$ una curva simple y conexa, entonces existen 2 opciones excluyentes para $C$:
    \begin{enumerate}[label=\alph*)]
        \item $C$ es una curva de Jordan ($C \cong S^1$) que se obtiene como la traza de una curva regular $L-$periódica ($L>0$) que es inyectiva en $[0,L)$.
        \item $C$ es la traza de una sola parametrización ($C\cong \bb{R}$)
    \end{enumerate}
\end{teo}

\subsection{Número de componentes conexas del complementario}

\begin{observacion}
    Supongamos que $C$ es una curva simple, conexa y cerrada en $\bb{R}^2$. Entonces $\bb{R}^2 \setminus C$ es abierto en $\bb{R}^2$. Queremos estudiar cuántas componentes conexas tiene este conjunto. Denotamos $\Omega$ a una c.c. cualquiera de $\bb{R}^2 \setminus C$ (que sabemos que es localmente conexo por ser abierto de $\bb{R}^2$). Por tanto $\Omega$ es abierto (en $\bb{R}^2\setminus C$ y por tanto en $\bb{R}^2$). Llamamos $\Omega'$ al conjunto de todas las demás c.c. de $\bb{R}^2 \setminus C$. Podemos escribir
    \begin{gather*}
        \bb{R}^2 \setminus C = \Omega \dot{\cup} \Omega' \quad \Rightarrow \quad \bb{R}^2 = C \dot{\cup} \Omega \dot{\cup} \Omega'
    \end{gather*}
    Y de esta forma tendremos $C\cup \Omega = \bb{R}^2 \setminus \Omega'$, lo que nos dice que $C\cup \Omega$ es cerrado en $\bb{R}^2$. Podemos tomar adherencias y considerar
    \begin{gather*}
        \overline{\Omega} \subset \overline{\Omega \cup C} = \Omega \cup C 
    \end{gather*}
    Además, sabemos que 
    \begin{gather*}
        Fr(\Omega) = \overline{\Omega} \setminus \overset{\circ}{\Omega} = \overline{\Omega} \setminus \Omega \quad \Rightarrow \quad Fr(\Omega) \subset (\Omega \cup C)\setminus \Omega = C
    \end{gather*}
    De esta forma hemos probado que $Fr(\Omega) \subset C$.\\

    Sabemos que $Fr(\Omega)$ es cerrado en $C$. Nos preguntamos si $Fr(\Omega)=\emptyset$ pero ya habíamos escrito $Fr(\Omega) = \overline{\Omega} \setminus \Omega$ por lo que si la frontera fuera vacía necesariamente tendríamos
    \begin{gather*}
        \overline{\Omega} = \Omega
    \end{gather*}
    donde $\overline{\Omega}$ es cerrado en $\bb{R}^2$ y $\Omega$ es abierto. Por la conexión de $\bb{R}^2$ necesariamente o $\Omega$ es el vacío o el total. No puede ser el vacío ya que es una componente conexa. Si $\Omega=\bb{R}^2$ entonces tendríamos que $C=\emptyset$ lo cual también es contradictorio. Llegamos entonces a que $Fr(\Omega)\subset C$ y es un cerrado no vacío de $C$.\\

    Veamos entonces que $Fr(\Omega)$ es abierto en $C$. Lo haremos viendo que todos los puntos de $Fr(\Omega)$ son interiores a $Fr(\Omega)$ (en la topología de $C$). Cogemos un punto $p\in Fr(\Omega)\subset C$ y por ser un punto de $C$ podemos aplicarle el lema visto anteriormente (Lema~\ref{lema:local}). Tenemos que existe $W$ un entorno de $p$ abierto en $\bb{R}^2$ de forma que $W\setminus C$ tiene 2 componentes conexas. Denotaremos 
    \begin{gather*}
        W \setminus C = A_1 \cup A_2
    \end{gather*}
    donde $A_1$ y $A_2$ son las c.c. De esta forma podemos escribir
    \begin{gather*}
        Fr^W(A_1) = Fr^W(A_2) = W \cap C
    \end{gather*}
    Tenemos por tanto
    \begin{gather*}
        W \setminus C = A_1 \cup A_2 \subset \bb{R}^2 \setminus C = \Omega \cup \Omega'
    \end{gather*}
    Por conexión tendremos que necesariamente $A_i$ con $i=1,2$ está en $\Omega$ o en $\Omega'$. Supongamos por un instante (porque no va a ser posible) que $A_1, A_2\subset \Omega'$, lo que nos diría que $A_1\cup A_2 = W \setminus C \subset  \Omega'$. Tenemos
    \begin{gather*}
        W \cap \Omega = (W \setminus C) \cap \Omega = (A_1\cup A_2) \cap \Omega = \emptyset
    \end{gather*}
    Hemos encontrado entonces un entorno $W$ de $p\in Fr(\Omega)$ abierto en $\bb{R}^2$ y tal que $W\cap \Omega = \emptyset$. Esto nos dice que $p\not\in \overline{\Omega}$ lo cual claramente es contradictorio ya que $p\in Fr(\Omega) = \overline{\Omega} \setminus \Omega$.\\

    Por tanto llegamos a la conclusión de que o bien $A_1$ o bien $A_2$ están en $\Omega$. Supongamos sin pérdida de generalidad que $A_1 \subset \Omega$. Podemos expresar
    \begin{gather*}
        W \cap C = Fr^W(A_1) = Fr(A_1) \cap W \subset Fr(A_1) \subset \overline{A_1} \subset \overline{\Omega}
    \end{gather*}
    Sabiendo que $W\cap C \subset \overline{\Omega} \setminus \Omega = Fr(\Omega)$. \\
    
    Hemos probado que existe un entorno abierto $W\cap C$ de $p$ en $C$ tal que $W\cap C \subset Fr(\Omega)$, luego $p$ es interior a $Fr(\Omega)$. La arbitrariedad de $p$ nos dice que $Fr(\Omega)$ es abierto y cerrado de $C$ y no es vacío, es decir, que $Fr(\Omega)=C$ (recordemos que $C$ es conexo).\\

    Sospechamos que $\bb{R}^2 \setminus C$ tiene exactamente 2 componentes conexas. Supongamos que hay 3 distintas (o más) en $\bb{R}^2 \setminus C$. Notamos entonces $\Omega_1, \Omega_2, \Omega_3$ tres componentes conexas distintas y fijamos un $p_0\in C$. El lema estudiado (Lema~\ref{lema:local}) nos dice que $\exists W_0$ entorno abierto de $p_0$ en $\bb{R}^2$ de forma que $W_0\setminus C = A_1 \dot{\cup} A_2$ donde $A_1$ y $A_2$ son las 2 c.c. Como $p_0\in C=Fr(\Omega_j)$ para $j=1,2,3$, se tiene que $W_0\cap \Omega_j\neq\emptyset$ y, por lo estudiado anteriormente, para cada $j\in\{1,2,3\}$ se verifica $A_1 \subset \Omega_j$ o bien $A_2\subset \Omega_j$. Esto nos dice que $\exists i_0 \in \{1,2\}$ y $j_1\neq j_2 \in \{1,2,3\}$ de forma que $A_{i_0} \subset (\Omega_{j_1}\cap \Omega_{j_2})=\emptyset$. Esto nos diría que $A_{i_0}=\emptyset$ lo que nos lleva a contradicción. Llegamos a la siguiente proposición:
\end{observacion}

\begin{prop}\label{prop:max2}
    Sea $C\subset \bb{R}^2$ una curva simple, conexa y cerrada. Entonces $\bb{R}^2 \setminus C$ tiene a lo sumo 2 componentes conexas cuya frontera común es $C$.
\end{prop}

\subsection{Distinción de las componentes: paridad de cortes con rectas}

Para probar que son dos exactamente hay que distinguir entre ellas. Vamos a suponer que $C$ es compacta, es decir, homeomorfa a $S^1$. Comenzamos viendo el caso más simple posible, $C=S^1$.\\

Comenzamos escribiendo $C$ de forma explícita:
\begin{gather*}
    x^2 + y^2 = 1, \qquad F(x,y) = x^2 + y^2, \qquad S^1 = F^{-1}(\{1\})
\end{gather*}
De esta forma tendremos que 
\begin{gather*}
    \bb{R}^2 \setminus S^1 = F^{-1}((-\infty, 1)) \dot{\cup} F^{-1}((1,+\infty))
\end{gather*}
donde ambos conjuntos disjuntos son conexos y abiertos. Esto lo podemos hacer porque $S^1$ viene dado por una ecuación implícita, es decir, es imagen inversa de un valor regular (apartado~\ref{ej:nivel} del Ejemplo~\ref{ej:construccion}) de un $F:\bb{R}^2 \to \bb{R}$. Sin embargo, en general para cualquier curva parece que no es viable encontrar una función $F$ con las características anteriores.\\

Veamos un enfoque distinto. Consideramos un punto cualquiera $p$ que quede en la región interior a $C$. Cualquier semirrecta que tracemos desde $p$ en una cierta dirección cortará en un único punto a $C$ (habrá infinitas semirrectas que verifiquen esto, bastará variar la dirección). Sin embargo, si consideramos ahora que $p$ queda fuera de la circunferencia, estas semirrectas ahora pueden cortar en $0,1$ o $2$ puntos a $C$ dependiendo de la dirección escogida (infinitas direcciones para cortar $0$ o $2$ veces y solo 2 direcciones para cortar $1$ vez, las tangentes). De momento vamos a olvidar que pueden cortar en las tangentes por lo que solo cortarán en $0$ o en $2$ puntos distintos.

Por tanto, en $S^1$ podemos distinguir también en qué c.c. está un punto dependiendo de la paridad de $\#(L \cap C)$ donde $L$ son semirrectas \underline{no tangentes} a $S^1$. Para que esto funcione para una curva cualquiera $C$ necesitaremos 2 ingredientes: uno geométrico (\hyperref[def:transv_p]{\textbf{transversalidad}}) y otro analítico (\hyperref[teo:sard]{\textbf{Teorema de Sard}}).\\

Consideramos así $C$ una curva simple y $L$ una recta (o semirrecta con punto base en $\bb{R}^2 \setminus C$) de $\bb{R}^2$.\\

Sabemos que $L$ es tangente a $C$ en $p$ si $T_pC=L$ (Usamos la notación $L\|_p C$). Podemos pensar en el resto de casos posibles y hacemos la siguiente definición:

\begin{definicion}\label{def:transv_p}
    Diremos que $L$ es \textbf{transversa} a $C$ en $p$ si $T_pC \neq L$ y lo notaremos por $L\transv_pC$.
\end{definicion}

Nos preguntamos ahora qué condición ($\|_p$ o bien $\transv_p$) es más ``estable''. En principio parece que ante pequeñas variaciones la condición de transversalidad es mucho más ``estable'' y que la tangencia es ``inestable''.\\

Consideramos así una curva $C$ y $L$ una recta con $C\transv_p L$. De esta forma tendremos que $L$ viene dada por una ecuación lineal 
\begin{gather*}
    L \equiv \langle x,n \rangle + c = 0
\end{gather*}
donde $x\in \bb{R}^2$ y $|n|=1$. El haz $H$ de rectas paralelas a $L$ tiene ecuación 
\begin{gather*}
    H = \langle x, n \rangle + c + t = 0 \quad \forall t \in \bb{R}
\end{gather*}
Si $t \in \bb{R}$, escribimos $L_t$ a la recta de $H$ correspondiente a dicho valor de $t$, de forma que $L_0=L$. Consideramos $\alpha:I \to \bb{R}^2$ curva regular, un intervalo $I$ con $0\in I$ de forma que $\alpha(0)=p$. Nos planteamos cómo escribir en función de $\alpha$ el hecho de que $C\transv_pL$. Sabemos que $T_pC \neq L$ lo cual podemos traducir como 
\begin{gather*}
    \langle \alpha'(0),n \rangle \neq 0
\end{gather*}
Definimos $F: I \times \bb{R} \to \bb{R}$ de forma que 
\begin{gather*}
    F(s,t) = \langle \alpha(s), n \rangle + c + t
\end{gather*}
y es diferenciable. Es claro también que
\begin{gather*}
    F(0,0) = \langle \alpha(0), n \rangle + c + 0 = \langle p,n \rangle + c = 0\\[0.8em]
    \frac{\partial F}{\partial s} (s,t) = \langle \alpha'(s), n \rangle \quad \Rightarrow \quad
    \frac{\partial F}{\partial s}(0,t) = \langle \alpha'(0), n \rangle \neq 0
\end{gather*}
y por continuidad de $\alpha'$ existirá un entorno de $0$ suficientemente pequeño en el que se tiene que $\frac{\partial F}{\partial s}(s,t) = \langle \alpha'(s), n \rangle \neq 0$. Por tanto podemos considerar el intervalo $I$ lo suficientemente pequeño como para que $\langle \alpha'(s), n \rangle\neq 0$ para cualquier $s\in I$ lo que nos dice que 
\begin{gather*}
    \frac{\partial F}{\partial s}(s,t) = \langle \alpha'(s), n \rangle \neq 0 \qquad \forall (s,t) \in I\times \bb{R}
\end{gather*}
Estamos en condiciones de aplicar el Teorema de la Función Implícita a $F$ en $(0,0)$. Este nos dirá que existe un $W$ entorno abierto de $(0,0)$ en $I \times \bb{R}$ de forma que $W=(-\delta , \delta) \times (-\veps, \veps)$ y tal que 
\begin{align*}
    W\cap F^{-1}(\{0\}) &= \text{ grafo de una }f:(-\veps, \veps) \to \bb{R}\\
    &= \{(s,t)\in (-\delta , \delta) \times (-\veps, \veps)\ : \ (s,t) = (f(t), t) \}
\end{align*}
Tenemos que 
\begin{gather*}
    F^{-1}(\{0\}) = \{(s,t)\in I \times \bb{R} \ : \ \langle \alpha(s), n \rangle + c + t = 0\}
\end{gather*}
por lo que $W \cap F^{-1}(\{0\})$ serán los puntos con $|s| < \delta$, $|t|<\veps$ escritos de la forma $\alpha((-\delta, \delta))\cap H_\veps$ donde $H_\veps$ es una banda de $H$ de radio $\veps$. Además
\begin{gather*}
    F(f(t), t) = 0 \qquad \forall |t|<\veps
\end{gather*}
Derivamos esta expresión respecto de $t$ y nos queda
\begin{align*}
    \frac{\partial F}{\partial s}(f(t), t)f'(t) + \frac{\partial F}{\partial t}(f(t), t) = 0\quad
    &\Rightarrow \quad
    \langle \alpha'(f(t)), n \rangle \cdot f'(t) + 1 = 0\quad
    \Rightarrow\\
    &\Rightarrow \quad
    f'(t) = - \frac{1}{\langle \alpha'(f(t)), n \rangle}
\end{align*}
Por lo que $f$ es monótona y biyectiva sobre la imagen. Llegamos a que 
para cada $t\in (-\veps, \veps)$ existe un único $p_t\in \alpha((-\delta, \delta))$ de forma que $\alpha((-\delta, \delta))\cap L_t = \{p_t\}$. Además, sabemos que $\alpha((-\delta, \delta)) \transv_{p_t} L_t$ y $\alpha((-\delta, \delta))$ pasa en $p_t$ de un semiplano determinado por $L_t$ al otro. Finalmente tenemos que
\begin{gather*}
    V=\alpha((-\delta, \delta)) \text{ es un entorno abierto de $p$ en $C$}
\end{gather*}

\subsection{Rectas tangentes y transversas a una curva}

\begin{lema}\label{lema:transv}
    Sea $C$ una curva simple, $L$ una recta (o semirrecta) de $\bb{R}^2$, $p\in C\cap L$ tal que $C \transv_p L$. Entonces existe un entorno abierto $V$ de $p$ en $C$ y $\exists \veps >0$ tales que 
    \begin{gather*}
        \forall |t| < \veps \text{ se tiene que } V\cap L_t = \{p_t\}
    \end{gather*}
    Además, $V \transv_{p_t}L_t$ y $V$ pasa en $p_t$ de un semiplano determinado por $L_t$ al otro.
\end{lema}

\begin{definicion}\label{def:tang}
    Sea $C\subset \bb{R}^2$ una curva simple y $L$ una recta diremos que $L$ es \textbf{tangente} a $C$ cuando lo sea en algún punto\footnote{aunque haya más puntos de corte, con que haya uno tangente ya diremos que estas curvas son tangentes} y escribiremos $L \| C$. Equivalentemente $L$ será tangente a $C$ si
    \begin{gather*}
        \exists p\in L\cap C \text{ tal que } L \|_p C
    \end{gather*}
\end{definicion}

\begin{definicion}\label{def:transv}
    Sea $C\subset \bb{R}^2$ una curva simple y $L$ una recta diremos que $L$ es \textbf{transversa} a $C$ cuando lo sea en todo punto de la intersección y escribiremos $L \transv C$. Equivalentemente $L$ será transversa a $C$ si
    \begin{gather*}
        \forall p\in L\cap C \text{ se tiene que } L \transv_p C
    \end{gather*}
\end{definicion}

\begin{observacion}
    Comenzamos viendo una serie de consecuencias que podemos extraer del lema anterior (Lema~\ref{lema:transv}):
    \begin{enumerate}
        \item\label{cons:1} Dada $C\subset \bb{R}^2$ una curva simple, $L$ una recta, $p\in L \cap C $ con $L \transv_p C$, entonces existe $V$ un entorno abierto de $p$ en $C$ tal que 
        \begin{gather*}
            V \cap L = \{p\}
        \end{gather*}
        \begin{proof}
            Sabemos por el Lema~\ref{lema:transv} que $\exists V$ entorno abierto de $p$ en $C$ y $\exists \veps>0$ de forma que 
            \begin{gather*}
                V \cap L_t = \{p_t\}
            \end{gather*}
            Sabemos que $|0|<\veps$ por lo que $V\cap L_0 = \{p_0\}$ pero $L_0=L$, es decir, $V\cap L=\{p_0\}$. Sabemos que $p\in V\cap L$ luego necesariamente $p_0=p$ y lo tenemos.
        \end{proof}
        \item\label{cons:2} Dada $C\subset \bb{R}^2$ una curva simple, $L$ una recta y $p\in L\cap C$ con $L\transv_pC$. Entonces existe un entorno $U_p$ abierto de $p$ en $\bb{R}^2$ tal que 
        \begin{gather*}
            U_p \cap (L\cap C) = \{p\}
        \end{gather*}
        \begin{proof}
            Por la consecuencia~\ref{cons:1} sabemos que $\exists V$ entorno abierto de $p$ en $C$ con $V\cap L = \{p\}$. Por ser $V$ entorno abierto relativo a $C$ sabemos que existe un $U_p$ entorno abierto de $p$ en $\bb{R}^2$ con $V=U_p\cap C$ y sustituyendo en lo que ya sabemos tenemos
            \begin{gather*}
                V\cap L =(U_p \cap C)\cap L = U_p \cap (C\cap L) = \{p\}
            \end{gather*}
        \end{proof}

        \item\label{cons:3} Tenemos $C\subset \bb{R}^2$ una curva simple, $L$ una recta y vamos a suponer que $L \transv C$. Entonces $L\cap C$ es discreto (en $\bb{R}^2$, en $L$ y en $C$).
        \begin{proof}
            Sea $p\in L\cap C$, sabemos que $L\transv_p C$. Por la consecuencia~\ref{cons:2} $\exists U_p$ entorno abierto de $p$ en $\bb{R}^2$ tal que 
            \begin{gather*}
                U_p \cap (L\cap C) = \{p\}
            \end{gather*}
            por lo que $\{p\}$ es abierto en $L\cap C$, lo cual es equivalente a decir que $L\cap C$ es discreto.
        \end{proof}
        \item\label{cons:4} Sea $C\subset \bb{R}^2$ una curva simple compacta, $L$ una recta con $L\transv C$. Entonces $L\cap C$ es discreto (por la consecuencia~\ref{cons:3}) y cerrado en $C$. Como además $C$ es compacto, $L\cap C$ será compacto. Por tanto $L\cap C$ es finito y podemos hablar de
        \begin{gather*}
            \# (L\cap C)\in \bb{N}
        \end{gather*}
    \end{enumerate}
\end{observacion}

\begin{observacion}
    Nos planteamos ahora si dada $C$ existen $L$ con $L\transv C$, si existen $L$ con $L\|C$ y cómo se relacionan. Consideramos así $C$ una curva simple con $C=\alpha(I)$ siendo $\alpha: I \to \bb{R}^2$ p.p.a. y $p\in \bb{R}^2 \setminus C$ y $v\in S^1\subset \bb{R}^2 \cong \bb{C}$. A la semirrecta que sale de $p$ en dirección $v$ la representamos por $L_{p,v}$. Con esta notación queremos saber cuándo $L_{p,v} \| C$ lo cual por definición (Definición~\ref{def:tang}) sucederá si
    \begin{gather*}
        \exists s \in I \text{ de forma que } L_{p,v} \|_{\alpha(s)} C \sii \left\{ 
            \begin{array}{c l}
                (1) & \alpha(s)\in L_{p,v}\\
                (2) & \alpha'(s)\ \| \ v
            \end{array}
        \right.
    \end{gather*}
    Veamos cada uno de estos puntos por separado
    \begin{enumerate}[label=(\arabic*)]
        \item $\alpha(s) \in L_{p,v} \sii \alpha(s) = p + \lambda v$ con $\lambda > 0$ (ya que $p\notin C$) lo cual sucederá si y solo si 
        \begin{gather*}
            \alpha(s) - p = \lambda v,\quad \lambda > 0
        \end{gather*}
        Tomando módulo en la expresión anterior obtenemos
        \begin{gather*}
            |\alpha(s) - p| = \lambda |v| = \lambda \sii \alpha(s) - p = |\alpha(s) - p |v \sii v = \frac{\alpha(s)-p}{|\alpha(s)-p|}
        \end{gather*}
        Por lo que concluimos que 
        \begin{gather*}
            \alpha(s) \in L_{p,v} \sii v = \frac{\alpha(s)-p}{|\alpha(s)-p|} := f_p(s)
        \end{gather*}
        Tenemos que $f_p$ es una función $f_p:I \to S^1 \subset \bb{R}^2 \equiv \bb{C}$ y
        \begin{gather*}
            \alpha(s) \in L_{p,v} \sii v = f_p(s)
        \end{gather*}

        \item Tenemos que 
        \begin{gather*}
            \alpha'(s) \ \| \ v \sii \alpha'(s)=\lambda v = \lambda \frac{\alpha(s) - p}{|\alpha(s)-p|} 
        \end{gather*}
        Si multiplicamos ahora escalarmente por $\alpha(s) - p$ obtenemos
        \begin{gather*}
            \langle \alpha'(s), \alpha(s) - p \rangle = \lambda \frac{|\alpha(s)-p|^2}{|\alpha(s)-p|} = \lambda |\alpha(s)-p| \quad \Rightarrow \quad \lambda = \frac{\langle \alpha'(s), \alpha(s) - p \rangle}{|\alpha(s) - p|}
        \end{gather*}
        Así, $\alpha'(s) \ \| \ v$ si y solo si 
        \begin{gather*}
            \alpha'(s) = \frac{\langle \alpha'(s), \alpha(s) - p\rangle}{|\alpha(s) - p |^2}(\alpha(s)-p) \sii \frac{\alpha'(s)}{|\alpha(s) - p|} - \frac{\langle \alpha'(s), \alpha(s) - p\rangle}{|\alpha(s) - p|}(\alpha(s) -p) = 0
        \end{gather*}
        Si ahora calculamos y desarrollamos $f'_p(s)$ llegamos a que 
        \begin{gather*}
            f_p'(s) = \frac{\alpha'(s)}{|\alpha(s) - p|} - \frac{\langle \alpha'(s), \alpha(s) - p \rangle}{|\alpha(s) - p|^3}(\alpha(s) - p) = 0
        \end{gather*}
    \end{enumerate}
    Juntando lo obtenido para (1) y (2) llegamos a que 
    \begin{gather*}
        L_{p,v} \ \| \ \sii (1) \text{ y } (2) \sii \exists s \in I \text{ tal que (1) } v=f_p(s) \text{ y (2) } f_p'(s)=0 \sii v\in f_p(C_f)
    \end{gather*}
    donde
    \begin{gather*}
        C_{f_p}=\{\text{puntos críticos de $f_p$}\}
    \end{gather*}
    Es decir, $L_{p,v}\ \| \ C$ si y solo si $v\in f_p(C_{f_p})$ lo cual ocurrirá si y solo si $v$ es un valor singular de $f_p$.\\
\end{observacion}
\subsection{Teorema de Sard}
A partir de aquí, para avanzar necesitaremos emplear el siguiente resultado del análisis (que nosotros no vamos a demostrar):

\begin{teo}[Teorema de Sard]\label{teo:sard}
    Sea $O\subset \bb{R}$ un abierto y $f:O\to \bb{R}$ con $f\in C^1(O)$. Entonces $f(C_f)$ tiene medida nula en $\bb{R}$. En particular, $f(C_f)$ tiene interior vacío. Por tanto $\bb{R}\setminus f(C_f)$ es denso en $\bb{R}$ y además 
    \begin{gather*}
        \bb{R}\setminus f(C_f) = VR(f) = \{\text{valores regulares de $f$}\}
    \end{gather*}
\end{teo}

\begin{observacion}
    El hecho de que $\bb{R}\setminus f(C_f)$ sea denso en $\bb{R}$ se deduce de que si $\overset{\circ}{A}=\emptyset$, entonces 
    \begin{gather*}
        \overline{X\setminus A} = X\setminus A = X\setminus \emptyset = X
    \end{gather*}
    Existe además una versión más general del teorema para dimensiones mayores.
\end{observacion}

No obstante, de momento no podemos aplicar directamente el teorema al caso que estábamos estudiando debido a que nuestra función es $f:I \to S^1 \subset \bb{R}^2 \cong \bb{C}$ y, por tanto, el \hyperref[teo:sard]{Teorema de Sard} no se puede aplicar, pues llega a $\bb{C}$ y no a $\bb{R}$.\\

Para solventar este problema trataremos de parametrizar la circunferencia usando números complejos y la función argumento. Consideramos así $z\in \bb{C}$ con $|z|=1$ y podremos escribir $z = e^{i\theta}$ donde $\theta = arg(z)$. Sin embargo, la función argumento, $arg(\cdot)$ no es continua en toda la circunferencia. Por este motivo vamos a trabajar con dos ``ramas'' del argumento definiendo
\begin{align*}
    arg^1 &: S^1 \setminus \{(1,0)\} \to (0,2\pi)\\
    arg^2 &: S^1 \setminus \{(-1,0)\} \to (-\pi,\pi)
\end{align*}
De esta forma se verificará $arg^2 = arg^1 - \pi$. Además, 
\begin{gather*}
    z(s) = e^{i\theta(s)}\\
    z'(s) = i\theta'(s)e^{i\theta(s)} = i\theta'(s)z(s) = i(arg^{1,2}(z))'(s) \cdot z(s)
\end{gather*}
de donde aplicando que $|z| = 1$ y $z(s) \neq 0$ tendremos que 
\begin{gather*}
    \boxed{\theta'(s) = 0 \sii z'(s) = 0}
\end{gather*}
Con esto podemos definir dos nuevas funciones reales:
\begin{align*}
    h^1_p(s) &= arg^1 (f_p(s)) \text{ definida para } f_p(s) \neq (1,0)\\
    h^2_p(s) &= arg^2 (f_p(s)) \text{ definida para } f_p(s) \neq (-1,0)
\end{align*}
Tomando un $\theta\in [0,2\pi[$ tendremos que la dirección que marca el argumento será 
\begin{gather*}
    v_\theta = (\cos(\theta), \sen(\theta)) = e^{i\theta}
\end{gather*}
y cualquier dirección (vector de $S^1$) en la que ``lancemos'' la semirrecta se podrá representar con un ángulo $\theta$. De esta forma tendremos que $L_{p,v} \ \| \ C$ si y solo si 
\begin{gather*}
    v=v_\theta \in f_p(C_{f_p}) \sii \theta \in h_p^1(C_{h_p^1}) \cup h_p^2(C_{h_p^2})
\end{gather*}
donde ambos conjuntos de la unión serán de medida nula ya que tanto a $h_p^1$ como a $h_p^2$ sí le podremos aplicar el \hyperref[teo:sard]{Teorema de Sard}.\\

Vamos a desglosar este último paso. Recordemos que la notación compleja era para expresar $z(s)=f_p(s)$ y que nos interesaban los valores singulares de $f_p$, es decir, los puntos $s$ que verifican $f_p'(s)=0$. Por tano, lo que queríamos ver era cuándo $z'(s)=0$ y hemos probado que es equivalente a cuándo $\theta'(s)=0$. Además, según el valor de $f_p(s)$ tendremos 
\begin{gather*}
    \theta'(s) = (arg^1(z))'(s)  \text{ o bien } \theta'(s) = (arg^2(z))'(s)
\end{gather*}
De esta forma, por la definición de las funciones $h_p^1$ y $h_p^2$ tendremos que 
\begin{gather*}
    \theta'(s) = 0 \sii
    \left\{
        \begin{array}{l}
            (arg^1(z))'(s) = 0 \sii \theta(s)\in h_p^1(C_{h_p^1})\\[1.25em]
            (arg^2(z))'(s) = 0 \sii \theta(s)\in h_p^2(C_{h_p^2})
        \end{array}
    \right.
\end{gather*}
y con esto tenemos justificada la última expresión, equivalente con la unión.\\

En conclusión, como la unión de dos conjuntos de medida nula tendrá también mediad nula, entonces
\begin{gather*}
    \{\theta\in[0,2\pi) \ : \ L_{p,v_\theta} \ \| \ C\} \text{ tiene medida nula } \\
    \Rightarrow A = \{\theta \in [0,2\pi) \ : \ L_{p,v_\theta} \transv C\} \text{ es denso en $[0,2\pi)$}
\end{gather*}
Todo esto lo podemos resumir en una nueva consecuencia:

\begin{enumerate}[start=5]
    \item\label{cons:5} Sea $C \subset \bb{R}^2$ una curva simple y $p\in \bb{R}^2 \setminus C$, entonces existen ``muchos'' vectores $v\in S^1$ (de hecho un conjunto denso) que hacen que $L_{p,v}\transv C$.
\end{enumerate}

Ahora nos vamos a centrar en tomar una curva simple $C\subset \bb{R}^2$ compacta y en separar exactaemente en dos regiones $\bb{R}^2 \setminus C$. \\

Consideramos $C\subset \bb{R}^2$ una curva simple, conexa y compacta. Definimos:
\begin{align*}
    \Omega_0 &= \{p\in \bb{R}^2 \setminus C \ : \ \exists v \in S^1 \text{ con } L_{p,v}\transv C \text{ y } \#(L_{p,v} \cap C) \text{ par}\} \subset \bb{R}^2 \setminus C\notag\\
    \Omega_1 &= \{p\in \bb{R}^2 \setminus C \ : \ \exists v \in S^1 \text{ con } L_{p,v}\transv C \text{ y } \#(L_{p,v} \cap C) \text{ impar}\} \subset \bb{R}^2 \setminus C
\end{align*}
Tenemos que $\Omega_0 \cup \Omega_1 \subset \bb{R}^2 \setminus C$ y nos planteamos si se dará la otra inclusión (y por tanto la igualdad). Sea $p\in \bb{R}^2 \setminus C$, por la consecuencia~\ref{cons:5} tenemos que existe un $v\in S^1$ tal que $L_{p,v} \transv C$. Sabemos por la consecuencia~\ref{cons:4} que $\#(L_{p,v}\cap C) \in \bb{N}$. Si este cardinal es par tendremos $p\in \Omega_0$ y en otro caso $p\in \Omega_1$. En cualquier caso
\begin{gather*}
    p\in \Omega_0\cup\Omega_1
\end{gather*}
Lo que nos da la otra inclusión y prueba la igualdad
\begin{gather*}
    \boxed{\Omega_0\cup\Omega_1 = \bb{R}^2 \setminus C}
\end{gather*}
Recordemos que $C$ era compacta, luego existe un $R>0$ tal que $C\subset D(R)$. Elegimos un $p\notin D(R)$ y definimos $v=\frac{p}{|p|}\in S^1$. Dicho $v$ es un vector que apunta a $p$ hacia el infinito desde el origen. Por ello es claro que $L_{p,v}\cap D(R) = \emptyset$ y por tanto
\begin{gather*}
    L_{p,v}\cap C \subset L_{p,v}\cap D(R) = \emptyset \Rightarrow L_{p,v}\cap C = \emptyset
\end{gather*}
y concluimos que $L_{p,v} \transv C$ y que $\#(L_{p,v}\cap C)=0$, que al ser par nos dirá que $p\in \Omega_0$ por lo que 
\begin{gather*}
    \bb{R}^2 \setminus D(R)\subset \Omega_0
\end{gather*}
llegando finalmente a que 
\begin{gather*}
    \boxed{\Omega_0 \text{ es no acotado}}
\end{gather*}
En particular, $\boxed{\Omega_0\neq \emptyset}$.\\

Veamos ahora que $\Omega_1$ también es no vacío. Para ello de nuevo $p\notin D(R)$. Como $C$ es compacta entonces existe un $q\in C$ tal que 
\begin{gather*}
    |p-q| = dist(p,C) >0
\end{gather*}
ya que $p\notin D(R) \supset C$. Esto nos dice que en la semirrecta que une $q$ con $p$ no puede haber más puntos de $C$, pues de haberlos contradiría que 
\begin{gather*}
    |p-q| = dist(p,C)
\end{gather*}

Además, necesariamente $p-q \perp T_qC$. Consideramos ahora 
\begin{gather*}
    v = \frac{p-q}{|p-q|}
\end{gather*}
que está bien definido, ya que $p\neq q$ ($p\notin C$ y $q\in C$).

Tenemos que 
\begin{gather*}
    L_{q,v} \cap C = \{q\}
\end{gather*}
y nos planteamos si $L_{q,v} \transv C$. La respuesta es afirmativa ya que la ortogonalidad es el caso ``extremo'' de transversalidad, en otras palabras, no es tangencia. Pero $q\notin \Omega_1$ ya que $q\in C$, por lo que buscaremos alguna solución.\\

Por la consecuencia~\ref{cons:2} sabemos que existe un $V$ entorno abierto de $q$ en $\bb{R}^2$ de forma que $V\cap C = \{q\}$. Como $V$ es entorno abierto de $q$ tendremos que 
\begin{gather*}
    \exists \delta >0 \ : \ B(q, \delta)\subset V \Rightarrow B(q, \delta)\cap C = \{q\}
\end{gather*}
De esta forma, tomando $0<\veps<\delta$ podemos considerar $r=q - \veps v \in B(q, \delta)\subset V$ y tendremos
\begin{gather*}
    L_{r,v}\cap C = ([r,q]\cup L_{q,v})\cap C = \{q\}
\end{gather*}
Además, $L_{r,v}\transv C$ claramente. Igualmente $\#(L_{r,v}\cap C)=1$
que es impar, luego $r\in \Omega_1$ lo que nos lleva que 
\begin{gather*}
    \boxed{\Omega_1 \neq \emptyset}
\end{gather*}

\subsection{Teorema de la Curva de Jordan}

Ya habíamos estudiado que $\bb{R}^2 \setminus C = \Omega_0 \cup \Omega_1$  y que $\Omega_0\neq \emptyset$ y $\Omega_1 \neq \emptyset$ . Además vimos que $\Omega_0$ no está acotado. Veamos ahora que $\Omega_0$ y $\Omega_1$ son disjuntos. Para ello suponemos que no, es decir que $\exists p\in \Omega_0\cap \Omega_1$. Entonces tendremos que
\begin{gather*}
    \exists v_0\in S^1 \text{ de forma que } L_{p,v_0}\transv C \text{ y } \#(L_{p,v_0} \cap C) \text{ par }\\
    \exists v_1\in S^1 \text{ de forma que } L_{p,v_1}\transv C \text{ y } \#(L_{p,v_1} \cap C) \text{ impar }
\end{gather*}
Es trivial que $v_0\neq v_1$ ya que un número no puede ser simultáneamente par e impar. Consideramos entonces\footnote{Lo llamamos $A$ porque es un ángulo}
\begin{gather*}
    A = L_{p,v_0} \cup L_{p, v_1}
\end{gather*}
que será un ángulo no trivial por lo que divide a $\bb{R}^2$ en dos c.c. $A_0$ y $A_1$. Además tenemos que
\begin{align*}
    A\cap C &= (L_{p, v_0} \cup L_{p, v_1}) \cap C = (L_{p,v_0} \cap C) \dot{\cup} (L_{p, v_1}\cap C)\\
    \#(A\cap C) &= \# (L_{p,v_0} \cap C) + \#(L_{p,v_1} \cap C) \text{ impar}
\end{align*}
Vamos a considerar el conjunto de puntos de intersección escribiendo $A\cap C = \{p_1, p_2, ..., p_k\}$. En estos puntos, ordenados según se recorre $C$ y en los que $C$ cruza de un lado de $A$ al otro por ser transversa, tendremos
\begin{itemize}[label=-, noitemsep]
    \item En $p_1$ se tiene que $C$ pasa de $A_1$ a $A_0$.
    \item En $p_2$ se tiene que $C$ pasa de $A_0$ a $A_1$.
    \item En $p_3$ se tiene que $C$ pasa de $A_1$ a $A_0$.
    \item ...
    \item En $p_k$ $C$ pasa de $A_0$ a $A_1$ o de $A_1$ a $A_0$, según la paridad de $k$.
\end{itemize}
Podemos concluir que en los $p_i$ con $i$ impar se pasa de $A_1$ a $A_0$ y los $p_i$ con $i$ par se pasa de $A_0$ a $A_1$. Como $C$ es una curva cerrada, después de $p_k$ debe volver a estar en el mismo lado en el que estaba antes de $p_1$ (es decir, en $A_1$), lo que nos dice que $k$ tiene que ser par. Sin embargo teníamos que $\#(A\cap C)$ era impar llegando a contradicción. Por tanto $\Omega_0\cap \Omega_1 = \emptyset$.\\

Podemos afirmar que $(\Omega_0,\Omega_1)$ es una partición no trivial de $\bb{R}^2 \setminus C$. Ahora observamos lo siguiente (donde $D(R)$ es un disco de radio $R>0$ que contiene a $C$):
\begin{gather*}
    \bb{R}^2 \setminus D(R) \subset \Omega_0\\
    \bb{R}^2 \setminus \Omega_0 \subset D(R)\\
    \bb{R}^2 \setminus \Omega_0 = \Omega_1 \cup C
\end{gather*}
por lo que $\Omega_1$ es acotado. Queremos ver ahora que $\Omega_0$ y $\Omega_1$ son abiertos en $\bb{R}^2 \setminus C$ (y por tanto en $\bb{R}^2$). En este caso tendremos una partición no trivial de abiertos lo que nos dirá que $\bb{R}^2 \setminus C$ no es conexo teniendo exactamente 2 componentes conexas (ya vimos que a lo sumo tenía 2, Proposición~\ref{prop:max2}).\\

Veamos que $\Omega_0$ es abierto. Tomamos $p\in \Omega_0$. Entonces existe una dirección $v\in S^1$ de forma que $L_{p,v} \transv C$ con $\#(L_{p,v} \cap C)$ par. Podemos escribir
\begin{gather*}
    L_{p,v}\cap C = \{p_1,p_2,...,p_{2k}\}
\end{gather*}
Por la definición de transversalidad (Definición~\ref{def:transv}) tenemos que 
\begin{gather*}
    L_{p,v}\transv C \Rightarrow L_{p,v}\transv _{p_i} C \text{ con } i=1,2,...,2k
\end{gather*}
Por el lema anterior (Lema~\ref{lema:transv}) tenemos que existe un entorno abierto $V_i$ de $p_i$ en $C$ y existe un $\veps_i>0$ tal que si $|t|< \veps_i$ se verifica
\begin{gather*}
    L_t\transv V_i\\
    \#(L_t \cap V_i) = 1
\end{gather*}
De esta forma podemos suponer sin pérdida de generalidad que $V_i\cap V_j= \emptyset$ si $i\neq j$ ya que si no fueran disjuntos podemos tomar los $V_i$ más pequeños (y los $\veps_i$ correspondientes) hasta que lo sean.\\

Vamos a considerar entonces $\veps = \min\{\veps_1, \veps_2, ... , \veps_{2k}\}>0$ y al tomar $|t|<\veps$ tendremos
\begin{gather*}
    L_t \transv V_i\\
    \#(L_t \cap V_i) = 1
\end{gather*}
Vamos a considerar $V_1\cup V_2 \cup ... \cup V_{2k}$ que será abierto en $C$ por ser unión de abiertos. Consideramos también su complementario
\begin{gather*}
    K = C\setminus (V_1\cup V_2 \cup ... \cup V_{2k}) \text{ que será cerrado en $C$}
\end{gather*}
Sabemos que $C$ es compacto por lo que $K$ (cerrado dentro de un compacto) será compacto también. Estudiamos ahora
\begin{gather*}
    r = dist(L_{p,v},K) > 0
\end{gather*}
Aquí $r>0$ porque $L_{p,v}$ es cerrado, $K$ es compacto y $L_{p,v}\cap K=\emptyset$. De esta forma existirá un $\delta>0$ con $B_\delta(p) \subset \bb{R}^2 \setminus C$ e imponemos también $\delta < min\{\veps, r\}$. Al obligar esto tenemos que si cogemos un punto $q\in B_\delta(p)$ tendremos que $L_{q,v} \subset L_t$ para cierto $|t| < \veps$ (de hecho $|t|\leq |q-p| < \delta$). Entonces, reduciendo $\delta$ si fuera necesario para que los puntos de corte queden sobre la semirrecta (es decir, ``después'' de $q$), $L_{q,v} \transv C$ y $\#(L_{q,v} \cap V_i) = 1$. Además $L_{q,v} \cap K = \emptyset$ ya que todo punto de $L_{q,v}$ está a distancia menor que $\delta<r$ de $L_{p,v}$. Ya tenemos todo lo que necesitamos para concluir.\\

Dado $q\in B_\delta(p)$ elegimos el mismo $v$ original y tendremos que 
\begin{align*}
    L_{q,v}\cap C &= (L_{q,v} \cap (V_1\cup V_2 \cup ... \cup V_{2k})) \cup (L_{q,v}\cap K) =\\
    &= (L_{q,v} \cap V_1) \dot{\cup} (L_{q,v} \cap V_2) \dot{\cup} ... \dot{\cup} (L_{q,v} \cap V_{2k})
\end{align*}
De esta forma 
\begin{gather*}
    \#(L_{q,v}\cap C) = \#(L_{q,v} \cap V_1) + \#(L_{q,v} \cap V_2) + ... + \#(L_{q,v} \cap V_{2k}) = 1 + 1 + ... + 1 = 2k
\end{gather*}
y además $L_{q,v}\transv C$ por lo que $q\in \Omega_0$. Por la arbitrariedad de $q$ tendremos que $B_\delta(p) \subset \Omega_0$ lo que nos dice que $\Omega_0$ es abierto. Análogamente $\Omega_1$ también lo será. Además, si $\Omega_0$ no fuera conexo tendríamos que $\Omega_0$ tiene al menos 2 c.c. por lo que 
\begin{gather*}
    \bb{R}^2 \setminus C = \Omega_0 \cup \Omega_1 \text{ tendría al menos 3 c.c.}
\end{gather*}
Por lo que $\Omega_0$ es conexo (y análogamente $\Omega_1$). Tenemos finalmente que 
\begin{gather*}
    (\Omega_0, \Omega_1)
\end{gather*}
es una partición no trivial en abiertos conexos de $\bb{R}^2 \setminus C$ lo que concluye en el siguiente teorema.

\begin{teo}[Teorema de la Curva (Diferenciable) de Jordan]\label{teo:curva_jordan}
    Si $C\subset \bb{R}^2$ es una curva simple compacta y conexa, entonces $\bb{R}^2 \setminus C$ tiene exactamente 2 componentes conexas cuya frontera común es $C$ (esto es válido imponiendo solo que $C$ es conexa y cerrada). De dichas componentes conexas, solo una es acotada (si $C$ es compacta).
\end{teo}

\begin{definicion}
    La componente conexa de $\bb{R}^2\setminus C$ que es acotada se llama el \textbf{dominio interior determinado por $C$} y se representa por $\Omega$, cuya definición recordamos que era
    \begin{gather*}
        \Omega = \{p\in \bb{R}^2 \setminus C \ : \ \exists v \in S^1 \text{ con } L_{p,v}\transv C \text{ y } \#(L_{p,v} \cap C) \text{ impar}\}
    \end{gather*} 
\end{definicion}

\begin{definicion}

    \end{gather*}
    que llamaremos la \textbf{bola métrica} en torno a $C$.
\end{definicion}

    \end{gather*}
    que ya no es homeomorfo a los casos anteriores.

\begin{observacion}
    Sabemos que $C\subset B_\delta(C)$ y por teoría de espacios métricos sabemos que $B_\delta(C)$ será un entorno abierto de $C$.\\
    
    Dado un $p\in C$ representamos por $N_\delta(p)$ el segmento abierto de la recta normal a $C$ en $p$ que tiene centro $p$ y radio $\delta$. Sea 
    \begin{gather*}
        N_\delta(C) = \bigcup_{p\in C} N_\delta(p)
    \end{gather*}
    y tenemos que $C\subset N_\delta(C)$. Veamos que los conjuntos $B_\delta(C)$ y $N_\delta(C)$ son iguales.
    \begin{proof}
        Para ver que son iguales probaremos la doble inclusión
        \begin{itemize}
            \item[$\subseteq$ )] Consideramos $p\in B_\delta(C)$ y como $C$ es compacto tendremos que 
            \begin{gather*}
                \exists q \in C \quad \text{ tal que } \quad |p-q| = dist(p,C) < \delta
            \end{gather*}
            Como $q$ minimiza la distancia a $p$ sobre $C$, derivando $|p-\alpha(s)|^2$ en $s_0$ (con $\alpha(s_0)=q$) se obtiene $\langle \alpha(s_0)-p, \alpha'(s_0)\rangle = 0$, es decir, $p-q$ es ortogonal a $T_qC$, lo que nos dice que $p$ está en la recta normal a $C$ que pasa por $q$ con $|p-q| < \delta$ lo que nos dice que $p\in N_\delta(q)\subset N_\delta(C)$. Por tanto $p\in N_\delta(C)$.

            \item[$\supseteq$ )] Tomamos $p\in N_\delta(C)$. Por la definición de $N_\delta(C)$ tendremos que 
            \begin{gather*}
                \exists q \in C \quad \text{ tal que } p\in N_\delta(q)
            \end{gather*}
            Además, sabemos que $dist(p,C) \leq |p-q| < \delta$ lo que nos confirma que $p\in B_\delta(C)$.
        \end{itemize}
        Finalmente podemos afirmar que $\forall \delta > 0$ se tiene que 
        \begin{gather*}
            B_\delta(C) = N_\delta(C)
        \end{gather*}
    \end{proof}
    Una consecuencia inmediata es que $N_\delta(C)$ es un entorno abierto de $C$. 
\end{observacion}

\subsection{Existencia de Entornos Tubulares}
Si $C$ es una curva simple, compacta y conexa sabemos que 
\begin{gather*}
    \exists \alpha : \bb{R} \longrightarrow C \quad \text{ p.p.a. y $L$-periódica}
\end{gather*}
tal que $\alpha_{|_{[0,L)}}$ es inyectiva y $tr(\alpha)=C$. Vamos a definir 
\Func{F}{\bb{R}^2}{\bb{R}^2}{(s,t)}{\alpha(s) + tN(s)}
y es claro que $F$ es diferenciable como aplicación de $\bb{R}^2$ en $\bb{R}^2$. Estudiamos ahora esta función.
Tenemos que 
\begin{gather*}
    F(\bb{R}\times \{0\}) = C\\
    F(\{s\} \times \bb{R}) = \text{recta normal a $C$ en $\alpha(s)$}
\end{gather*}
A la función $F$ se le llama \textbf{exponencial} ya que si $C=S^1$ nos recuerda al sistema de coordenadas polares. De esta forma $F$ controla todas las rectas normales a la curva $C$ en cualquier punto $p\in C$. Estudiamos ahora sus derivadas parciales
\begin{align*}
    \frac{\partial F}{\partial s}(s,t) &= \alpha'(s) + tN'(s) = (1-t\kappa(s)) T(s)\\[1.5em]
    \frac{\partial F}{\partial t}(s,t) &= N(s)
\end{align*}
Y aplicándolo a $(s,0)$
\begin{align*}
    \left.
        \begin{array}{c}
            \dfrac{\partial F}{\partial s}(s,0) = T(s)\\[1.5em]
            \dfrac{\partial F}{\partial t}(s,0) = N(s)
        \end{array}
    \right\} \Rightarrow Jac(F)(s,0) \text{ es regular}
\end{align*}
por lo que son linealmente independientes lo que nos permite aplicar el Teorema de la Función Inversa en $(s,0)$. Este nos dirá que 
\begin{gather*}
    \exists \delta_s >0 , \ \ \exists \veps_s > 0 \ \ \text{ tal que } \ \ F_{|_{(s-\delta_s, s+\delta_s) \times (-\veps_s, \veps_s)}} : (s-\delta_s, s+\delta_s) \times (-\veps_s, \veps_s) \longrightarrow \ \ \bigcup_{\mathclap{r\in(s-\delta_s, s+\delta_s)}}\ N_{\veps_s}(\alpha(r))\\
    \exists V_{\alpha(s)} = \alpha(s-\delta_s, s+\delta_s) \text{ entorno abierto de $\alpha(s)$} \\[1.5em]
    \exists \veps_s > 0 \ \ : \ \ N_{\veps_s}(r)\cap N_{\veps_s}(q) = \emptyset \text{ para } r\neq q \in V_{\alpha(s)}
\end{gather*}
Resumiendo esta información en lo que nos interesa hemos probado que 
\begin{gather*}
    \forall p \in C \quad \exists \veps_p > 0, \quad \exists V_p \text{ entorno abierto de $p$ en $C$}
\end{gather*}
de forma que si tomamos $q,r\in V_p$ con $q\neq r$ se tiene que $N_{\veps_p}(q)\cap N_{\veps_p}(r)=\emptyset$.\\

Vamos a estudiar ahora la función $F$ en $[0,L]$ lo que nos permitirá estudiar la curva entera (recordemos que era $L$-periódica). Ya sabemos que 
\begin{gather*}
    F_{|_{(s-\delta_s, s+\delta_s) \times (-\veps_s, \veps_s)}} \text{ es un difeomorfismo}
\end{gather*}
Y podemos considerar $(s-\delta_s, s+\delta_s)$ variando $s\in [0,L]$ que será un recubrimiento por abiertos de $[0,L]$, que es compacto. Por la definición topológica de compacidad tenemos que existe un subrecubrimiento finito, es decir,
\begin{gather*}
    \exists s_1,...,s_k\in [0,L] \ \ : \ \ (s_i - \delta_{s_i}, s_i + \delta_{s_i}) \text{ con } i=1,...,k \ \text{ recubre a } [0,L]
\end{gather*}
Vamos a considerar ahora los $\veps$ asociados a esos puntos que notaremos como $\veps_{s_1}, ..., \veps_{s_k}$ todos ellos positivos. Como tenemos una cantidad finita podemos tomar
\begin{gather*}
    \veps = \min\{\veps_{s_1}, ..., \veps_{s_k}\} > 0
\end{gather*}
De esta forma, con la altura fijada podemos considerar
\begin{gather*}
    (s_i - \delta_{s_i}, s_i + \delta_{s_i})\times(-\veps, \veps)
\end{gather*}
y al restringir $F$ a cada uno de estos rectángulos seguiremos teniendo que es un difeomorfismo.\\

Consideramos ahora $F_{|_{[0,L]\times(-\veps, \veps)}}$ que sabemos que no puede ser difeomorfismo (debido a que no es inyectiva en los extremos), pero sí que es un difeomorfismo local.
De esta forma, extendiendo el intervalo $[0,L]$ a $\bb{R}$ tendremos que
\begin{gather*}
    F_{|_{\bb{R}\times(-\veps, \veps)}} : \bb{R}\times(-\veps, \veps) \longrightarrow \bigcup_{p\in C} N_\veps(p) = N_\veps(C)
\end{gather*}
es un difeomorfismo local sobreyectivo. Cabe destacar que $F$ no es inyectiva (es $L$-periódica en la variable $s$).

Vamos a seguir estudiando esta función $F$. Tomamos un $p\in C$ y sabemos que hay un entorno abierto suyo $V_p$ en $C$ de forma que si cogemos dos puntos distintos $q,r\in V_p$, tendremos que $N_\veps(q)\cap N_\veps(r) = \emptyset$. Veamos que $\exists 0 < \delta \leq \veps$ de forma que si tomamos $q,r\in C$ distintos tendremos que 
\begin{gather*}
    N_\delta(q) \cap N_\delta(r) = \emptyset
\end{gather*}
es decir, que esta propiedad no funciona solo a nivel local, sino que se verifica para toda la curva tomando una banda lo suficientemente estrecha.
\begin{proof}
    Supongamos que no, es decir que
    \begin{gather*}
        \forall n\in \bb{N} \text{ con } n\geq N \text{ se tiene que } 0<\frac{1}{n}\leq \veps \text{ y } N_{\frac{1}{n}}(p_n)\cap N_{\frac{1}{n}}(q_n) \neq \emptyset
    \end{gather*}
    para ciertos $p_n,q_n\in C$ con $p_n\neq q_n$. Elegimos $r_n\in  N_{\frac{1}{n}}(p_n)\cap N_{\frac{1}{n}}(q_n)$. Como $p_n$ es una sucesión de $C$ que es un compacto, tendremos que existe una parcial convergente de $p_n$ convergente. Análogamente existirá una parcial convergente de $q_n$ y tomando los índices comunes a ambas parciales tendremos que 
    \begin{gather*}
        p_n \longrightarrow p , \qquad q_n \longrightarrow q,\qquad p,q\in C
    \end{gather*}
    Por la desigualdad triangular tendremos que 
    \begin{gather*}
        |p_n - q_n| \leq |p_n - r_n| + |r_n-q_n| \leq \frac{1}{n} + \frac{1}{n} = \frac{2}{n}
    \end{gather*}
    Tomando límite en esta desigualdad tendremos 
    \begin{gather*}
        |p-q| \leq 0 \Rightarrow p=q
    \end{gather*}
    Consideramos $V_p=V_q$ correspondiente y tendremos que $p_n,q_n\in V_p=V_q$ para un $n$ lo suficientemente grande. De esta forma, por el resultado local anterior tendremos que 
    \begin{gather*}
        N_{\frac{1}{n}}(p_n)\cap N_{\frac{1}{n}}(q_n) = \emptyset
    \end{gather*} 
    pero sabemos que $r_n\in N_{\frac{1}{n}}(p_n)\cap N_{\frac{1}{n}}(q_n)$ y llegamos a la contradicción que buscábamos.
\end{proof}
Concluimos que para una curva simple $C$ simple, conexa y compacta se tiene que $\exists \veps>0$ de forma que 
\begin{gather*}
    F: \bb{R}\times (-\veps, \veps) \longrightarrow N_\veps(C) \qquad \text{ es un difeomorfismo local sobreyectivo}
\end{gather*}
y $\exists 0 < \delta \leq \veps$ de forma que $F$ es inyectiva en $[0,L) \times (-\delta, \delta)$. Podemos considerar $\delta=\veps$ de forma que $F_{|_{[0,L)\times(-\veps, \veps)}}$ es inyectiva. Podemos considerar el siguiente conjunto cociente:
\begin{gather*}
    \xymatrix{
        \bb{R}\times(-\veps, \veps) \ar[r]^{F} \ar[d]_\pi^{R_F} & B_\veps(C) \\
        \frac{\bb{R}\times (-\veps, \veps)}{R_F} & \hspace{-1.4cm} {{} = R/R_\alpha \times (-\veps, \veps)}
    }
\end{gather*}
donde definimos la relación de equivalencia $R_F$ como
\begin{gather*}
    (s,t)R_F (s',t') \sii F(s,t) = F(s',t') %\sii N_\veps(\alpha(s))\cap N_\veps(\alpha(s')) \neq \emptyset
\end{gather*}
Además, es fácil ver que con esta definición
\begin{gather*}
    (s,t)R_F (s',t') \sii \alpha(s)=\alpha'(s'),\ \ t=t' \sii sR_\alpha s',\ \ t=t'
\end{gather*}
De esta forma
\begin{gather*}
    \xymatrix@R=1.2em{
        S^1 \times (-\veps, \veps) \ar@{}[d]|{\rotatebox{-90}{$\cong$}} & \hspace{-0.9cm} \cong \frac{\bb{R}\times(-\veps, \veps)}{R_F} = R/R_\alpha \times (-\veps, \veps)\\
        C\times(-\veps, \veps)
    }
\end{gather*}
y sabemos que $F$ es sobreyectiva, continua y abierta (por ser difeomorfismo local) por lo que $F$ es una identificación y de esta forma $\tilde{F}$ será un homeomorfismo, es decir,
\begin{gather*}
    B_\veps(C) = N_\veps(C) \cong S^1 \times (-\veps, \veps)\cong C \times(-\veps, \veps)
\end{gather*}

\begin{teo}[Teorema de existencia de entornos tubulares]\label{teo:exist_ent_tub}
    Dada $C\subset \bb{R}^2$ una curva simple, conexa y compacta. Entonces $\exists \veps > 0$ tal que $F:\bb{R}\times(-\veps, \veps) \to B_\veps(C)$ es un difeomorfismo local sobreyectivo tal que $F_{|_{[0,L)\times(-\veps, \veps)}}$ es inyectiva, es decir,
    \begin{gather*}
        N_\veps(\alpha(s))\cap N_\veps(\alpha(s')) \neq \emptyset \quad \text{si} \quad s-s'\notin L\bb{Z}
    \end{gather*}
    Además, se tiene que 
    \begin{gather*}
        B_\veps(C) \cong C \times(-\veps, \veps) \cong S^1 \times (-\veps, \veps)
    \end{gather*}
    Por eso, a $B_\veps(C)$ se le llama \textbf{entorno tubular} de $C$ con radio $\veps>0$.
\end{teo}

% Dada una curva $C\subset \bb{R}^2$ simple, compacta y conexa existe un $\veps>0$ tal que la función $F:\bb{R}\times(-\veps, \veps) \to B_\veps(C)$ es un difeomorfismo local, sobreyectiva, su restricción $F_{|_{[0,L)\times(-\veps, \veps)}}$ es inyectiva (en vista del \hyperref[teo:exist_ent_tub]{Teorema de Existencia de Entornos Tubulares}). Además, sabemos que 
% \begin{gather*}
%     B_\veps(C) \cong S^1 \times (-\veps, \veps) \cong C \times (-\veps, \veps)
% \end{gather*}
\begin{ejercicio}
    Sea $C\subset \bb{R}^2$ una curva simple, conexa y compacta y sea $\veps>0$ tal que $B_\veps(C)$ es un entorno tubular. Entonces para cada $0<t<\veps$ vamos a definir
    \begin{gather*}
        C_t = \{p\in \bb{R}^2 \ : \ dist(p,C)=t,\ p\in \Omega\}
    \end{gather*}
    Demostrar que $C_t$ es una curva simple, compacta y conexa. Calcular además su longitud y su curvatura. A $C_t$ lo llamaremos \textbf{curva paralela} a $C$ a distancia $t>0$ en el interior de $C$.\\

    \begin{solucion}
        Podemos reescribir la curva paralela de la siguiente forma
        \begin{gather*}
            C_t = \{p\in \bb{R}^2 \ : \ p=F(s, t) \ \ \forall s \in \bb{R}\}
        \end{gather*}
        y sabemos que $F$ está asociada a un $\alpha:\bb{R} \to \bb{R}^2$ p.p.a $L$-periódica con $\alpha_{[0,L)}$ inyectiva de forma que 
        \begin{gather*}
            F(s,t) = \alpha(s) + tN(s)
        \end{gather*}
        Supongamos sin pérdida de generalidad que $\alpha$ orienta positivamente a $C$. Podemos definir
        \Func{\alpha_t}{\bb{R}}{\bb{R}^2}
             {s}{(F(s,t))}
        y a primera vista podemos ver que $\alpha_t$ es diferenciable por lo que es una curva parametrizada. Queremos ver si es regular, es decir, si su vector tangente es no nulo en cada punto. Calculemos dicho vector:
        \begin{gather*}
            \alpha_t'(s) = \alpha'(s) + tN'(s) = T(s) - t\kappa(s)T(s) = (1-t\kappa(s)) T(s) = \frac{\partial F}{\partial s}(s,t) \neq 0
        \end{gather*}
        donde $(1-t\kappa(s))$ es una función escalar y $T(s)$ es vectorial por lo que esta derivada será una función vectorial. Además, sabemos que es no nulo ya que $F$ es un difeomorfismo local (su derivada parcial no puede ser nula). Sabemos que dentro del entorno tubular se da que 
        \begin{gather*}
            1+ t \kappa(s) \neq 0
        \end{gather*}
        por lo que no cambia de signo. Si $t=0$ está expresión vale 1 por lo que necesariamente por continuidad $1 +t\kappa(s) > 0$. Como $\alpha_t(s) \neq 0$ para todo $s\in \bb{R}$ tendremos que $\alpha_t$ es regular. Además, $\alpha_t$ es por construcción 
        \begin{gather*}
            \alpha_t(s) = \alpha(s) + tN(s) = \alpha(s) + tJ T(s) = \alpha(s) + tJ\alpha'(s)
        \end{gather*}
        y como $\alpha$ es periódica de período $L$ necesariamente $\alpha_t$ será también $L$-periódica\footnote{el mismo período que $\alpha$ original}. Si consideramos la restricción
        \begin{gather*}
            \alpha_{t|_{[0,L)}} = F_{|_{[0,L)\times \{t\}}}
        \end{gather*}
        tenemos que $\alpha_{t|_{[0,L)}}$ es inyectiva. Podemos afirmar hasta el momento que $C_t=traza(\alpha_t)$ con $\alpha_t$ regular, $L$-periódica y con $\alpha_{t|_{[0,L)}}$ inyectiva. Por las construcciones de curvas estudiadas (apartado~\ref{ej:periodica} del Ejemplo~\ref{ej:construccion}) concluímos que $C_t$ es una curva simple. Por ser cerrada y estar dentro de $\Omega$, que es acotado, será también compacta. Como $\bb{R}$ es conexo y $\alpha_t$ es continua tendremos finalmente que $C_t$ es conexa. Hemos probado que 
        \begin{gather*}
            \boxed{C_t \text{ es una curva simple compacta y conexa}}
        \end{gather*}
        Nos queda por estudiar su longitud y su curvatura.
        \begin{align*}
            \alpha_t'(s) &= (1-t\kappa(s))T(s) \Rightarrow |\alpha_t'(s)| = |1-t\kappa(s)| = 1-t\kappa(s)\\[1.25em]
            L(C_t) &= \int_0^L |\alpha_t'(s)| ds = \int_0^L (1-t\kappa(s))ds = L - t\int_0^L\kappa(s) ds = L(C) - t\int_0^L \kappa(s) ds
        \end{align*}
        Concluimos que 
        \begin{gather*}
            \boxed{L(C_t) = L(C) - t\int_0^L \kappa(s) ds}
        \end{gather*}
        Nos queda por estudiar la curvatura. Recordemos que 
        \begin{gather*}
            \kappa_t(s) = \frac{det(\alpha_t'(s), \alpha_t''(s))}{|\alpha_t'(s)|^3}
        \end{gather*}
        De los cálculos realizados hasta el momento solo nos queda calcular $\alpha_t''(s)$. Aplicando la regla de la cadena y las ecuaciones de Frenet:
        \begin{align*}
            \alpha_t''(s) &= ((1-t\kappa(s))T(s))' = -t\kappa'(s)T(s) + (1-t\kappa(s))T'(s) = \\
            &= -t\kappa'(s)T(s) + \kappa(s)(1-t\kappa(s))N(s)
        \end{align*}
        Continuamos con los cálculos, notando $\ast = -t\kappa'(s)$ (ya que solo es un escalar por lo que no nos importará para los cálculos)
        \begin{gather*}
            det(\alpha_t'(s),\alpha_t''(s)) = det((1-t\kappa(s))T(s), \ast T(s) + \kappa(s)(1-t\kappa(s))N(s))
        \end{gather*}
        por lo linealidad del determinante tendremos que 
        \begin{align*}
            det(\alpha_t'(s),\alpha_t''(s)) &= \cancel{det((1-t\kappa(s))T(s), \ast T(s) )} + det((1-t\kappa(s))T(s), \kappa(s)(1-t\kappa(s))N(s)) =\\
            &= \kappa(s)(1-t\kappa(s))^2det(T(s),N(s)) =\\
            &= \kappa(s)(1-\kappa(s))^2
        \end{align*}
        donde hemos aplicado que $\{T(s), N(s)\}$ es una base ortonormal positivamente orientada de $\bb{R}^2$ (diedro de Frenet) por lo que su determinante es $1$. Aplicando esto a la fórmula de la curvatura
        \begin{gather*}
            \kappa_t(s) = \frac{\kappa(s)(1-t\kappa(s))^2}{(1-t\kappa(s))^3}
        \end{gather*}
        y llegamos finalmente a 
        \begin{gather*}
            \boxed{\kappa_t(s) = \frac{\kappa(s)}{1-t\kappa(s)} \qquad \forall s \in \bb{R}}
        \end{gather*}
        Una observación que podemos hacer aquí es que si $t\to 0$ tendremos que $\kappa_t \to \kappa$. Sin embargo, si $t\to \nicefrac{1}{k}$ tendremos que $\kappa_t\to \infty$, lo que nos dice que las curvas colapsan.\\

        Nos queda por estudiar la relación de las áreas. Consideramos $0<t<\veps$ y $B_\veps(C)$ un entorno tubular. Llamamos $\Omega_t$ al dominio determinado por $C_t$ y queremos estudiar la relación entre $\Omega$ y $\Omega_t$.\\

        Sabemos que $C_t\subset \Omega$. Tomando complemento en la inclusión tenemos
        \begin{gather*}
            \bb{R}^2 \setminus \Omega \subset \bb{R}^2 \setminus C_t
        \end{gather*}
        Sabemos que $\bb{R}^2 \setminus \Omega = C\cup \Omega^{ext}$ y aplicando de nuevo el \hyperref[teo:clasif]{Teorema de la Curva de Jordan} tenemos que $\bb{R}^2 \setminus C_t = \Omega_t \cup \Omega_t^{ext}$. Reescribimos la expresión anterior:
        \begin{gather*}
            C\cup \Omega^{ext} \subset \Omega_t \cup \Omega_t^{ext} \sii \overline{\Omega^{ext}} \subset \Omega_t \dot{\cup} \Omega_t^{ext}
        \end{gather*}
        y sabemos que $\overline{\Omega^{ext}}$ es no acotado y conexo. Por tanto tendríamos solo dos posibilidades
        \begin{gather*}
            \overline{\Omega^{ext}} \subset \Omega_t \qquad \text{ o bien } \qquad \overline{\Omega^{ext}} \subset \Omega_t^{ext}
        \end{gather*}
        y descartamos la primera ya que $\Omega_t$ está acotado por lo que solo puede ser la segunda. Tomando complementarios de nuevo
        \begin{gather*}
            \bb{R}^2 \setminus \Omega_t^{ext} \subset \bb{R}^2 \setminus \overline{\Omega^{ext}} \sii C_t\cup \Omega_t \subset \Omega
        \end{gather*}
        De donde se deduce claramente que
        \begin{gather*}
            \Omega_t \subset \Omega
        \end{gather*} 
        Vamos a estudiar ahora el área de estos conjuntos (con el área de Lebesgue). Deducimos que 
        \begin{gather*}
            A(\Omega) = A(\Omega_t) + A(B_t^+(C))
        \end{gather*}
        Veamos si podemos extraer alguna conclusión de $A(B_t^+(C))$, lo cual nos permitirá estudiar la relación entre las áreas de $\Omega$ y $\Omega_t$. Podemos escribir
        \begin{gather*}
            A(B_t^+(C)) = A(F([0,L)\times (0,T]))
        \end{gather*}
        Aplicando la definición de área a esta igualdad tenemos
        \begin{gather*}
            A(B_t^+(C)) = A(F((0,L)\times (0,t))) = \int_{F((0,L)\times (0,t))} dsdr
        \end{gather*}
        Estamos en condiciones de aplicar el \hyperref[teo:cambio_variable]{Teorema de Cambio de Variable} (ya que $F$ es un difeomorfismo en $(0,L)\times (0,t)$) y tenemos
        \begin{gather*}
            A(B_t^+(C)) = \int_{(0,L)\times (0,t)} (1 \circ F)(s,r) |Jac(F(s,r))|drds = \int_{(0,L)\times (0,t)} |Jac(F(s,r))|drds
        \end{gather*}
        sabemos que 
        \begin{gather*}
            \frac{\partial F}{\partial s}(s,r) = (1-r\kappa(s))T(s) \qquad \qquad \frac{\partial F}{\partial r}(s,r) = N(s)
        \end{gather*}
        por lo que el Jacobiano de $F$ en el punto $(s,r)$ será 
        \begin{gather*}
            Jac(F(s,r)) = det 
            \begin{pmatrix}
                \frac{\partial F}{\partial s} \\[1em] \frac{\partial F}{\partial r}
            \end{pmatrix}
            (s,r) = (1-r\kappa(s))
        \end{gather*}
        por lo que volviendo al cálculo del Área
        \begin{gather*}
            A(B_t^+(C)) = \int_{(0,L)\times (0,t)} |Jac(F(s,r))|drds = \int_{(0,L)\times (0,t)} (1-r\kappa(s)) drds 
        \end{gather*}
        Por el Teorema de Fubini tenemos
        \begin{gather*}
            A(B_t^+(C)) = \int_0^t \left(L-r \int_0^L \kappa(s) ds\right) dr
        \end{gather*}
        donde podemos pensar en que
        \begin{gather*}
            L-r \int_0^L \kappa(s) ds = \int_0^t L(C_r)dr
        \end{gather*}
        Sin embargo, nos interesa seguir calculando la integral:
        \begin{gather*}
            A(B_t^+(C)) = \int_0^t \left(L-r \int_0^L \kappa(s) ds\right) dr = Lt - (\int_0^L \kappa(s)ds) \frac{t^2}{2}
        \end{gather*}
        Uniendo esto con la relación que estábamos estudiando
        \begin{gather*}
            A(\Omega) = A(\Omega_t) + Lt - \left(\int_0^L \kappa(s)ds\right) \frac{t^2}{2}
        \end{gather*}
        de donde finalmente llegamos a la fórmula buscada
        \begin{gather*}
            \boxed{A(\Omega_t) = A(\Omega) + Lt - \left(\int_0^L \kappa(s)ds\right) \frac{t^2}{2}}
        \end{gather*}
        que observamos que es un polinomio de grado 2.
    \end{solucion}
\end{ejercicio}

\section{Teorema de la Divergencia}

\subsection{Teorema del Área}

Vamos a comenzar recordando un teorema que ya deberíamos todos de conocer:

\begin{teo}[Teorema de Cambio De Variable]\label{teo:cambio_variable}
    Consideramos los intervalos abiertos $I,J\subset \bb{R}$ y un difeomorfismo $\phi : I \to J$. Si consideramos ahora una función $f$ cualquiera con
    \begin{gather*}
        \xymatrix{
            I \ar[r]^\phi & J \ar[r]^f & \bb{R}
        }
    \end{gather*}
    Entonces $f$ es integrable en $J$ si y solo si $(f\circ \phi)|\phi'|$ es integrable en $I$. En tal caso
    \begin{gather*}
        \int_J f = \int_I (f \circ \phi)|\phi'|
    \end{gather*}
\end{teo}

A pesar de la utilidad de este teorema, nosotros necesitamos uno más general para el problema que estamos estudiando, en el que nos gustaría que desapareciera $J$. Inicialmente pensamos en la siguiente aproximación:\\

Dada una función $h: I \to \bb{R}$ elegimos $f=h\circ \phi^{-1} : J \to \bb{R}$. Sustituyendo $f$ en el \hyperref[teo:cambio_variable]{Teorema de Cambio de Variable} obtenemos que 
\begin{gather*}
    \begin{array}{c}
        h\circ \phi^{-1} \\
        \text{integrable en $J$}
    \end{array} \sii
    \begin{array}{c}
        (h \circ \phi^{-1} \circ \phi)|\phi'|\\
        \text{es integrable en $I$}
    \end{array} \sii
    \begin{array}{c}
        h|\phi'|\\
        \text{es integrable en $I$}
    \end{array}
\end{gather*}
en cuyo caso 
\begin{gather*}
    \int_J h \circ \phi^{-1} = \int_I h|\phi'|
\end{gather*}
donde $\phi: I \to J$ es un difeomorfismo y $h:I \to \bb{R}$ es una función.
Definimos así una función nueva\footnote{se llama $n$ porque ``cuenta''} $n(\phi, h)$ dada por
\begin{gather*}
    n(\phi, h)(x) = \left\{
        \begin{array}{l c l}
            0 & \text{ si } & x\notin J = im(\phi)\\
            h(\phi^{-1}(x)) & \text{ si } & x\in J = im(\phi)
        \end{array}
    \right.
\end{gather*}
Esta función podemos reescribirla como 
\begin{gather*}
    n(\phi, h)(n) = \sum_{\phi(s)=x} h(s)
\end{gather*}
donde la suma se entiende $0$ si $\phi^{-1}(x)=\emptyset$. Con esta nueva aplicación podemos reescribir el teorema como sigue:\\

Dado $I \subset \bb{R}$ un intervalo abierto de $\bb{R}$, un difeomorfismo $\phi: I \to \phi(I)$ y $h:I \to \bb{R}$ una función cualquiera. Entonces 
\begin{gather*}
    h|\phi'| \text{ integrable en $I$ } \sii n(\phi, h) \text{ integrable en $\bb{R}$ }
\end{gather*}
En dicho caso, se tendrá que 
\begin{gather*}
    \int_I h|\phi'| = \int_\bb{R} n(\phi, h) 
\end{gather*}

La equivalencia y la igualdad anteriores tienen sentido si $\phi$ es solamente $\cc{C}^1$. Pues en este caso el teorema también es válido. Esto nos lleva a la redacción un nuevo Teorema:

\begin{teo}[Teorema del Área]\label{teo:area}
    Dado $I \subset \bb{R}$ un conjunto medible de, $\phi: I \to \phi(I)$ una función $\cc{C}^1$ y $h:I \to \bb{R}$ una función cualquiera. Entonces 
    \begin{gather*}
        h|\phi'| \text{ integrable en $I$ } \sii n(\phi, h) \text{ integrable en $\bb{R}$ }
    \end{gather*}
    donde $n(\phi, h):\bb{R}\to \bb{R}$ está definida por
    \begin{gather*}
        n(\phi, h)(n) = \sum_{\phi(s)=x} h(s)
    \end{gather*}
    donde se considera que la suma es $0$ si no hay ningún $s$ tal que $\phi(s)=x$. En cualquiera de los casos, se tendrá que 
    \begin{gather*}
        \int_I h|\phi'| = \int_\bb{R} n(\phi, h) 
    \end{gather*}
\end{teo}

Esto es claramente una generalización del \hyperref[teo:cambio_variable]{Teorema de Cambio de Variable}.

\begin{ejemplo}
    Consideramos $C\subset \bb{R}^2$ una curva simple, compacta y conexa y una parametrización por el arco $L$-periódica $\alpha: \bb{R} \to \bb{R}^2$ tal que $\alpha_{|_{[0,L)}}$ es inyectiva y tal que $\alpha$ orienta positivamente a $C$. Elegimos $I=[0,L]$ un conjunto medible y definimos
    \Func{\phi}{[0,L]}{\bb{R}}
         {s}{arg(N(s))}
    Sin embargo, vamos a considerar $\phi: J\subset [0,L] \to \bb{R}$ ya que hay que quitar los $s\in [0,L]$ que verifiquen $N(s)=(1,0)$, es decir
    \begin{gather*}
        J = I \setminus \{ s\in I \ : \ N(s)=(1,0) \}\subset I
    \end{gather*}
    En cualquier caso tenemos que su dominio es medible y $\phi$ es $\cc{C}^1$. Consideramos $h: I \to \bb{R}$ dado por $\chi_A$ (la función característica del conjunto $A$) donde 
    \begin{gather*}
        A = \{s\in [0,L] \ : \ \kappa(s) \geq 0 \}
    \end{gather*}
    Por el \hyperref[teo:area]{Teorema del Área} tenemos que $\chi_A|arg'(N)|$ es integrable en $J$ si y solo si $n(arg(N), \chi_A)$ es integrable en $\bb{R}$. En tal caso
    \begin{gather*}
        \int_J \chi_A |arg'N| = \int_\bb{R} n(arg(N), \chi_A)
    \end{gather*}
    Dado $s\in J$ tenemos que 
    \begin{gather*}
        \chi_A(s) \cdot |arg'(N(s))| = \chi_A(s) \cdot \left| \frac{1}{i} \cdot \frac{N'(s)}{N(s)} \right| = \chi_A(s) \cdot |N'(s)| = \chi_A(s)|\kappa(s) = \kappa^+(s)
    \end{gather*}
    donde $\kappa^+$ es la parte positiva de $\kappa$. Tenemos entonces que $\kappa^+$ es integrable en $J$ si y solo si $n(arg(N), \chi_A)$ es integrable en $\bb{R}$. Sabemos que $\kappa^+$ es integrable en $J$ ya que es continua y está acotada dentro de un medible lo que nos garantiza que se da la equivalencia. Tenemos entonces justificado el siguiente cálculo:
    \begin{gather*}
        \int_J \kappa^+ = \int_\bb{R} n(arg(N), \chi_A)
    \end{gather*}
    Vamos a tomar un $\theta \in \bb{R}$ y calculamos
    \begin{gather*}
        n(arg(N), \chi_A)(\theta) = \sum_{\smash{arg(N(s))=\theta}} \chi_A = 
    \end{gather*}
    Nos planteamos qué ocurrirá si $\theta\notin ]0,2\pi[$ y llegamos a la conclusión de que no habrá sumandos (en cuyo caso, pon convención considerábamos que $n(arg(N), \chi_A)=0$). Por tanto podemos escribir
    \begin{gather*}
        \int_J \kappa^+ = \int_\bb{R} n(arg(N), \chi_A) = \int_0^{2\pi} n(arg(N), \chi_A)
    \end{gather*}
    De esta forma, podemos considerar $v_\theta = (\cos(\theta), \sen(\theta))$
    \begin{gather*}
        n(arg(N), \chi_A)(\theta) = \sum_{arg(N(s))=\theta} \chi_A = \sum_{N(s)=v_\theta} \chi_A(s)
    \end{gather*}
    Dado $\theta\in]0,2\pi[$ y $v_\theta = (\cos(\theta), \sen(\theta))$ definimos 
    \Func{f}{[0,2\pi]}{\bb{R}}
         {s}{f(s) = \langle \alpha(s), v_\theta \rangle}
    Esta función $f$ medirá la altura orientada de los puntos de la curva $C$ respecto de $R_\theta$ donde
    \begin{gather*}
        R_\theta \perp v_\theta
    \end{gather*}
    Como $[0,2\pi]$ es compacto y $f$ es continua tendrá mínimo, es decir
    \begin{gather*}
        \exists s_0\in [0,2\pi] \text{ tal que $f$ alcanza su mínimo en $s_0$} 
    \end{gather*}
    Además, por resultados básicos de análisis sabemos que por ser $s_0$ mínimo se tiene
    \begin{gather*}
        f'(s_0)=0 \qquad \qquad f''(s_0)\geq 0
    \end{gather*}
    De estos resultados analíticos buscamos obtener información geométrica. 
    %TODO: En el dibujo se ve muy bien
    Vamos a comenzar con la primera condición:
    \begin{gather*}
        0=f'(s_0) = \langle \alpha'(s_0), v_\theta \rangle
    \end{gather*}
    Lo que nos dice que el vector tangente a la curva en $\alpha(s_0)$ es ortogonal a $v_\theta$, es decir, $\alpha'(s_0)\perp v_\theta$ por lo que solo nos quedan dos posibilidades para el vector normal. 
    \begin{gather*}
        N(s_0) = \pm v_\theta
    \end{gather*}
    Nos gustaría concretar su signo y, como $v_\theta$ apunta a $\Omega$ por ser $s_0$ mínimo, es decir
    \begin{gather*}
        N(s_0)=v_\theta
    \end{gather*}
    Estudiamos ahora la segunda condición usando las ecuaciones de Frenet
    \begin{gather*}
        0 \leq f''(s_0) = \langle \alpha''(s_0), v_\theta \rangle = \langle T'(s_0), v_\theta \rangle = \kappa(s_0)\langle N(s_0), v_\theta \rangle = \kappa(s_0)
    \end{gather*}
    Donde hemos aplicado que $N(s_0) = v_\theta$ por lo que $\langle N(s_0), v_\theta\rangle = 1$. Por tanto, dado $\theta\in]0,2\pi[$ y $v_\theta=(\cos(\theta), \sen(\theta))$ entonces 
    \begin{gather*}
        \exists s_0\in[0,L] \text{ tal que } N(s_0) = v_\theta \quad \text{ y } \quad \kappa(s_0)\geq 0
    \end{gather*}
    o lo que es lo mismo
    \begin{gather*}
        \exists s_0\in[0,L] \text{ tal que } N(s_0) = v_\theta \quad \text{ y } \quad \chi_A(s_0)=1
    \end{gather*}
    Con esta nueva información podemos escribir
    \begin{gather*}
        n(arg(N), \chi_A)(s) = \sum_{N(s)=v_\theta} \chi_A(s) \geq \chi_A(s_0) = 1
    \end{gather*}
    Lo que nos dice que 
    \begin{gather*}
        \int_J \kappa^+ = \int_0^{2\pi} n(arg(N), \chi_A) \geq \int_0^{2\pi} 1 = 2\pi
    \end{gather*}
    Como $J\subset I$ y $\kappa^+$ es positiva tenemos que 
    \begin{gather*}
        \boxed{\int_0^{2\pi} \kappa^+(s) ds \geq 2\pi}
    \end{gather*}
    En una circunferencia de radio $R$ es fácil ver que vale justo $2\pi$ ya que $\kappa(s)=\frac{1}{R}>0$ para todo $s\in (0,2\pi)$ por lo que
    \begin{gather*}
        \int_0^{2\pi R} \kappa^+(s)ds = \int_0^{2\pi R} \frac{1}{R} ds = \frac{1}{R}\cdot 2\pi R = 2\pi
    \end{gather*}
    Intuitivamente esto nos dice que al ``abollar'' la circunferencia conseguiremos puntos con curvatura negativa, pero este resultado nos dice que entonces aparecen puntos con curvatura positiva que de alguna forma ``compensan'' esa curvatura negativa.\\

    Este último resultado es una copia en dimensión $1$ de un famoso Teorema de superficies (el Teorema de Chern-Lashoff).\\

    Finalmente sabemos que en general $|\kappa| \geq \kappa^+$, luego
    \begin{gather*}
        \int_0^L |\kappa(s)| ds \geq \int_0^L \kappa^+(s) ds \geq 2\pi
    \end{gather*}
    En caso de que se dé la igualdad tendremos que $\kappa = \kappa^+ \geq 0$ en todo punto de la curva $C$, en cuyo caso diremos que $C$ es un \textbf{óvalo}\footnote{se puede demostrar que un óvalo es la frontera de un convexo pero nosotros no lo haremos}.
\end{ejemplo}

\subsection{Teorema de la Divergencia}

Dada $C\subset \bb{R}^2$ una curva simple, compacta y conexa, $\alpha$ una parametrización local $L$-periódica parametrizada por el arco que orienta positivamente a $C$ y dado también $\Omega$ el dominio interior de $C$, tenemos $I=[0,L]\subset \bb{R}$ medible y vamos a escribir $\alpha=(x_1, x_2)$ definiendo $\phi: [0,L]\to \bb{R}$ dada por
\begin{gather*}
    \phi = x_1
\end{gather*}
y claramente $\phi$ es diferenciable. Definimos también $h:[0,L]\to \bb{R}$ dada por
\begin{gather*}
    h = (signo(N_2))X^2
\end{gather*}
donde $N = (N_1, N_2)$ es el vector normal a la curva y $X=(X^1, X^2)$ un campo definido en $\overline{\Omega}$, es decir, $X: \overline{\Omega} \to \bb{R}^2$ campo. \\

En este contexto, el \hyperref[teo:area]{Teorema del Área} nos dice que 
\begin{gather*}
    h|\phi'| \text{ integrable en $I$ } \sii n(\phi, h) \text{ integrable en $\bb{R}$} 
\end{gather*}
y que en tal caso sus integrales coincide. Veamos ahora quiénes son exactamente esas funciones en la situación planteada. Tenemos que 
\begin{gather*}
    h|\phi'| = (signo(N_2))X^2|x_1'|\\[1.25em]
    N = (N_1, N_2) = J(x_1', x_2') = (-x_2', x_1') \Rightarrow N_2 = x_1'
\end{gather*}
De esta forma tenemos que podemos expresar
\begin{gather*}
    h|\phi'| = (signo(N_2)) \cdot X^2 |N_2| = X^2\cdot N_2
\end{gather*}
Nos planteamos si esto es integrable en $I$. Sabemos que tanto $N_2$ como $X^2$ son diferenciables por lo que, en particular, $X^2 \cdot N_2$ es continua y además $I$ es compacto lo que finalmente nos confirma que $X^2 N_2$ es integrable en $I$. Aplicamos así el \hyperref[teo:area]{Teorema del Área} donde ambas integrales tienen sentido:
\begin{gather*}
    \int_0^L X^2N_2 = \int_\bb{R} n(\phi, h)
\end{gather*}
Nos planteamos ahora quién es $n(\phi, h)$. Tomamos $a\in \bb{R}$ y aplicando la definición tenemos
\begin{gather*}
    n(\phi, h)(a) = \sum_{x_1(s)=a} (signo(N_2(s)))\cdot X^2(\alpha(s))
\end{gather*}
Buscamos los $s\in I$ tales que $\alpha(s)=(x_1(s), x_2(s)) = (a, x_2(s))$, es decir,
\begin{gather*}
    n(\phi, a) = \sum_{(a,y)\in C}(signo(N_2(a,y)))X^2(a,y)
\end{gather*}
Buscar $s$ es lo mismo que buscar imágenes en $C$, es decir, buscar $(a,y)\in C$. Esto es, en el dibujo de ejemplo, ver los puntos de corte de $C$ con la recta vertical $x=a$.

En notación nos interesa el conjunto
\begin{gather*}
    \{(a,y)\in C : y\in \bb{R}\} = \{x=a\}\cap C
\end{gather*}
y para casi todo $a$ tendremos que $\{x=a\}\transv C$. Esto nos dice que el número de puntos de corte con $C$ es finito y, de hecho, como estamos considerando la intersección de $C$ no solo con una sermiirecta sino con la recta entera, entonces $\#(\{x0a\}\cap C)=2k$ con $k\in \bb{N}$ (ya que dado cualquier punto de la recta la paridad de las dos semirrectas que determina deberá coincidir para respetar el \hyperref[teo:clasif]{Teorema de la Curva de Jordan}).\\

Como $\#(\{x=a\} \cap C)<\infty$ podemos ordenar este conjunto y lo hacemos de la siguiente forma
\begin{gather*}
    \{x=a\}\cap C = \{(a,y_1), (a,y_2), ..., (a,y_{2k})\}
\end{gather*}
de forma que $y_1<y_2<...<y_{2k}$. Así podemos expresar mejor qué es $h(\phi, h)(a)$ como
\begin{gather*}
    n(\phi, h)(a) = \sum_{i=1, 2, ..., 2k} (signo (N_2(a.y_i)))X^2(a,y_i)
\end{gather*}
Notemos que $N_2 = \langle N, (0,1) \rangle$. Como para el primer punto, $(a,y_1)$, el normal entrará dentro de $\Omega$ y hay transversalidad tanto de ese normal como de la recta $\{x=a\}$ entonces, en este caso, están tanto $N(a_1, y_1)$ como el vector $(0,1)$ sobre el punto $(a,y_1)$ al mismo lado de un semiplano (el determinado por la recta tangente en $(a,y_1)$). Esto nos dice que el ángulo formado por $N(a,y_1)$ y $(0,1)$ es agudo, es decir,
\begin{gather*}
    \langle N(a, y_1) , (0,1) \rangle > 0
\end{gather*}
Siguiendo un razonamiento similar, pero concreto para este otro caso siguiente, también es sencillo ver que
\begin{gather*}
    \langle N(a, y_2) , (0,1) \rangle < 0
\end{gather*}
De hecho tendremos 
\begin{align*}
    \langle N(a, y_j) , (0,1) \rangle > 0 &\qquad 1 \leq j < 2k,\ \ j \text{ impar}\\
    \langle N(a, y_j) , (0,1) \rangle < 0 &\qquad 1 < j \leq 2k,\ \ j \text{ par}
\end{align*}

Por tanto, se verifica
\begin{align*}
    n(\phi, h)(a) &= X^2(a,y_1)-X^2(a,y_2) + ... + X^2(a,y_{2k-1}) - X^2(a,y_{2k}) =\\
    &\overset{(\ast)}{=} - \int_{y_1}^{y_2} \frac{\partial X^2}{\partial x_2} (a,y)dy - ... - \int_{y_{2k-1}}^{y_{2k}} \frac{\partial X^2}{\partial x_2} (a,y)dy =\\
    &= -\int_\bb{R} \frac{\partial \tilde{X}^2}{\partial x_2}(a,y) dy
\end{align*}
donde en $(\ast)$ hemos agrupado los términos consecutivos $2$ a $2$ y hemos aplicado el Teorema Fundamental del Cálculo (1 variable) y $\tilde{X}$ es la extensión de $X$ por cero fuera de $\overline{\Omega}$, es decir
\begin{gather*}
    \tilde{X} = 
    \left\{
        \begin{array}{l l}
            0 & \text{ fuera de $\overline{\Omega}$}\\
            X & \text{ en  $\overline{\Omega}$}
        \end{array}
    \right.
\end{gather*}
El resultado al que hemos llegado hasta ahora es 
\begin{gather*}
    n(\phi, h)(a) = -\int_\bb{R} \frac{\partial \tilde{X}^2}{\partial x_2}(a,y) dy
\end{gather*}
y recordemos que por aplicar el \hyperref[teo:area]{Teorema del Área} nos interesaba realmente
\begin{gather*}
    \int_\bb{R} n(\phi, h) \left( = \int_0^L X^2N_2 \right)
\end{gather*}
Así que
\begin{gather*}
    \int_0^L X^2N_2 = - \int_\bb{R} \left(\int_\bb{R} \frac{\partial \tilde{X}^2}{\partial x_2}(a,y) dy\right)da \overset{(\ast)}{=} -\int_{\bb{R}^2} \frac{\partial \tilde{X}^2}{\partial x_2} \overset{(\ast\ast)}{=} - \int_\Omega \frac{\partial \tilde{X}^2}{\partial x_2} \overset{(\ast\ast\ast)}{=} -\int_\Omega \frac{\partial X^2}{\partial x_2} 
\end{gather*}
donde en $(\ast)$ hemos aplicado el Teorema de Fubini, en $(\ast\ast)$ hemos aplicado que en $\bb{R}^2 \setminus\Omega$ se tiene $\tilde{X}=0$ y en $(\ast\ast\ast)$ hemos aplicado que en $\Omega$ se tiene $\tilde{X}=X$.\\

En conclusión, tenemos que 
\begin{gather*}
    \boxed{\int_\Omega \frac{\partial X^2}{\partial x_2} = - \int_0^L X^2 N_2}
\end{gather*}
Podemos ver que en esta fórmula aparece el número $2$ en todos lados y nos planteamos si podríamos repetir el mismo razonamiento pero cambiando los papeles de $1$ y $2$ y la respuesta es que sí (compruébese) por lo que también es cierto que
\begin{gather*}
    \boxed{\int_\Omega \frac{\partial X^1}{\partial x_1} = - \int_0^L X^1 N_1}
\end{gather*}
Para obtener una fórmula que no dependa de las dos expresiones pensamos en sumar ambasa obteniendo así
\begin{gather*}
    \int_\Omega \left( \frac{\partial X^1}{\partial x_1} + \frac{\partial X^2}{\partial x_2} \right) = - \int_0^L \langle X, N \rangle
\end{gather*}
A este primer término le pondremos nombre
\begin{definicion}
    Dado $A\subset \bb{R}^2$, un campo $X:A \to \bb{R}^2$, $C$ una curva simple, compacta y conexa con $\alpha: I=[0,L]$ una parametrización por el arco suya que la orienta positivamente y $\Omega$ el dominio interior de $C$ definimos la \textbf{divergencia} de $X$ como $div(X):\overline{\Omega}\to \bb{R}$ dada por
    \begin{gather*}
        div(X) = \left( \frac{\partial X^1}{\partial x_1} + \frac{\partial X^2}{\partial x_2} \right)
    \end{gather*}
\end{definicion}

Con esta definición, la expresión anterior queda como 
\begin{gather*}
    \boxed{\int_\Omega div(X) = - \int_0^L \langle X, N \rangle}
\end{gather*}
lo cual es el Teorema de la Divergencia, que enunciaremos debidamente más adelante. Notemos que es fundamental que $\alpha$ oriente positivamente a $C$ pues en caso contrario la fórmula saldría con el signo opuesto. Se suele denotar el segundo término por 
\begin{gather*}
    - \int_0^L \langle X, N \rangle = - \int_C \langle X, N \rangle = - \oint \langle X, N \rangle
\end{gather*}
(Los físicos dicen que este teorema es trivial, más adelante veremos también por qué) Fijémonos por último que este teorema es como una generalización del Teorema Fundamental del Cálculo, pues dice que la integral de una cierta ``derivada'' (la divergencia) se calcula como una integral en el borde (la curva $C$). Formalicemos el Teorema:

\begin{teo}[Teorema de la Divergencia (Gauss, Grin, Strokes, Ostrogradski)]\label{teo:divergencia}
    Sea $C\subset \bb{R}^2$ una curva simple, compacta y conexa donde $\Omega$ es su dominio (interior) y $X: \overline{\Omega} \to \bb{R}$ un campo $C^1$, entonces
    \begin{gather*}
        \int_\Omega div(X) = -\int_C \langle X, N \rangle = -\int_0^L \langle X_{\alpha(s)}, N(s) \rangle ds
    \end{gather*}
    donde $div(X)= \dfrac{\partial X^1}{\partial x_1} + \dfrac{\partial X^1}{\partial x_1}$ y $\alpha$ parametriza a $C$ (por el arco) orientándola positivamente.
\end{teo}

\begin{observacion}
    Vamos a hacer una serie de observaciones-ejercicios acerca de lo que acabamos de ver
    \begin{enumerate}
        \item ¿Por qué los físicos ``dicen'' que este teorema es trivial?\\
        Para un $|t|<\veps$ definimos $\phi_t : \overline{\Omega} \to \bb{R}^2$ dado por $\phi_t=I+tX$. Si lo pensamos, esta función hace
        \begin{gather*}
            p \underset{\phi_t}{\mapsto} p +t X_p
        \end{gather*}
        Suponemos que $\veps$ es suficientemente pequeño para que 
        \begin{gather*}
            \phi_t : \overline{\Omega} \to \phi_t(\overline{\Omega})
        \end{gather*}
        sea difeomorfismo. Creemos que existe dicho $\veps>0$ y queremos estudiar cómo cambia el área al modificar $t$. De esta forma llamamos $A(t)$ al área de $\phi_t(\Omega)$, es decir,
        \begin{gather*}
            A(t) = Area(\phi_t(\Omega)) = \int_{\phi_t(\Omega)} 1
        \end{gather*}
        Tenemos la siguiente situación
        \begin{gather*}
            \xymatrix{
                \Omega \ar[r]^{\phi_t}_{dif.} & \phi_t(\Omega) \ar[r]^1 &\bb{R}^2
            }
        \end{gather*}
        Lo que nos confirma que podemos aplicar el \hyperref[teo:cambio_variable]{Teorema de Cambio de Variable}. De esta forma
        \begin{gather*}
            A(t) = \int_\Omega (1\circ \phi_t)|Jac(\phi_t)| = \int_\Omega |Jac(\phi_t)|
        \end{gather*}
        y vemos que así es mucho más fácil manejar una integral dependiente de un parámetro. Vamos ahora a intentar calcular el valor de $Jac(\phi_t)$. Vamos a escribir primero
        \begin{gather*}
            (x_1, x_2) \overset{\phi_t}{\longmapsto} (x_1, x_2) + t(X^1(x_1, x_2), X^2(x_1, x_2))
        \end{gather*}
        de donde
        \begin{gather*}
            Jac(\phi_t) = Jac(I + tX) = det
            \begin{pmatrix}
                1 + t\dfrac{\partial X^1}{\partial x_1} & t \dfrac{\partial X^2}{\partial x_1}\\[1em]
                t \dfrac{\partial X^1}{\partial x_2} & 1 + t\dfrac{\partial X^2}{\partial x_2}
            \end{pmatrix}
            = 1 + t(div(X)) + t^2(Jac(X))
        \end{gather*}
        Por lo que ya podemos ver el valor del área
        \begin{gather*}
            A(t) = \int_\Omega (1+t\cdot div(X) + t^2 \cdot Jac(X))
        \end{gather*}
        Vamos a estudiar ahora cómo varía el área. Para ello vamos a calcular la siguiente derivada
        \begin{gather*}
            \left.\frac{d}{dt} \right|_{t=0} A(t) = \left.\frac{d}{dt} \right|_{t=0} \int_\Omega (1+t\cdot div(X) + t^2 \cdot Jac(X)) = \int_\Omega div(X)
        \end{gather*}
        y acabamos de dar una interpretación de la divergencia. Esto además justifica el hecho de que llamemos a esto divergencia. Ahora queremos saber por qué para los físicos es ``trivial'' el hecho de que 
        \begin{gather*}
            \left.\frac{d}{dt} \right|_{t=0} A(t) = \int_\Omega div(X) = -\int_C \langle X, N \rangle
        \end{gather*}
        Una interpretación que podemos hacer es que $\left.\frac{d}{dt} \right|_{t=0} A(t)$ es lo que sale menos lo que entra en $\Omega$ por $C$.

        \item ¿Qué ocurre si no encuentro una parametrización por el arco de $C$?\\
        
        Supongamos que $\alpha: \bb{R} \to \bb{R}^2$ es una parametrización regular \underline{no parametrizada por el arco} en general, $A$-periódica, con $\alpha_{|_{[0,A)}}$ inyectiva, $tr(\alpha)=C$ y tal que $\alpha$ orienta positivamente a $C$. El teorema de la divergencia se escribirá de la siguiente forma:
        \begin{gather*}
            \int_\Omega div(X) = - \int_C \langle X,N \rangle = - \int_0^A \langle X_{\alpha(t)}, J\alpha'(t)\rangle dt
        \end{gather*}
        Esto se puede ver aplicando el \hyperref[teo:cambio_variable]{Teorema de Cambio de Variable} al \hyperref[teo:area]{Teorema del Área} (se deja como ejercicio el cálculo concreto).

        \item Supongamos que $A\subset \bb{R}^2$ y que tenemos el campo diferenciable $X: A \to \bb{R}^2$. Sabemos que si $p\in A$, entonces
        \begin{gather*}
            div(X)(p) = traza((dX)_p) = \sum_{i=1}^2 \langle (dX)_p(e_i), e_i \rangle
        \end{gather*}
        donde $\{e_1, e_2\}$ es una base ortonormal de $\bb{R}^2$.
    \end{enumerate}
\end{observacion}

\begin{propiedades}\
    \begin{enumerate}
        \item Si $X,Y$ campos, entonces $div(X+Y)=div(X)+div(Y)$.
        \item Si $\lambda\in \bb{R}$ y $X$ campo, entonces $div(\lambda X) = \lambda\ div(X)$.
        \item Si $X$ campo en $A$ y $f$ una función diferenciable en $A$, entonces
        \begin{gather*}
            div(fX) = \langle \nabla f, X \rangle + f\ div(X)
        \end{gather*}
        \item Si $X=\nabla f$ donde $f:A\to \bb{R}$ diferenciable, entonces
        \begin{gather*}
            \nabla f_p = \left\langle \frac{\partial f}{\partial x_1}(p), \frac{\partial f}{\partial x_2}(p) \right\rangle
        \end{gather*}
        por lo que 
        \begin{gather*}
            (div (\nabla f))(p) = \frac{\partial^2 f}{\partial x_1^2}(p) + \frac{\partial^2 f}{\partial x_2^2}(p) = (\Delta f)(p)
        \end{gather*}
        donde $\Delta f$ es el \textbf{Laplaciano} de $f$.
        \item $div(J(\nabla f)) = 0$ para todo $f$.
    \end{enumerate}
\end{propiedades}

\begin{observacion}
    Continuando con las observaciones anteriores:
    \begin{enumerate}[start=4]
        \item Consideramos $C$ y $\Omega$ y el campo $X:\bb{R}^2 \to \bb{R}^2$ dado por 
        \begin{gather*}
            X(p)=a \qquad \forall p \in \bb{R}^2
        \end{gather*}
        donde $a\in \bb{R}^2$. Tenemos que $(dX)_p=0$ por lo que 
        \begin{gather*}
            (div(X))(p) = tr(dX)_p = 0
        \end{gather*}
        y por el \hyperref[teo:divergencia]{Teorema de la Divergencia} tenemos que 
        \begin{align*}
            0 &= \oint \langle N, a \rangle = \int_C \langle C, a \rangle = \int_0^L \langle N(s), a \rangle ds
            = a_1 \int_0^L N_1(s) ds + a_2\int_0^L N_2(s)ds =\\
            &= \left\langle (a_1, a_2), \left(  \int_0^L N_1(s) ds,  \int_0^L N_2(s) ds \right) \right\rangle = \left\langle a,  \int_0^L N(s) ds \right\rangle =\\
            &= \left\langle \int_0^L N(s) ds, a \right\rangle
        \end{align*}
        Como esto ocurre para todo $a\in \bb{R}^2$, necesariamente tendremos $\int_0^L N(s) ds = 0$

        \item Consideramos $C$ y $\Omega$ y el campo $X:\bb{R}^2 \to \bb{R}^2$ dado por 
        \begin{gather*}
            X(p)=p-a \qquad \forall p \in \bb{R}^2
        \end{gather*}
        donde $a\in \bb{R}^2$ fijo. Vamos a tratar de calcular la divergencia:
        \begin{gather*}
            (div(X))(p) = tr(dX)_p
        \end{gather*}
        Vamos a calcular entonces $(dX)_p$
        \begin{gather*}
            (dX)_p = I_{\bb{R}^2} \Rightarrow (div(X))(p) = tr(I_{\bb{R}^2})=2
        \end{gather*}
        Esto nos dice que 
        \begin{gather*}
            \int_\Omega 2 = - \int_0^L \langle X_{\alpha(s)}, N(s) \rangle ds = -\int_0^L \langle \alpha(s)-a, N(s) \rangle ds
        \end{gather*}
        de donde
        \begin{gather*}
            2A(\Omega) =  -\int_0^L \langle \alpha(s)-a, N(s) \rangle ds = -\int_0^L \langle \alpha(s), N(s) \rangle ds
        \end{gather*}
        lo que nos lleva a la siguiente fórmula para calcular el área de $\Omega$:
        \begin{gather*}
            \boxed{A(\Omega) = -\frac{1}{2} \int_0^L \langle \alpha(s), N(s)\rangle ds}
        \end{gather*}
        Vamos a usar esta fórmula en un ejemplo. Consideramos $C=S^1(b, r)$ y tendríamos $\Omega = B(b,r)$ y aplicando la fórmula obtenemos
        \begin{gather*}
            A(B(b,r)) = - \frac{1}{2} \int_0^{2\pi r} \langle \alpha(s)-b, N(s)\rangle ds
        \end{gather*}
        donde $N(s) = \frac{b-\alpha(s)}{r}$ y sustituyendo
        \begin{align*}
            A(B(b,r)) &= - \frac{1}{2} \int_0^{2\pi r} \langle \alpha(s)-b, N(s)\rangle ds = - \frac{1}{2} \int_0^{2\pi r} \left\langle \alpha(s)-b, \frac{b-\alpha(s)}{r}\right\rangle ds =\\
            &= \frac{1}{2} \int_0^{2\pi r} \frac{|\alpha(s)-b|^2}{r} ds = \frac{1}{2} \int_0^{2\pi r} \frac{r^{\cancel{2}}}{\cancel{r}} ds = \frac{1}{{2}} r \int_0^{2\pi r} 1ds = \frac{r}{2}\cdot 2\pi r = \pi r^2
        \end{align*}
        Y vemos que la fórmula funciona para este caso que ya conocemos.

        \item Consideramos $C$ y $\Omega$ y el campo $X:\bb{R}^2 \to \bb{R}^2$ dado por 
        \begin{gather*}
            X(p)=\langle p, a \rangle b \qquad \forall p \in \bb{R}^2
        \end{gather*}
        donde vamos a llamar $f(p)=\langle p, a \rangle$ y $Y(p)=b$ donde $a,b\in \bb{R}^2$ fijos. Calculamos de nuevo la divergencia
        \begin{gather*}
            div(X) = div(fY) = \langle \nabla f, Y \rangle + \cancel{f\ div(Y)} = \langle \nabla f, Y \rangle
        \end{gather*}
        Vamos a calcular dicho gradiente para cualquier vector $v\in \bb{R}^2$
        \begin{gather*}
            \langle \nabla f, Y \rangle = (df)_p(v) =\left. \frac{d}{dt}\right|_{t=0} \langle p+tv, a\rangle = \langle v,a\rangle
        \end{gather*}
        por lo que $(\nabla f)_p = a$ para todo $p\in \bb{R}^2$ y
        \begin{gather*}
            div(X) = \langle \nabla f, Y \rangle = \langle a, b \rangle
        \end{gather*}
        Por el \hyperref[teo:divergencia]{Teorema de la Divergencia}
        \begin{gather*}
            \int_\Omega \langle a,b \rangle = -\int_0^L \langle \langle \alpha(s), a \rangle b, N(s) \rangle ds = - \int_0^L \langle \alpha(s), a \rangle \langle N(s), b \rangle ds
        \end{gather*}
        Podemos escribir
        \begin{gather*}
            \langle a,b\rangle A(\Omega) = -\int_0^L \langle \alpha(s), a \rangle \langle N(s), b \rangle ds \qquad \forall b\in \bb{R}^2
        \end{gather*}
        Por lo que 
        \begin{gather*}
            \langle A(\Omega)a, n \rangle = -\int_0^L \langle \alpha(s), a \rangle \langle N(s), b \rangle ds \qquad \forall b\in \bb{R}^2
        \end{gather*}
        Como esto ocurre para todo $b$ podemos escribir
        \begin{gather*}
            \boxed{A(\Omega) a = - \int_0^L \langle \alpha(s), a \rangle N(s) ds \qquad \forall a \in \bb{R}^2}
        \end{gather*}
        que se conoce como \textbf{Teorema de Arquímedes}.

        \item Consideramos $a\in \bb{R}^2$ fijo, y el campo $X:\bb{R}^2 \setminus \{a\}\to \bb{R}^2$ dado por 
        \begin{gather*}
            X(p)=\frac{p-a}{|p-a|^2}
        \end{gather*}
        Para interpretar mejor cómo actúa este campo vamos a escribir
        \begin{gather*}
            X(p)= \frac{1}{|p-a|} \cdot \frac{p-a}{|p-a|}
        \end{gather*}
        y vemos que actúa en cada punto $p$ en la dirección de la recta que une $p$ con $a$ y es inversamente proporcional a la distancia de dicho $p$ con $a$. Esto nos puede recordar a un campo de fuerzas centrales de los que estudiamos en la física (pero la distancia no aparece al cuadrado). Queremos ver ahora que la divergencia de este campo es $0$. Para ello
        \begin{gather*}
            (div(X))(p) = traza(dX)_p = (dX)_p
        \end{gather*}
        donde $(dX)_p:\bb{R}^2\to \bb{R}^2$ es un endomorfismo del plano. Para ver cómo actúa tomamos $v\in \bb{R}^2$ y calculamos la derivada direccional
        \begin{gather*}
            (dX)_p(v) = \left.\frac{d}{dt}\right|_{t=0} X(p+tv) = \frac{v}{|p-a|^2} - \frac{2\langle v, p-a \rangle}{|p-a|^4}(p-a) \qquad \forall v \in \bb{R}^2
        \end{gather*}
        y podemos calcular ya la divergencia
        \begin{align*}
            (div(X))(p) &= tr(dX)_p = \sum_{i=1}^2 \langle (dX)_p(e_i), e_i \rangle =\\
            &= \sum_{i=1}^2 \left\langle \frac{e_i}{|p-a|}-2\frac{\langle e_i, p-a \rangle}{|p-a|}(p-a), e_i \right\rangle =\\
            &= \sum_{i=1}^2 \left(\frac{|e_i|^2}{|p-a|^2} - 2\frac{\langle e_i, p-a\rangle^2}{|p-a|^4}\right) =\\
            &= \frac{2}{|p-a|^2} - 2 \frac{|p-a|^2}{|p-a|^4} = \frac{2}{|p-a|^2} - \frac{2}{|p-a|^2} =\\
            &= 0
        \end{align*}
        y efectivamente tenemos que para cualquier punto $p\in \bb{R}^2 \setminus \{a\}$ se tiene $(div(X))(p)=0$. Se deja como ejercicio ver que $X$ es el gradiente de $f$ para cierta $f:\bb{R}^2\setminus\{a\} \to \bb{R}^2$, es decir
        \begin{gather*}
            X = \nabla f
        \end{gather*}
        y veremos que dicha $f$ es la energía potencial. En dicho caso tendremos $div(X)=\Delta f=0$ y diremos que $f$ es una \textbf{función armónica}.\\

        Hacemos ahora aparecer la curva $C\subset \bb{R}^2$ simple, compacta y conexa. Veamos una distinción de casos
        \begin{enumerate}
            \item Supongamos que $a\notin \overline{\Omega}$, entonces $X$ está bien definido en $\overline{\Omega}$ y es diferenciable. Podemos aplicar por tanto el \hyperref[teo:divergencia]{Teorema de la Divergencia}.
            \begin{gather*}
                0 = \int_\Omega div(X) = - \int_0^L \frac{\langle \alpha(s)-a, N(s)\rangle}{|\alpha(s)-a|^2} ds
            \end{gather*}

            \item Si $\Omega$, entonces no podemos aplicar el Teorema, ya que $X$ no está definido en $a$. Veamos una definición y una observación para poder continuar:
            \begin{definicion}[Dominio Regular]
                Sea $D\subset \bb{R}^2$ un abierto acotado. Si 
                \begin{gather*}
                    Fr(D) = C_1 \dot{\cup} ... \dot{\cup} C_k
                \end{gather*}
                donde $C_i$ son curvas simples compactas y conexas diremos que $D$ es un \textbf{dominio regular}.
            \end{definicion}
            \begin{observacion}
                En un dominio regular es válido el \hyperref[teo:divergencia]{Teorema de la Divergencia} (y su demostración).
            \end{observacion}

            Por tanto vamos a tratar de excluir el punto $a$ del dominio $\Omega$ encerrándolo en una curva. Como $a\in \Omega$ que es abierto tendremos que existe un $\veps>0$ tal que 
            \begin{gather*}
                B(a,\veps) \subset \Omega
            \end{gather*}
            Vamos a considerar entonces $\Omega_\veps = \Omega \setminus \overline{B}(a,\veps)$ que es un abierto en $\bb{R}^2$ y acotado (ya que está dentro de $\Omega$). Además,
            \begin{gather*}
                Fr(\Omega_\veps) = C \cup S^1(a,\veps)
            \end{gather*}
            Aplicamos el \hyperref[teo:divergencia]{Teorema de la Divergencia} en $\Omega_\veps$ para el campo $X$, ya que está bien definido en $\Omega_\veps$.

            \begin{figure}[H]
                \centering
                \begin{tikzpicture}[scale=0.8]
                    \shorthandoff{>}

                    % Puntos principales
                    \coordinate (P1) at (0, 1);   
                    \coordinate (P2) at (2, 3);
                    \coordinate (P3) at (5, 2);   
                    \coordinate (P4) at (4, -2);
                    \coordinate (P5) at (2, 0);

                    % Dominio \Omega_\veps
                    \filldraw[thick, ForestGreen, fill=ForestGreen!5] plot [smooth cycle, tension=0.7] coordinates {
                        (P1) (P2) (P3) (P4) (P5)
                    };
                    \node[ForestGreen] at (4,1) {$\Omega_\veps$};
                    \filldraw[thick, ForestGreen, fill=white] (2,1.5) circle (0.5);
                    \node[ForestGreen] at (5.5,2) {$C$};
                    \node[ForestGreen] at (2, 0.75) {$S(a,\veps)$};

                    % Punto a
                    \fill[red!80!black] (2,1.5) circle (2.5pt); 
                    \node[red!80!black] at (2.25, 1.5) {$a$};     
                \end{tikzpicture}
            \end{figure}
            Si consideramos $\beta(s)\in S^1(a,\veps)$ donde $\beta$ es una parametrización de la segunda curva que ha aparecido.
            \begin{align*}
                0 &= \int_0^L \frac{\langle \alpha(s)-a, N(s)\rangle}{|\alpha(s)-a|^2} ds +\int_0^{2\pi \veps} \frac{\langle \beta(s)-a, N(s)\rangle}{|\beta(s)-a|^2} = \\
                &= \int_0^L \frac{\langle \alpha(s)-a, N(s)\rangle}{|\alpha(s)-a|^2} ds +\int_0^{2\pi \veps} \frac{1}{\veps} = \\
                &= \int_0^L \frac{\langle \alpha(s)-a, N(s)\rangle}{|\alpha(s)-a|^2} ds + 2 \pi
            \end{align*}
            y llegamos a la siguiente expresión
            \begin{gather*}
                \int_0^L \frac{\langle \alpha(s)-a, N(s) \rangle}{|\alpha(s)-a|^2} ds = 
                \left\{
                    \begin{array}{l c c}
                        0 & \text{ si } & a\notin \overline{\Omega}\\
                        -2\pi & \text{ si } & a\in \Omega
                    \end{array}
                \right.
            \end{gather*}
        \end{enumerate}
    \end{enumerate}
\end{observacion}
