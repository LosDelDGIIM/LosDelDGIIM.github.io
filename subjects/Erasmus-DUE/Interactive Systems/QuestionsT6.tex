\begin{ejercicio}
    Give an explanation of the concept of internal and external attribution according to the Attribution Theory and provide an example for each in the context of HCI.
\end{ejercicio}

\begin{ejercicio}
    Explain the main concepts of Activity Theory.
\end{ejercicio}

\begin{ejercicio}
    Draw an activity theory triangle for the action ``writing the exam'' regarding the activity ``visiting the course Interactive Systems''. Then analyse your solution and name two different actions and operations in the activity.
\end{ejercicio}

    \end{itemize}
\end{ejercicio}
